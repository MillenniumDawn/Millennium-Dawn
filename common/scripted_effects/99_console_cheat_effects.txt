# Console cheats, run on the player country with `effect <name>`.
# Nothing outside this file calls these, so validate_unused_scripted skips it.

cheat_reset_economy = {
	cheat_reset_currency_strength = yes
	cheat_reset_inflation = yes
	cheat_clear_debt = yes
	cheat_reset_treasury = yes
}

cheat_reset_inflation = {
	set_variable = { inflation_rate_var = 0 }
	clear_array = year_quarter_inflation_tracker
	resize_array = { array = year_quarter_inflation_tracker value = 0 size = 4 }
	calculate_inflation_effects = yes
	ingame_update_setup = yes
}

cheat_clear_debt = {
	set_variable = { debt = 0 }
	ingame_update_setup = yes
}

cheat_reset_treasury = {
	set_variable = { treasury = 0 }
	update_money_dirty_variable = yes
}

cheat_reset_currency_strength = {
	set_variable = { currency_strength = 1 }
	set_variable = { currency_inflation_feed = 0 }
	calculate_inflation_effects = yes
	ingame_update_setup = yes
}
