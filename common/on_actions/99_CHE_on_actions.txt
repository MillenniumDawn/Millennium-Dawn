#---------# Chechnya On Actions: #---------#
#------------------------------------------#
on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = {
					tag = CHE
				}
				CHE_init_devastation_effect = yes
			}
		}
	}

	on_weekly_CHE = {
		effect = {
			CHE_apply_weekly_devastation_drift_effect = yes
		}
	}
}
