	}
	else_if = {
		limit = { check_variable = { MD_ftc_academy_level = 1 } }

		if = {
			limit = { has_idea = MD_ftc_academy_blue_water_1 }
			swap_ideas = { remove_idea = MD_ftc_academy_blue_water_1 add_idea = MD_ftc_academy_blue_water_2 }
		}
		if = {

		}
		if = {
			limit = { has_idea = MD_ftc_academy_guerre_de_course_1 }
			swap_ideas = { remove_idea = MD_ftc_academy_guerre_de_course_1 add_idea = MD_ftc_academy_guerre_de_course_2 }
		}
	}
	else = {
		navy_experience = @MD_ftc_upgrade_cost_3_neg
		if = {
			limit = { has_idea = MD_ftc_academy_blue_water_2 }
			swap_ideas = { remove_idea = MD_ftc_academy_blue_water_2 add_idea = MD_ftc_academy_blue_water_3 }
		}
		if = {
			limit = { has_idea = MD_ftc_academy_green_water_2 }
			swap_ideas = { remove_idea = MD_ftc_academy_green_water_2 add_idea = MD_ftc_academy_green_water_3 }

		if = {
			limit = { has_idea = MD_ftc_academy_jeune_ecole_2 }
			swap_ideas = { remove_idea = MD_ftc_academy_jeune_ecole_2 add_idea = MD_ftc_academy_jeune_ecole_3 }
		}
		if = {
			limit = { has_idea = MD_ftc_academy_guerre_de_course_2 }
			swap_ideas = { remove_idea = MD_ftc_academy_guerre_de_course_2 add_idea = MD_ftc_academy_guerre_de_course_3 }

MD_ftc_record_doctrine = {
	if = {
		limit = { has_doctrine = blue_water }
		set_country_flag = MD_ftc_doctrine_blue_water
	}
	if = {
		limit = { has_doctrine = green_water }
		set_country_flag = MD_ftc_doctrine_green_water
	}
	if = {
		limit = { has_doctrine = jeune_ecole }
		set_country_flag = MD_ftc_doctrine_jeune_ecole
	}
	if = {
		limit = { has_doctrine = guerre_de_course }
		set_country_flag = MD_ftc_doctrine_guerre_de_course
	}
}

MD_ftc_reset_on_doctrine_change = {
	hidden_effect = {
		if = {
			limit = { MD_ftc_doctrine_changed = yes }
			set_variable = { MD_ftc_academy_level = 0 }
			set_variable = { MD_ftc_fleet_level = 0 }
			set_variable = { MD_ftc_command_level = 0 }
			clr_country_flag = MD_ftc_doctrine_blue_water
			clr_country_flag = MD_ftc_doctrine_green_water
			clr_country_flag = MD_ftc_doctrine_jeune_ecole
			clr_country_flag = MD_ftc_doctrine_guerre_de_course
		}
		MD_ftc_record_doctrine = yes
	}
}

MD_ftc_toggle_window = {
	MD_ftc_reset_on_doctrine_change = yes
	if = {
		limit = { NOT = { has_country_flag = MD_ftc_open } }
		set_country_flag = MD_ftc_open
	}
	else = {
		clr_country_flag = MD_ftc_open
	}
	update_MD_ftc_dirty_variable = yes
}

update_MD_ftc_dirty_variable = {
	if = {
		limit = { check_variable = { global.MD_ftc_dirty_update_var < 10000 } }
		add_to_variable = { global.MD_ftc_dirty_update_var = 1 }
	}
	else = {
		set_variable = { global.MD_ftc_dirty_update_var = 1 }
	}
}
