	artsakh_screen_gui = {
		context_type = decision_category
		window_name = "artsakh_screen_gui"
		dirty = global.ARM_artsakh_dirty_update_var

		effects = {
			region_stepanakert_click = {
				set_country_flag = gu_stepanakert_select
				clr_country_flag = gu_hadrut_select
				clr_country_flag = gu_karvachar_select
				update_arm_artsakh_dirty_variable = yes
			}
			region_karvachar_click = {
				clr_country_flag = gu_stepanakert_select
				clr_country_flag = gu_hadrut_select
				set_country_flag = gu_karvachar_select
				update_arm_artsakh_dirty_variable = yes
			}
			region_hadrut_click = {
				clr_country_flag = gu_stepanakert_select
				set_country_flag = gu_hadrut_select
				clr_country_flag = gu_karvachar_select
				update_arm_artsakh_dirty_variable = yes
			}
			artsakh_cancel_button_click = {
				clr_country_flag = gu_stepanakert_select
				clr_country_flag = gu_hadrut_select
				clr_country_flag = gu_karvachar_select
				update_arm_artsakh_dirty_variable = yes
			}
