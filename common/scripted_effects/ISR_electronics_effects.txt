# Israeli electronics corporate history. Five milestones set the route; the
# closing milestone resolves it into one of four futures for the sector.

ISR_electronics_apply_sovereign_silicon = {
	if = {
		limit = { ISR_electronics_outcome_settled = no }
		add_ideas = ISR_electronics_sovereign_silicon
	}
}

# One entry point per milestone, called from the yearly pulse. Full presents the
# choice; Outcomes Only writes the historical route silently and grants nothing.
ISR_electronics_milestone_2001 = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		country_event = { id = israel_economy.300 days = 212 }
	}
	else_if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
		set_country_flag = ISR_electronics_wireless_consolidation
	}
}
