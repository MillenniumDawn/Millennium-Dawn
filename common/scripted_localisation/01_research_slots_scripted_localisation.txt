# v is the slot number of the research slot screen row
defined_text = {
	name = gdp_slot_status

	text = {
		trigger = { check_variable = { next_slot_from_gdp > v } }
		localization_key = research_slot_unlocked
	}
	text = {
		trigger = {
			check_variable = { next_slot_from_gdp = v }
			has_decision = gdp_gain_slot
		}
		localization_key = research_slot_in_progress
	}
	text = {
		trigger = {
			OR = {
				check_variable = { next_slot_from_gdp < v }
				check_variable = { gdp_total < GLOBAL.gdp_slot^v }
			}
		}
		localization_key = research_slot_locked
	}
	text = {
		localization_key = research_slot_available
	}
}

defined_text = {
	name = gdp_c_slot_status

	text = {
		trigger = { check_variable = { next_slot_from_gdp_c > v } }
		localization_key = research_slot_unlocked
	}
	text = {
		trigger = {
			check_variable = { next_slot_from_gdp_c = v }
			has_decision = gdp_c_gain_slot
		}
		localization_key = research_slot_in_progress
	}
	text = {
		trigger = {
			OR = {
				check_variable = { next_slot_from_gdp_c < v }
				check_variable = { gdp_per_capita < GLOBAL.gdp_c_slot^v }
				check_variable = { industrial_complex_total < GLOBAL.gdp_c_slot_civs^v }
			}
		}
		localization_key = research_slot_locked
	}
	text = {
		localization_key = research_slot_available
	}
}
