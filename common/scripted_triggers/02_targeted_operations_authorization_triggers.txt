	check_variable = { TOP_proposal_location > 59 }
	check_variable = { TOP_proposal_pattern > 59 }
	check_variable = { TOP_proposal_lead_age < 57 }
	NOT = { check_variable = { TOP_proposal_state = 0 } }
	check_variable = { TOP_proposal_host > 0 }

	set_temp_variable = { TOP_arg_target = TOP_proposal_target }
	set_temp_variable = { TOP_arg_host = TOP_proposal_host }
	TOP_case_can_review = yes
	check_variable = { TOP_case_sequence^TOP_proposal_target = TOP_proposal_case_sequence }
	if = {
		limit = { check_variable = { TOP_proposal_method < 3 } }
		set_temp_variable = { TOP_arg_target = TOP_proposal_target }
		set_temp_variable = { TOP_arg_method = TOP_proposal_method }
		set_temp_variable = { TOP_arg_state = TOP_proposal_state }
		TOP_native_binding_unused = yes
	}
	set_temp_variable = { TOP_review_expected_host = TOP_proposal_host }
	var:TOP_proposal_state = { is_controlled_by = var:PREV.TOP_review_expected_host }
	var:TOP_proposal_host = { exists = yes }
	set_temp_variable = { TOP_target = TOP_proposal_target }
	TOP_person_operational_eligible = yes
	set_temp_variable = { TOP_access_method = TOP_proposal_method }
	set_temp_variable = { TOP_access_state = TOP_proposal_state }
	set_temp_variable = { TOP_access_host = TOP_proposal_host }
	TOP_person_access_available = yes
	check_variable = { TOP_doctrine = TOP_proposal_doctrine }
	if = {
		limit = { check_variable = { TOP_proposal_method = 7 } }
		check_variable = { TOP_proposal_identity > 89 }
	}
	if = {
		limit = { check_variable = { global.TOP_leader_role^TOP_proposal_target > 0 } check_variable = { global.TOP_leader_role^TOP_proposal_target < 4 } }
		check_variable = { TOP_proposal_identity > 79 }
		check_variable = { TOP_proposal_location > 79 }
		check_variable = { TOP_proposal_pattern > 79 }
	}
}
