scripted_map_modes = {
	SCO_map_mode = {
		top = {
			type = country
			color = {
				if = { limit = { FROM = { is_sco = yes } }
					set_temp_variable = { red = 0.872 }
					set_temp_variable = { green = 0.389 }
					set_temp_variable = { blue = 0.101 }
				}
				else_if = { limit = { FROM = { has_idea = CHI_sco_observer } }
					set_temp_variable = { red = 0.551 }
					set_temp_variable = { green = 0.322 }
					set_temp_variable = { blue = 0.792 }
				}
				else_if = { limit = { FROM = { has_idea = CHI_sco_dialogue_partner } }
					set_temp_variable = { red = 0.461 }
					set_temp_variable = { green = 0.698 }
					set_temp_variable = { blue = 0.798 }
				}
				else = {
					set_temp_variable = { red = 0.927 }
					set_temp_variable = { green = 0.950 }
					set_temp_variable = { blue = 0.051 }
				}
				set_temp_variable = { alpha = 1 }
			}

			targets = { target_array = global.possible_sco_member_state }
		}
		bottom = { type = none }
		far_text = country
		near_text = none
		update_daily = no
	}
}
