scripted_gui = {
	CHE_devastation_meter_screen = {
		context_type = decision_category
		window_name = "show_devastation_meter"
		dirty = global.md_gui_dirty

		effects = {
			activate_tier1_button_click = {
				if = {
					limit = { NOT = { has_country_flag = chechnya_stance_tier_1 } }
					CHE_activate_stance_tier_1_effect = yes
				}
			}
			activate_tier2_button_click = {
				if = {
					limit = { NOT = { has_country_flag = chechnya_stance_tier_2 } }
					CHE_activate_stance_tier_2_effect = yes
				}
			}
			activate_tier3_button_click = {
				if = {
					limit = { NOT = { has_country_flag = chechnya_stance_tier_3 } }
					CHE_activate_stance_tier_3_effect = yes
				}
			}
			enact_salvage_button_click = {
				if = {
					limit = {
						NOT = { has_country_flag = chechnya_salvage_active }
						has_political_power = 30
					}
					add_political_power = -30
					CHE_enact_salvage_and_rebuild_effect = yes
				}
			}
		}

		properties = {
			devastation_bar = {
				frame = devastation_frame
				image = "[devastation_bar_sprite]"
			}
		}
	}
}
