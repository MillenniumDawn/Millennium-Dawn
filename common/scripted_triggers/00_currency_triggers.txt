# Currency System — Scripted Triggers

is_reserve_currency_issuer = {
	OR = {
		has_country_flag = usd_reserve_currency_issuer
		has_country_flag = eur_reserve_currency_issuer
		has_country_flag = rub_reserve_currency_issuer
	}
}

has_strong_currency = {
	check_variable = { currency_strength > 1.1 }
}

has_weak_currency = {
	check_variable = { currency_strength < 0.7 }
}

has_currency_crisis = {
	has_country_flag = in_currency_crisis
}
