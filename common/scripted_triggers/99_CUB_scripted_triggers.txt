CUB_ai_can_attack_hai_trigger = {
	has_country_flag = CUB_ai_can_attack_hai
}

CUB_ai_can_attack_hai_uncached_trigger = {
	has_wargoal_against = HAI
	NOT = { has_war_with = HAI }
	country_exists = HAI
	NOT = { is_in_faction_with = HAI }
	HAI = {
		is_in_faction = no
		is_subject = no
		has_war = no
		NOT = {
			any_of_scopes = {
				array = HAI.guaranteeing_nations
				exists = yes
				has_guaranteed = HAI
			}
		}
	}
	strength_ratio = { tag = HAI ratio > 1.5 }
}
