# Economic Events

country_event = {
	id = turkish_events.2 # The Rates Collapse
	title = turkish_events.2.t
	desc = turkish_events.2.d
	is_triggered_only = yes

	option = {
		name = turkish_events.2.a
		log = "[GetDateText]: [This.GetName]:turkish_events.2.a executed"
		set_temp_variable = { treasury_change = 200.00 }
		modify_treasury_effect = yes
		newline = yes
		add_stability = 0.05
	}

	option = {
		name = turkish_events.2.b
		log = "[GetDateText]: [This.GetName]:turkish_events.2.b executed"
		add_ideas = TUR_honeymoon
	}
}

country_event = {
	id = turkish_events.3 # Five Plus Five
	title = turkish_events.3.t
	desc = turkish_events.3.d
	is_triggered_only = yes

	option = {
		name = turkish_events.3.a
		log = "[GetDateText]: [This.GetName]:turkish_events.3.a executed"
		random_list = { # Keeping Suleyman for 3 more years will delay the rise of AKP and potentially prevent black wednesday
			20 = {
				custom_effect_tooltip = TUR_3_more_years_TT
			}
			80 = {
				country_event = {
					id = turkish_events.4
					days = 3
				}
			}
		}
		newline = yes
		add_stability = -0.05
		newline	= yes
		increase_corruption = yes
	}

	option = {
		name = turkish_events.3.b
		log = "[GetDateText]: [This.GetName]:turkish_events.3.b executed"
		hidden_effect = {
			country_event = {
			id = turkish_events.4
			days = 3
		}
		}
	}
}
