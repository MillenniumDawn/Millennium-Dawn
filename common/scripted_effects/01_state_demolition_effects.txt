state_demolition_add_type = {
	if = { limit = { $TYPE$ > 0 }
		add_to_array = { ROOT.state_demolition_buildings = token:$TYPE$ }
	}
}

state_demolition_rebuild_list = {
	clear_array = ROOT.state_demolition_buildings
	state_demolition_add_type = { TYPE = fossil_powerplant }
	state_demolition_add_type = { TYPE = nuclear_reactor }
	state_demolition_add_type = { TYPE = renewable_energy_infra }
	state_demolition_add_type = { TYPE = air_base }
	state_demolition_add_type = { TYPE = radar_station }
	state_demolition_add_type = { TYPE = anti_air_building }
	state_demolition_add_type = { TYPE = fuel_silo }
}

state_demolition_remove_type = {
	if = {
		limit = {
			check_variable = { ROOT.state_demolition_building = token:$TYPE$ }
			$TYPE$ > 0
		}
		remove_building = { type = $TYPE$ level = 1 }
	}
}

state_demolition_confirm = {
	if = { limit = { state_demolition_can_confirm = yes }
		log = "[GetDateText]: [ROOT.GetName]: Demolished one level of [?ROOT.state_demolition_building.GetTokenKey] in [THIS.GetName]"
		state_demolition_remove_type = { TYPE = fossil_powerplant }
		state_demolition_remove_type = { TYPE = nuclear_reactor }
		state_demolition_remove_type = { TYPE = renewable_energy_infra }
		state_demolition_remove_type = { TYPE = air_base }
		state_demolition_remove_type = { TYPE = radar_station }
		state_demolition_remove_type = { TYPE = anti_air_building }
		state_demolition_remove_type = { TYPE = fuel_silo }
		ROOT = {
			clear_variable = state_demolition_building
			ingame_update_setup = yes
		}
		state_demolition_rebuild_list = yes
	}
}
