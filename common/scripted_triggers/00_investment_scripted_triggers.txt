# Prevents reactor construction in countries that have banned it (ITA) or gone carbon-neutral (GER).
# Consulted by both the player investment GUI and the AI scorer.
country_can_build_nuclear_reactors = {
	NOT = { has_idea = ITA_nuclear_power_banned }
	NOT = { has_idea = GER_idea_carbon_neutral }
}

# Parameters: investment_slots_needed = infrastructure levels to build in THIS state.
investment_infrastructure_unlocked = {
	set_temp_variable = { required_infrastructure_level = { value = building_level@infrastructure add = investment_slots_needed } }
	# The controller's research can cap infrastructure below the state's physical cap.
	CONTROLLER = {
		if = { limit = { check_variable = { required_infrastructure_level > 5 } }
			has_tech = special_project_tech_smart_grid
		}
		else_if = { limit = { check_variable = { required_infrastructure_level > 4 } }
			has_tech = railway_system_06
		}
		else_if = { limit = { check_variable = { required_infrastructure_level > 3 } }
			has_tech = railway_system_05
		}
		else_if = { limit = { check_variable = { required_infrastructure_level > 2 } }
			has_tech = railway_system_04
		}
		else_if = { limit = { check_variable = { required_infrastructure_level > 1 } }
			has_tech = railway_system_02
		}
		else = { has_tech = railway_system_01 }
	}
}
