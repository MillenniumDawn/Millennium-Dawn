	IBR_raise_language_funding = {
		icon = GFX_decision_ISR_education_button
		cost = 25
		visible = { check_variable = { IBR_language_funding < 3 } }

		available = {
			custom_trigger_tooltip = {
				tooltip = IBR_language_funding_curriculum_tt
				OR = {
					check_variable = { IBR_language_funding < 1 }
					has_completed_focus = IBR_common_curriculum
				}
			}
		}

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision IBR_raise_language_funding"
			if = {
				limit = { check_variable = { IBR_language_funding < 1 } }
				add_ideas = IBR_language_classes_1
			}
			else_if = {
				limit = { check_variable = { IBR_language_funding < 2 } }
				swap_ideas = { remove_idea = IBR_language_classes_1 add_idea = IBR_language_classes_2 }
			}
			else = {
				swap_ideas = { remove_idea = IBR_language_classes_2 add_idea = IBR_language_classes_3 }
			}
			hidden_effect = {
				add_to_variable = { IBR_language_funding = 1 }
				IBR_update_map_dirty = yes
			}
		}

		ai_will_do = {
			base = 10
			modifier = {
				factor = 0
				has_active_mission = bankruptcy_incoming_collapse
			}
		}
	}
