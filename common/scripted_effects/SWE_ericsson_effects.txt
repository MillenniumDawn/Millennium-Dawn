SWE_ericsson_ensure_digital_modifier = {
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = SWE_digital_modifier } } }
		add_dynamic_modifier = { modifier = SWE_digital_modifier }
	}
}

# Outcomes are idempotent so a silent reconstruction and a live choice cannot stack.
SWE_ericsson_apply_joint_venture_outcome = {
	if = {
		limit = {
			NOT = { has_country_flag = SWE_sony_ericsson_joint_venture }
			NOT = { has_country_flag = SWE_ericsson_retained_handsets }
		}
		SWE_ericsson_ensure_digital_modifier = yes
		custom_effect_tooltip = { localization_key = modifies_dynamic_modifier_tt MODIFIER = SWE_digital_modifier }
		add_to_variable = { SWE_Digital_receiving_investment_cost_modifier = -0.01 tooltip = receiving_investment_cost_modifier_tt }
		set_country_flag = SWE_sony_ericsson_joint_venture
	}
}

# One entry point per milestone, called from the yearly pulse. Full presents the
# choice; Outcomes Only writes the historical route silently and grants nothing.
SWE_ericsson_milestone_2001 = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		country_event = { id = Sweden.300 days = 273 }
	}
	else_if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
		SWE_ericsson_apply_joint_venture_outcome = yes
	}
}
