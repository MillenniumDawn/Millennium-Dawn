scripted_map_modes = {
	subideology_map_mode = {
		top = {
			type = country
			color = {
				if = {
					limit = {
						check_variable = { FROM.ruling_party > -1 }
						check_variable = { FROM.ruling_party < 24 }
					}
					set_temp_variable = { red = global.ruling_party_map_red^FROM.ruling_party }
					set_temp_variable = { green = global.ruling_party_map_green^FROM.ruling_party }
					set_temp_variable = { blue = global.ruling_party_map_blue^FROM.ruling_party }
				}
				else = {
					set_temp_variable = { red = 0.244 }
					set_temp_variable = { green = 0.304 }
					set_temp_variable = { blue = 0.398 }
				}

				set_temp_variable = { alpha = 1.0 }
			}
		}

		bottom = {
			type = none
		}

		far_text = country
		near_text = none

		update_daily = no
	}
}
