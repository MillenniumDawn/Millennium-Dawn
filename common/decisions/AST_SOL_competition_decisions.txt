# Solomon Islands influence competition - ongoing engagement
#
# The original competition was three bursts of construction spending in the
# early 2000s and then nothing at all until a single tally in 2022. These
# decisions give Australia, the United States and Honiara something to do
# across the whole span, and give the winner something to hold afterwards.
#
# Scoring runs through common/scripted_effects/AST_SOL_competition_effects.txt.

AST_solomon_engagement_category = {

	# The bread and butter of Australian Pacific policy: aid budgets, scholarships,
	# and the seasonal worker programme. Always available while the contest runs.
	AST_SOL_development_assistance = {
		icon = GFX_decision_generic_construction

		allowed = {
			original_tag = AST
		}

		available = {
			country_exists = SOL
			SOL_competition_active = yes
			NOT = { has_war_with = SOL }
			check_variable = { treasury > 12 }
			NOT = { has_active_mission = bankruptcy_incoming_collapse }
		}

		visible = {
			country_exists = SOL
			SOL_competition_active = yes
		}

		fire_only_once = no
		cost = 30
		days_remove = 120
		days_re_enable = 365

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision AST_SOL_development_assistance"
			set_temp_variable = { treasury_change = -12.00 }
			modify_treasury_effect = yes
		}

		remove_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision remove AST_SOL_development_assistance"
			set_temp_variable = { comp_points = 3 }
			SOL = { SOL_award_AST_points = yes }
			set_temp_variable = { percent_change = 3 }
			set_temp_variable = { tag_index = AST }
			set_temp_variable = { influence_target = SOL.id }
			change_influence_percentage = yes
			SOL = {
				add_opinion_modifier = {
					target = AST
					modifier = recently_invest
				}
			}
		}

		ai_will_do = {
			base = 10
			modifier = {
				factor = 0
				has_active_mission = bankruptcy_incoming_collapse
			}
			modifier = {
				factor = 2
				NOT = { SOL_competition_caller_leads = yes }
			}
			# Realigning on Beijing does not take Australia out of the contest,
			# it changes who is standing on the other side of it. Canberra's own
			# column still carries the bloc, and the rival is now a great power,
			# so a China-aligned Australia has more reason to keep spending in
			# Honiara rather than less.
			modifier = {
				factor = 1.8
				SOL_competition_is_usa_rival = yes
			}
		}
	}

			}

			# Australian police officers takes an Australian relationship, so
			# this reads Canberra's own engagement at the same threshold
			# Beijing's Pacific branch uses for its own hard content.
			custom_trigger_tooltip = {
				tooltip = AST_pacific_engagement_gt_7_tt
				check_variable = { AST_pacific_engagement > 7 }
			}
			check_variable = { treasury > 8 }
		}

		visible = {
			country_exists = SOL
			SOL_competition_active = yes
			has_completed_focus = AST_Solomon_Islands_Restore_Government
		}

		fire_only_once = no
		cost = 40
		days_remove = 150
		days_re_enable = 545

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision AST_SOL_police_partnership"
			set_temp_variable = { treasury_change = -8.00 }
			modify_treasury_effect = yes
		}

		remove_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision remove AST_SOL_police_partnership"
			set_temp_variable = { comp_points = 4 }
			SOL = {
				SOL_award_AST_points = yes
				add_stability = 0.03
				add_opinion_modifier = {
					target = AST
					modifier = SOL_ramsi_support
				}
			}
		}

		ai_will_do = {
			base = 12
			modifier = {
				factor = 0
				has_active_mission = bankruptcy_incoming_collapse
			}

			# Honiara, and it matters most when the rival across the table is

			}
		}

			}
