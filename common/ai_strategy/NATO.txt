# NATO AI strategies. Membership is decision-driven (common/decisions/NATO.txt), so the
# befriend strategies key on the NATO_member idea and NATO_Aspirant flag instead of
# historical accession dates. allowed uses the NATO_possible_countries trigger, which
# mirrors the joiner gate on the NATO decision categories. One befriend block per target
# lets a member drop it while at war with, or holding a war goal on, that target.

nato_befriend_bel = {
	allowed = {
		NOT = { has_game_rule = { rule = rule_disable_nato option = yes } }
		NATO_possible_countries = yes
	}
	enable = {
		OR = {
			has_idea = NATO_member
			has_country_flag = NATO_Aspirant
		}
		NOT = { has_war_with = BEL }
		NOT = { has_wargoal_against = BEL }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = befriend id = "BEL" value = 50 }
}
