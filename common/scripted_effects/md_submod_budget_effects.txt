md_submod_budget_graph_init = {
	set_temp_variable = { m_rev = display_income }
	multiply_temp_variable = { m_rev = 4.33 }
	set_variable = { md_current_monthly_revenue = m_rev }

	set_temp_variable = { m_exp = display_expense }
	multiply_temp_variable = { m_exp = 4.33 }
	set_variable = { md_current_monthly_expense = m_exp }

	set_temp_variable = { m_bal = treasury_rate }
	multiply_temp_variable = { m_bal = 4.33 }
	set_variable = { md_current_monthly_balance = m_bal }

	resize_array = { array = revenue_array value = md_current_monthly_revenue size = 120 }
	resize_array = { array = expense_array value = md_current_monthly_expense size = 120 }
	resize_array = { array = budget_balance_array value = md_current_monthly_balance size = 120 }

	set_variable = { revenue_array^0 = md_current_monthly_revenue }
	set_variable = { expense_array^0 = md_current_monthly_expense }
	set_variable = { budget_balance_array^0 = md_current_monthly_balance }
}

# 1. THE BUDGET MONTHLY SHIFTER (Self-Healing)
md_submod_budget_graph_on_monthly = {
	set_temp_variable = { m_rev = display_income }
	multiply_temp_variable = { m_rev = 4.33 }
	set_variable = { md_current_monthly_revenue = m_rev }

	set_temp_variable = { m_exp = display_expense }
	multiply_temp_variable = { m_exp = 4.33 }
	set_variable = { md_current_monthly_expense = m_exp }

	set_temp_variable = { m_bal = treasury_rate }
	multiply_temp_variable = { m_bal = 4.33 }
	set_variable = { md_current_monthly_balance = m_bal }

	if = {
		limit = { NOT = { check_variable = { revenue_array^num > 0 } } }
		md_submod_budget_graph_init = yes
	}

	set_variable = { revenue_array^current_gdp_graph_end = md_current_monthly_revenue }
	set_variable = { expense_array^current_gdp_graph_end = md_current_monthly_expense }
	set_variable = { budget_balance_array^current_gdp_graph_end = md_current_monthly_balance }

	# When month 1 arrives, calibrate index 0 to true calculated baseline
	if = {
		limit = { check_variable = { current_gdp_graph_end = 1 } }
		set_variable = { revenue_array^0 = md_current_monthly_revenue }
		set_variable = { expense_array^0 = md_current_monthly_expense }
		set_variable = { budget_balance_array^0 = md_current_monthly_balance }
	}
}

# 2. THE MATH TRANSLATOR & FAILSAFE
update_md_flaxbeard_budget_graph = {
	set_temp_variable = { m_rev = display_income }
	multiply_temp_variable = { m_rev = 4.33 }
	set_variable = { md_current_monthly_revenue = m_rev }

	set_temp_variable = { m_exp = display_expense }
	multiply_temp_variable = { m_exp = 4.33 }
	set_variable = { md_current_monthly_expense = m_exp }

	set_temp_variable = { m_bal = treasury_rate }
	multiply_temp_variable = { m_bal = 4.33 }
	set_variable = { md_current_monthly_balance = m_bal }

	if = {
		limit = { NOT = { check_variable = { revenue_array^num > 0 } } }
		md_submod_budget_graph_init = yes
	}

	clear_array = md_flax_offsets_revenue_green
	clear_array = md_flax_offsets_revenue_red
	clear_array = md_flax_slopes_revenue

	clear_array = md_flax_offsets_expense_red
	clear_array = md_flax_offsets_expense_green
	clear_array = md_flax_slopes_expense

	clear_array = md_flax_offsets_balance_green
	clear_array = md_flax_offsets_balance_red
	clear_array = md_flax_slopes_balance

	# --- YOY (JANUARY-ONLY SNAPSHOT) ---
	set_temp_variable = { months_since_jan = global.month }
	subtract_from_temp_variable = { months_since_jan = 1 }

	set_temp_variable = { recent_jan_idx = current_gdp_graph_end }
	subtract_from_temp_variable = { recent_jan_idx = months_since_jan }
	if = { limit = { check_variable = { recent_jan_idx < 0 } } set_temp_variable = { recent_jan_idx = 0 } }

	set_temp_variable = { prev_jan_idx = recent_jan_idx }
	subtract_from_temp_variable = { prev_jan_idx = 12 }
	if = { limit = { check_variable = { prev_jan_idx < 0 } } set_temp_variable = { prev_jan_idx = 0 } }

	set_temp_variable = { recent_jan_bal = budget_balance_array^recent_jan_idx }
	set_temp_variable = { prev_jan_bal = budget_balance_array^prev_jan_idx }

	set_variable = { md_yoy_budget_change_billions = recent_jan_bal }
	subtract_from_variable = { md_yoy_budget_change_billions = prev_jan_bal }
	# ---------------------------------------------

	# FIND BOUNDS (Revenue, Expense, and Balance all share same axis in $B/mo)
	set_temp_variable = { true_max = revenue_array^0 }
	set_temp_variable = { true_min = budget_balance_array^0 } 
	set_temp_variable = { max_jump = 0 }

	for_each_loop = {
		array = revenue_array
		value = v
		index = i

		if = {
			limit = { NOT = { check_variable = { i > current_gdp_graph_end } } }
			if = { limit = { check_variable = { v > true_max } } set_temp_variable = { true_max = v } }
			if = { limit = { check_variable = { v < true_min } } set_temp_variable = { true_min = v } }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }
			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { jump = revenue_array^next_i }
				subtract_from_temp_variable = { jump = v }
				if = { limit = { check_variable = { jump < 0 } } multiply_temp_variable = { jump = -1 } }
				if = { limit = { check_variable = { jump > max_jump } } set_temp_variable = { max_jump = jump } }
			}
		}
	}

	for_each_loop = {
		array = expense_array
		value = v
		index = i

		if = {
			limit = { NOT = { check_variable = { i > current_gdp_graph_end } } }
			if = { limit = { check_variable = { v > true_max } } set_temp_variable = { true_max = v } }
			if = { limit = { check_variable = { v < true_min } } set_temp_variable = { true_min = v } }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }
			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { jump = expense_array^next_i }
				subtract_from_temp_variable = { jump = v }
				if = { limit = { check_variable = { jump < 0 } } multiply_temp_variable = { jump = -1 } }
				if = { limit = { check_variable = { jump > max_jump } } set_temp_variable = { max_jump = jump } }
			}
		}
	}

	for_each_loop = {
		array = budget_balance_array
		value = v
		index = i

		if = {
			limit = { NOT = { check_variable = { i > current_gdp_graph_end } } }
			if = { limit = { check_variable = { v > true_max } } set_temp_variable = { true_max = v } }
			if = { limit = { check_variable = { v < true_min } } set_temp_variable = { true_min = v } }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }
			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { jump = budget_balance_array^next_i }
				subtract_from_temp_variable = { jump = v }
				if = { limit = { check_variable = { jump < 0 } } multiply_temp_variable = { jump = -1 } }
				if = { limit = { check_variable = { jump > max_jump } } set_temp_variable = { max_jump = jump } }
			}
		}
	}

	set_temp_variable = { data_diff = true_max }
	subtract_from_temp_variable = { data_diff = true_min }

	# FAILSAFE PADDING
	if = {
		limit = { check_variable = { data_diff < 1 } }
		set_temp_variable = { high = true_max }
		add_to_temp_variable = { high = 1.0 }
		set_temp_variable = { low = true_min }
		subtract_from_temp_variable = { low = 1.0 }
	}
	else = {
		set_temp_variable = { pad = data_diff }
		multiply_temp_variable = { pad = 0.1 } 
		set_temp_variable = { high = true_max }
		add_to_temp_variable = { high = pad }
		set_temp_variable = { low = true_min }
		subtract_from_temp_variable = { low = pad }
	}

	set_temp_variable = { diff = high }
	subtract_from_temp_variable = { diff = low }
	if = { limit = { check_variable = { diff < 0.01 } } set_temp_variable = { diff = 0.01 } }

	# DYNAMIC ANTI-TEAR ZOOM 
	set_temp_variable = { safe_diff = max_jump }
	multiply_temp_variable = { safe_diff = 45 }
	divide_temp_variable = { safe_diff = 10 }

	if = {
		limit = { check_variable = { diff < safe_diff } }
		set_temp_variable = { shortfall = safe_diff }
		subtract_from_temp_variable = { shortfall = diff }
		divide_temp_variable = { shortfall = 2 }

		add_to_temp_variable = { high = shortfall }
		subtract_from_temp_variable = { low = shortfall }
		set_temp_variable = { diff = safe_diff }
	}

	# Y-AXIS LABELS ($B/mo)
	set_temp_variable = { step = diff }
	multiply_temp_variable = { step = 0.1 } 
	set_temp_variable = { buf = low }
	for_loop_effect = {
		start = 0 end = 11
		set_variable = { md_budget_y_axis_label^v = buf }
		add_to_temp_variable = { buf = step }
	}

	# 1. DRAW REVENUE LINE (Green on up/flat, Red on down)
	for_each_loop = {
		array = revenue_array
		value = v
		index = i

		if = {
			limit = { check_variable = { i > current_gdp_graph_end } }
			add_to_array = { md_flax_offsets_revenue_green = 1000 } 
			add_to_array = { md_flax_offsets_revenue_red = 1000 } 
			add_to_array = { md_flax_slopes_revenue = 50 }
		}
		else = {
			set_temp_variable = { y1_val = v }
			subtract_from_temp_variable = { y1_val = low }
			divide_temp_variable = { y1_val = diff } 
			set_temp_variable = { y1_px = y1_val }
			multiply_temp_variable = { y1_px = 450 }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }

			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } } 
				set_temp_variable = { y2_val = revenue_array^next_i }
			}
			else = {
				set_temp_variable = { y2_val = v } 
			}

			subtract_from_temp_variable = { y2_val = low }
			divide_temp_variable = { y2_val = diff } 
			set_temp_variable = { y2_px = y2_val }
			multiply_temp_variable = { y2_px = 450 } 

			set_temp_variable = { offset = y1_px }
			multiply_temp_variable = { offset = -1 }

			if = {
				limit = { check_variable = { y2_val < y1_val } } # Revenue down = Red
				add_to_array = { md_flax_offsets_revenue_red = offset }
				add_to_array = { md_flax_offsets_revenue_green = 1000 }
			}
			else = { # Revenue up/flat = Green
				add_to_array = { md_flax_offsets_revenue_green = offset }
				add_to_array = { md_flax_offsets_revenue_red = 1000 }
			}

			set_temp_variable = { slope = y2_px }
			subtract_from_temp_variable = { slope = y1_px }
			divide_temp_variable = { slope = 2 }
			add_to_temp_variable = { slope = 50 }

			if = { limit = { check_variable = { slope > 100 } } set_temp_variable = { slope = 100 } }
			if = { limit = { check_variable = { slope < 1 } } set_temp_variable = { slope = 1 } }

			round_variable = slope
			add_to_array = { md_flax_slopes_revenue = slope }
		}
	}

	# 2. DRAW EXPENSE LINE (Red on up, Green on down)
	for_each_loop = {
		array = expense_array
		value = v
		index = i

		if = {
			limit = { check_variable = { i > current_gdp_graph_end } }
			add_to_array = { md_flax_offsets_expense_red = 1000 } 
			add_to_array = { md_flax_offsets_expense_green = 1000 } 
			add_to_array = { md_flax_slopes_expense = 50 }
		}
		else = {
			set_temp_variable = { y1_val = v }
			subtract_from_temp_variable = { y1_val = low }
			divide_temp_variable = { y1_val = diff } 
			set_temp_variable = { y1_px = y1_val }
			multiply_temp_variable = { y1_px = 450 }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }

			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } } 
				set_temp_variable = { y2_val = expense_array^next_i }
			}
			else = {
				set_temp_variable = { y2_val = v } 
			}

			subtract_from_temp_variable = { y2_val = low }
			divide_temp_variable = { y2_val = diff } 
			set_temp_variable = { y2_px = y2_val }
			multiply_temp_variable = { y2_px = 450 } 

			set_temp_variable = { offset = y1_px }
			multiply_temp_variable = { offset = -1 }

			if = {
				limit = { check_variable = { y2_val > y1_val } } # Expense up = Red
				add_to_array = { md_flax_offsets_expense_red = offset }
				add_to_array = { md_flax_offsets_expense_green = 1000 }
			}
			else = { # Expense down/flat = Green
				add_to_array = { md_flax_offsets_expense_green = offset }
				add_to_array = { md_flax_offsets_expense_red = 1000 }
			}

			set_temp_variable = { slope = y2_px }
			subtract_from_temp_variable = { slope = y1_px }
			divide_temp_variable = { slope = 2 }
			add_to_temp_variable = { slope = 50 }

			if = { limit = { check_variable = { slope > 100 } } set_temp_variable = { slope = 100 } }
			if = { limit = { check_variable = { slope < 1 } } set_temp_variable = { slope = 1 } }

			round_variable = slope
			add_to_array = { md_flax_slopes_expense = slope }
		}
	}

	# 3. DRAW NET BALANCE LINE (Green on Surplus / Increasing, Red on Deficit / Decreasing)
	for_each_loop = {
		array = budget_balance_array
		value = v
		index = i

		if = {
			limit = { check_variable = { i > current_gdp_graph_end } }
			add_to_array = { md_flax_offsets_balance_green = 1000 } 
			add_to_array = { md_flax_offsets_balance_red = 1000 } 
			add_to_array = { md_flax_slopes_balance = 50 }
		}
		else = {
			set_temp_variable = { y1_val = v }
			subtract_from_temp_variable = { y1_val = low }
			divide_temp_variable = { y1_val = diff } 
			set_temp_variable = { y1_px = y1_val }
			multiply_temp_variable = { y1_px = 450 }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }

			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } } 
				set_temp_variable = { y2_val = budget_balance_array^next_i }
			}
			else = {
				set_temp_variable = { y2_val = v } 
			}

			subtract_from_temp_variable = { y2_val = low }
			divide_temp_variable = { y2_val = diff } 
			set_temp_variable = { y2_px = y2_val }
			multiply_temp_variable = { y2_px = 450 } 

			set_temp_variable = { offset = y1_px }
			multiply_temp_variable = { offset = -1 }

			if = {
				limit = { check_variable = { y1_val < 0 } } # Deficit = Red
				add_to_array = { md_flax_offsets_balance_red = offset }
				add_to_array = { md_flax_offsets_balance_green = 1000 }
			}
			else = { # Surplus = Green
				add_to_array = { md_flax_offsets_balance_green = offset }
				add_to_array = { md_flax_offsets_balance_red = 1000 }
			}

			set_temp_variable = { slope = y2_px }
			subtract_from_temp_variable = { slope = y1_px }
			divide_temp_variable = { slope = 2 }
			add_to_temp_variable = { slope = 50 }

			if = { limit = { check_variable = { slope > 100 } } set_temp_variable = { slope = 100 } }
			if = { limit = { check_variable = { slope < 1 } } set_temp_variable = { slope = 1 } }

			round_variable = slope
			add_to_array = { md_flax_slopes_balance = slope }
		}
	}
}
