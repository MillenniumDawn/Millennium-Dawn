BOS_mark_gui_dirty_effect = {
	if = {
		limit = { is_ai = no }
		add_to_variable = { global.md_gui_dirty = 1 }
	}
}

BOS_refresh_internal_faction_modifiers_effect = {
	set_variable = { BOS_internal_faction_stability_factor = 0 }
	set_variable = { BOS_internal_faction_manpower_factor = 0 }

	if = {
		limit = { check_variable = { BOS_bosniak_stability < 2 } }
		add_to_variable = { BOS_internal_faction_stability_factor = -0.15 }
	}
	else_if = {
		limit = { check_variable = { BOS_bosniak_stability > 2 } }
		add_to_variable = { BOS_internal_faction_stability_factor = 0.05 }
		if = {
			limit = { check_variable = { BOS_bosniak_stability > 3 } }
			add_to_variable = { BOS_internal_faction_stability_factor = 0.05 }
		}
	}
	if = {
		limit = { check_variable = { BOS_croat_stability < 2 } }
		add_to_variable = { BOS_internal_faction_stability_factor = -0.15 }
	}
	else_if = {
		limit = { check_variable = { BOS_croat_stability > 2 } }
		add_to_variable = { BOS_internal_faction_stability_factor = 0.05 }
		if = {
			limit = { check_variable = { BOS_croat_stability > 3 } }
			add_to_variable = { BOS_internal_faction_stability_factor = 0.05 }
		}
	}
	if = {
		limit = { check_variable = { BOS_srpska_stability < 2 } }
		add_to_variable = { BOS_internal_faction_stability_factor = -0.15 }
	}
	else_if = {
		limit = { check_variable = { BOS_srpska_stability > 2 } }
		add_to_variable = { BOS_internal_faction_stability_factor = 0.05 }
		if = {
			limit = { check_variable = { BOS_srpska_stability > 3 } }
			add_to_variable = { BOS_internal_faction_stability_factor = 0.05 }
		}
	}

	if = {
		limit = { check_variable = { BOS_bosniak_manpower < 2 } }
		add_to_variable = { BOS_internal_faction_manpower_factor = -0.15 }
	}
	else_if = {
		limit = { check_variable = { BOS_bosniak_manpower > 2 } }
		add_to_variable = { BOS_internal_faction_manpower_factor = 0.15 }
		if = {
			limit = { check_variable = { BOS_bosniak_manpower > 3 } }
			add_to_variable = { BOS_internal_faction_manpower_factor = 0.05 }
		}
	}
	if = {
		limit = { check_variable = { BOS_croat_manpower < 2 } }
		add_to_variable = { BOS_internal_faction_manpower_factor = -0.15 }
	}
	else_if = {
		limit = { check_variable = { BOS_croat_manpower > 2 } }
		add_to_variable = { BOS_internal_faction_manpower_factor = 0.15 }
		if = {
			limit = { check_variable = { BOS_croat_manpower > 3 } }
			add_to_variable = { BOS_internal_faction_manpower_factor = 0.05 }
		}
	}
	if = {
		limit = { check_variable = { BOS_srpska_manpower < 2 } }
		add_to_variable = { BOS_internal_faction_manpower_factor = -0.15 }
	}
	else_if = {
		limit = { check_variable = { BOS_srpska_manpower > 2 } }
		add_to_variable = { BOS_internal_faction_manpower_factor = 0.15 }
		if = {
			limit = { check_variable = { BOS_srpska_manpower > 3 } }
			add_to_variable = { BOS_internal_faction_manpower_factor = 0.05 }
		}
	}

	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = BOS_internal_faction_balance } } }
		add_dynamic_modifier = { modifier = BOS_internal_faction_balance }
	}
	BOS_mark_gui_dirty_effect = yes
}

		limit = {

				OR = {
					has_idea = intervention_regional_interventionism

			}

BOS_update_national_reconciliation_effect = {
	if = {
		limit = {
			NOT = { has_idea = BOS_national_reconciliation }
			NOT = { has_country_flag = civil_war_bosnia }
			check_variable = { BOS_bosniak_attitude > 2 }
			check_variable = { BOS_croat_attitude > 2 }
			check_variable = { BOS_srpska_attitude > 2 }
		}
		add_ideas = BOS_national_reconciliation
	}
	else_if = {
		limit = {
			has_idea = BOS_national_reconciliation
			OR = {
				has_country_flag = civil_war_bosnia
				check_variable = { BOS_bosniak_attitude < 3 }
				check_variable = { BOS_croat_attitude < 3 }
				check_variable = { BOS_srpska_attitude < 3 }
			}
		}
		remove_ideas = BOS_national_reconciliation
	}
}
