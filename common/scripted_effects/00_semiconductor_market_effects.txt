# The semiconductor market is a gameplay abstraction. Existing microchip resources
# and GPU demand remain authoritative; this layer only derives bounded pressure.

semiconductor_market_initialize_state = {
	if = {
		limit = {
			has_game_rule = { rule = rule_corporate_history option = full }
			NOT = { has_global_flag = GLOBAL_semiconductor_market_state_initialized }
		}
		set_variable = { global.semiconductor_market_demand = 5 }
		set_variable = { global.semiconductor_market_tool_backlog = 0 }
		set_variable = { global.semiconductor_market_access_tension = 2 }
		set_global_flag = GLOBAL_semiconductor_market_state_initialized
	}
}

semiconductor_market_clamp_state = {
	clamp_variable = { var = global.semiconductor_market_demand min = 0 max = 10 }
	clamp_variable = { var = global.semiconductor_market_tool_backlog min = 0 max = 10 }
	clamp_variable = { var = global.semiconductor_market_access_tension min = 0 max = 10 }
}

# Owner callbacks call these shared effects instead of writing market state.
semiconductor_market_register_tool_order = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		semiconductor_market_initialize_state = yes
		add_to_variable = { global.semiconductor_market_tool_backlog = 1 }
		semiconductor_market_clamp_state = yes
	}
}

semiconductor_market_complete_tool_order = {
	if = {
		limit = {
			has_game_rule = { rule = rule_corporate_history option = full }
			has_global_flag = GLOBAL_semiconductor_market_state_initialized
		}
		subtract_from_variable = { global.semiconductor_market_tool_backlog = 1 }
		semiconductor_market_clamp_state = yes
	}
}

semiconductor_market_update_demand = {
	# Historical dates set targets; the values themselves are gameplay abstractions.
	if = {
		limit = { date < 2001.1.1 }
		set_temp_variable = { semiconductor_market_demand_target = 5 }
	}
	else_if = {
		limit = { date < 2003.1.1 }
		set_temp_variable = { semiconductor_market_demand_target = 3 }
	}
	else_if = {
		limit = { date < 2008.1.1 }
		set_temp_variable = { semiconductor_market_demand_target = 5 }
	}
	else_if = {
		limit = { date < 2010.1.1 }
		set_temp_variable = { semiconductor_market_demand_target = 3 }
	}
	else_if = {
		limit = { date < 2017.1.1 }
		set_temp_variable = { semiconductor_market_demand_target = 5 }
	}
	else_if = {
		limit = { date < 2020.1.1 }
		set_temp_variable = { semiconductor_market_demand_target = 6 }
	}
	else_if = {
		limit = { date < 2023.1.1 }
		set_temp_variable = { semiconductor_market_demand_target = 9 }
	}
	else_if = {
		limit = { date < 2024.1.1 }
		set_temp_variable = { semiconductor_market_demand_target = 6 }
	}
	else = {
		set_temp_variable = { semiconductor_market_demand_target = 8 }
	}

	if = {
		limit = {
			country_exists = KOR
			KOR = {
				OR = {
					check_variable = { resource@microchips < 0 }
					has_idea = gpu_accelerator_demand
				}
			}
		}
		add_to_temp_variable = { semiconductor_market_demand_target = 1 }
	}
	else_if = {
		limit = {
			country_exists = KOR
			KOR = {
				check_variable = { resource@microchips > 100 }
				check_variable = { microchip_plant_total > 9 }
				NOT = { has_idea = gpu_accelerator_demand }
			}
		}
		subtract_from_temp_variable = { semiconductor_market_demand_target = 1 }
	}

	clamp_temp_variable = { var = semiconductor_market_demand_target min = 0 max = 10 }
	if = {
		limit = {
			check_variable = {
				var = global.semiconductor_market_demand
				value = semiconductor_market_demand_target
				compare = less_than
			}
		}
		add_to_variable = { global.semiconductor_market_demand = 1 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = global.semiconductor_market_demand
				value = semiconductor_market_demand_target
				compare = greater_than
			}
		}
		subtract_from_variable = { global.semiconductor_market_demand = 1 }
	}
}

semiconductor_market_update_access_tension = {
	if = {
		limit = { date < 2019.1.1 }
		set_temp_variable = { semiconductor_market_access_target = 2 }
	}
	else_if = {
		limit = { date < 2023.1.1 }
		set_temp_variable = { semiconductor_market_access_target = 4 }
	}
	else = {
		set_temp_variable = { semiconductor_market_access_target = 6 }
	}

	if = {
		limit = {
			country_exists = HOL
			HOL = {
				NOT = { has_country_flag = collapsed_nation }
				has_country_flag = HOL_asml_open_customer_allocation
			}
		}
		subtract_from_temp_variable = { semiconductor_market_access_target = 1 }
	}
	else_if = {
		limit = {
			country_exists = HOL
			HOL = {
				NOT = { has_country_flag = collapsed_nation }
				has_country_flag = HOL_asml_strategic_licensing
			}
		}
		add_to_temp_variable = { semiconductor_market_access_target = 1 }
	}

	clamp_temp_variable = { var = semiconductor_market_access_target min = 0 max = 10 }
	if = {
		limit = {
			check_variable = {
				var = global.semiconductor_market_access_tension
				value = semiconductor_market_access_target
				compare = less_than
			}
		}
		add_to_variable = { global.semiconductor_market_access_tension = 1 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = global.semiconductor_market_access_tension
				value = semiconductor_market_access_target
				compare = greater_than
			}
		}
		subtract_from_variable = { global.semiconductor_market_access_tension = 1 }
	}
}

semiconductor_market_deliver_tools = {
	if = {
		limit = { check_variable = { global.semiconductor_market_tool_backlog > 0 } }
		subtract_from_variable = { global.semiconductor_market_tool_backlog = 1 }

		if = {
			limit = {
				check_variable = { global.semiconductor_market_tool_backlog > 0 }
				country_exists = HOL
				HOL = {
					NOT = { has_country_flag = collapsed_nation }
					check_variable = { var = HOL_asml_euv_maturity value = 7 compare = greater_than_or_equals }
					check_variable = { var = HOL_asml_supplier_resilience value = 7 compare = greater_than_or_equals }
				}
			}
			subtract_from_variable = { global.semiconductor_market_tool_backlog = 1 }
		}
	}
}

semiconductor_market_refresh_owner_pressure = {
	if = {
		limit = {
			country_exists = KOR
			KOR = { NOT = { has_country_flag = collapsed_nation } }
		}
		KOR = { KOR_samsung_update_market_pressure = yes }
	}
	if = {
		limit = {
			country_exists = HOL
			HOL = { NOT = { has_country_flag = collapsed_nation } }
		}
		HOL = { HOL_asml_update_market_pressure = yes }
	}
}

semiconductor_market_quarterly_update = {
	semiconductor_market_initialize_state = yes
	semiconductor_market_update_demand = yes
	semiconductor_market_update_access_tension = yes
	semiconductor_market_deliver_tools = yes
	semiconductor_market_clamp_state = yes
	semiconductor_market_refresh_owner_pressure = yes
}

# KOR and HOL both call this from their native monthly hosts. Timed global guards
# make the update idempotent without a global poll or country iteration.
semiconductor_market_quarterly_driver = {
	if = {
		limit = {
			has_game_rule = { rule = rule_corporate_history option = full }
			NOT = { has_country_flag = collapsed_nation }
			check_variable = { global.month = 1 }
			NOT = { has_global_flag = GLOBAL_semiconductor_market_q1_processed }
		}
		set_global_flag = { flag = GLOBAL_semiconductor_market_q1_processed value = 1 days = 80 }
		semiconductor_market_quarterly_update = yes
	}
	else_if = {
		limit = {
			has_game_rule = { rule = rule_corporate_history option = full }
			NOT = { has_country_flag = collapsed_nation }
			check_variable = { global.month = 4 }
			NOT = { has_global_flag = GLOBAL_semiconductor_market_q2_processed }
		}
		set_global_flag = { flag = GLOBAL_semiconductor_market_q2_processed value = 1 days = 80 }
		semiconductor_market_quarterly_update = yes
	}
	else_if = {
		limit = {
			has_game_rule = { rule = rule_corporate_history option = full }
			NOT = { has_country_flag = collapsed_nation }
			check_variable = { global.month = 7 }
			NOT = { has_global_flag = GLOBAL_semiconductor_market_q3_processed }
		}
		set_global_flag = { flag = GLOBAL_semiconductor_market_q3_processed value = 1 days = 80 }
		semiconductor_market_quarterly_update = yes
	}
	else_if = {
		limit = {
			has_game_rule = { rule = rule_corporate_history option = full }
			NOT = { has_country_flag = collapsed_nation }
			check_variable = { global.month = 10 }
			NOT = { has_global_flag = GLOBAL_semiconductor_market_q4_processed }
		}
		set_global_flag = { flag = GLOBAL_semiconductor_market_q4_processed value = 1 days = 80 }
		semiconductor_market_quarterly_update = yes
	}
}
