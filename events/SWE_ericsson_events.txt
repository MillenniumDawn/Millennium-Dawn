	title = SWE_ericsson_events.1.t
	desc = SWE_ericsson_events.1.d
	picture = GFX_computer
	is_triggered_only = yes

	trigger = {
		corporate_history_full_enabled = yes
		original_tag = SWE
		NOT = { has_country_flag = collapsed_nation }
		NOT = { has_country_flag = SWE_ericsson_event_1_resolved }
		NOT = { has_country_flag = SWE_sony_ericsson_joint_venture }
		NOT = { has_country_flag = SWE_ericsson_retained_handsets }
	}

	option = {
		name = SWE_ericsson_events.1.a
		log = "[GetDateText]: [Root.GetName]: SWE_ericsson_events.1.a executed"
		if = {
			limit = {
				hidden_trigger = {
					NOT = { has_country_flag = SWE_sony_ericsson_joint_venture }
					NOT = { has_country_flag = SWE_ericsson_retained_handsets }
				}
			}
			add_political_power = -5
			add_tech_bonus = {
				name = SWE_ericsson_events.1.joint_venture_bonus
				bonus = 0.10
				uses = 1
				category = CAT_computing_tech
			}
			SWE_ericsson_apply_joint_venture_outcome = yes
		}
		SWE_ericsson_complete_event_1 = yes
		SWE_ericsson_schedule_due_successor = yes
		ai_chance = {
			base = 50
			modifier = {
				factor = 10
				is_historical_focus_on = yes
			}
		}
	}

	option = {
		name = SWE_ericsson_events.1.b
		log = "[GetDateText]: [Root.GetName]: SWE_ericsson_events.1.b executed"
		if = {
			limit = {
				hidden_trigger = {
					NOT = { has_country_flag = SWE_sony_ericsson_joint_venture }
					NOT = { has_country_flag = SWE_ericsson_retained_handsets }
				}
			}
			set_temp_variable = { treasury_change = -2.00 }
			modify_treasury_effect = yes

				name = SWE_ericsson_events.1.handsets_bonus
				bonus = 0.15

				category = CAT_computing_tech
			}
			SWE_ericsson_apply_retained_handsets_outcome = yes
		}
