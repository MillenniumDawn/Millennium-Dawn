		limit = { linux_system_enabled = yes }
		set_country_flag = linux_system_dirty

linux_system_clear_adoption_ideas = {
	remove_ideas = linux_system_experimental_adoption
	remove_ideas = linux_system_institutional_adoption
	remove_ideas = linux_system_infrastructure_standard
	remove_ideas = linux_system_broad_economic_adoption
}

linux_system_clear_support_ideas = {
	remove_ideas = linux_system_mixed_linux_estate
	remove_ideas = linux_system_upstream_partnership
	remove_ideas = linux_system_enterprise_distribution
	remove_ideas = linux_system_national_baseline
}

linux_system_update_adoption_idea = {
	if = {
		limit = { original_tag = USA }
		linux_system_clear_adoption_ideas = yes
	}
	else_if = {
		limit = { check_variable = { linux_system_effective_deployment < 3 } }
		if = {
			limit = { NOT = { has_idea = linux_system_experimental_adoption } }
			linux_system_clear_adoption_ideas = yes
			add_ideas = linux_system_experimental_adoption
		}
	}
	else_if = {
		limit = { check_variable = { linux_system_effective_deployment < 6 } }
		if = {
			limit = { NOT = { has_idea = linux_system_institutional_adoption } }
			linux_system_clear_adoption_ideas = yes
			add_ideas = linux_system_institutional_adoption
		}
	}
	else_if = {
		limit = { check_variable = { linux_system_effective_deployment < 9 } }
		if = {
			limit = { NOT = { has_idea = linux_system_infrastructure_standard } }
			linux_system_clear_adoption_ideas = yes
			add_ideas = linux_system_infrastructure_standard
		}
	}
	else = {
		if = {
			limit = { NOT = { has_idea = linux_system_broad_economic_adoption } }
			linux_system_clear_adoption_ideas = yes
			add_ideas = linux_system_broad_economic_adoption
		}
	}
}

	set_variable = { linux_system_adapter_deployment = 0 }
	set_variable = { linux_system_adapter_stewardship = 0 }
	set_variable = { linux_system_adapter_assurance = 0 }
	set_variable = { linux_system_adapter_support_model = 0 }

		limit = {
			original_tag = USA
			corporate_history_enabled = yes
		}
		linux_system_refresh_usa_adapter = yes
	}
	else_if = {
		limit = { original_tag = ENG }
		linux_system_refresh_eng_adapter = yes
	}
	else_if = {
		limit = { original_tag = GER }
		linux_system_refresh_ger_adapter = yes
	}
	else_if = {
		limit = {
			corporate_history_enabled = yes
			original_tag = FRA
		}
		linux_system_refresh_fra_adapter = yes
	}
	else_if = {
		limit = { original_tag = RAJ }
		linux_system_refresh_raj_adapter = yes
	}
	else_if = {
		limit = {
			corporate_history_enabled = yes
			original_tag = SOV
		}
		linux_system_refresh_sov_adapter = yes
	}
	else_if = {
		limit = {
			corporate_history_enabled = yes
			original_tag = CHI

	}
	else_if = {
		limit = {
			corporate_history_enabled = yes

		linux_system_refresh_pol_adapter = yes

	else_if = {

linux_system_refresh_adapter_if_changed = {
	set_temp_variable = { linux_system_previous_adapter_deployment = linux_system_adapter_deployment }
	set_temp_variable = { linux_system_previous_adapter_stewardship = linux_system_adapter_stewardship }
	set_temp_variable = { linux_system_previous_adapter_assurance = linux_system_adapter_assurance }
	set_temp_variable = { linux_system_previous_adapter_support_model = linux_system_adapter_support_model }

	linux_system_refresh_adapter = yes

	if = {
		limit = {
			OR = {
				check_variable = { var = linux_system_adapter_deployment value = linux_system_previous_adapter_deployment compare = not_equals }
				check_variable = { var = linux_system_adapter_stewardship value = linux_system_previous_adapter_stewardship compare = not_equals }
				check_variable = { var = linux_system_adapter_assurance value = linux_system_previous_adapter_assurance compare = not_equals }
				check_variable = { var = linux_system_adapter_support_model value = linux_system_previous_adapter_support_model compare = not_equals }
			}
		}
		linux_system_mark_dirty = yes
	}
}

linux_system_recalculate_state = {
	if = {
		limit = {
			linux_system_enabled = yes
			has_country_flag = linux_system_initialized
		}

		clamp_variable = { var = linux_system_base_deployment min = 0 max = 10 }
		clamp_variable = { var = linux_system_base_stewardship min = 0 max = 10 }
		clamp_variable = { var = linux_system_base_assurance min = 0 max = 10 }
		clamp_variable = { var = linux_system_base_support_model min = 0 max = 3 }

		linux_system_refresh_adapter = yes

		set_variable = { linux_system_effective_deployment = linux_system_base_deployment }
		add_to_variable = { linux_system_effective_deployment = linux_system_adapter_deployment }
		set_variable = { linux_system_effective_stewardship = linux_system_base_stewardship }
		add_to_variable = { linux_system_effective_stewardship = linux_system_adapter_stewardship }
		set_variable = { linux_system_effective_assurance = linux_system_base_assurance }
		add_to_variable = { linux_system_effective_assurance = linux_system_adapter_assurance }
		if = {
			limit = { check_variable = { linux_system_base_support_model > 0 } }
			set_variable = { linux_system_effective_support_model = linux_system_base_support_model }
		}
		else = { set_variable = { linux_system_effective_support_model = linux_system_adapter_support_model } }

		clamp_variable = { var = linux_system_effective_deployment min = 0 max = 10 }
		clamp_variable = { var = linux_system_effective_stewardship min = 0 max = 10 }
		clamp_variable = { var = linux_system_effective_assurance min = 0 max = 10 }
		clamp_variable = { var = linux_system_effective_support_model min = 0 max = 3 }

		set_variable = { linux_system_milestone_stage = 0 }
		if = { limit = { has_country_flag = linux_system_event_1_resolved } set_variable = { linux_system_milestone_stage = 1 } }
		if = { limit = { has_country_flag = linux_system_event_2_resolved } set_variable = { linux_system_milestone_stage = 2 } }
		if = { limit = { has_country_flag = linux_system_event_3_resolved } set_variable = { linux_system_milestone_stage = 3 } }
		if = { limit = { has_country_flag = linux_system_event_4_resolved } set_variable = { linux_system_milestone_stage = 4 } }
		if = { limit = { has_country_flag = linux_system_event_5_resolved } set_variable = { linux_system_milestone_stage = 5 } }
		clamp_variable = { var = linux_system_milestone_stage min = 0 max = 5 }

		linux_system_refresh_ideas = yes
		if = {
			limit = {
				original_tag = USA
				corporate_history_enabled = yes
			}
			USA_corporate_systems_update_economic_bridge = yes
		}
		clr_country_flag = linux_system_dirty
	}
}

	if = {
		limit = {
			linux_system_enabled = yes
			has_country_flag = linux_system_initialized
			linux_system_country_needs_reconstruction = yes
			OR = {
				linux_system_outcomes_only_enabled = yes
				NOT = { has_country_flag = linux_system_country_bootstrapped }
			}
		}

		if = {
			limit = {
				date > 2001.1.3
				NOT = { has_country_flag = linux_system_event_1_resolved }
				NOT = { has_country_flag = linux_system_event_1_expected }
				NOT = { has_country_flag = linux_system_event_1_pending }
			}
			linux_system_reconstruct_event_1 = yes
		}
		if = {
			limit = {
				date > 2008.5.31
				has_country_flag = linux_system_event_1_resolved
				NOT = { has_country_flag = linux_system_event_2_resolved }
				NOT = { has_country_flag = linux_system_event_2_expected }
				NOT = { has_country_flag = linux_system_event_2_pending }
			}
			linux_system_reconstruct_event_2 = yes
		}
		if = {
			limit = {
				date > 2015.12.9
				has_country_flag = linux_system_event_2_resolved
				NOT = { has_country_flag = linux_system_event_3_resolved }
				NOT = { has_country_flag = linux_system_event_3_expected }
				NOT = { has_country_flag = linux_system_event_3_pending }
			}
			linux_system_reconstruct_event_3 = yes
		}
		if = {
			limit = {
				date > 2019.7.8
				has_country_flag = linux_system_event_3_resolved
				NOT = { has_country_flag = linux_system_event_4_resolved }
				NOT = { has_country_flag = linux_system_event_4_expected }
				NOT = { has_country_flag = linux_system_event_4_pending }
			}
			linux_system_reconstruct_event_4 = yes
		}
		if = {
			limit = {
				date > 2024.4.29
				has_country_flag = linux_system_event_4_resolved
				NOT = { has_country_flag = linux_system_event_5_resolved }
				NOT = { has_country_flag = linux_system_event_5_expected }
				NOT = { has_country_flag = linux_system_event_5_pending }
			}
			linux_system_reconstruct_event_5 = yes
		}

		linux_system_recalculate_state = yes
	}
}

linux_system_monthly_driver = {
	if = {
		limit = {
			linux_system_enabled = yes
			exists = yes
			NOT = { tag = ABK }
			NOT = { tag = ZOM }
			NOT = { MD_special_countries = yes }
			NOT = { has_country_flag = collapsed_nation }
		}
		linux_system_initialize_state = yes
		if = {
			limit = { NOT = { has_country_flag = linux_system_country_bootstrapped } }
			linux_system_reconstruct_country = yes
			set_country_flag = linux_system_country_bootstrapped
		}
		else_if = {
			limit = { linux_system_outcomes_only_enabled = yes }
			linux_system_reconstruct_country = yes
		}
		linux_system_refresh_adapter_if_changed = yes
		if = {
			limit = { linux_system_full_enabled = yes }
			linux_system_schedule_year_events = yes
			linux_system_recover_due_events = yes
		}
		if = {
			limit = { has_country_flag = linux_system_dirty }
			linux_system_recalculate_state = yes
		}
	}
	else_if = {
		limit = {
			NOT = { linux_system_enabled = yes }
			linux_system_has_owned_artifacts = yes
		}
		linux_system_clear_owned_artifacts = yes
	}
}
