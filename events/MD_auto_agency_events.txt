add_namespace = md_auto_agency

# Deferred update for the auto-agency button handlers.
# Processing the queue immediately on Upgrades On / Queue Only click caused three
# original bugs (queue vanishing, wrong upgrade starting, first queue-only entry
# disappearing). The original FFTF mod fixed this by deferring to on_daily; we use
# a per-click delayed event instead so we don't pay a per-day per-country cost.
country_event = {
	id = md_auto_agency.1
	hidden = yes
	is_triggered_only = yes
	immediate = {
		log = "[GetDateText]: [This.GetName]: md_auto_agency.1 executed"
		MD_auto_agency_update_state = yes
	}
}
