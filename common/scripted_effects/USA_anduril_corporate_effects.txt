USA_anduril_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = USA_anduril_state_initialized } }
		set_variable = { USA_anduril_autonomy_depth = 3 }
		set_variable = { USA_anduril_production_scale = 1 }
		set_variable = { USA_anduril_procurement_reform = 4 }
		set_variable = { USA_anduril_allied_integration = 1 }
		set_variable = { USA_anduril_civil_oversight = 5 }
		set_country_flag = USA_anduril_state_initialized
	}
	USA_anduril_clamp_state = yes
}

USA_anduril_resolve_capstone = {
	USA_anduril_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = USA_anduril_capstone_resolved } }
		if = {
			limit = { check_variable = { USA_anduril_capstone_choice = 1 } }
			USA_anduril_apply_autonomous_arsenal = yes
		}
		else_if = {
			limit = { check_variable = { USA_anduril_capstone_choice = 2 } }
			USA_anduril_apply_allied_autonomy_bloc = yes
		}
		else_if = {
			limit = { check_variable = { USA_anduril_capstone_choice = 3 } }
			USA_anduril_apply_procurement_insurgency = yes
		}
		else_if = {
			limit = { check_variable = { USA_anduril_capstone_choice = 4 } }
			USA_anduril_apply_unaccountable_platform = yes
		}
		else = { USA_anduril_apply_disciplined_challenger = yes }
	}
}
