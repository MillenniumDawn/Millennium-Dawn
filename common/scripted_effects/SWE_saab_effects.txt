# Saab corporate history. Five milestones set the route; the closing milestone
# resolves the export platform against alliance integration into one of four outcomes.
SWE_saab_apply_allied_arms_exporter = {
	if = {
		limit = { SWE_saab_outcome_settled = no }
		add_ideas = SWE_saab_allied_arms_exporter
	}
}
