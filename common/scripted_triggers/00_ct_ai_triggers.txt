# AI Counter-Terrorism trigger definitions
#
# These triggers support the CT AI brain in 00_ct_ai_effects.txt.
# Scoping: called from the on_weekly var:v loop, so THIS = the AI country.

# Nation power classification for CT prioritisation
is_ct_major_power = {
	OR = {
		has_idea = super_power
		has_idea = great_power
	}
}

is_ct_regional_power = {
	OR = {
		has_idea = large_power
		has_idea = regional_power
	}
}

# Region match check.
# Expects temp var `ct_check_region` to be set by the calling effect
# before this trigger is evaluated.
# ct_check_region values: 0=Europe, 1=Sub-Saharan Africa, 2=Middle East, 3=Asia, 4=Americas
ct_ai_same_region_as_org = {
	OR = {
		AND = {
			check_variable = { ct_check_region = 0 }
			has_country_flag = european_nation_flag
		}
		AND = {
			check_variable = { ct_check_region = 1 }
			has_country_flag = africa_nation_flag
		}
		AND = {
			check_variable = { ct_check_region = 2 }
			has_country_flag = middle_east_flag
		}
		AND = {
			check_variable = { ct_check_region = 3 }
			has_country_flag = asian_nation_flag
		}
		AND = {
			check_variable = { ct_check_region = 4 }
			NOT = {
				OR = {
					has_country_flag = european_nation_flag
					has_country_flag = africa_nation_flag
					has_country_flag = middle_east_flag
					has_country_flag = asian_nation_flag
				}
			}
		}
	}
}
