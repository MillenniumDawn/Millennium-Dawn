add_namespace = faction_goal_cache

# Runs after weekly country updates so faction totals use one update generation.
country_event = {
	id = faction_goal_cache.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { is_faction_leader = yes }
			log = "[GetDateText]: [This.GetName]: faction_goal_cache.1 executed"
			faction_goal_update_weekly_caches = yes
		}
	}
}

# Establishes a country ROOT because the global startup scope has none.

			faction_goal_update_daily_caches = yes
		}
	}
}
