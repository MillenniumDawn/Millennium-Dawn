ALG_morocco_tensions = {
	allowed = { original_tag = ALG }
	enable = {
		country_exists = SHA
		country_exists = MOR
		MOR = { any_owned_state = { is_core_of = SHA } }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = antagonize id = "MOR" value = 50 }
	ai_strategy = { type = contain id = "MOR" value = 50 }
	ai_strategy = { type = support id = "SHA" value = 50 }
}

#Libya supports Algeria

	allowed = { original_tag = ALG }
	enable = {
		LBA = { has_completed_focus = LBA_support_algeria }
