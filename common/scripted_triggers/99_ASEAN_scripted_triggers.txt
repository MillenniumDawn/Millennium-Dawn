# Decision `allowed` locks at game start, so every nation the tree can ever admit has to be
# named here. A country admitted through Branch 4 that is missing from this list joins the
# member array and then never sees a single ASEAN decision, including the one to leave.
can_hold_asean_decisions = {
	OR = {
		is_asean_nation = yes
		original_tag = PAP		#Papua New Guinea
		original_tag = FIJ		#Fiji
		original_tag = SOL		#Solomon Islands
		original_tag = PAU		#Palau
		original_tag = VAN		#Vanuatu
		original_tag = TON		#Tonga
		original_tag = MAR		#Marshall Islands
		original_tag = BAN		#Bangladesh
		original_tag = SRI		#Sri Lanka
	}
}

is_asean_member = {
	OR = {
		has_idea = ASEAN_Member
		has_idea = ASEAN_Member_drifting
		has_idea = ASEAN_Member_detached
		has_idea = ASEAN_Member_estranged
	}
}

is_asean_member_committed = {
	has_idea = ASEAN_Member
}

is_asean_member_active = {
	OR = {
		has_idea = ASEAN_Member
		has_idea = ASEAN_Member_drifting
	}
}

# No power-tier gate: half the founding membership starts at non_power, so any tier floor
# locks them out of the whole tree. Standing is the gate instead.
can_do_asean_focus = {
	is_asean_member_active = yes
}
