lar_recruitment_capital_trigger = {
	OR = {
		AND = {
			is_capital = yes
			NOT = { is_controlled_by = ROOT }
		}
		AND = {
			has_variable = historical_capital_for_country
			var:historical_capital_for_country = {
				NOT = { tag = ROOT }
				OR = {
					has_capitulated = yes
					exists = no
				}
			}
		}
	}
}
