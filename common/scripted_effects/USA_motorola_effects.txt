			limit = {

			limit = {

					has_country_flag = USA_motorola_freescale_retained
					has_country_flag = USA_motorola_freescale_strategic_foundry
					has_country_flag = USA_motorola_freescale_us_control

				check_variable = { var = USA_motorola_semiconductor_control value = 7 compare = greater_than_or_equals }
			}

		}
		else_if = {

				}
