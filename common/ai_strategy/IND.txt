IND_singapore_is_expansionary = {
	allowed = { original_tag = IND }
	enable = {
		SIN = {
			has_completed_focus = SIN_pressure_the_south_asian_seas
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = contain id = "SIN" value = 50 }
	ai_strategy = { type = antagonize id = "SIN" value = 50 }
}

# Losing-war brakes for the wargoals the focus tree hands out - the regional purification branch (PHI, SIN, VIE, CBD, SIA), the Riau and Australian reparations demands (MAY, AST) and the Aceh settlement (ACE) - so Indonesia backs off a war it is losing instead of fighting to the last man
IND_cancel_war_PHI = {
	allowed = { original_tag = IND }
	enable = {
		has_war_with = PHI
		strength_ratio = { tag = PHI ratio < 1 }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = declare_war id = PHI value = -4000 }
}
