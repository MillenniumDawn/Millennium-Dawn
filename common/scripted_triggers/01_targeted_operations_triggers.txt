# Reads TOP_target.
TOP_person_operational_eligible = {
	TOP_human_offense = yes
	check_variable = { TOP_target > 0 }
	check_variable = { TOP_target < global.TOP_status^num }
	check_variable = { global.TOP_status^TOP_target = 1 }
	set_temp_variable = { TOP_role_target = TOP_target }
	TOP_authored_role_eligible = yes
	set_temp_variable = { TOP_person_group = global.TOP_affiliation^TOP_target }
	set_temp_variable = { TOP_person_class = global.TOP_group_class^TOP_person_group }
	OR = {
		AND = {
			TOP_full_sandbox = yes
			OR = {
				check_variable = { TOP_person_class = 1 }
				NOT = { check_variable = { global.TOP_group_host^TOP_person_group = THIS } }
				AND = {
					check_variable = { TOP_mandates^TOP_target > global.TOP_clock }
					set_temp_variable = { TOP_civilian_valid_target = TOP_target }
					OR = {
						has_civil_war = yes
						TOP_authored_civilian_mandate_valid = yes
					}
				}
			}
		}
		AND = {
			TOP_limited_sandbox = yes
			OR = {
				check_variable = { TOP_person_class = 1 }
				AND = {
					check_variable = { TOP_person_class = 2 }
					OR = {
						check_variable = { TOP_mandates^TOP_target > global.TOP_clock }
						check_variable = { TOP_org_mandates^TOP_person_group > global.TOP_clock }
						var:global.TOP_group_host^TOP_person_group = { has_war_with = PREV }
					}
				}
				AND = {
					check_variable = { TOP_person_class > 2 }
					OR = {
						check_variable = { TOP_mandates^TOP_target > global.TOP_clock }
						check_variable = { TOP_org_mandates^TOP_person_group > global.TOP_clock }
					}
				}
			}
		}
	}
}

# Reads TOP_access_method, TOP_access_state and TOP_access_host.
TOP_person_access_available = {
	NOT = { check_variable = { TOP_access_state = 0 } }
	check_variable = { TOP_access_host > 0 }
	var:TOP_access_host = { exists = yes }
	OR = {
		AND = {
			check_variable = { TOP_access_method = 1 }
			# Native launch revalidates base, equipment, range, and DLC gates.
			always = yes
		}
		AND = {
			check_variable = { TOP_access_method = 2 }
			has_tech = special_forces_tech_1
		}
		AND = {
			check_variable = { TOP_access_method = 3 }
			has_tech = decryption_1
			NOT = { check_variable = { TOP_access_host = THIS } }
		}
		AND = {
			check_variable = { TOP_access_method = 4 }
			has_tech = special_forces_tech_1
			NOT = { check_variable = { TOP_access_host = THIS } }
		}
		AND = {
			check_variable = { TOP_access_method = 5 }
			NOT = { check_variable = { TOP_access_host = THIS } }
		}
		AND = {
			check_variable = { TOP_access_method = 7 }
			original_tag = SOV
			has_tech = decryption_2
			NOT = { check_variable = { TOP_access_host = THIS } }
		}
	}
}
