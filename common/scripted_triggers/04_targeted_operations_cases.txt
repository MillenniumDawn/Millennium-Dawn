# Reads TOP_arg_target and TOP_arg_host.
TOP_host_case_available = {
	NOT = {
		any_of = {
			array = TOP_active_cases
			value = TOP_host_check
			check_variable = { TOP_case_host^TOP_host_check = TOP_arg_host }
			check_variable = { TOP_case_phase^TOP_host_check > 0 }
			NOT = { check_variable = { TOP_host_check = TOP_arg_target } }
		}
	}
}

TOP_can_designate = {
	TOP_country_eligible = yes
	TOP_has_selected_target = yes
	check_variable = { global.TOP_status^TOP_selected = 1 }
	check_variable = { TOP_case_phase^TOP_selected = 0 }
	check_variable = { TOP_lead_host^TOP_selected > 0 }
	var:TOP_lead_host^TOP_selected = { exists = yes }
	set_temp_variable = { TOP_role_target = TOP_selected }
	TOP_authored_role_eligible = yes
	set_temp_variable = { TOP_arg_target = TOP_selected }
	set_temp_variable = { TOP_arg_host = TOP_lead_host^TOP_selected }
	TOP_host_case_available = yes
	set_temp_variable = { TOP_target = TOP_selected }
	TOP_exceptional_authority = yes
}

}

# Reads TOP_arg_target, TOP_arg_method and TOP_arg_state.
TOP_native_binding_unused = {
	set_temp_variable = { TOP_binding_state = TOP_arg_state }
	check_variable = { TOP_binding_state > 0 }
	check_variable = { TOP_binding_state < 10000 }
	set_temp_variable = { TOP_binding_key = { value = TOP_arg_target multiply = 20000 } }
	set_temp_variable = { TOP_binding_key = { value = TOP_binding_key add = { value = TOP_arg_state multiply = 2 } } }
	set_temp_variable = { TOP_binding_key = { value = TOP_binding_key add = TOP_arg_method } }
	NOT = { is_in_array = { TOP_retired_native_bindings = TOP_binding_key } }
}

# Reads TOP_arg_target, TOP_arg_method and TOP_arg_state.
TOP_case_binding_valid = {
	TOP_country_eligible = yes
	set_temp_variable = { TOP_binding_target = TOP_arg_target }
	check_variable = { TOP_binding_target > 0 }
	check_variable = { TOP_binding_target < global.TOP_registry_capacity }
	OR = { check_variable = { TOP_case_phase^TOP_arg_target = 2 } check_variable = { TOP_case_phase^TOP_arg_target = 3 } }
	check_variable = { TOP_case_method^TOP_arg_target = TOP_arg_method }
	check_variable = { TOP_case_state^TOP_arg_target = TOP_arg_state }
	check_variable = { TOP_case_until^TOP_arg_target > global.TOP_clock }
	check_variable = { global.TOP_status^TOP_arg_target = 1 }
	check_variable = { TOP_lead_age^TOP_arg_target < 57 }
	check_variable = { TOP_confidence^TOP_arg_target > 59 }
	set_temp_variable = { TOP_expected_host = TOP_case_host^TOP_arg_target }
	var:TOP_case_state^TOP_arg_target = { is_controlled_by = var:TOP_expected_host }
	set_temp_variable = { TOP_role_target = TOP_arg_target }
	TOP_authored_role_eligible = yes
	set_temp_variable = { TOP_target = TOP_arg_target }
	set_temp_variable = { TOP_case_visit_target = TOP_arg_target }
	TOP_case_visit_execution_valid = yes
	TOP_exceptional_authority = yes
}
