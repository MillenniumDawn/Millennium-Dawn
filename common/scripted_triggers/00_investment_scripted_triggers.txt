# Prevents reactor construction in countries that have banned it (ITA) or gone carbon-neutral (GER).
# Consulted by both the player investment GUI and the AI scorer.
country_can_build_nuclear_reactors = {
	NOT = { has_idea = ITA_nuclear_power_banned }
	NOT = { has_idea = GER_idea_carbon_neutral }
}

# Building slot availability triggers for investment projects.
