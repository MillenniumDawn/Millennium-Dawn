add_namespace = HKG_contract

# Contract offer from Beijing. Fired by on_weekly (AI PRC), the CHI commission
# decision (player PRC), or HKG's own solicit decision. The immediate block
# rolls the offer terms so the description can display them.
country_event = {
	id = HKG_contract.1
	title = HKG_contract.1.t
	picture = GFX_politics_negotiations

	is_triggered_only = yes

	desc = { text = HKG_contract.1.d_normal  trigger = { check_variable = { HKG_contract_offer_license = 0 } } }
	desc = { text = HKG_contract.1.d_license trigger = { check_variable = { HKG_contract_offer_license = 1 } } }

	immediate = {
		if = {
			limit = {
				NOT = { has_country_flag = HKG_contract_offer_pending }
				check_variable = { HKG_contract_count < HKG_contract_max_slots }
			}
			HKG_contract_generate_offer = yes
			set_country_flag = { flag = HKG_contract_offer_cooldown value = 1 days = 150 }
		}
	}

	option = {
		name = HKG_contract.1.a
		ai_chance = {
			base = 80
			modifier = { factor = 0 has_war = yes }
		}
		log = "[GetDateText]: [Root.GetName]: Event HKG_contract.1 Option a"
		custom_effect_tooltip = HKG_contract_tt_accept_advance
		if = {
			limit = { check_variable = { HKG_contract_offer_license = 1 } }
			custom_effect_tooltip = HKG_contract_tt_accept_license
		}
		custom_effect_tooltip = HKG_contract_tt_accept_onfulfil
		HKG_contract_accept_offer = yes
	}

	option = {
		name = HKG_contract.1.b
		ai_chance = { base = 20 }
		log = "[GetDateText]: [Root.GetName]: Event HKG_contract.1 Option b"
		custom_effect_tooltip = HKG_contract_tt_decline
		HKG_contract_decline_offer = yes
	}
}
