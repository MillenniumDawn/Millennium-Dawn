# Hidden weekend router: resolves the private-consortium attempt so that the federal
# rescue and the October escalation each keep a single visible caller.
country_event = {
	id = USA_aig_events.2
	is_triggered_only = yes
	hidden = yes
	fire_only_once = yes

	trigger = {
		corporate_history_full_enabled = yes
		original_tag = USA
		NOT = { has_country_flag = collapsed_nation }
	}

	immediate = {
		USA_aig_initialize_state = yes
		if = {
			limit = { has_country_flag = USA_aig_consortium_attempted }
			random_list = {
				35 = {
					set_country_flag = USA_aig_consortium_succeeded
					set_country_flag = USA_aig_rescue_active
					add_to_variable = { USA_aig_banking_capital = 1 }
					USA_aig_clamp_state = yes
					country_event = { id = USA_aig_events.5 days = 14 }
				}
				65 = {
					add_to_variable = { USA_aig_financial_panic = 1 }
					add_to_variable = { USA_aig_public_anger = 1 }
					USA_aig_clamp_state = yes
					country_event = { id = USA_aig_events.3 days = 1 }
				}
			}
		}
		else = { country_event = { id = USA_aig_events.3 days = 1 } }
	}
}
