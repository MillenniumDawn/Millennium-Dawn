TOP_process_organization_cases = {
	clear_array = TOP_organization_cases_to_close
	for_each_loop = {
		array = TOP_active_organization_cases
		value = top_org_case
		set_temp_variable = { TOP_group_target = top_org_case }
		if = {
			limit = { check_variable = { TOP_org_case_phase^top_org_case < 4 } NOT = { TOP_group_operational = yes } }
			add_to_array = { TOP_organization_cases_to_close = top_org_case }
		}
		else_if = {
			limit = {
				check_variable = { TOP_org_case_phase^top_org_case = 3 }
				check_variable = { var = global.TOP_clock value = TOP_org_case_due^top_org_case compare = greater_than_or_equals }
			}
			set_temp_variable = { TOP_arg_group = top_org_case }
			if = { limit = { TOP_organization_mission_binding_valid = yes } TOP_resolve_organization_operation = yes }
			else = { add_to_array = { TOP_organization_cases_to_close = top_org_case } }
		}
		else_if = {
			limit = {
				check_variable = { TOP_org_case_phase^top_org_case < 3 }
				check_variable = { var = global.TOP_clock value = TOP_org_case_until^top_org_case compare = greater_than_or_equals }
			}
			add_to_array = { TOP_organization_cases_to_close = top_org_case }
		}
	}
	for_each_loop = {
		array = TOP_organization_cases_to_close
		value = top_org_case
		set_temp_variable = { TOP_arg_group = top_org_case }
		set_temp_variable = { TOP_arg_sequence = TOP_org_case_sequence^top_org_case }
		TOP_close_organization_case = yes
	}
}

TOP_resolve_organization_operation = {
	set_temp_variable = { TOP_intelligence_roll = 0 }
	set_temp_variable = { TOP_report_enqueue_impact = 0 }
	randomize_temp_variable = { var = TOP_intelligence_roll min = 1 max = 100 distribution = uniform }
