HOL_lake_help = {
	allowed = { original_tag = HOL }
	enable = {
		country_exists = GLC
		HOL = {
			has_war = no
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = send_volunteers_desire id = "GLC" value = 1000 }
	ai_strategy = { type = send_lend_lease_desire id = "GLC" value = 400 }
}
