# - crimea_requested_influence_change: Total points for the hulls in this order or delivery.
UKR_crimean_bop_apply_naval_reorganization_influence = {
	if = {
		limit = {
			tag = UKR
			has_country_flag = UKR_crimean_bop_initialized_flag
			NOT = { has_country_flag = UKR_crimean_bop_settlement_resolved_flag }
			669 = { is_owned_by = UKR }
			country_exists = SOV
		}
		custom_effect_tooltip = UKR_crimean_bop_naval_reorganization_influence_tt
		hidden_effect = {
			set_temp_variable = { crimea_influence_history_source = 13 }
			UKR_crimean_bop_transfer_influence_from_russia_to_ukraine = yes
		}
	}
}

UKR_crimean_bop_transfer_influence_from_russia_to_ukraine = {
	set_temp_variable = { crimea_applied_influence_change = crimea_requested_influence_change }
	set_temp_variable = { crimea_available_influence_gain = 65 }
	if = {
		limit = {
			OR = {
				check_variable = { var = UKR_crimean_bop_crimean_tatar_influence value = 25 compare = less_than }
				UKR_crimean_bop_historical_ai_before_2014 = yes
			}
		}
		set_temp_variable = { crimea_available_influence_gain = 45 }
	}
	subtract_from_temp_variable = { crimea_available_influence_gain = UKR_crimean_bop_ukrainian_influence }
	if = { limit = { check_variable = { var = crimea_applied_influence_change value = crimea_available_influence_gain compare = greater_than } } set_temp_variable = { crimea_applied_influence_change = crimea_available_influence_gain } }
	if = { limit = { check_variable = { var = crimea_applied_influence_change value = UKR_crimean_bop_russian_influence compare = greater_than } } set_temp_variable = { crimea_applied_influence_change = UKR_crimean_bop_russian_influence } }
	if = { limit = { check_variable = { var = crimea_applied_influence_change value = 0 compare = greater_than } }
		add_to_variable = { UKR_crimean_bop_ukrainian_influence = crimea_applied_influence_change }
		subtract_from_variable = { UKR_crimean_bop_russian_influence = crimea_applied_influence_change }
		add_to_variable = { UKR_crimean_bop_influence_changes_count = 1 }
		set_temp_variable = { crimea_ukrainian_influence_change = 0 }
		set_temp_variable = { crimea_crimean_tatar_influence_change = 0 }
		set_temp_variable = { crimea_russian_influence_change = 0 }
		set_temp_variable = { crimea_ukrainian_influence_change = crimea_applied_influence_change }
		set_temp_variable = { crimea_russian_influence_change = crimea_applied_influence_change }
		multiply_temp_variable = { crimea_russian_influence_change = -1 }
		UKR_crimean_bop_record_influence_change = yes
		UKR_crimean_bop_update_display = yes
		UKR_crimean_bop_check_settlement_outcome = yes
	}
}

# Once Crimean Tatar influence reaches 30%, supporting the Tatar bloc
# draws half of the marginal influence from Kyiv rather than exclusively from Russia.
# Effect: UKR_crimean_bop_limit_historical_tatar_gain
# Purpose: Cap Tatar influence from a historical AI sponsor before 2014.
# Parameters:
# - crimea_tatar_influence_sponsor: Country ID of the government that started the program.
# - crimea_requested_influence_change: Requested Tatar influence gain, reduced in place.
UKR_crimean_bop_limit_historical_tatar_gain = {
	if = {
		limit = { var:crimea_tatar_influence_sponsor = { UKR_crimean_bop_historical_ai_before_2014 = yes } }
		set_temp_variable = { crimea_historical_tatar_influence_room = 15 }
		subtract_from_temp_variable = { crimea_historical_tatar_influence_room = UKR_crimean_bop_crimean_tatar_influence }
		clamp_temp_variable = { var = crimea_historical_tatar_influence_room min = 0 }
		clamp_temp_variable = { var = crimea_requested_influence_change max = crimea_historical_tatar_influence_room }
	}
}

UKR_crimean_bop_apply_tatar_coalition_influence_gain = {
	if = { limit = { check_variable = { var = UKR_crimean_bop_crimean_tatar_influence value = 30 compare = greater_than_or_equals } }
		set_temp_variable = { crimea_original_influence_change = crimea_requested_influence_change }
		set_temp_variable = { crimea_tatar_coalition_history_source = crimea_influence_history_source }
		divide_temp_variable = { crimea_requested_influence_change = 2 }
		UKR_crimean_bop_transfer_influence_from_ukraine_to_crimean_tatars = yes
		set_temp_variable = { crimea_influence_history_source = crimea_tatar_coalition_history_source }
		set_temp_variable = { crimea_requested_influence_change = crimea_original_influence_change }
		divide_temp_variable = { crimea_requested_influence_change = 2 }
		UKR_crimean_bop_transfer_influence_from_russia_to_crimean_tatars = yes
	}
	else = { UKR_crimean_bop_transfer_influence_from_russia_to_crimean_tatars = yes }
}

UKR_crimean_bop_update_monthly = {
	if = { limit = { NOT = { has_country_flag = UKR_crimean_bop_initialized_flag } } UKR_crimean_bop_initialize = yes }
	# Track foreign-held Sevastopol before a return; owning it at game start
	# is NOT a return and must not grant a free +15.
	if = { limit = { 1112 = { NOT = { is_owned_by = UKR } } }
		set_country_flag = UKR_crimean_bop_sevastopol_was_foreign_flag
	}
	# A peaceful Crimean settlement may already be resolved by the time the
	# native Sevastopol negotiation hands state 1112 back to Ukraine.
	if = { limit = { 1112 = { is_owned_by = UKR } has_country_flag = UKR_crimean_bop_sevastopol_was_foreign_flag NOT = { has_country_flag = UKR_crimean_bop_sevastopol_bonus_applied_flag } }
		UKR_crimean_bop_sevastopol_return_bonus = yes
	}
	if = { limit = { NOT = { has_country_flag = UKR_crimean_bop_settlement_resolved_flag } UKR_crimean_bop_can_negotiate_settlement = yes }
		add_to_variable = { UKR_crimean_bop_no_admin_months = 1 }
		if = { limit = { check_variable = { var = UKR_crimean_bop_no_admin_months value = 18 compare = greater_than_or_equals } NOT = { has_active_mission = UKR_crimean_bop_corruption_mission } }
			set_variable = { UKR_crimean_bop_no_admin_months = 0 }
			activate_mission = UKR_crimean_bop_corruption_mission
		}
		if = { limit = {
				check_variable = { var = UKR_crimean_bop_returnees_population value = 8000 compare = greater_than_or_equals }
				check_variable = { var = UKR_crimean_bop_housing_completed value = 1 compare = less_than }
				NOT = { has_active_mission = UKR_crimean_bop_land_disputes_mission }
				NOT = { has_country_flag = UKR_crimean_bop_land_disputes_cooldown_flag }
			}
			activate_mission = UKR_crimean_bop_land_disputes_mission
			set_country_flag = { flag = UKR_crimean_bop_land_disputes_cooldown_flag days = 365 value = 1 }
		}
		if = { limit = { check_variable = { var = UKR_crimean_bop_ukrainian_influence value = 35 compare = greater_than_or_equals } check_variable = { var = UKR_crimean_bop_crimean_tatar_influence value = 25 compare = greater_than_or_equals } check_variable = { var = UKR_crimean_bop_russian_influence value = 25 compare = greater_than_or_equals } NOT = { has_active_mission = UKR_crimean_bop_joint_resistance_mission } }
			random = { chance = 5 activate_mission = UKR_crimean_bop_joint_resistance_mission }
		}
		UKR_crimean_bop_check_settlement_outcome = yes
	}
}

# Recalculate diplomatic percentages from the MD parallel influence arrays.
# One unit of raw influence is shared by all foreign contributors and domestic independence.
UKR_crimean_bop_update_repatriation_diplomatic_influence = {
	set_variable = { UKR_crimean_bop_repatriation_ukrainian_influence_raw = 0 }
	set_variable = { UKR_crimean_bop_repatriation_russian_influence_raw = 0 }
	set_temp_variable = { crimea_repatriation_total_influence = domestic_influence_amount }
	for_each_loop = { array = influence_array_val value = crimea_foreign_influence_entry
		add_to_temp_variable = { crimea_repatriation_total_influence = crimea_foreign_influence_entry }
	}
	for_each_loop = { array = influence_array value = crimea_influencing_country_id index = crimea_foreign_influence_index
		if = { limit = { check_variable = { var = crimea_influencing_country_id value = UKR.id } }
			set_variable = { UKR_crimean_bop_repatriation_ukrainian_influence_raw = influence_array_val^crimea_foreign_influence_index }
		}
		if = { limit = { check_variable = { var = crimea_influencing_country_id value = SOV.id } }
			set_variable = { UKR_crimean_bop_repatriation_russian_influence_raw = influence_array_val^crimea_foreign_influence_index }
		}
	}
	set_variable = { UKR_crimean_bop_repatriation_ukrainian_influence_share = 0 }
	if = { limit = { check_variable = { var = crimea_repatriation_total_influence value = 0 compare = greater_than } }
		set_variable = { UKR_crimean_bop_repatriation_ukrainian_influence_share = UKR_crimean_bop_repatriation_ukrainian_influence_raw }
		divide_variable = { var = UKR_crimean_bop_repatriation_ukrainian_influence_share value = crimea_repatriation_total_influence }
		multiply_variable = { UKR_crimean_bop_repatriation_ukrainian_influence_share = 100 }
	}
	set_variable = { UKR_crimean_bop_repatriation_russian_influence_limit = UKR_crimean_bop_repatriation_ukrainian_influence_raw }
	multiply_variable = { UKR_crimean_bop_repatriation_russian_influence_limit = 3 }
}

# The only action permitted to raise Ukrainian influence over the standard 65-point ceiling.
# Must be called from UKR after state 1112 has ACTUALLY been transferred to UKR.
# Idempotent across the existing Sevastopol event and on_monthly synchronization.
UKR_crimean_bop_sevastopol_return_bonus = {
	if = {
		limit = {
			has_country_flag = UKR_crimean_bop_initialized_flag
			has_country_flag = UKR_crimean_bop_sevastopol_was_foreign_flag
			NOT = { has_country_flag = UKR_crimean_bop_sevastopol_bonus_applied_flag }
			OR = {
				669 = { is_owned_by = UKR }
				AND = { 669 = { is_owned_by = CRM } CRM = { is_subject_of = UKR } }
			}
			1112 = { is_owned_by = UKR }
		}
		set_temp_variable = { crimea_ukrainian_influence_before_sevastopol_bonus = UKR_crimean_bop_ukrainian_influence }
		set_temp_variable = { crimea_crimean_tatar_influence_before_sevastopol_bonus = UKR_crimean_bop_crimean_tatar_influence }
		set_temp_variable = { crimea_russian_influence_before_sevastopol_bonus = UKR_crimean_bop_russian_influence }
		set_country_flag = UKR_crimean_bop_sevastopol_bonus_applied_flag
		set_temp_variable = { crimea_sevastopol_remaining_influence_bonus = 15 }
		set_temp_variable = { crimea_sevastopol_available_influence_gain = 100 }
		subtract_from_temp_variable = { crimea_sevastopol_available_influence_gain = UKR_crimean_bop_ukrainian_influence }
		if = {
			limit = { check_variable = { var = crimea_sevastopol_remaining_influence_bonus value = crimea_sevastopol_available_influence_gain compare = greater_than } }
			set_temp_variable = { crimea_sevastopol_remaining_influence_bonus = crimea_sevastopol_available_influence_gain }
		}
		# Primary donor is SOV / pro-Russian bloc.
		set_temp_variable = { crimea_sevastopol_influence_from_russia = crimea_sevastopol_remaining_influence_bonus }
		if = {
			limit = { check_variable = { var = crimea_sevastopol_influence_from_russia value = UKR_crimean_bop_russian_influence compare = greater_than } }
			set_temp_variable = { crimea_sevastopol_influence_from_russia = UKR_crimean_bop_russian_influence }
		}
		if = {
			limit = { check_variable = { var = crimea_sevastopol_influence_from_russia value = 0 compare = greater_than } }
			add_to_variable = { UKR_crimean_bop_ukrainian_influence = crimea_sevastopol_influence_from_russia }
			subtract_from_variable = { UKR_crimean_bop_russian_influence = crimea_sevastopol_influence_from_russia }
			subtract_from_temp_variable = { crimea_sevastopol_remaining_influence_bonus = crimea_sevastopol_influence_from_russia }
		}
		# Only if Russia's share is insufficient, transfer the remainder from Tatar bloc.
		if = {
			limit = { check_variable = { var = crimea_sevastopol_remaining_influence_bonus value = 0 compare = greater_than } }
			if = {
				limit = { check_variable = { var = crimea_sevastopol_remaining_influence_bonus value = UKR_crimean_bop_crimean_tatar_influence compare = greater_than } }
				set_temp_variable = { crimea_sevastopol_remaining_influence_bonus = UKR_crimean_bop_crimean_tatar_influence }
			}
			add_to_variable = { UKR_crimean_bop_ukrainian_influence = crimea_sevastopol_remaining_influence_bonus }
			subtract_from_variable = { UKR_crimean_bop_crimean_tatar_influence = crimea_sevastopol_remaining_influence_bonus }
		}
		set_temp_variable = { crimea_ukrainian_influence_change = UKR_crimean_bop_ukrainian_influence }
		subtract_from_temp_variable = { crimea_ukrainian_influence_change = crimea_ukrainian_influence_before_sevastopol_bonus }
		set_temp_variable = { crimea_crimean_tatar_influence_change = UKR_crimean_bop_crimean_tatar_influence }
		subtract_from_temp_variable = { crimea_crimean_tatar_influence_change = crimea_crimean_tatar_influence_before_sevastopol_bonus }
		set_temp_variable = { crimea_russian_influence_change = UKR_crimean_bop_russian_influence }
		subtract_from_temp_variable = { crimea_russian_influence_change = crimea_russian_influence_before_sevastopol_bonus }
		if = { limit = { check_variable = { var = crimea_ukrainian_influence_change value = 0 compare = greater_than } }
			set_temp_variable = { crimea_influence_history_source = 11 }
			UKR_crimean_bop_record_influence_change = yes
		}
		UKR_crimean_bop_update_display = yes
		UKR_crimean_bop_check_settlement_outcome = yes
	}
}

# Reuse CRM as the ONLY Crimean tag. CRM_TATAR is a cosmetic tag, never a new country tag.
# Each settlement explicitly handles core and claims on BOTH states.
# IMPORTANT: these effects should run in UKR country scope, not in state or other country scope.

UKR_crimean_bop_ukrainian_settlement = {
	669 = {
		add_core_of = UKR
		remove_claim_by = UKR
		remove_core_of = SOV
		remove_claim_by = SOV
		remove_core_of = CRM
		remove_claim_by = CRM
	}
	1112 = {
		# The Russian-owned leased bay is not confiscated by a peaceful settlement.
		if = { limit = { is_owned_by = UKR }
			add_core_of = UKR
			remove_claim_by = UKR
		}
		else = { add_claim_by = UKR }
		remove_claim_by = CRM
		remove_core_of = CRM
	}
}

UKR_crimean_bop_return_bay_to_ukraine = {
	# Mark a foreign-to-Ukrainian return BEFORE the ownership transition.
	if = { limit = { 1112 = { NOT = { is_owned_by = UKR } } }
		set_country_flag = UKR_crimean_bop_sevastopol_was_foreign_flag
	}
	1112 = {
		transfer_state_to = UKR
		add_core_of = UKR
		remove_claim_by = UKR
		remove_core_of = SOV
		remove_claim_by = SOV
		remove_core_of = CRM
		remove_claim_by = CRM
	}
	UKR_crimean_bop_sevastopol_return_bonus = yes
}

# Stage the autonomous CRM territory before an actual release; do not let the base be
# accidentally transferred to CRM through pre-existing Crimean cores.
UKR_crimean_bop_stage_crm_release = {
	669 = {
		add_core_of = CRM
		remove_core_of = SOV
		remove_claim_by = SOV
		remove_claim_by = CRM
	}
	1112 = {
		remove_core_of = CRM
		remove_claim_by = CRM
	}
}

UKR_crimean_bop_release_tatar_ukr_subject = {
	UKR_crimean_bop_stage_crm_release = yes
	# UKR still owns 669 at this stage. Engine's release_puppet only does anything when CRM doesn't exist.
	if = { limit = { NOT = { country_exists = CRM } 669 = { is_owned_by = UKR } }
		release_puppet = CRM
	}
	if = { limit = { country_exists = CRM }
		669 = { transfer_state_to = CRM }
		puppet = CRM
		CRM = {
			set_cosmetic_tag = CRM_TATAR
			set_country_flag = CRM_crimean_tatar_government_flag
			mark_focus_tree_layout_dirty = yes
		}
		669 = {
			add_core_of = CRM
			remove_core_of = UKR
			add_claim_by = UKR
			remove_claim_by = SOV
		}
		1112 = {
			add_claim_by = CRM
			# Ukraine retains a territorial claim where Russia retains its naval base.
			if = { limit = { is_owned_by = SOV } add_claim_by = UKR }
			if = { limit = { is_owned_by = UKR } add_core_of = UKR remove_claim_by = UKR }
		}
	}
}

# Read-only, cached *indicative* forecast based on currently active missions.
# Uses the same 65% caps and Ukrainian 45% coalition rule as the source system.
# Completion order is not guaranteed and conditional missions may fail: never portray as certain.
UKR_crimean_bop_forecast_transfer_influence_from_russia_to_ukraine = {
	set_temp_variable = { crimea_forecast_applied_influence_change = crimea_forecast_expected_influence_change }
	set_temp_variable = { crimea_forecast_available_influence_gain = 65 }
	if = {
		limit = {
			OR = {
				check_variable = { var = UKR_crimean_bop_forecast_crimean_tatar_influence value = 25 compare = less_than }
				UKR_crimean_bop_historical_ai_before_2014 = yes
			}
		}
		set_temp_variable = { crimea_forecast_available_influence_gain = 45 }
	}
	subtract_from_temp_variable = { crimea_forecast_available_influence_gain = UKR_crimean_bop_forecast_ukrainian_influence }
	if = { limit = { check_variable = { var = crimea_forecast_applied_influence_change value = crimea_forecast_available_influence_gain compare = greater_than } } set_temp_variable = { crimea_forecast_applied_influence_change = crimea_forecast_available_influence_gain } }
	if = { limit = { check_variable = { var = crimea_forecast_applied_influence_change value = UKR_crimean_bop_forecast_russian_influence compare = greater_than } } set_temp_variable = { crimea_forecast_applied_influence_change = UKR_crimean_bop_forecast_russian_influence } }
	if = { limit = { check_variable = { var = crimea_forecast_applied_influence_change value = 0 compare = greater_than } }
		add_to_variable = { UKR_crimean_bop_forecast_ukrainian_influence = crimea_forecast_applied_influence_change }
		subtract_from_variable = { UKR_crimean_bop_forecast_russian_influence = crimea_forecast_applied_influence_change }
	}
}

# Effect: UKR_crimean_bop_limit_historical_tatar_forecast
# Purpose: Cap Tatar influence from a historical AI sponsor before 2014.
# Parameters:
# - crimea_tatar_influence_sponsor: Country ID of the government that started the program.
# - crimea_forecast_expected_influence_change: Requested Tatar influence gain, reduced in place.
UKR_crimean_bop_limit_historical_tatar_forecast = {
	if = {
		limit = { var:crimea_tatar_influence_sponsor = { UKR_crimean_bop_historical_ai_before_2014 = yes } }
		set_temp_variable = { crimea_historical_tatar_influence_room = 15 }
		subtract_from_temp_variable = { crimea_historical_tatar_influence_room = UKR_crimean_bop_forecast_crimean_tatar_influence }
		clamp_temp_variable = { var = crimea_historical_tatar_influence_room min = 0 }
		clamp_temp_variable = { var = crimea_forecast_expected_influence_change max = crimea_historical_tatar_influence_room }
	}
}
