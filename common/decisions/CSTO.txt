CSTO_management = {
	SOV_convert_csto_to_offensive = {
		icon = GFX_decision_faction
		fire_only_once = yes
		cost = 100
		days_remove = 60

		visible = {
			NOT = { has_global_flag = CSTO_member_offensive }
		}

		available = {
			has_war = no
			num_faction_members > 1
		}

		remove_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision SOV_convert_csto_to_offensive"
			set_global_flag = CSTO_member_offensive
			if = {
				limit = { has_dlc = "No Compromise, No Surrender" }
				set_faction_rule = call_to_war_rule_independent_only
			}
			newline = yes
			news_event = { id = CSTO.6 hours = 6 }
			newline = yes
			add_named_threat = {
				threat = 2
				name = SOV_convert_csto_to_offensive_threat
			}
		}

		ai_will_do = {
			base = 0
			modifier = {
				add = 10
				threat > 0.25
				has_war_support > 0.5
				any_neighbor_country = {
					NOT = { is_in_faction_with = ROOT }
					has_opinion = {
						target = ROOT
						value < -50
					}
				}
			}
			modifier = {
				add = 5
				threat > 0.75
				any_of_scopes = {
					array = global.nato_members
					is_neighbor_of = ROOT
				}
			}
			modifier = {
				factor = 0.05
				any_of_scopes = {
					array = global.nato_members
					is_faction_leader = yes
					strength_ratio = {
						tag = ROOT
						ratio > 1.05
					}
				}
			}
			modifier = {
				factor = 0
				SOV_ai_historical_path = yes
			}
		}
	}
}
