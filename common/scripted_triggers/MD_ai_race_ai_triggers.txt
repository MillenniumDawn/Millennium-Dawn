ai_race_ai_candidate_eligible = {
	is_ai = yes
	exists = yes
	NOT = { has_country_flag = disabled_economic_system }
	NOT = { has_country_flag = collapsed_nation }
	check_variable = { gdp_total > 0 }
	amount_research_slots > 0
	check_variable = { var = ai_race_ai_candidate_civs value = 10 compare = greater_than_or_equals }
}

ai_race_ai_country_operable = {
	ai_race_full_mode = yes
	is_ai = yes
	exists = yes
	date > 2012.12.31
	NOT = { has_country_flag = disabled_economic_system }
	has_variable = ai_race_ai_target_stage
	NOT = { has_country_flag = collapsed_nation }
	check_variable = { gdp_total > 0 }
	OR = {
		check_variable = { ai_race_stage > 0 }
		AND = {
			amount_research_slots > 0
			any_controlled_state = { always = yes }
		}
	}
}

# These inputs support domestic production; refineries never satisfy the race's oil requirement.
ai_race_ai_rubber_feedstock_needed = {
	ai_race_ai_support_permitted = yes
	check_variable = { ai_race_ai_composites_ratio < 1.1 }
	OR = {
		check_variable = { ai_race_ai_composites_ratio < 1 }
		check_variable = { ai_race_ai_composites_latched = 1 }
	}
	OR = {
		AND = {
			has_tech = composite_production_1
			check_variable = { ai_race_ai_slots_composites > 0 }
			can_staff_an_composite_plant = yes
			check_variable = { resource@rubber < 1 }
			check_variable = { var = resource@chromium value = 1 compare = greater_than_or_equals }
			check_variable = { var = resource@oil value = 1 compare = greater_than_or_equals }
		}
		AND = {
			NOT = { has_tech = composite_production_1 }
			NOT = { is_special_project_completed = sp:sp_composite_production }
			check_variable = { resource@rubber < 9 }
			check_variable = { var = resource@chromium value = 9 compare = greater_than_or_equals }
		}
	}
}
