scripted_map_modes = {
	influence_map_mode = {
		bottom = {
			type = none
		}
		top = {
			type = country
			color = {
				if = { limit = { check_variable = { FROM.influence_array^0 = ROOT.id } }
					set_temp_variable = { red = 0.776 }
					set_temp_variable = { green = 0.959 }
					set_temp_variable = { blue = 0.883 }
				}
				else_if = {
					limit = {
						OR = {
							check_variable = { FROM.influence_array^1 = ROOT.id }
							check_variable = { FROM.influence_array^2 = ROOT.id }
						}
					}
					set_temp_variable = { red = 0.029 }
					set_temp_variable = { green = 0.698 }
					set_temp_variable = { blue = 0.838 }
				}
				else_if = {
					limit = {
						OR = {
							check_variable = { FROM.influence_array^3 = ROOT.id }
							check_variable = { FROM.influence_array^4 = ROOT.id }
							check_variable = { FROM.influence_array^5 = ROOT.id }
							check_variable = { FROM.influence_array^6 = ROOT.id }
						}
					}
					set_temp_variable = { red = 0.134 }
					set_temp_variable = { green = 0.388 }
					set_temp_variable = { blue = 1.000 }
				}
				else_if = {
					limit = { tag = FROM }
					set_temp_variable = { red = 0.924 }
					set_temp_variable = { green = 0.867 }
					set_temp_variable = { blue = 0.117 }
				}
				else = {
					set_temp_variable = { red = 0.25 }
					set_temp_variable = { green = 0.25 }
					set_temp_variable = { blue = 0.25 }
				}

				if = { limit = { ROOT = { has_country_flag = spread_influence@FROM } }
					set_temp_variable = { red = 0.839 }
					set_temp_variable = { green = 0.024 }
					set_temp_variable = { blue = 0.077 }
				}

				if = { limit = { ROOT = { is_in_array = { auto_influence_array = FROM.id } } }
					set_temp_variable = { red = 0.747 }
					set_temp_variable = { green = 0.097 }
					set_temp_variable = { blue = 0.544 }
				}


				set_temp_variable = { alpha = 1.0 }
			}
		}

		far_text = country
		near_text = none

		update_daily = no
	}
}
