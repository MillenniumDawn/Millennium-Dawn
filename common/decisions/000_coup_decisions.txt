# COUP PROGRESS DECISIONS
coup_progress_category = {
	# MAIN MISSION
		# this is a rolling ticker that displays the current timeframe until the plan is carried out, it will be possible to carry out the plan immediately (with event to double check) or a decision that prolongs the mission with the chance of failure increasing each time (longer the plans, the more of a paper trail)
		coup_planning_mission = {
			icon = revolt
			fire_only_once = no
			days_remove = 1

			available = {
				OR = {
					is_debug = yes
					has_country_flag = coup_initiated
				}
			}

			is_good = yes
			selectable_mission = yes
			days_mission_timeout = 180

			complete_effect = {
				add_political_power = 100
			}

			timeout_effect = {
				clr_country_flag = coup_initiated
				add_political_power = 150
			}
		}
}
