# Effect: update_focus_tree_obsolete_branches
# Purpose: Updates the focus tree to hide branches that are no longer available via the game for optimizing the Focus Tree windows.
update_focus_tree_obsolete_branches = {
	hidden_effect = {
		if = {
			limit = {
				has_game_rule = {
					rule = obsolete_focus_branches_visibility
					option = HIDE
				}
			}
			mark_focus_tree_layout_dirty = yes
		}
	}
}

# Effect: diplomatic_request_display
# Purpose: Replaces need for custom_effect_tooltips for each "if they accept" and other related tooltips.
# Conditionals:
#	- diplomatic_target_generic: specifies if the effect is for a collective "they", defaults to no, needs to be boolean no if set yes earlier in code block
#	- diplomatic_target_multi: specifies if the effect is for multiple tags, defaults to no, needs to be boolean no if set yes earlier in code block
#	- diplomatic_target_self: specifies if the effect is for self, defaults to no, needs to be boolean no if set yes earlier in code block
#	- diplomatic_reply_accept: specifies if the effect is shown for accepting a proposition, defaults to no, needs to be boolean no if set yes earlier or if decline is used in code block
#	- diplomatic_reply_decline: specifies if the effect is shown for declining a proposition, defaults to no, needs to be boolean no if set yes earlier or if accept is used in code block
diplomatic_request_display = {
	custom_effect_tooltip = diplomatic_request_display_TT
}
