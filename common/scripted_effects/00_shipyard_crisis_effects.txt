# Country scope. Needs the setup variables of the TAG_shipyard_setup effect.
# Refreshes cost, drift and modifier variables from shipyard_crisis_level.
shipyard_recalc = {
	set_temp_variable = { shipyard_years_late = 0 }
	if = {
		limit = { shipyard_is_late_phase = yes }
		set_temp_variable = { shipyard_years_late = global.year }
		subtract_from_temp_variable = { shipyard_years_late = shipyard_late_year }
		clamp_temp_variable = { var = shipyard_years_late min = 0 max = 10 }
		multiply_temp_variable = { shipyard_years_late = 0.15 }
	}
	set_temp_variable = { shipyard_severity = shipyard_crisis_level }
	divide_temp_variable = { shipyard_severity = 100 }
	set_temp_variable = { shipyard_level_part = shipyard_severity }
	multiply_temp_variable = { shipyard_level_part = 0.5 }

	set_variable = { shipyard_cost_mult = 1 }
	add_to_variable = { shipyard_cost_mult = shipyard_years_late }
	add_to_variable = { shipyard_cost_mult = shipyard_level_part }
	set_variable = { shipyard_cost_drydock = shipyard_cost_base }
	multiply_variable = { shipyard_cost_drydock = shipyard_cost_mult }

	if = {
		limit = { shipyard_is_late_phase = yes }
		set_variable = { shipyard_drift = shipyard_drift_late }
	}
	else = {
		set_variable = { shipyard_drift = shipyard_drift_early }
	}
	set_temp_variable = { shipyard_relief = 1 }
	set_temp_variable = { shipyard_relief_part = shipyard_drydock_stage }
	multiply_temp_variable = { shipyard_relief_part = 0.1 }
	subtract_from_temp_variable = { shipyard_relief = shipyard_relief_part }
	set_temp_variable = { shipyard_relief_part = shipyard_workforce_stage }
	multiply_temp_variable = { shipyard_relief_part = 0.15 }
	subtract_from_temp_variable = { shipyard_relief = shipyard_relief_part }
	set_temp_variable = { shipyard_relief_part = shipyard_consolidation_stage }
	multiply_temp_variable = { shipyard_relief_part = 0.1 }
	subtract_from_temp_variable = { shipyard_relief = shipyard_relief_part }
	clamp_temp_variable = { var = shipyard_relief min = 0 max = 1 }
	if = {
		limit = { shipyard_is_recovering = yes }
		multiply_temp_variable = { shipyard_relief = -1 }
		add_to_temp_variable = { shipyard_relief = 2 }
	}
	multiply_variable = { shipyard_drift = shipyard_relief }
	if = {
		limit = {
			shipyard_is_recovering = yes
			check_variable = { shipyard_crisis_level = 0 }
		}
		set_variable = { shipyard_drift = 0 }
	}

	set_variable = { shipyard_mod_output = shipyard_cap_output }
	multiply_variable = { shipyard_mod_output = shipyard_severity }
	multiply_variable = { shipyard_mod_output = -1 }
	if = {
		limit = {
			shipyard_is_recovering = yes
			check_variable = { shipyard_growth_start > 0 }
		}
		set_variable = { shipyard_growth_output = shipyard_growth_start }
		subtract_from_variable = { shipyard_growth_output = shipyard_crisis_level }
		clamp_variable = { var = shipyard_growth_output min = 0 max = 100 }
		divide_variable = { shipyard_growth_output = shipyard_growth_start }
		multiply_variable = { shipyard_growth_output = shipyard_growth_cap }
		add_to_variable = { shipyard_mod_output = shipyard_growth_output }
	}
	set_variable = { shipyard_mod_repair = shipyard_cap_repair }
	multiply_variable = { shipyard_mod_repair = shipyard_severity }
	multiply_variable = { shipyard_mod_repair = -1 }
	set_variable = { shipyard_mod_refit = shipyard_cap_refit }
	multiply_variable = { shipyard_mod_refit = shipyard_severity }
	multiply_variable = { shipyard_mod_refit = -1 }
	set_variable = { shipyard_mod_refit_cost = shipyard_cap_refit }
	multiply_variable = { shipyard_mod_refit_cost = shipyard_severity }

	# Dockyards on a production line or a repair job. The engine exposes the free ones only.
	set_variable = { shipyard_dockyards = num_of_naval_factories }
	set_variable = { shipyard_dockyards_in_use = num_of_naval_factories }
	subtract_from_variable = { shipyard_dockyards_in_use = num_of_available_naval_factories }

	set_variable = { shipyard_capital_yard = 0 }
	if = {
		limit = {
			any_owned_state = {
				is_controlled_by = PREV
				non_damaged_building_level = { building = large_slipway level > 0 }
			}
		}
		set_variable = { shipyard_capital_yard = 1 }
	}

	set_variable = { shipyard_submarine_yard = 0 }
	if = {
		limit = {
			any_owned_state = {
				is_controlled_by = PREV
				non_damaged_building_level = { building = submarine_hall level > 0 }
			}
		}
		set_variable = { shipyard_submarine_yard = 1 }
	}

	shipyard_country_recalc = yes
	force_update_dynamic_modifier = yes
	shipyard_sync_equipment = yes
	shipyard_sync_yard_locks = yes
	shipyard_update_dirty_variable = yes
}

shipyard_monthly = {
	add_to_variable = { shipyard_crisis_level = shipyard_drift }
	clamp_variable = { var = shipyard_crisis_level min = 0 max = 100 }
	shipyard_recalc = yes

	if = {
		limit = { is_ai = no }
		shipyard_overload_check = yes
	}
}

# Country scope. Takes the signed point change in the temp variable shipyard_crisis_change.
shipyard_change_crisis = {
	if = {
		limit = { has_dynamic_modifier = { modifier = shipyard_crisis_modifier } }
		custom_effect_tooltip = shipyard_crisis_change_tt
		hidden_effect = {
			add_to_variable = { shipyard_crisis_level = shipyard_crisis_change }
			clamp_variable = { var = shipyard_crisis_level min = 0 max = 100 }
			shipyard_recalc = yes
		}
	}
}

# State scope entered from the country that will own and control the facility (PREV).
shipyard_build_large_slipway = {
	if = {
		limit = {
			is_owned_and_controlled_by = PREV
			is_coastal = yes
			check_variable = { building_level@large_slipway = 0 }
		}
		if = {
			limit = { NOT = { free_building_slots = { building = large_slipway size > 0 include_locked = no } } }
			add_extra_state_shared_building_slots = 1
		}
		add_building_construction = { type = large_slipway level = 1 instant_build = yes }
	}
}

# Warn once, then penalize the player each month that dockyards in use exceed capacity.
shipyard_overload_check = {
	if = {
		limit = { check_variable = { shipyard_dockyards_in_use > shipyard_effective_dockyards } }
		add_to_variable = { shipyard_overload_months = 1 }
		if = {
			limit = { check_variable = { shipyard_overload_months = 1 } }
			country_event = { id = md_shipyard.1 days = 1 }
		}
		else_if = {
			limit = { NOT = { has_country_flag = shipyard_overload_grace_flag } }
			add_to_variable = { shipyard_crisis_level = 1 }
			clamp_variable = { var = shipyard_crisis_level min = 0 max = 100 }
			add_stability = -0.01
			every_owned_state = {
				limit = {
					check_variable = { building_level@dockyard > 0 }
					NOT = { has_dynamic_modifier = { modifier = shipyard_overload_modifier } }
				}
				add_dynamic_modifier = { modifier = shipyard_overload_modifier days = 60 }
			}
			shipyard_recalc = yes
		}
	}
	else = {
		set_variable = { shipyard_overload_months = 0 }
	}
}

# Default for a country whose dockyards are all usable: the backlog takes up to half of
# them out of productive work at a full crisis.
shipyard_effective_from_crisis = {
	set_variable = { shipyard_effective_dockyards = shipyard_crisis_level }
	divide_variable = { shipyard_effective_dockyards = 100 }
	multiply_variable = { shipyard_effective_dockyards = -0.5 }
	add_to_variable = { shipyard_effective_dockyards = 1 }
	multiply_variable = { shipyard_effective_dockyards = shipyard_dockyards }
	set_temp_variable = { shipyard_eff_fraction = shipyard_effective_dockyards }
	modulo_temp_variable = { var = shipyard_eff_fraction value = 1 }
	subtract_from_variable = { shipyard_effective_dockyards = shipyard_eff_fraction }
	clamp_variable = { var = shipyard_effective_dockyards min = 1 max = 1000 }
}

# add_equipment_bonus cannot be removed and cannot read variables, so the penalty moves in
# whole steps of 10 crisis points and is reverted with a negative step. The step count is
# floored, so a starting level below 10 costs the player nothing.
shipyard_sync_equipment = {
	set_temp_variable = { shipyard_eq_target = shipyard_crisis_level }
	divide_temp_variable = { shipyard_eq_target = 10 }
	set_temp_variable = { shipyard_eq_fraction = shipyard_eq_target }
	modulo_temp_variable = { var = shipyard_eq_fraction value = 1 }
	subtract_from_temp_variable = { shipyard_eq_target = shipyard_eq_fraction }
	while_loop_effect = {
		limit = { check_variable = { shipyard_eq_steps < shipyard_eq_target } }
		shipyard_equipment_step_up = yes
		add_to_variable = { shipyard_eq_steps = 1 }
	}
	while_loop_effect = {
		limit = { check_variable = { shipyard_eq_steps > shipyard_eq_target } }
		shipyard_equipment_step_down = yes
		subtract_from_variable = { shipyard_eq_steps = 1 }
	}
}

# A country with no large slipway can still lay a capital ship down, at a price nobody
# pays twice. The same for submarines without a submarine hall. add_equipment_bonus cannot
# be removed, so each lock is toggled with its own negative.
# Carrier bonuses also reach helicopter operators (type = carrier), so their entries hold only the difference.
shipyard_sync_yard_locks = {
	if = {
		limit = {
			check_variable = { shipyard_capital_yard = 0 }
			check_variable = { shipyard_capital_locked = 0 }
		}
		shipyard_capital_lock_on = yes
		set_variable = { shipyard_capital_locked = 1 }
	}
	else_if = {
		limit = {
			check_variable = { shipyard_capital_yard = 1 }
			check_variable = { shipyard_capital_locked = 1 }
		}
		shipyard_capital_lock_off = yes
		set_variable = { shipyard_capital_locked = 0 }
	}
	if = {
		limit = {
			check_variable = { shipyard_submarine_yard = 0 }
			check_variable = { shipyard_submarine_locked = 0 }
		}
		shipyard_submarine_lock_on = yes
		set_variable = { shipyard_submarine_locked = 1 }
	}
	else_if = {
		limit = {
			check_variable = { shipyard_submarine_yard = 1 }
			check_variable = { shipyard_submarine_locked = 1 }
		}
		shipyard_submarine_lock_off = yes
		set_variable = { shipyard_submarine_locked = 0 }
	}
}
