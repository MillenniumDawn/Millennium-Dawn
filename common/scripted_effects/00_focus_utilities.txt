# Effect: update_focus_tree_obsolete_branches
# Purpose: Updates the focus tree to hide branches that are no longer available via the game for optimizing the Focus Tree windows.
update_focus_tree_obsolete_branches = {
	hidden_effect = {
		if = {
			limit = {
				has_game_rule = {
					rule = obsolete_focus_branches_visibility
					option = HIDE
				}
			}
			mark_focus_tree_layout_dirty = yes
		}
	}
}

# Effect: diplomatic_request_display
# Purpose: Replaces need for custom_effect_tooltips for each "if they accept" and other related tooltips.
diplomatic_request_display = {
	# TARGETING
	if = {
		limit = { is_targeting_self_diplo = no }
		if = {
			limit = { is_targeting_multi_diplo = yes }
			custom_effect_tooltip = diplomatic_request_multiple_display_TT
		}
		else = {
			custom_override_tooltip = {
				tooltip = {
					localization_key = diplomatic_request_display_TT
					TAG = [?var:diplo_target.GetNameWithFlag]
				}
			}
		}
	}
	# SELF
	else = {
		custom_effect_tooltip = diplomatic_request_self_display_TT
	}
}
