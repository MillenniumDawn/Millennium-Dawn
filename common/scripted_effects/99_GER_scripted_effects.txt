# BAVARIA MECHANIC #
GER_update_bavaria_standing = {
	if = {
		limit = {
			OR = {
				has_idea = stagnation
				has_idea = recession
				has_idea = depression
			}
		}
		random = {
			chance = 40
			add_to_variable = { GER_bavarian_nationalism_var = 1 }
		}
	}
	if = {
		limit = {
			OR = {
				has_idea = fast_growth
				has_idea = economic_boom
			}
			check_variable = { GER_bavarian_nationalism_var > 0 }
		}
		random = {
			chance = 40
			add_to_variable = { GER_bavarian_nationalism_var = -1 }
		}
	}
}

GER_update_dirty_variable = {
	if = {
		limit = { is_ai = no }
		if = {
			limit = { NOT = { has_variable = global.update_ger_ui } }
			set_variable = { global.update_ger_ui = 0 }
		}
		add_to_variable = { global.update_ger_ui = 1 }
	}
}

GER_government_change = {
	if = {
		limit = {
			original_tag = GER
			has_cosmetic_tag = GER_franco_german_union
			emerging_anarchist_communism_are_in_power = no
		}
		drop_cosmetic_tag = yes
		clear_variable = dynamic_flag_variant
	}
	if = {
		limit = {
			original_tag = GER
			has_cosmetic_tag = GER_fourth_reich
			nationalist_fascist_are_in_power = no
		}
		drop_cosmetic_tag = yes
		clear_variable = dynamic_flag_variant
	}
	if = {
		limit = {
			original_tag = GER
			OR = {
				has_cosmetic_tag = GER_german_empire
				has_cosmetic_tag = GER_holy_german_empire
			}
			nationalist_monarchists_are_in_power = no
		}
		drop_cosmetic_tag = yes
		clear_variable = dynamic_flag_variant
	}
}

# Siemens corporate history. Five milestones set the route; the closing
# milestone resolves it into one of four shapes for the group.

GER_siemens_apply_focused_industrials = {
	if = {
		limit = { GER_siemens_outcome_settled = no }
		add_timed_idea = { idea = GER_siemens_focused_industrials days = 365 }
	}
}

# One entry point per milestone, called from the yearly pulse. Full presents the
# choice; Outcomes Only writes the historical route silently and grants nothing.
GER_siemens_milestone_2001 = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		country_event = { id = germany_yearly.300 days = 119 }
	}
	else_if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
		set_country_flag = GER_siemens_integrated_conglomerate
	}
}
