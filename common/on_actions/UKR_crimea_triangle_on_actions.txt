on_actions = {
	on_startup = {
		effect = {
			if = { limit = { country_exists = UKR } UKR = { UKR_crimean_bop_initialize = yes } }
			if = { limit = { country_exists = UZB } UZB = { UKR_crimean_bop_update_repatriation_diplomatic_influence = yes } }
			if = { limit = { country_exists = KAZ } KAZ = { UKR_crimean_bop_update_repatriation_diplomatic_influence = yes } }
			if = { limit = { country_exists = KYR } KYR = { UKR_crimean_bop_update_repatriation_diplomatic_influence = yes } }
			if = { limit = { country_exists = TAJ } TAJ = { UKR_crimean_bop_update_repatriation_diplomatic_influence = yes } }
			if = { limit = { country_exists = TRK } TRK = { UKR_crimean_bop_update_repatriation_diplomatic_influence = yes } }
		}
	}

	on_daily_UKR = {
		effect = {
			if = {
				limit = { has_country_flag = UKR_crimean_bop_initialized_flag }
				UKR_crimean_bop_update_influence_forecast = yes
				UKR_crimean_bop_update_marker_for_viewers = yes
			}
		}
	}

	on_daily_UZB = {
		effect = { UKR_crimean_bop_update_repatriation_diplomatic_influence = yes }
	}

	on_daily_KAZ = {
		effect = { UKR_crimean_bop_update_repatriation_diplomatic_influence = yes }
	}

	on_daily_KYR = {
		effect = { UKR_crimean_bop_update_repatriation_diplomatic_influence = yes }
	}

	on_daily_TAJ = {
		effect = { UKR_crimean_bop_update_repatriation_diplomatic_influence = yes }
	}

	on_daily_TRK = {
		effect = { UKR_crimean_bop_update_repatriation_diplomatic_influence = yes }
	}

	on_monthly_UKR = {
		effect = { UKR_crimean_bop_update_monthly = yes }
	}
}
