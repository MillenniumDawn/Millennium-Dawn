# --- Post-capture: PAK vs RAJ ---
# Fires on the side that lost a state to a grab. Player picks the response:
# accept, escalate, or negotiate return.
country_event = {
	id = border_war_post_capture.1
	title = border_war_post_capture.1.t
	desc = border_war_post_capture.1.desc
	picture = GFX_Military_Reform
	is_triggered_only = yes

	option = {
		name = border_war_post_capture.1.a
		log = "[GetDateText]: [This.GetName]: border_war_post_capture.1.a executed"
		add_war_support = 0.10
		add_stability = -0.10
		hidden_effect = {
			clr_country_flag = border_war_lost_state_recent
			clear_variable = border_war_lost_state
		}
		ai_chance = {
			base = 5
			modifier = {
				factor = 0
				strength_ratio = { tag = FROM ratio < 0.8 }
			}
		}
	}

	option = {
		name = border_war_post_capture.1.b
		log = "[GetDateText]: [This.GetName]: border_war_post_capture.1.b executed"
		if = {
			limit = {
				NOT = { has_global_flag = border_war_crisis_PAK_RAJ }
			}
			set_global_flag = border_war_crisis_PAK_RAJ
			set_country_flag = border_war_crisis_escalator
			country_event = { id = border_war_crisis.1 days = 1 }
		}
		ai_chance = {
			base = 3
			modifier = {
				factor = 0
				has_global_flag = border_war_crisis_PAK_RAJ
			}
			modifier = {
				factor = 0
				strength_ratio = { tag = FROM ratio < 1.5 }
			}
		}
	}

	option = {
		name = border_war_post_capture.1.c
		log = "[GetDateText]: [This.GetName]: border_war_post_capture.1.c executed"
		add_political_power = 75
		add_stability = 0.03
		FROM = { add_political_power = -50 }
		ai_chance = {
			base = 2
			modifier = {
				factor = 3
				has_stability < 0.40
			}
		}
	}

	option = {
		name = border_war_post_capture.1.d
		log = "[GetDateText]: [This.GetName]: border_war_post_capture.1.d executed"
		add_timed_idea = {
			idea = border_war_fortification_idea
			days = 365
		}
		ai_chance = {
			base = 2
			modifier = {
				factor = 0
				has_idea = border_war_fortification_idea
			}
		}
	}
}

# 1: fires to the escalator (lost a state, or leads on tension+progress).
country_event = {
	id = border_war_crisis.1
	title = border_war_crisis.1.t
	desc = border_war_crisis.1.d
	picture = GFX_Military_Reform
	is_triggered_only = yes

	option = {
		name = border_war_crisis.1.a
		log = "[GetDateText]: [This.GetName]: border_war_crisis.1.a executed"
		set_temp_variable = { border_war_add_tension_amount = 8 }
		border_war_add_tension = yes
		if = {
			limit = { original_tag = RAJ }
			PAK = { country_event = { id = border_war_crisis.2 days = 1 } }
		}
		else_if = {
			limit = { original_tag = PAK }
			RAJ = { country_event = { id = border_war_crisis.2 days = 1 } }
		}
		custom_effect_tooltip = TT_IF_THEY_ACCEPT
		effect_tooltip = { subtract_from_variable = { global.border_tension = 10 } }
		custom_effect_tooltip = TT_IF_THEY_REJECT
		ai_chance = {
			base = 5
			modifier = {
				factor = 2
				has_global_flag = border_war_ai_aggressive
			}
			modifier = {
				factor = 0.3
				is_historical_focus_on = yes
				NOT = { has_global_flag = border_war_ai_aggressive }
			}
			modifier = {
				factor = 0.3
				has_global_flag = border_war_nuclear_standoff
			}
		}
	}

	option = {
		name = border_war_crisis.1.b
		log = "[GetDateText]: [This.GetName]: border_war_crisis.1.b executed"
		add_stability = 0.03
		hidden_effect = {
			border_war_crisis_resolve_pak_raj = yes
			set_global_flag = border_war_crisis_resolved_peacefully
			if = {
				limit = { original_tag = RAJ }
				PAK = { country_event = { id = border_war_crisis.9 days = 1 } }
			}
			else_if = {
				limit = { original_tag = PAK }
				RAJ = { country_event = { id = border_war_crisis.9 days = 1 } }
			}
		}
		ai_chance = {
			base = 3
			modifier = {
				factor = 3
				has_stability < 0.40
			}
			modifier = {
				factor = 2
				is_historical_focus_on = yes
				NOT = { has_global_flag = border_war_ai_aggressive }
			}
			modifier = {
				factor = 0.2
				has_global_flag = border_war_ai_aggressive
			}
		}
	}
}
