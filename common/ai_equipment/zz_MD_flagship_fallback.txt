# Universal fallback flagship. The invasion reach budget needs one capital_ship hull,
# but every authored cruiser design gates on module techs its nation may never research
# (PER's Moudge needs VLS; the generic cruiser needs tech_vls_surface_3 and is
# blocked_for every authored-navy nation, with a 10-dockyard kill switch besides).
# This design is buildable the moment cruiser_hull_1 is: engine only — every other
# cruiser slot is optional, and tech_naval_systems_engineering (module_surface_oil_power)
# is in all 393 country history files. High priority ONLY while the nation owns no
# capital ship, so authored designs take over for every hull after the first.
MD_flagship_fallback = {
	category = naval
	roles = { naval_cruiser }
	priority = {
		base = -1
		modifier = { add = 500 has_no_capital_ships = yes }
	}

	MD_fallback_flagship = {
		role_icon_index = 2

		priority = {
			base = -1
			modifier = { add = 900 has_tech = cruiser_hull_1 }
		}

		target_variant = {
			type = cruiser_hull_1
			match_value = 200

			modules = {
				fixed_ship_engine_slot = module_surface_oil_power
			}
		}
	}
}
