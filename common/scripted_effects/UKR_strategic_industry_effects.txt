# Ukrainian strategic industry. Five milestones set the route; the wartime
# milestone resolves it into one of four industrial futures.

UKR_strategic_industry_apply_western_integration = {
	if = {
		limit = { UKR_strategic_industry_outcome_settled = no }
		add_ideas = UKR_strategic_industry_western_integrated_defense_industry
	}
}

	}
}

UKR_strategic_industry_apply_chinese_corridor = {

	}
}

# One entry point per milestone, called from the yearly pulse. Full presents the
# choice; Outcomes Only writes the historical route silently and grants nothing.
UKR_strategic_industry_milestone_2001 = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		country_event = { id = ukraine_tech.300 days = 78 }
	}
	else_if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
		set_country_flag = UKR_strategic_industry_state_retained
	}
}
