@MD_ftc_upgrade_cost_2 = 650

@MD_ftc_min_convoys = 100
@MD_ftc_command_power_cost = 75

MD_ftc_can_start_exercise_capital = {
	has_navy_experience > @MD_ftc_exercise_cost
	command_power > @MD_ftc_command_power_cost
	NOT = { has_idea = MD_ftc_exercise_capital }
	NOT = { has_navy_size = { size < @MD_ftc_fleet_min_ships } }
	OR = {
		has_navy_size = { unit = carrier size > 0 }
		has_navy_size = { unit = helicopter_operator size > 0 }
		has_navy_size = { unit = battleship size > 0 }
		has_navy_size = { unit = battle_cruiser size > 0 }
		has_navy_size = { unit = cruiser size > 0 }
		has_navy_size = { unit = destroyer size > 0 }
		has_navy_size = { unit = stealth_destroyer size > 0 }
		has_navy_size = { unit = heavy_frigate size > 0 }
	}
}

MD_ftc_can_upgrade_academy = {
	OR = {
		AND = {
			check_variable = { MD_ftc_academy_level < 1 }
			has_navy_experience > @MD_ftc_upgrade_cost_1
		}
		AND = {
			check_variable = { MD_ftc_academy_level = 1 }
			has_navy_experience > @MD_ftc_upgrade_cost_2
		}
		AND = {
			check_variable = { MD_ftc_academy_level = 2 }
			has_navy_experience > @MD_ftc_upgrade_cost_3
		}
	}
}

MD_ftc_doctrine_changed = {
	OR = {
		AND = {
			has_country_flag = MD_ftc_doctrine_blue_water
			NOT = { has_doctrine = blue_water }
		}
		AND = {
			has_country_flag = MD_ftc_doctrine_green_water
			NOT = { has_doctrine = green_water }
		}
		AND = {
			has_country_flag = MD_ftc_doctrine_jeune_ecole
			NOT = { has_doctrine = jeune_ecole }
		}
		AND = {
			has_country_flag = MD_ftc_doctrine_guerre_de_course
			NOT = { has_doctrine = guerre_de_course }
		}
	}
}
