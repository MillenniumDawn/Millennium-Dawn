# Exports push the world artifact price down and purchases push it up, within fixed bounds.
STALKER_sell_artifacts_abroad = {
	add_to_variable = { STALKER_artifact_stockpile = -5 }
	add_to_variable = { global.STALKER_market_artifacts = 5 }
	set_temp_variable = { treasury_change = { value = global.STALKER_artifact_price multiply = 5 } }
	modify_treasury_effect = yes
	add_to_variable = { global.STALKER_artifact_price = -0.02 }
	clamp_variable = { var = global.STALKER_artifact_price min = 0.1 max = 0.5 }
}

STALKER_purchase_market_artifacts = {
	add_to_variable = { global.STALKER_market_artifacts = -5 }
	set_temp_variable = { treasury_change = { value = global.STALKER_artifact_price multiply = -5 } }
	modify_treasury_effect = yes
	add_dynamic_modifier = { modifier = STALKER_artifact_studies days = 180 }
	add_to_variable = { global.STALKER_artifact_price = 0.02 }
	clamp_variable = { var = global.STALKER_artifact_price min = 0.1 max = 0.5 }
}

# The world price drifts back toward 0.2 once the post-Kaymanov market has set it.
STALKER_monthly_market_pulse = {
	if = {
		limit = {
			check_variable = { global.STALKER_artifact_price > 0 }
			check_variable = { global.STALKER_artifact_price < 0.195 }
		}
		add_to_variable = { global.STALKER_artifact_price = 0.005 }
	}
	else_if = {
		limit = { check_variable = { global.STALKER_artifact_price > 0.205 } }
		add_to_variable = { global.STALKER_artifact_price = -0.005 }
	}
}

STALKER_add_consortium_member = {
	add_to_array = { global.STALKER_consortium_members = THIS }
	add_dynamic_modifier = { modifier = STALKER_consortium_membership }
}
