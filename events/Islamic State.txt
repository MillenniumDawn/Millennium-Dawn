	id = isis.2
	title = isis.2.t
	desc = isis.2.d
	picture = GFX_SYR_tahrir_al_sham
	is_triggered_only = yes
	trigger = {
		original_tag = ISI
		NOT = { has_country_flag = ISI_infiltrated_al_nusra }
		ISI = {
			is_neighbor_of = NUS
			NOT = { has_war_with = NUS }

		}

			surrender_progress < 0.5
		}

		set_country_flag = ISI_infiltrated_al_nusra

		name = isis.2.a
		log = "[GetDateText]: [This.GetName]: isis.2.a executed"
		ai_chance = { base = 100 }
		random_list = {
			70 = {
				ISI = { country_event = { days = 28 id = isis.4 } }
				hidden_effect = {
					NUS = { country_event = { days = 30 id = isis.3 } }
				}
			}
			30 = {
				custom_effect_tooltip = isis.2.a_tt

			}
		}
	}

	# Avoid Provocations
	option = {
		name = isis.2.b
		ai_chance = {
			base = 0
		}
	}
}

		hidden_effect = { news_event = { id = isisNews.1 hours = 6 } }

		ai_chance = {
			base = 100
		}
	}
}
