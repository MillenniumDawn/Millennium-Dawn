# Reads TOP_arg_sequence, TOP_arg_target.
TOP_close_case = {
	if = {
		limit = { check_variable = { TOP_case_phase^TOP_arg_target > 0 } check_variable = { TOP_case_sequence^TOP_arg_target = TOP_arg_sequence } }
		set_temp_variable = { TOP_history_current_event = 8 }
		if = { limit = { check_variable = { TOP_history_current_reason = 0 } check_variable = { TOP_case_result^TOP_arg_target = 0 } check_variable = { global.TOP_status^TOP_arg_target > 1 } } set_temp_variable = { TOP_history_current_reason = 23 } }
		if = { limit = { check_variable = { TOP_history_current_reason = 0 } check_variable = { TOP_case_phase^TOP_arg_target > 3 } } set_temp_variable = { TOP_history_current_reason = 24 } }
		if = { limit = { check_variable = { TOP_history_current_reason = 0 } check_variable = { var = global.TOP_clock value = TOP_case_until^TOP_arg_target compare = greater_than_or_equals } check_variable = { TOP_case_phase^TOP_arg_target < 3 } } set_temp_variable = { TOP_history_current_reason = 22 } }
		if = { limit = { check_variable = { TOP_history_current_reason = 0 } check_variable = { TOP_case_phase^TOP_arg_target = 3 } } set_temp_variable = { TOP_history_current_reason = 23 } }
		TOP_log_person_milestone = yes
		if = {
			limit = { check_variable = { TOP_case_native_prepared^TOP_arg_target = 1 } check_variable = { TOP_case_method^TOP_arg_target > 0 } check_variable = { TOP_case_method^TOP_arg_target < 8 } }
			# Only an unresolved prepared raid can still deliver a stale callback.
			set_temp_variable = { TOP_binding_key = { value = TOP_arg_target multiply = 40000 } }
			add_to_temp_variable = { TOP_binding_key = { value = TOP_case_state^TOP_arg_target multiply = 4 } }
			set_temp_variable = { TOP_binding_family = TOP_case_method^TOP_arg_target }
			if = { limit = { OR = { check_variable = { TOP_binding_family = 4 } check_variable = { TOP_binding_family = 5 } } } set_temp_variable = { TOP_binding_family = 2 } }
			else_if = { limit = { check_variable = { TOP_binding_family = 7 } } set_temp_variable = { TOP_binding_family = 3 } }
			add_to_temp_variable = { TOP_binding_key = TOP_binding_family }
			if = {
				limit = { NOT = { is_in_array = { TOP_retired_native_bindings = TOP_binding_key } } }
				add_to_array = { TOP_retired_native_bindings = TOP_binding_key }
			}
		}
		if = {
			limit = { check_variable = { TOP_package_state^TOP_arg_target = 3 } }
			set_variable = { TOP_package_state^TOP_arg_target = 1 }
		}
		set_variable = { TOP_case_phase^TOP_arg_target = 0 }
		set_variable = { TOP_case_method^TOP_arg_target = 0 }
		set_variable = { TOP_case_due^TOP_arg_target = 0 }
		set_variable = { TOP_case_collecting^TOP_arg_target = 0 }
		set_variable = { TOP_case_native_prepared^TOP_arg_target = 0 }
		set_variable = { TOP_case_native_success_bonus^TOP_arg_target = 0 }
		set_variable = { TOP_case_native_success_penalty^TOP_arg_target = 0 }
		if = {
			limit = {
				check_variable = { TOP_operation_subject_kind = 1 }
				check_variable = { TOP_operation_subject_id = TOP_arg_target }
				check_variable = { TOP_operation_sequence = TOP_case_sequence^TOP_arg_target }
			}
			TOP_trace_slot_end = yes
			set_variable = { TOP_operation_subject_kind = 0 }
			set_variable = { TOP_operation_subject_id = 0 }
			set_variable = { TOP_operation_sequence = 0 }
		}
		set_variable = { TOP_case_visit_token^TOP_arg_target = 0 }
		set_variable = { TOP_case_visit_status^TOP_arg_target = 0 }
		set_variable = { TOP_case_visit_story^TOP_arg_target = 0 }
		remove_from_array = { TOP_active_cases = TOP_arg_target }
		if = {
			limit = { check_variable = { TOP_authorized_target = TOP_arg_target } }
			set_variable = { TOP_authorized_target = 0 }
			set_variable = { TOP_pending_target = 0 }
		}
		TOP_build_view = yes
	}
}

	}

TOP_process_cases = {
	clear_array = TOP_cases_to_close
	for_each_loop = {
		array = TOP_active_cases
		value = top_timer_case
		set_temp_variable = { TOP_target = top_timer_case }
		set_temp_variable = { TOP_role_target = TOP_target }
		if = {
			limit = {
				check_variable = { TOP_case_phase^TOP_target > 0 }
				check_variable = { TOP_case_phase^TOP_target < 3 }
				check_variable = { TOP_case_visit_status^TOP_target > 0 }
				OR = {
					check_variable = { global.TOP_visit_status^TOP_target = 0 }
					NOT = { check_variable = { TOP_case_visit_token^TOP_target = global.TOP_visit_token^TOP_target } }
					NOT = { check_variable = { TOP_case_visit_story^TOP_target = global.TOP_visit_story^TOP_target } }
					NOT = { TOP_authored_role_eligible = yes }
				}
			}
			add_to_array = { TOP_cases_to_close = top_timer_case }
		}
		if = {
			limit = {
				check_variable = { TOP_case_phase^TOP_target > 0 }
				check_variable = { TOP_case_phase^TOP_target < 4 }
				check_variable = { TOP_case_visit_status^TOP_target > 0 }
				check_variable = { TOP_case_visit_token^TOP_target = global.TOP_visit_token^TOP_target }
				check_variable = { TOP_case_visit_story^TOP_target = global.TOP_visit_story^TOP_target }
			}
			set_variable = { TOP_case_until^TOP_target = global.TOP_visit_execution_until^TOP_target }
			if = {
				limit = {
					check_variable = { TOP_case_phase^TOP_target = 3 }
					check_variable = { TOP_case_due^TOP_target < global.TOP_visit_start^TOP_target }
				}
				set_variable = { TOP_case_due^TOP_target = global.TOP_visit_start^TOP_target }
			}
		}
		if = {
			limit = { check_variable = { TOP_case_phase^TOP_target < 4 } check_variable = { global.TOP_status^TOP_target > 1 } }
			if = {
				limit = {
					check_variable = { TOP_operation_subject_kind = 1 }
					check_variable = { TOP_operation_subject_id = TOP_target }
					check_variable = { TOP_operation_sequence = TOP_case_sequence^TOP_target }
				}
				TOP_trace_slot_end = yes
				set_variable = { TOP_operation_subject_kind = 0 }
				set_variable = { TOP_operation_subject_id = 0 }
				set_variable = { TOP_operation_sequence = 0 }
			}
			# The case is no longer actionable, but another actor's physical result remains unknown.
			set_variable = { TOP_assessment^TOP_target = 6 }
			set_variable = { TOP_case_phase^TOP_target = 5 }
		}

		if = {
			limit = {
				OR = {
					AND = { check_variable = { TOP_case_phase^top_timer_case < 3 } check_variable = { var = global.TOP_clock value = TOP_case_until^top_timer_case compare = greater_than_or_equals } }
					AND = { is_ai = yes check_variable = { TOP_case_phase^top_timer_case = 5 } }
				}
			}
			if = { limit = { NOT = { is_in_array = { TOP_cases_to_close = top_timer_case } } } add_to_array = { TOP_cases_to_close = top_timer_case } }
		}
	}
	for_each_loop = {
		array = TOP_cases_to_close
		value = top_timer_case
		set_temp_variable = { TOP_arg_target = top_timer_case } set_temp_variable = { TOP_arg_sequence = TOP_case_sequence^top_timer_case } TOP_close_case = yes
	}
}

		limit = { TOP_visit_can_end = yes }
		set_temp_variable = { TOP_visit_restore_state = 0 }
		set_temp_variable = { TOP_visit_restore_host = global.TOP_visit_return_host^top_visit_target }
		if = {
			limit = {
				check_variable = { global.TOP_status^top_visit_target = 1 }
				check_variable = { global.TOP_host^top_visit_target = global.TOP_visit_host^top_visit_target }
				check_variable = { global.TOP_state^top_visit_target = global.TOP_visit_state^top_visit_target }
				NOT = { check_variable = { global.TOP_visit_return_state^top_visit_target = 0 } }
			}
			var:global.TOP_visit_return_state^top_visit_target = {
				if = {
					limit = { is_controlled_by = var:TOP_visit_restore_host }
					set_temp_variable = { TOP_visit_restore_state = THIS }
				}
			}
		}
		if = {
			limit = {
				check_variable = { global.TOP_status^top_visit_target = 1 }
				check_variable = { global.TOP_host^top_visit_target = global.TOP_visit_host^top_visit_target }

				check_variable = { TOP_visit_restore_host > 0 }
				var:TOP_visit_restore_host = { exists = yes num_of_controlled_states > 0 }
			}
			var:TOP_visit_restore_host = {
				random_controlled_state = { set_temp_variable = { TOP_visit_restore_state = THIS } }

			limit = { NOT = { check_variable = { TOP_visit_restore_state = 0 } } }

			set_variable = { global.TOP_host^top_visit_target = TOP_visit_restore_host }

	clear_array = global.TOP_planned_visits_next

		array = global.TOP_planned_visits

		if = {
			limit = { check_variable = { TOP_visit_revalidated = 1 } }
			add_to_array = { global.TOP_planned_visits_next = top_visit_target }
		}
		else_if = {
			limit = {
				check_variable = { global.TOP_visit_status^top_visit_target = 1 }
				check_variable = { global.TOP_visit_planned_token^top_visit_target = global.TOP_visit_token^top_visit_target }
			}
			set_temp_variable = { TOP_arg_target = top_visit_target }
			TOP_clear_visit = yes
		}
	}
	clear_array = global.TOP_planned_visits
	for_each_loop = {
		array = global.TOP_planned_visits_next
		value = top_visit_target
		add_to_array = { global.TOP_planned_visits = top_visit_target }
	}

	clear_array = global.TOP_active_visits_next
	for_each_loop = {
		array = global.TOP_active_visits
		value = top_visit_target
		set_temp_variable = { TOP_arg_target = top_visit_target }
		TOP_revalidate_active_visit = yes
		if = {
			limit = { check_variable = { TOP_visit_revalidated = 1 } }
			add_to_array = { global.TOP_active_visits_next = top_visit_target }
		}
		else_if = {
			limit = {
				check_variable = { global.TOP_visit_status^top_visit_target = 2 }
				check_variable = { global.TOP_visit_active_token^top_visit_target = global.TOP_visit_token^top_visit_target }

		}
		set_variable = { global.TOP_crisis_victim_coalition_anchor = global.TOP_crisis_protection_faction_leader }
	}
	# -1 favors the victim coalition, 0 is parity, and 1 favors the actor coalition.
	set_variable = { global.TOP_crisis_coalition_balance = 0 }
	if = {
		limit = {
			NOT = { check_variable = { global.TOP_crisis_actor_coalition_anchor = global.TOP_crisis_victim_coalition_anchor } }
			var:global.TOP_crisis_victim_coalition_anchor = {
				var:global.TOP_crisis_actor_coalition_anchor = {
					strength_ratio = { tag = PREV ratio > 1.25 }
				}
			}
		}
		set_variable = { global.TOP_crisis_coalition_balance = 1 }
	}
	else_if = {
		limit = {
			NOT = { check_variable = { global.TOP_crisis_actor_coalition_anchor = global.TOP_crisis_victim_coalition_anchor } }
			var:global.TOP_crisis_victim_coalition_anchor = {
				var:global.TOP_crisis_actor_coalition_anchor = {
					strength_ratio = { tag = PREV ratio < 0.80 }
				}
			}
		}
		set_variable = { global.TOP_crisis_coalition_balance = -1 }
	}
}

# Stories 1 to 5 are dispatched once each from trigger_year_2032_events, so
# before 2032 nothing ever schedules a visit and every target sits still. This
# offers the open story instead, throttled globally rather than per country so
# the rate does not scale with how many hosts are eligible.
TOP_visit_country_opportunities = {
	if = {
		limit = {
			TOP_country_eligible = yes
			check_variable = { global.TOP_visit_open_after < global.TOP_clock }
			# Free again once the held pick has no visit left, so a chain that
			# stalls because the person was taken or killed does not wedge the
			# host out of every later invitation.
			OR = {
				check_variable = { TOP_open_visit_target = 0 }
				check_variable = { global.TOP_visit_status^TOP_open_visit_target = 0 }
			}
		}
		set_temp_variable = { TOP_visit_open_candidate = 0 }
		set_temp_variable = { TOP_visit_open_state = 0 }
		random_owned_state = {
			limit = { is_controlled_by = ROOT }
			set_temp_variable = { TOP_visit_open_state = THIS }
		}
		if = {
			limit = { NOT = { check_variable = { TOP_visit_open_state = 0 } } }
			for_loop_effect = {
				start = 1
				end = global.TOP_registry_capacity
				value = top_visit_person
				if = {
					limit = {
						check_variable = { TOP_visit_open_candidate = 0 }
						set_temp_variable = { TOP_visit_trigger_target = top_visit_person }
						set_temp_variable = { TOP_visit_trigger_state = TOP_visit_open_state }
						set_temp_variable = { TOP_visit_trigger_story = 6 }
						TOP_visit_can_invite = yes
					}
					set_temp_variable = { TOP_visit_open_candidate = top_visit_person }
				}
			}
		}
		if = {
			limit = { check_variable = { TOP_visit_open_candidate > 0 } }
			set_variable = { global.TOP_visit_open_after = { value = global.TOP_clock add = global.TOP_visit_open_interval } }
			# The chain spans a 60 day notice and a 21 day stay, so the pick is
			# held in country variables rather than temporaries.
			set_variable = { TOP_open_visit_target = TOP_visit_open_candidate }
			set_variable = { TOP_open_visit_state = TOP_visit_open_state }
			country_event = { id = TOP_visit.6 days = 1 }
		}
	}
}

	set_temp_variable = { TOP_arg_target = TOP_target }
	set_temp_variable = { TOP_history_current_event = 10 }
	set_temp_variable = { TOP_history_current_reason = { value = TOP_method add = 30 } }
	TOP_log_person_milestone = yes
	set_temp_variable = { TOP_case_result_context = 1 }
	set_temp_variable = { TOP_case_result_sequence = TOP_case_sequence^TOP_target }
	add_to_variable = { global.TOP_attempts^TOP_target = 1 }
	add_to_variable = { TOP_attempts^TOP_target = 1 }
	set_temp_variable = { TOP_result = 9 }
	set_temp_variable = { TOP_physical_changed = 0 }
	TOP_calculate_person_consequences = yes
	if = {
		limit = { NOT = { check_variable = { global.TOP_state^TOP_target = TOP_case_state^TOP_target } } }
		set_temp_variable = { TOP_result = 7 }
		set_variable = { TOP_location_confidence^TOP_target = 0 }
		set_variable = { TOP_location_source^TOP_target = 0 }
		set_variable = { TOP_package_state^TOP_target = 1 }
	}
	else_if = {
		limit = { check_variable = { TOP_tier > 1 } }
		set_temp_variable = { TOP_result = 3 }
		if = { limit = { OR = { check_variable = { TOP_method = 2 } check_variable = { TOP_method = 4 } check_variable = { TOP_method = 5 } } } set_temp_variable = { TOP_result = 2 } }
		set_temp_variable = { TOP_physical_changed = 1 }
		set_temp_variable = { TOP_core_result = 1 }
		TOP_resolve_target = yes
		set_temp_variable = { TOP_core_result = 0 }
	}
	else_if = { limit = { check_variable = { TOP_tier = 1 } } set_temp_variable = { TOP_result = 8 } }
	if = { limit = { check_variable = { TOP_physical_changed = 0 } } set_variable = { TOP_assessment^TOP_target = TOP_result } }

	if = {
		limit = { NOT = { check_variable = { TOP_result = 7 } } }
		TOP_find_org = yes
		if = {
			limit = { check_variable = { TOP_org_slot > -1 } }
			set_temp_variable = { targeted_org = TOP_org_slot }
			if = { limit = { check_variable = { TOP_tier = 0 } } ct_hvt_raid_failure = yes }
			else_if = { limit = { check_variable = { TOP_tier = 1 } } ct_hvt_raid_limited_success = yes }
			else_if = { limit = { check_variable = { TOP_tier = 2 } } ct_hvt_raid_success = yes }
			else = { ct_hvt_raid_critical_success = yes }
		}
	}
	set_variable = { TOP_case_result^TOP_target = TOP_result }
	set_temp_variable = { TOP_pressure_amount = 35 }
	TOP_add_person_pressure = yes
	if = {
		limit = { check_variable = { TOP_physical_changed = 0 } }
		TOP_archive_result = yes
	}
	TOP_apply_shared_consequences = yes
	TOP_finish_case = yes
	if = {
		limit = { check_variable = { TOP_result = 7 } }
		set_temp_variable = { TOP_arg_target = TOP_target }
		set_temp_variable = { TOP_arg_sequence = TOP_case_sequence^TOP_target }
		TOP_close_case = yes
	}
	set_temp_variable = { TOP_report_enqueue_kind = 1 }
	set_temp_variable = { TOP_report_enqueue_id = TOP_target }
	set_temp_variable = { TOP_report_enqueue_result = TOP_result }
	set_temp_variable = { TOP_report_enqueue_state = TOP_case_state^TOP_target }
	set_temp_variable = { TOP_report_enqueue_host = TOP_case_host^TOP_target }
	set_temp_variable = { TOP_report_enqueue_objective = 0 }
	set_temp_variable = { TOP_report_enqueue_impact = 0 }
	TOP_queue_field_report = yes
	set_temp_variable = { TOP_case_result_context = 0 }
	set_temp_variable = { TOP_case_result_sequence = 0 }
}



TOP_calculate_person_consequences = {
	set_temp_variable = { TOP_operation_protection_country = TOP_case_protection^TOP_target }
	set_temp_variable = { TOP_exposure_score = TOP_case_exposure_score^TOP_target }
	set_temp_variable = { TOP_exposure_roll = 0 }
	randomize_temp_variable = { var = TOP_exposure_roll min = 1 max = 100 distribution = uniform }
	set_variable = { TOP_case_attribution^TOP_target = 0 }
	if = { limit = { check_variable = { TOP_exposure_roll < TOP_exposure_score } } set_variable = { TOP_case_attribution^TOP_target = 1 } }
	set_temp_variable = { TOP_credible_floor = { value = TOP_exposure_score subtract = 20 } }
	if = { limit = { check_variable = { TOP_exposure_roll < TOP_credible_floor } } set_variable = { TOP_case_attribution^TOP_target = 2 } }
	set_temp_variable = { TOP_public_floor = { value = TOP_exposure_score subtract = 40 } }
	if = { limit = { check_variable = { TOP_exposure_roll < TOP_public_floor } } set_variable = { TOP_case_attribution^TOP_target = 3 } }
	if = { limit = { check_variable = { TOP_method < 3 } } set_variable = { TOP_case_attribution^TOP_target = 3 } set_temp_variable = { top_native_reaction_attribution = 3 } }
	set_variable = { TOP_case_attribution_due^TOP_target = { value = global.TOP_clock add = 14 } }
	set_variable = { TOP_case_attribution_investigated^TOP_target = 0 }
	if = { limit = { NOT = { is_in_array = { TOP_attribution_pending_people = TOP_target } } } add_to_array = { TOP_attribution_pending_people = TOP_target } }
	set_temp_variable = { TOP_harm_score = TOP_case_harm_risk^TOP_target }
	TOP_roll_harm_tier = yes
	set_variable = { TOP_case_harm^TOP_target = TOP_harm_tier }
}

TOP_roll_harm_tier = {
	set_temp_variable = { TOP_harm_tier = 0 }
	set_temp_variable = { TOP_harm_roll = 0 }
	randomize_temp_variable = { var = TOP_harm_roll min = 1 max = 100 distribution = uniform }
	if = { limit = { check_variable = { TOP_harm_roll < TOP_harm_score } } set_temp_variable = { TOP_harm_tier = 1 } }
	set_temp_variable = { TOP_severe_floor = { value = TOP_harm_score subtract = 10 } }
	if = { limit = { check_variable = { TOP_harm_roll < TOP_severe_floor } } set_temp_variable = { TOP_harm_tier = 2 } }
	set_temp_variable = { TOP_durable_floor = { value = TOP_harm_score subtract = 25 } }
	if = { limit = { check_variable = { TOP_harm_roll < TOP_durable_floor } } set_temp_variable = { TOP_harm_tier = 3 } }
}

TOP_process_attribution_investigations = {
	clear_array = TOP_attribution_people_remaining
	for_each_loop = {
		array = TOP_attribution_pending_people
		value = top_attribution_target
		if = {
			limit = { check_variable = { var = global.TOP_clock value = TOP_case_attribution_due^top_attribution_target compare = greater_than_or_equals } }
			if = {
				limit = { check_variable = { TOP_case_attribution_investigated^top_attribution_target = 0 } check_variable = { TOP_case_attribution^top_attribution_target < 3 } }
				set_variable = { TOP_case_attribution_investigated^top_attribution_target = 1 }
				set_temp_variable = { TOP_investigation_chance = 35 }
				if = { limit = { check_variable = { TOP_case_oversight^top_attribution_target = 3 } } add_to_temp_variable = { TOP_investigation_chance = 25 } }
				set_temp_variable = { TOP_preinvestigation_attribution = TOP_case_attribution^top_attribution_target }
				random = { chance = TOP_investigation_chance add_to_variable = { TOP_case_attribution^top_attribution_target = 1 } }
				if = {
					limit = { check_variable = { TOP_case_attribution^top_attribution_target > TOP_preinvestigation_attribution } }
					set_temp_variable = { TOP_archive_update_row = TOP_case_archive_row^top_attribution_target }
					if = {
						limit = {
							check_variable = { TOP_case_archive_token^top_attribution_target = TOP_case_sequence^top_attribution_target }
							check_variable = { TOP_archive_subject_kind^TOP_archive_update_row = 1 }
							check_variable = { TOP_archive_subject_id^TOP_archive_update_row = top_attribution_target }
							check_variable = { TOP_archive_sequence^TOP_archive_update_row = TOP_case_sequence^top_attribution_target }
						}
						set_variable = { TOP_archive_attribution^TOP_archive_update_row = TOP_case_attribution^top_attribution_target }
					}
					set_temp_variable = { TOP_target = top_attribution_target }
					set_temp_variable = { TOP_result = TOP_case_result^top_attribution_target }
					TOP_apply_shared_consequences = yes
				}
			}
		}
		else = { add_to_array = { TOP_attribution_people_remaining = top_attribution_target } }
	}
	clear_array = TOP_attribution_pending_people
	for_each_loop = { array = TOP_attribution_people_remaining value = top_attribution_target add_to_array = { TOP_attribution_pending_people = top_attribution_target } }
	clear_array = TOP_attribution_organizations_remaining
	for_each_loop = {
		array = TOP_attribution_pending_organizations
		value = top_attribution_group
		if = {
			limit = { check_variable = { var = global.TOP_clock value = TOP_org_case_attribution_due^top_attribution_group compare = greater_than_or_equals } }
			if = {
				limit = { check_variable = { TOP_org_case_attribution_investigated^top_attribution_group = 0 } check_variable = { TOP_org_case_attribution^top_attribution_group < 3 } }
				set_variable = { TOP_org_case_attribution_investigated^top_attribution_group = 1 }
				set_temp_variable = { TOP_investigation_chance = 35 }
				if = { limit = { check_variable = { TOP_org_case_oversight^top_attribution_group = 3 } } add_to_temp_variable = { TOP_investigation_chance = 25 } }
				set_temp_variable = { TOP_preinvestigation_attribution = TOP_org_case_attribution^top_attribution_group }
				random = { chance = TOP_investigation_chance add_to_variable = { TOP_org_case_attribution^top_attribution_group = 1 } }
				if = {
					limit = { check_variable = { TOP_org_case_attribution^top_attribution_group > TOP_preinvestigation_attribution } }
					set_temp_variable = { TOP_archive_update_row = TOP_org_case_archive_row^top_attribution_group }
					if = {
						limit = {
							check_variable = { TOP_org_case_archive_token^top_attribution_group = TOP_org_case_sequence^top_attribution_group }
							check_variable = { TOP_archive_subject_kind^TOP_archive_update_row = 2 }
							check_variable = { TOP_archive_subject_id^TOP_archive_update_row = top_attribution_group }
							check_variable = { TOP_archive_sequence^TOP_archive_update_row = TOP_org_case_sequence^top_attribution_group }
						}
						set_variable = { TOP_archive_attribution^TOP_archive_update_row = TOP_org_case_attribution^top_attribution_group }
					}
					set_temp_variable = { TOP_group_target = top_attribution_group }
					set_temp_variable = { TOP_result = TOP_org_case_result^top_attribution_group }
					TOP_apply_organization_consequences = yes
				}
			}
		}
		else = { add_to_array = { TOP_attribution_organizations_remaining = top_attribution_group } }
	}
	clear_array = TOP_attribution_pending_organizations
	for_each_loop = { array = TOP_attribution_organizations_remaining value = top_attribution_group add_to_array = { TOP_attribution_pending_organizations = top_attribution_group } }
}

TOP_apply_shared_consequences = {
	set_temp_variable = { TOP_attribution_level = TOP_case_attribution^TOP_target }
	set_temp_variable = { TOP_operation_protection_country = TOP_case_protection^TOP_target }
	set_variable = { global.TOP_consequence_actor = THIS }
	if = { limit = { is_subject = yes } OVERLORD = { set_variable = { global.TOP_consequence_actor = THIS } } }
	if = {
		limit = {
			check_variable = { TOP_case_crisis_opened^TOP_target = 1 }
			check_variable = { global.TOP_crisis_active = 1 }
			check_variable = { global.TOP_crisis_actor = global.TOP_consequence_actor }
			check_variable = { global.TOP_crisis_protection = TOP_operation_protection_country }
			check_variable = { TOP_attribution_level > global.TOP_crisis_attribution }
		}
		set_variable = { global.TOP_crisis_attribution = TOP_attribution_level }
	}
	if = {
		limit = { check_variable = { TOP_case_crisis_opened^TOP_target = 1 } check_variable = { global.TOP_crisis_queue_targets^num > 0 } }
		for_loop_effect = {
			start = 0
			end = global.TOP_crisis_queue_targets^num
			compare = less_than
			value = top_crisis_queue_index
			if = {
				limit = {
					check_variable = { global.TOP_crisis_queue_actors^top_crisis_queue_index = global.TOP_consequence_actor }
					check_variable = { global.TOP_crisis_queue_targets^top_crisis_queue_index = TOP_target }
					check_variable = { global.TOP_crisis_queue_sequences^top_crisis_queue_index = TOP_case_sequence^TOP_target }
					check_variable = { global.TOP_crisis_queue_attributions^top_crisis_queue_index < TOP_attribution_level }
				}
				set_variable = { global.TOP_crisis_queue_attributions^top_crisis_queue_index = TOP_attribution_level }
			}
		}
	}
	if = {
		limit = { check_variable = { TOP_attribution_level > 0 } check_variable = { TOP_operation_protection_country > 0 }
			OR = { check_variable = { TOP_case_consent^TOP_target = 0 } NOT = { check_variable = { TOP_operation_protection_country = TOP_case_host^TOP_target } } }
			OR = {
				NOT = { AND = { check_variable = { TOP_case_method^TOP_target > 0 } check_variable = { TOP_case_method^TOP_target < 3 } } }
				NOT = { check_variable = { TOP_operation_protection_country = TOP_case_host^TOP_target } }
			}
		}
		var:TOP_operation_protection_country = { add_opinion_modifier = { target = PREV modifier = TOP_sovereignty_violation } }
	}
	if = {
		limit = {
			check_variable = { TOP_attribution_level > 1 }
			check_variable = { TOP_case_crisis_opened^TOP_target = 0 }
			OR = { check_variable = { TOP_result = 2 } check_variable = { TOP_result = 3 } }
			check_variable = { global.TOP_leader_role^TOP_target > 0 }
			check_variable = { global.TOP_leader_role^TOP_target < 3 }
		}
		set_temp_variable = { TOP_crisis_enqueued_or_started = 0 }
		if = { limit = { check_variable = { TOP_result = 2 } } TOP_start_capture_crisis = yes }
		else = { TOP_start_strategic_leader_crisis = yes }
		if = { limit = { check_variable = { TOP_crisis_enqueued_or_started = 1 } } set_variable = { TOP_case_crisis_opened^TOP_target = 1 } }
	}
	if = {
		limit = {
			OR = {
				check_variable = { TOP_case_harm^TOP_target > 1 }
				AND = { check_variable = { TOP_attribution_level > 1 } check_variable = { global.TOP_leader_role^TOP_target > 0 } check_variable = { global.TOP_leader_role^TOP_target < 4 } }
				AND = { check_variable = { TOP_attribution_level > 0 } OR = { check_variable = { TOP_result = 8 } check_variable = { TOP_result = 9 } } }
			}
		}
		TOP_open_oversight = yes
	}
	set_variable = { global.TOP_consequence_actor = 0 }
}

TOP_apply_organization_consequences = {
	set_temp_variable = { TOP_attribution_level = TOP_org_case_attribution^TOP_group_target }
	set_temp_variable = { TOP_operation_protection_country = TOP_org_case_host^TOP_group_target }
	if = {
		limit = { check_variable = { TOP_attribution_level > 0 } check_variable = { TOP_operation_protection_country > 0 } }
		var:TOP_operation_protection_country = { add_opinion_modifier = { target = PREV modifier = TOP_sovereignty_violation } }
	}
	if = {
		limit = {
			OR = {
				check_variable = { TOP_org_case_harm^TOP_group_target > 1 }
				AND = {
					check_variable = { TOP_attribution_level > 0 }
					OR = {
						check_variable = { TOP_result = 8 }
						check_variable = { TOP_result = 9 }
					}
				}
			}
		}
		TOP_open_organization_oversight = yes
	}
}
