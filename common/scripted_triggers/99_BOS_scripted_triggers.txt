# Rotation flag marks the seat the NEXT rotation installs, so the current chair is the previous ethnicity in the serb -> croat -> bosniak cycle: flag serb = a Bosniak sits, flag croat = a Serb sits, flag bosniak = a Croat sits. The BOS_pro_srpska / BOS_pro_bosnia timed ideas from the lead-government focuses override the rotation while active.
BOS_srpska_leads_government_trigger = {
	NOT = { has_country_flag = civil_war_bosnia }
	OR = {
		has_idea = BOS_pro_srpska
		AND = {
			NOT = { has_idea = BOS_pro_srpska }
			NOT = { has_idea = BOS_pro_bosnia }
			has_country_flag = BOS_a_croat_is_in_power
		}
	}
}
BOS_bosniak_leads_government_trigger = {
	NOT = { has_country_flag = civil_war_bosnia }
	NOT = { has_idea = BOS_pro_srpska }
	OR = {
		has_idea = BOS_pro_bosnia
		AND = {
			NOT = { has_idea = BOS_pro_bosnia }
			has_country_flag = BOS_a_serb_is_in_power
		}
	}
}
BOS_croat_leads_government_trigger = {
	NOT = { has_country_flag = civil_war_bosnia }
	NOT = { has_idea = BOS_pro_srpska }
	NOT = { has_idea = BOS_pro_bosnia }
	has_country_flag = BOS_a_bosniak_is_in_power
}
