		set_variable = { KOR_samsung_foundry_capability = 4 }

KOR_samsung_update_market_pressure = {
	remove_ideas = {
		KOR_samsung_market_pressure_light
		KOR_samsung_market_pressure_moderate
		KOR_samsung_market_pressure_severe
	}
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		set_variable = { KOR_samsung_market_pressure = global.semiconductor_market_demand }
		add_to_variable = { KOR_samsung_market_pressure = global.semiconductor_market_tool_backlog }
		add_to_variable = { KOR_samsung_market_pressure = global.semiconductor_market_access_tension }
		add_to_variable = { KOR_samsung_market_pressure = 10 }
		subtract_from_variable = { KOR_samsung_market_pressure = KOR_samsung_foundry_capability }
		subtract_from_variable = { KOR_samsung_market_pressure = KOR_samsung_supply_resilience }
		divide_variable = { KOR_samsung_market_pressure = 3 }
		clamp_variable = { var = KOR_samsung_market_pressure min = 0 max = 10 }
		if = {
			limit = { check_variable = { KOR_samsung_market_pressure > 6 } }
			add_ideas = KOR_samsung_market_pressure_severe
		}
		else_if = {
			limit = { check_variable = { KOR_samsung_market_pressure > 3 } }
			add_ideas = KOR_samsung_market_pressure_moderate
		}
		else_if = {
			limit = { check_variable = { KOR_samsung_market_pressure > 0 } }
			add_ideas = KOR_samsung_market_pressure_light
		}
	}
}

KOR_samsung_resolve_capstone = {
	KOR_samsung_initialize_state = yes
	KOR_samsung_clear_capstone_outcome = yes
	if = {
		limit = {
			has_country_flag = KOR_samsung_integrated_stack_strategy
			check_variable = { KOR_samsung_memory_leadership > 8 }
			check_variable = { KOR_samsung_foundry_capability > 7 }
			check_variable = { KOR_samsung_consumer_ecosystem > 7 }
			check_variable = { KOR_samsung_supply_resilience > 6 }
		}
		KOR_samsung_apply_integrated_ai_semiconductor_stack = yes
	}
	else_if = {
		limit = {
			has_country_flag = KOR_samsung_merchant_foundry_strategy
			check_variable = { KOR_samsung_foundry_capability > 8 }
			check_variable = { KOR_samsung_supply_resilience > 6 }
		}
		KOR_samsung_apply_merchant_foundry_challenger = yes
	}
	else_if = {
		limit = {
			check_variable = { KOR_samsung_memory_leadership > 8 }
			check_variable = { KOR_samsung_supply_resilience > 5 }
		}
		KOR_samsung_apply_memory_ai_infrastructure_champion = yes
	}
	else_if = {
		limit = {
			has_country_flag = KOR_samsung_galaxy_ecosystem_strategy
			check_variable = { KOR_samsung_consumer_ecosystem > 8 }
			check_variable = { KOR_samsung_governance_trust > 4 }
		}
		KOR_samsung_apply_global_galaxy_ecosystem = yes
	}
	else = { KOR_samsung_apply_disciplined_diversified_champion = yes }
	KOR_samsung_clamp_state = yes
}
