CHI_lenovo_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = CHI_lenovo_state_initialized } }
		set_country_flag = CHI_lenovo_state_initialized
		add_ideas = CHI_lenovo_domestic_champion
	}
}

CHI_lenovo_schedule_cfius_followup = {
	if = {
		limit = {
			corporate_history_full_enabled = yes
			original_tag = CHI
			NOT = { has_country_flag = collapsed_nation }
			has_country_flag = CHI_lenovo_ibm_bid_accepted
			NOT = { has_country_flag = CHI_lenovo_cfius_cleared }
			NOT = { has_country_flag = CHI_lenovo_event_2_delivered }
			NOT = { has_country_flag = CHI_lenovo_event_2_pending }
			country_exists = USA
			NOT = { has_war_with = USA }
			USA = {
				NOT = { has_country_flag = collapsed_nation }
				has_country_flag = USA_ibm_lenovo_cfius_cleared
			}
		}
		set_country_flag = { flag = CHI_lenovo_event_2_pending value = 1 days = 40 }
		country_event = CHI_lenovo_events.2
	}
}

CHI_lenovo_begin_ibm_bridge = {

	set_country_flag = CHI_lenovo_service_alliance
	add_ideas = CHI_lenovo_inherited_ibm_reputation
	add_ideas = CHI_lenovo_global_integration_costs
	if = {
		limit = {
			country_exists = USA
			NOT = { has_war_with = USA }
			USA = {
				NOT = { has_country_flag = collapsed_nation }
				has_country_flag = USA_ibm_rtp_transition_managed
			}
		}
		add_ideas = CHI_lenovo_strategic_bridge
	}
}

		}
	}
}

CHI_lenovo_reconstruct_history = {
	CHI_lenovo_initialize_state = yes
	if = {
		limit = {
			date > 2005.5.1
			NOT = { has_country_flag = CHI_lenovo_pc_path_chosen }
			NOT = { has_country_flag = CHI_lenovo_event_1_delivered }
		}
		if = {
			limit = {
				date > 2005.5.1
				NOT = { has_country_flag = CHI_lenovo_pc_path_chosen }
				country_exists = USA
				NOT = { has_war_with = USA }
				USA = {
					NOT = { has_country_flag = collapsed_nation }
					has_country_flag = USA_ibm_pc_divested
					has_country_flag = USA_ibm_lenovo_cfius_cleared
				}
			}
			set_country_flag = CHI_lenovo_pc_path_chosen
			set_country_flag = CHI_lenovo_ibm_bid_accepted
			set_country_flag = CHI_lenovo_cfius_cleared
			CHI_lenovo_begin_ibm_bridge = yes
		}
		else_if = {
			limit = {
				date > 2005.5.1
				NOT = { has_country_flag = CHI_lenovo_pc_path_chosen }
				country_exists = USA
				USA = {
					OR = {
						has_country_flag = USA_ibm_pc_retained
						has_country_flag = USA_ibm_pc_spinout
					}
				}
			}
			set_country_flag = CHI_lenovo_pc_path_chosen
			set_country_flag = CHI_lenovo_organic_globalization
		}
		else_if = {
			limit = {
				date > 2006.1.1
				NOT = { has_country_flag = CHI_lenovo_pc_path_chosen }
				OR = {
					NOT = { country_exists = USA }
					AND = {
						country_exists = USA
						has_war_with = USA
					}
					AND = {
						country_exists = USA
						USA = { has_country_flag = collapsed_nation }
					}
				}
			}
			set_country_flag = CHI_lenovo_pc_path_chosen
			set_country_flag = CHI_lenovo_organic_globalization
		}
	}
	if = {
		limit = {
			corporate_history_outcomes_only_enabled = yes
			date > 2005.5.1
			has_country_flag = CHI_lenovo_pc_path_chosen
			has_country_flag = CHI_lenovo_ibm_bid_accepted
			NOT = { has_country_flag = CHI_lenovo_ibm_pc_acquired }
		}
		if = {
			limit = {
				date > 2005.5.1
				has_country_flag = CHI_lenovo_pc_path_chosen
				has_country_flag = CHI_lenovo_ibm_bid_accepted
				NOT = { has_country_flag = CHI_lenovo_ibm_pc_acquired }
				country_exists = USA
				NOT = { has_war_with = USA }
				USA = {
					NOT = { has_country_flag = collapsed_nation }
					has_country_flag = USA_ibm_pc_divested
					has_country_flag = USA_ibm_lenovo_cfius_cleared
				}
			}
			set_country_flag = CHI_lenovo_cfius_cleared
			CHI_lenovo_begin_ibm_bridge = yes
		}
		else_if = {
			limit = {
				date > 2005.5.1
				has_country_flag = CHI_lenovo_pc_path_chosen
				has_country_flag = CHI_lenovo_ibm_bid_accepted
				NOT = { has_country_flag = CHI_lenovo_ibm_pc_acquired }
				OR = {
					AND = {
						country_exists = USA
						USA = {
							OR = {
								has_country_flag = USA_ibm_pc_retained
								has_country_flag = USA_ibm_pc_spinout
							}
						}
					}
					AND = {
						date > 2006.1.1
						OR = {
							NOT = { country_exists = USA }
							AND = {
								country_exists = USA
								has_war_with = USA
							}
							AND = {
								country_exists = USA
								USA = { has_country_flag = collapsed_nation }
							}
						}
					}
				}
			}
			clr_country_flag = CHI_lenovo_ibm_bid_accepted
			clr_country_flag = CHI_lenovo_narrow_ibm_bridge
			set_country_flag = CHI_lenovo_organic_globalization
		}
	}
	if = {
		limit = {
			date > 2007.10.1
			NOT = { has_country_flag = CHI_lenovo_brand_path_chosen }
			NOT = { has_country_flag = CHI_lenovo_event_4_delivered }
			NOT = { has_country_flag = CHI_lenovo_event_4_pending }
			NOT = { has_country_flag = CHI_lenovo_event_5_delivered }
			NOT = { has_country_flag = CHI_lenovo_event_5_pending }
			OR = {
				has_country_flag = CHI_lenovo_ibm_pc_acquired
				AND = {
					has_country_flag = CHI_lenovo_narrow_ibm_bridge
					NOT = { has_country_flag = CHI_lenovo_ibm_bid_accepted }
				}
				has_country_flag = CHI_lenovo_organic_globalization
			}
		}
		set_country_flag = CHI_lenovo_integration_doctrine_chosen
		set_country_flag = CHI_lenovo_brand_path_chosen
		CHI_lenovo_complete_pc_integration = yes
	}
	if = {
		limit = {
			date > 2010.5.1
			has_country_flag = CHI_lenovo_service_alliance
			has_country_flag = CHI_lenovo_brand_path_chosen
			NOT = { has_country_flag = CHI_lenovo_alliance_matured }
			NOT = { has_country_flag = CHI_lenovo_event_6_delivered }
			NOT = { has_country_flag = CHI_lenovo_event_6_pending }
		}
		set_country_flag = CHI_lenovo_alliance_matured
		remove_ideas = CHI_lenovo_strategic_bridge
	}
	if = {
		limit = {
			corporate_history_outcomes_only_enabled = yes
			date > 2014.10.1
			has_country_flag = CHI_lenovo_systemx_path_chosen
			NOT = { has_country_flag = CHI_lenovo_systemx_path_resolved }
		}
		if = {
			limit = {
				date > 2014.10.1
				has_country_flag = CHI_lenovo_systemx_path_chosen
				has_country_flag = CHI_lenovo_systemx_acquisition_route
				NOT = { has_country_flag = CHI_lenovo_systemx_path_resolved }
			}
			set_country_flag = CHI_lenovo_systemx_path_resolved
			CHI_lenovo_complete_systemx_integration = yes
		}
		else_if = {
			limit = {
				date > 2014.10.1
				has_country_flag = CHI_lenovo_systemx_path_chosen
				NOT = { has_country_flag = CHI_lenovo_systemx_path_resolved }
				OR = {
					has_country_flag = CHI_lenovo_systemx_alliance_route
					has_country_flag = CHI_lenovo_systemx_organic_route
				}
			}
			set_country_flag = CHI_lenovo_systemx_path_resolved
		}
	}
	if = {
		limit = {
			date > 2014.10.1
			NOT = { has_country_flag = CHI_lenovo_systemx_path_chosen }
			NOT = { has_country_flag = CHI_lenovo_systemx_path_resolved }
			NOT = { has_country_flag = CHI_lenovo_event_7_delivered }
			NOT = { has_country_flag = CHI_lenovo_event_7_pending }
		}
		if = {
			limit = {
				date > 2014.10.1
				NOT = { has_country_flag = CHI_lenovo_systemx_path_resolved }
				country_exists = USA
				NOT = { has_war_with = USA }
				USA = {
					NOT = { has_country_flag = collapsed_nation }
					OR = {
						has_country_flag = USA_ibm_x86_divested
						has_country_flag = USA_ibm_silicon_retained
					}
				}
			}
			set_country_flag = CHI_lenovo_systemx_path_chosen
			set_country_flag = CHI_lenovo_systemx_acquisition_route
			clr_country_flag = CHI_lenovo_systemx_alliance_route
			clr_country_flag = CHI_lenovo_systemx_organic_route
			set_country_flag = CHI_lenovo_systemx_path_resolved
			CHI_lenovo_complete_systemx_integration = yes
		}
		else_if = {
			limit = {
				date > 2014.10.1
				NOT = { has_country_flag = CHI_lenovo_systemx_path_resolved }
				country_exists = USA
				USA = { has_country_flag = USA_ibm_x86_retained }
			}
			set_country_flag = CHI_lenovo_systemx_path_chosen
			clr_country_flag = CHI_lenovo_systemx_acquisition_route
			clr_country_flag = CHI_lenovo_systemx_alliance_route
			set_country_flag = CHI_lenovo_systemx_organic_route
			set_country_flag = CHI_lenovo_systemx_path_resolved
		}
		else_if = {
			limit = {
				date > 2015.1.1
				NOT = { has_country_flag = CHI_lenovo_systemx_path_resolved }
				OR = {
					NOT = { country_exists = USA }
					AND = {
						country_exists = USA
						has_war_with = USA
					}
					AND = {
						country_exists = USA
						USA = { has_country_flag = collapsed_nation }
					}
				}
			}
			set_country_flag = CHI_lenovo_systemx_path_chosen
			clr_country_flag = CHI_lenovo_systemx_acquisition_route
			clr_country_flag = CHI_lenovo_systemx_alliance_route
			set_country_flag = CHI_lenovo_systemx_organic_route
			set_country_flag = CHI_lenovo_systemx_path_resolved
		}
	}

	CHI_lenovo_update_motorola_mobility = yes
	if = {
		limit = {
			date > 2014.10.1
			has_country_flag = CHI_lenovo_pc_path_chosen
			has_country_flag = CHI_lenovo_brand_path_chosen
			has_country_flag = CHI_lenovo_systemx_path_resolved
			NOT = { has_country_flag = CHI_lenovo_reconstruct_complete }
		}
		set_country_flag = CHI_lenovo_reconstruct_complete
	}
}

CHI_lenovo_schedule_current_year_events = {
	if = {
		limit = { NOT = { has_country_flag = CHI_lenovo_start_year_events_scheduled } }

		if = {
			limit = {
				NOT = { has_start_date < 2014.1.1 }
				has_start_date < 2014.1.2
				NOT = { has_country_flag = CHI_lenovo_systemx_path_resolved }
				NOT = { has_country_flag = CHI_lenovo_event_7_delivered }
			}
			set_country_flag = { flag = CHI_lenovo_event_7_pending value = 1 days = 230 }
			country_event = { id = CHI_lenovo_events.7 days = 190 }
		}
	}

	set_country_flag = CHI_lenovo_start_year_events_scheduled
}
