# Currency System — Scripted Triggers

is_reserve_currency_issuer = {
	OR = {
		check_variable = { global.usd_issuer = THIS }
		check_variable = { global.eur_issuer = THIS }
		check_variable = { global.cny_issuer = THIS }
		check_variable = { global.rub_issuer = THIS }
		check_variable = { global.gbp_issuer = THIS }
		check_variable = { global.jpy_issuer = THIS }
		check_variable = { global.chf_issuer = THIS }
	}
}

has_strong_currency = {
	check_variable = { currency_strength > 1.1 }
}

has_weak_currency = {
	check_variable = { currency_strength < 0.7 }
}

has_currency_crisis = {
	has_country_flag = in_currency_crisis
}
