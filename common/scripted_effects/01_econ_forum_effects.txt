# Country scope, temp ef_i set. Makes this country the forum's host, seats it in the capital and seeds its tracks.
econ_forum_register_host = {
	set_variable = { global.econ_forum_host^ef_i = THIS.id }
	set_variable = { econ_forum_hosted = ef_i }
	capital_scope = { set_variable = { global.econ_forum_seat^ef_i = THIS } }
	econ_forum_init_track_prestige = yes
}

# A forum's own tracks start at its prestige, the others at half.
econ_forum_init_track_prestige = {
	for_loop_effect = {
		end = 6
		value = ef_t

		set_temp_variable = { ef_idx = { value = ef_i multiply = 6 add = ef_t } }
		set_variable = { global.econ_forum_track_prestige^ef_idx = { value = global.econ_forum_prestige^ef_i multiply = 0.5 } }
		if = {
			limit = { econ_forum_track_in_identity = yes }
			set_variable = { global.econ_forum_track_prestige^ef_idx = global.econ_forum_prestige^ef_i }
		}
	}
}

# Global monthly tick. Preparation opens two months before a forum's summit month.
econ_forum_monthly_update = {
	for_loop_effect = {
		end = global.econ_forum_host^num
		value = ef_i

		set_temp_variable = { ef_host = global.econ_forum_host^ef_i }
		if = {
			limit = { NOT = { check_variable = { global.econ_forum_seat^ef_i = 0 } } }
			econ_forum_check_succession = yes
		}
		if = {
			limit = {
				check_variable = { ef_host > 0 }
				var:ef_host = { exists = yes }
			}
			set_temp_variable = { ef_gap = { value = global.econ_forum_month^ef_i subtract = global.month add = 12 } }
			modulo_temp_variable = { ef_gap = 12 }

			if = {
				limit = { check_variable = { ef_gap = 2 } }
				econ_forum_open_preparation = yes
			}
			else_if = {
				limit = {
					check_variable = { ef_gap = 0 }
					check_variable = { global.econ_forum_preparing^ef_i = 1 }
				}
				econ_forum_convene_summit = yes
			}
			# Personal invitations close a month out so every reply resolves before the summit.
			else_if = {
				limit = { check_variable = { ef_gap = 1 } }
				set_variable = { global.econ_forum_invites_left^ef_i = 0 }
			}
		}
	}
}

# A forum follows its seat. When the seat state's owner is not the host, the old host loses the forum, alive or
# not, and the cycle in progress is cancelled. The seat owner takes it over unless it already hosts a forum; then
# the forum stands vacant until its seat owner is free. Updates ef_host.
econ_forum_check_succession = {
	set_temp_variable = { ef_seat_owner = 0 }
	var:global.econ_forum_seat^ef_i = {
		OWNER = { set_temp_variable = { ef_seat_owner = THIS.id } }
	}
	if = {
		limit = {
			check_variable = { ef_seat_owner > 0 }
			NOT = { check_variable = { ef_seat_owner = ef_host } }
		}
		if = {
			limit = { check_variable = { ef_host > 0 } }
			var:ef_host = {
				clear_variable = econ_forum_hosted
				clr_country_flag = econ_forum_preparing
			}
		}
		set_variable = { global.econ_forum_preparing^ef_i = 0 }
		if = {
			limit = { var:ef_seat_owner = { NOT = { has_variable = econ_forum_hosted } } }
			set_variable = { global.econ_forum_host^ef_i = ef_seat_owner }
			set_temp_variable = { ef_host = ef_seat_owner }
			var:ef_host = { set_variable = { econ_forum_hosted = ef_i } }
			set_variable = { global.econ_forum_news_forum = ef_i }
			set_variable = { global.econ_forum_news_host = ef_host }
			news_event = econ_forum_news.6
		}
		else = {
			set_variable = { global.econ_forum_host^ef_i = 0 }
			set_temp_variable = { ef_host = 0 }
		}
	}
}

# Country scope. One slot per forum; raise the sizes with the registry.
econ_forum_ensure_country_arrays = {
	if = {
		limit = { check_variable = { econ_forum_status^num < 8 } }
		resize_array = { array = econ_forum_status value = 0 size = 8 }
		resize_array = { array = econ_forum_last value = 0 size = 8 }
		resize_array = { array = econ_forum_streak value = 0 size = 8 }
	}
}

econ_forum_open_preparation = {
	set_variable = { global.econ_forum_preparing^ef_i = 1 }
	set_variable = { global.econ_forum_sponsors^ef_i = 0 }
	set_variable = { global.econ_forum_speakers^ef_i = 0 }
	set_variable = { global.econ_forum_track_count^ef_i = 0 }
	set_variable = { global.econ_forum_invites_left^ef_i = { value = global.econ_forum_prestige^ef_i divide = 25 round = yes add = 3 } }
	# Opened one month out by a founding or reschedule: no time for personal invitations.
	if = {
		limit = { check_variable = { ef_gap = 1 } }
		set_variable = { global.econ_forum_invites_left^ef_i = 0 }
	}
	for_loop_effect = {
		end = 6
		value = ef_t

		set_temp_variable = { ef_idx = { value = ef_i multiply = 6 add = ef_t } }
		set_variable = { global.econ_forum_tracks^ef_idx = 0 }
	}

	# Recurring guests and core members are invited automatically; an indispensable forum invites every regional power.
	every_country = {
		limit = {
			OR = {
				check_variable = { econ_forum_status^num > ef_i }
				econ_forum_is_core_member = yes
				AND = {
					check_variable = { global.econ_forum_prestige^ef_i > 59 }
					is_in_array = { global.PR_regional_or_greater_powers = THIS.id }
				}
			}
		}
		econ_forum_ensure_country_arrays = yes
		set_variable = { econ_forum_status^ef_i = 0 }
		if = {
			limit = { check_variable = { ef_host = THIS.id } }
			set_variable = { econ_forum_status^ef_i = 4 }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { econ_forum_last^ef_i > 0 }
					econ_forum_is_core_member = yes
					AND = {
						check_variable = { global.econ_forum_prestige^ef_i > 59 }
						is_in_array = { global.PR_regional_or_greater_powers = THIS.id }
					}
				}
			}
			econ_forum_send_invitation = yes
		}
	}

	var:ef_host = {
		econ_forum_ensure_country_arrays = yes
		set_variable = { econ_forum_status^ef_i = 4 }
		set_country_flag = econ_forum_preparing
		if = {
			limit = { is_ai = no }
			econ_forum_refresh_view = yes
			country_event = econ_forum.1
		}
		else = {
			econ_forum_ai_prepare = yes
		}
	}
}

# Invitee scope. status 1 is a standing invitation, 2 a personal one from the host.
econ_forum_send_invitation = {
	econ_forum_ensure_country_arrays = yes
	if = {
		limit = { check_variable = { econ_forum_status^ef_i = 0 } }
		set_variable = { econ_forum_status^ef_i = 1 }
		if = {
			limit = { is_ai = no }
			econ_forum_notify_invitee = yes
		}
	}
}

# A personal invitation upgrades a standing one; a human invitee who already has the event is not asked twice.
econ_forum_send_personal_invitation = {
	econ_forum_ensure_country_arrays = yes
	if = {
		limit = {
			is_ai = no
			check_variable = { econ_forum_status^ef_i = 0 }
		}
		econ_forum_notify_invitee = yes
	}
	set_variable = { econ_forum_status^ef_i = 2 }
}

# Invitee scope. Queues the forum id; each econ_forum.2 claims the oldest queued forum in its immediate, so
# invitations from several forums never overwrite each other.
econ_forum_notify_invitee = {
	add_to_array = { econ_forum_invite_queue = ef_i }
	country_event = econ_forum.2
}

# Invitee scope, from econ_forum.2. ef_answer is 3 (ministers), 4 (head of government) or -1 (decline); ef_cost is
# the political power the option charged, refunded when the forum changed hands or closed meanwhile.
econ_forum_answer_invitation = {
	set_temp_variable = { ef_i = -1 }
	event_target:econ_forum_inviter = {
		if = {
			limit = { has_variable = econ_forum_hosted }
			set_temp_variable = { ef_i = econ_forum_hosted }
			set_temp_variable = { ef_inviter_id = THIS.id }
		}
	}
	if = {
		limit = {
			check_variable = { ef_i > -1 }
			check_variable = { global.econ_forum_host^ef_i = ef_inviter_id }
			check_variable = { global.econ_forum_preparing^ef_i = 1 }
			check_variable = { econ_forum_status^ef_i > 0 }
			check_variable = { econ_forum_status^ef_i < 3 }
		}
		set_variable = { econ_forum_status^ef_i = ef_answer }
	}
	else = {
		add_political_power = ef_cost
	}
}

# Selects track ef_t for forum ef_i while fewer than three are on the program.
econ_forum_select_track = {
	set_temp_variable = { ef_idx = { value = ef_i multiply = 6 add = ef_t } }
	if = {
		limit = {
			check_variable = { global.econ_forum_tracks^ef_idx = 0 }
			check_variable = { global.econ_forum_track_count^ef_i < 3 }
		}
		set_variable = { global.econ_forum_tracks^ef_idx = 1 }
		add_to_variable = { global.econ_forum_track_count^ef_i = 1 }
	}
}

# Host scope. Copies the forum's program and sector prestige into arrays the decision text can read.
econ_forum_refresh_view = {
	resize_array = { array = econ_forum_view_selected value = 0 size = 6 }
	resize_array = { array = econ_forum_view_prestige value = 0 size = 6 }
	for_loop_effect = {
		end = 6
		value = ef_t

		set_temp_variable = { ef_idx = { value = ef_i multiply = 6 add = ef_t } }
		set_variable = { econ_forum_view_selected^ef_t = global.econ_forum_tracks^ef_idx }
		set_variable = { econ_forum_view_prestige^ef_t = global.econ_forum_track_prestige^ef_idx }
	}
}

# Host scope, AI only. Runs the forum's three strongest tracks and spends what the host can afford.
econ_forum_ai_prepare = {
	for_loop_effect = {
		end = 3
		value = ef_pick

		set_temp_variable = { ef_best = -1 }
		set_temp_variable = { ef_best_prestige = -1 }
		for_loop_effect = {
			end = 6
			value = ef_t

			set_temp_variable = { ef_idx = { value = ef_i multiply = 6 add = ef_t } }
			if = {
				limit = {
					check_variable = { global.econ_forum_tracks^ef_idx = 0 }
					check_variable = { global.econ_forum_track_prestige^ef_idx > ef_best_prestige }
				}
				set_temp_variable = { ef_best = ef_t }
				set_temp_variable = { ef_best_prestige = global.econ_forum_track_prestige^ef_idx }
			}
		}
		if = {
			limit = { check_variable = { ef_best > -1 } }
			set_temp_variable = { ef_t = ef_best }
			econ_forum_select_track = yes
		}
	}
	if = {
		limit = { check_variable = { treasury > 10 } }
		set_temp_variable = { treasury_change = -3 }
		modify_treasury_effect = yes
		add_to_variable = { global.econ_forum_sponsors^ef_i = 5 }
	}
	if = {
		limit = { has_political_power > 150 }
		add_political_power = -50
		econ_forum_try_book_speaker = yes
	}
	for_loop_effect = {
		end = global.econ_forum_invites_left^ef_i
		value = ef_n

		random_scope_in_array = {
			array = global.PR_regional_or_greater_powers
			limit = {
				NOT = { tag = PREV }
				NOT = { has_war_with = PREV }
				has_opinion = { target = PREV value > 0 }
				econ_forum_can_be_invited = yes
			}
			econ_forum_send_personal_invitation = yes
		}
	}
	set_variable = { global.econ_forum_invites_left^ef_i = 0 }
}

# Host scope. A headline speaker is harder to land the further the strongest rival forum is ahead.
# Landing one while the World Economic Forum is also preparing takes one of its speakers.
econ_forum_try_book_speaker = {
	econ_forum_find_rival = yes
	set_temp_variable = { ef_chance = { value = global.econ_forum_prestige^ef_i subtract = ef_rival_prestige divide = 2 add = 60 clamp = { min = 20 max = 90 } } }
	set_temp_variable_to_random = { var = ef_roll min = 0 max = 100 }
	set_variable = { econ_forum_speaker_result = 0 }
	if = {
		limit = { check_variable = { ef_roll < ef_chance } }
		add_to_variable = { global.econ_forum_speakers^ef_i = 1 }
		set_variable = { econ_forum_speaker_result = 1 }
		if = {
			limit = {
				NOT = { check_variable = { ef_i = 0 } }
				check_variable = { global.econ_forum_preparing^0 = 1 }
				check_variable = { global.econ_forum_speakers^0 > 0 }
			}
			add_to_variable = { global.econ_forum_speakers^0 = -1 }
			set_variable = { econ_forum_speaker_result = 2 }
		}
	}
	if = {
		limit = { is_ai = no }
		country_event = econ_forum.4
	}
}

# Sets ef_rival to the id of the most prestigious other active forum and ef_rival_prestige to its prestige.
econ_forum_find_rival = {
	set_temp_variable = { ef_rival = -1 }
	set_temp_variable = { ef_rival_prestige = 0 }
	for_loop_effect = {
		end = global.econ_forum_host^num
		value = ef_j

		if = {
			limit = {
				NOT = { check_variable = { ef_j = ef_i } }
				check_variable = { global.econ_forum_host^ef_j > 0 }
				check_variable = { global.econ_forum_prestige^ef_j > ef_rival_prestige }
			}
			set_temp_variable = { ef_rival = ef_j }
			set_temp_variable = { ef_rival_prestige = global.econ_forum_prestige^ef_j }
		}
	}
}

# Host scope. The monthly tick only opens preparation two months out, so a summit that is now one or two months
# away opens it at once. One month out leaves no time for personal invitations.
econ_forum_open_if_due = {
	set_temp_variable = { ef_gap = { value = global.econ_forum_month^ef_i subtract = global.month add = 12 } }
	modulo_temp_variable = { ef_gap = 12 }
	if = {
		limit = {
			check_variable = { ef_gap > 0 }
			check_variable = { ef_gap < 3 }
		}
		set_temp_variable = { ef_host = THIS.id }
		econ_forum_open_preparation = yes
	}
}

# Host scope. Moves the summit ef_offset months after the strongest rival's: 1 counter-programs it, 6 avoids it.
econ_forum_reschedule = {
	econ_forum_find_rival = yes
	if = {
		limit = { check_variable = { ef_rival > -1 } }
		set_temp_variable = { ef_new_month = { value = global.econ_forum_month^ef_rival add = ef_offset subtract = 1 } }
		modulo_temp_variable = { ef_new_month = 12 }
		add_to_temp_variable = { ef_new_month = 1 }
	}
	if = {
		limit = {
			OR = {
				check_variable = { ef_rival = -1 }
				check_variable = { global.econ_forum_month^ef_i = ef_new_month }
			}
		}
		add_political_power = 50
	}
	else = {
		set_variable = { global.econ_forum_month^ef_i = ef_new_month }
		econ_forum_open_if_due = yes
		set_country_flag = { flag = econ_forum_rescheduled value = 1 days = 730 }
	}
}

econ_forum_convene_summit = {
	set_temp_variable = { ef_host = global.econ_forum_host^ef_i }
	set_temp_variable = { ef_prestige = global.econ_forum_prestige^ef_i }
	set_temp_variable = { ef_points = 0 }
	set_temp_variable = { ef_heads = 0 }
	set_temp_variable = { ef_ministers = 0 }
	set_temp_variable = { ef_poached = 0 }
	set_temp_variable = { ef_skip_leader = 0 }
	clear_temp_array = ef_attendees

	# A more prestigious forum meeting within a month of ours draws attention away. If it met just before us,
	# we are counter-programming it and draw headlines and heads of government from it.
	set_temp_variable = { ef_clash = 0 }
	set_temp_variable = { ef_clash_forum = -1 }
	for_loop_effect = {
		end = global.econ_forum_host^num
		value = ef_j

		if = {
			limit = {
				NOT = { check_variable = { ef_j = ef_i } }
				check_variable = { global.econ_forum_host^ef_j > 0 }
				check_variable = { global.econ_forum_prestige^ef_j > ef_prestige }
			}
			set_temp_variable = { ef_distance = { value = global.econ_forum_month^ef_j subtract = global.econ_forum_month^ef_i add = 12 } }
			modulo_temp_variable = { ef_distance = 12 }
			if = {
				limit = {
					OR = {
						check_variable = { ef_distance < 2 }
						check_variable = { ef_distance > 10 }
					}
				}
				set_temp_variable = { ef_clash = 20 }
				if = {
					limit = {
						OR = {
							check_variable = { ef_distance > 10 }
							AND = {
								check_variable = { ef_distance = 0 }
								check_variable = { ef_j < ef_i }
							}
						}
					}
					set_temp_variable = { ef_clash_forum = ef_j }
				}
			}
		}
	}

	var:ef_host = { save_event_target_as = econ_forum_host_target }

	every_country = {
		limit = {
			check_variable = { econ_forum_status^num > ef_i }
			NOT = { check_variable = { econ_forum_status^ef_i = 0 } }
		}
		if = {
			limit = {
				is_ai = yes
				check_variable = { econ_forum_status^ef_i > 0 }
				check_variable = { econ_forum_status^ef_i < 3 }
			}
			econ_forum_ai_decide_attendance = yes
		}
		econ_forum_record_attendance = yes
	}

	# Leading every other active forum in a track on the program adds 3 corporate points.
	set_temp_variable = { ef_sector_points = 0 }
	for_loop_effect = {
		end = 6
		value = ef_t

		set_temp_variable = { ef_idx = { value = ef_i multiply = 6 add = ef_t } }
		if = {
			limit = { check_variable = { global.econ_forum_tracks^ef_idx = 1 } }
			set_temp_variable = { ef_leads = 1 }
			for_loop_effect = {
				end = global.econ_forum_host^num
				value = ef_j

				set_temp_variable = { ef_jdx = { value = ef_j multiply = 6 add = ef_t } }
				if = {
					limit = {
						NOT = { check_variable = { ef_j = ef_i } }
						check_variable = { global.econ_forum_host^ef_j > 0 }
						NOT = { check_variable = { global.econ_forum_track_prestige^ef_idx > global.econ_forum_track_prestige^ef_jdx } }
					}
					set_temp_variable = { ef_leads = 0 }
				}
			}
			if = {
				limit = { check_variable = { ef_leads = 1 } }
				add_to_temp_variable = { ef_sector_points = 3 }
			}
		}
	}

	set_temp_variable = { ef_government = { value = ef_points multiply = 0.6 min = 60 } }
	set_temp_variable = { ef_corporate = { value = ef_prestige multiply = 0.15 add = global.econ_forum_sponsors^ef_i add = ef_sector_points min = 30 } }
	set_temp_variable = { ef_speaker_points = { value = global.econ_forum_speakers^ef_i multiply = 5 add = { value = ef_prestige multiply = 0.1 } } }
	set_temp_variable = { ef_score = { value = ef_government add = ef_corporate add = ef_speaker_points } }
	if = {
		limit = { check_variable = { ef_clash_forum > -1 } }
		add_to_temp_variable = { ef_score = 5 }
	}
	clamp_temp_variable = { var = ef_score min = 0 max = 100 }

	set_variable = { global.econ_forum_prestige^ef_i = { value = ef_score subtract = ef_prestige multiply = 0.3 add = ef_prestige clamp = { min = 0 max = 100 } } }
	set_variable = { global.econ_forum_last_heads^ef_i = ef_heads }
	set_variable = { global.econ_forum_last_ministers^ef_i = ef_ministers }
	set_variable = { global.econ_forum_last_score^ef_i = ef_score }
	set_variable = { global.econ_forum_preparing^ef_i = 0 }
	set_variable = { global.econ_forum_invites_left^ef_i = 0 }

	# Tracks on the program move toward the score; the rest fade.
	for_loop_effect = {
		end = 6
		value = ef_t

		set_temp_variable = { ef_idx = { value = ef_i multiply = 6 add = ef_t } }
		if = {
			limit = { check_variable = { global.econ_forum_tracks^ef_idx = 1 } }
			set_variable = { global.econ_forum_track_prestige^ef_idx = { value = ef_score subtract = global.econ_forum_track_prestige^ef_idx multiply = 0.4 add = global.econ_forum_track_prestige^ef_idx } }
		}
		else = {
			multiply_variable = { global.econ_forum_track_prestige^ef_idx = 0.95 }
		}
	}

	# Heads of government we drew away from a clashing rival cost it a point of prestige each, up to 5.
	if = {
		limit = { check_variable = { ef_poached > 0 } }
		set_temp_variable = { ef_rival_loss = { value = ef_poached min = 5 } }
		subtract_from_variable = { global.econ_forum_prestige^ef_clash_forum = ef_rival_loss }
		clamp_variable = { var = global.econ_forum_prestige^ef_clash_forum min = 0 max = 100 }
		if = {
			limit = { check_variable = { ef_skip_leader > 0 } }
			set_variable = { global.econ_forum_news_forum = ef_i }
			set_variable = { global.econ_forum_news_rival = ef_clash_forum }
			set_variable = { global.econ_forum_news_leader = ef_skip_leader }
			news_event = econ_forum_news.3
		}
	}

	var:ef_host = {
		clr_country_flag = econ_forum_preparing
		set_temp_variable = { ef_deals = 0 }
		if = {
			limit = { check_variable = { global.econ_forum_state_led^ef_i = 1 } }
			econ_forum_reward_host = yes
		}
		if = {
			limit = { is_ai = no }
			set_variable = { econ_forum_report_prestige_before = ef_prestige }
			set_variable = { econ_forum_report_prestige = global.econ_forum_prestige^ef_i }
			set_variable = { econ_forum_report_heads = ef_heads }
			set_variable = { econ_forum_report_ministers = ef_ministers }
			set_variable = { econ_forum_report_score = ef_score }
			set_variable = { econ_forum_report_deals = ef_deals }
			econ_forum_refresh_view = yes
			country_event = econ_forum.3
		}
	}

	econ_forum_check_milestones = yes
}

# News for the rivalry with the World Economic Forum. Each fires once per campaign.
econ_forum_check_milestones = {
	if = {
		limit = {
			NOT = { check_variable = { ef_i = 0 } }
			NOT = { has_global_flag = econ_forum_challenger_rising }
			check_variable = { global.econ_forum_prestige^ef_i > 49 }
		}
		set_global_flag = econ_forum_challenger_rising
		set_variable = { global.econ_forum_news_forum = ef_i }
		set_variable = { global.econ_forum_news_host = ef_host }
		news_event = econ_forum_news.5
	}
	if = {
		limit = {
			NOT = { check_variable = { ef_i = 0 } }
			NOT = { has_global_flag = econ_forum_davos_overtaken }
			check_variable = { global.econ_forum_prestige^ef_i > global.econ_forum_prestige^0 }
		}
		set_global_flag = econ_forum_davos_overtaken
		set_variable = { global.econ_forum_news_forum = ef_i }
		set_variable = { global.econ_forum_news_host = ef_host }
		news_event = econ_forum_news.2
	}
	if = {
		limit = {
			has_global_flag = econ_forum_davos_overtaken
			NOT = { has_global_flag = econ_forum_davos_reasserted }
		}
		set_temp_variable = { ef_top_challenger = 0 }
		for_loop_effect = {
			start = 1
			end = global.econ_forum_host^num
			value = ef_j

			if = {
				limit = {
					check_variable = { global.econ_forum_host^ef_j > 0 }
					check_variable = { global.econ_forum_prestige^ef_j > ef_top_challenger }
				}
				set_temp_variable = { ef_top_challenger = global.econ_forum_prestige^ef_j }
			}
		}
		if = {
			limit = { check_variable = { global.econ_forum_prestige^0 > ef_top_challenger } }
			set_global_flag = econ_forum_davos_reasserted
			news_event = econ_forum_news.4
		}
	}
}

# Attendee scope, AI only. Turns an invitation into ministers, the head of government, or a refusal.
econ_forum_ai_decide_attendance = {
	set_temp_variable = { ef_chance = { value = ef_prestige multiply = 0.6 subtract = ef_clash } }
	if = {
		limit = { check_variable = { econ_forum_status^ef_i = 2 } }
		add_to_temp_variable = { ef_chance = 15 }
	}
	add_to_temp_variable = { ef_chance = { value = econ_forum_streak^ef_i multiply = 3 min = 15 } }
	if = {
		limit = { has_opinion = { target = event_target:econ_forum_host_target value > 25 } }
		add_to_temp_variable = { ef_chance = 10 }
	}
	else_if = {
		limit = { has_opinion = { target = event_target:econ_forum_host_target value < -25 } }
		add_to_temp_variable = { ef_chance = -15 }
	}
	if = {
		limit = { is_in_faction_with = event_target:econ_forum_host_target }
		add_to_temp_variable = { ef_chance = 10 }
	}
	if = {
		limit = { econ_forum_is_core_member = yes }
		add_to_temp_variable = { ef_chance = 15 }
	}
	if = {
		limit = { has_war_with = event_target:econ_forum_host_target }
		add_to_temp_variable = { ef_chance = -100 }
	}

	# Each track on the program that matters to us adds a tenth of the forum's standing in it.
	for_loop_effect = {
		end = 6
		value = ef_t

		set_temp_variable = { ef_idx = { value = ef_i multiply = 6 add = ef_t } }
		if = {
			limit = {
				check_variable = { global.econ_forum_tracks^ef_idx = 1 }
				econ_forum_interested_in_track = yes
			}
			add_to_temp_variable = { ef_chance = { value = global.econ_forum_track_prestige^ef_idx multiply = 0.1 } }
		}
	}

	# Heads of government spend their attention: each other forum they led a delegation to costs 10.
	for_loop_effect = {
		end = global.econ_forum_host^num
		value = ef_j

		if = {
			limit = {
				NOT = { check_variable = { ef_j = ef_i } }
				check_variable = { econ_forum_last^ef_j = 4 }
			}
			add_to_temp_variable = { ef_chance = -10 }
		}
	}
	clamp_temp_variable = { var = ef_chance min = 0 max = 95 }

	set_temp_variable = { ef_head_chance = { value = ef_chance multiply = 0.4 } }
	set_temp_variable_to_random = { var = ef_roll min = 0 max = 100 }
	if = {
		limit = { check_variable = { ef_roll < ef_chance } }
		set_variable = { econ_forum_status^ef_i = 3 }
		if = {
			limit = {
				check_variable = { ef_roll < ef_head_chance }
				OR = {
					check_variable = { ef_prestige > 29 }
					econ_forum_is_core_member = yes
				}
			}
			set_variable = { econ_forum_status^ef_i = 4 }
		}
	}
	else = {
		set_variable = { econ_forum_status^ef_i = -1 }
	}
}

# Attendee scope. Tallies the delegation, updates the streak and clears the invitation.
econ_forum_record_attendance = {
	if = {
		limit = { check_variable = { econ_forum_status^ef_i > 2 } }
		set_temp_variable = { ef_weight = { value = percentage_of_global_factories multiply = 100 } }
		if = {
			limit = { check_variable = { econ_forum_status^ef_i = 4 } }
			add_to_temp_variable = { ef_heads = 1 }
			add_to_temp_variable = { ef_points = { value = ef_weight multiply = 2 } }
			if = {
				limit = {
					check_variable = { ef_clash_forum > -1 }
					NOT = { check_variable = { econ_forum_last^ef_clash_forum = 4 } }
				}
				add_to_temp_variable = { ef_poached = 1 }
				if = {
					limit = {
						check_variable = { ef_skip_leader = 0 }
						OR = {
							has_idea = super_power
							has_idea = great_power
						}
					}
					set_temp_variable = { ef_skip_leader = THIS.id }
				}
			}
		}
		else = {
			add_to_temp_variable = { ef_ministers = 1 }
			add_to_temp_variable = { ef_points = ef_weight }
		}
		set_variable = { econ_forum_last^ef_i = econ_forum_status^ef_i }
		add_to_variable = { econ_forum_streak^ef_i = 1 }
		econ_forum_grant_delegation_bonus = yes
		if = {
			limit = { NOT = { check_variable = { ef_host = THIS.id } } }
			add_to_temp_array = { ef_attendees = THIS }
		}
	}
	else = {
		set_variable = { econ_forum_last^ef_i = 0 }
		set_variable = { econ_forum_streak^ef_i = 0 }
	}
	set_variable = { econ_forum_status^ef_i = 0 }
}

# Attendee scope. Investors follow the delegation home. A stronger bonus still running keeps its own expiry.
econ_forum_grant_delegation_bonus = {
	set_temp_variable = { ef_bonus = { value = ef_prestige multiply = -0.0005 } }
	if = {
		limit = { check_variable = { econ_forum_status^ef_i = 4 } }
		multiply_temp_variable = { ef_bonus = 2 }
	}
	if = {
		limit = {
			OR = {
				NOT = { has_dynamic_modifier = { modifier = econ_forum_delegation_modifier } }
				NOT = { check_variable = { econ_forum_investor_bonus < ef_bonus } }
			}
		}
		set_variable = { econ_forum_investor_bonus = ef_bonus }
		add_dynamic_modifier = { modifier = econ_forum_delegation_modifier days = 365 }
	}
}

# Host scope, state-led forums only. Delegations leave with a better view of the host, sign investment
# agreements that steer their AI investors toward it for a year, and the host gains influence scaled by prestige.
econ_forum_reward_host = {
	set_temp_variable = { ef_influence = { value = ef_prestige multiply = 0.02 } }
	for_each_scope_loop = {
		array = ef_attendees

		add_opinion_modifier = { target = PREV modifier = econ_forum_attended_summit }
		PREV = { set_country_flag = { flag = econ_forum_deal@PREV value = 1 days = 365 } }
		add_to_temp_variable = { ef_deals = 1 }
		set_temp_variable = { percent_change = ef_influence }
		if = {
			limit = { check_variable = { econ_forum_last^ef_i = 4 } }
			multiply_temp_variable = { percent_change = 2 }
		}
		set_temp_variable = { tag_index = ef_host }
		set_temp_variable = { influence_target = THIS.id }
		change_influence_percentage = yes
	}
	set_variable = { econ_forum_host_pp = { value = global.econ_forum_prestige^ef_i multiply = 0.005 } }
	set_variable = { econ_forum_host_influence = { value = global.econ_forum_prestige^ef_i multiply = 0.002 } }
	add_dynamic_modifier = { modifier = econ_forum_host_modifier days = 365 }
}

# Founder scope. Registers the founder as host of forum ef_i.
econ_forum_found = {
	set_variable = { global.econ_forum_prestige^ef_i = 15 }
	econ_forum_register_host = yes
	set_variable = { global.econ_forum_news_forum = ef_i }
	set_variable = { global.econ_forum_news_host = THIS.id }
	news_event = { id = econ_forum_news.1 days = 1 }
	econ_forum_open_if_due = yes
}
