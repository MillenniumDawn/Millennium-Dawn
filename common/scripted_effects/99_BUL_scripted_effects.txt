BUL_macedonia_revoult = {
	hidden_effect = {
		start_civil_war = {
			ruling_party = ROOT
			ideology = nationalist
			size = 0.1
			capital = 136
			states = { 136 }
			set_popularities = {
				democratic = 5
				neutrality = 0
				fascism = 0
				communism = 5
				nationalist = 90
			}
			start_politics_input = yes
			set_variable = { party_pop_array^2 = 0.05 }
			set_variable = { party_pop_array^4 = 0.05 }
			set_variable = { party_pop_array^23 = 0.90 }
			set_temp_variable = { rul_party_temp = 23 }
			set_temp_variable = { disable_elections = 1 }
			change_ruling_party_effect = yes
			recalculate_party = yes
			country_event = bulg.19
		}
		set_temp_variable = { percent_change = -30 }
		set_temp_variable = { influence_target = FYR }
		change_influence_percentage = yes
	}
}
