# The New Front wins the May 2000 election. ROOT = Suriname.
country_event = {
	id = suriname.1
	title = suriname.1.t
	desc = suriname.1.d
	picture = GFX_election_day
	is_triggered_only = yes

	trigger = { original_tag = SUR }

	immediate = {
		hidden_effect = {
			if = { limit = { has_elections = yes } set_elections_60_months = yes }
		}
	}

	# Govern with the new majority
	option = {
		name = suriname.1.a
		log = "[GetDateText]: [This.GetName]: suriname.1.a executed"
		add_stability = 0.03
		add_political_power = 25
		ai_chance = { base = 80 }
	}

	# Reach out to the opposition
	option = {
		name = suriname.1.b
		log = "[GetDateText]: [This.GetName]: suriname.1.b executed"
		set_temp_variable = { party_index = 1 }
		set_temp_variable = { party_popularity_increase = 0.03 }
		set_temp_variable = { temp_outlook_increase = 0.03 }
		change_relative_party_popularity = yes
		add_political_power = 10
		ai_chance = { base = 20 }
	}
}

# The National Assembly elects Desi Bouterse president after the 2010 election. ROOT = Suriname.
country_event = {
	id = suriname.2
	title = suriname.2.t
	desc = suriname.2.d
	picture = GFX_election_rally
	is_triggered_only = yes

	trigger = {
		original_tag = SUR
		has_elections = yes
		check_variable = { party_banned_array^1 = 0 }
	}

	# Elect Bouterse
	option = {
		name = suriname.2.a
		log = "[GetDateText]: [This.GetName]: suriname.2.a executed"
		if = {
			limit = { check_variable = { party_pop_array^1 < 0.5 } }
			set_temp_variable = { party_index = 1 }
			set_temp_variable = { party_popularity_increase = 0.25 }
			set_temp_variable = { temp_outlook_increase = 0.25 }
			change_relative_party_popularity = yes
		}
		if = {
			limit = { check_variable = { party_pop_array^20 < 0.04 } }
			set_temp_variable = { party_index = 20 }
			set_temp_variable = { party_popularity_increase = 0.04 }
			set_temp_variable = { temp_outlook_increase = 0.04 }
			change_relative_party_popularity = yes
		}
		set_variable = { conservatism_leader = 1 }
		if = {
			limit = { check_variable = { ruling_party = 1 } }
			hidden_effect = { set_leader = yes }
		}
		else = {
			set_temp_variable = { rul_party_temp = 1 }
			set_temp_variable = { col_one = 20 }
			set_temp_variable = { disable_elections = 0 }
			change_ruling_party_effect = yes
		}
		ai_chance = { base = 90 }
	}

	# Keep the New Front in office
	option = {
		name = suriname.2.b
		log = "[GetDateText]: [This.GetName]: suriname.2.b executed"
		add_stability = -0.03
		add_political_power = 25
		ai_chance = {
			base = 10
			modifier = { factor = 0 has_stability < 0.3 }
		}
	}
}
