# A treaty member's station runs a Radiation Laboratory right now: undamaged, fuelled and powered, the
# conditions Antarctica checks before it grants laboratory effects. Its weekly totals can lag a blizzard.
STALKER_has_working_radiation_lab = {
	has_idea = at_member
	any_of = {
		array = country_owned_station_ids
		value = STALKER_station
		check_variable = { STALKER_station > 0 }
		check_variable = { global.antarctica_station_controller^STALKER_station = THIS.id }
		OR = {
			AND = {
				check_variable = { global.antarctica_station_module_slot_6^STALKER_station = 7 }
				NOT = { check_variable = { global.antarctica_station_module_slot_6_blizzard_damaged^STALKER_station > 0 } }
			}
			AND = {
				check_variable = { global.antarctica_station_module_slot_7^STALKER_station = 7 }
				NOT = { check_variable = { global.antarctica_station_module_slot_7_blizzard_damaged^STALKER_station > 0 } }
			}
			AND = {
				check_variable = { global.antarctica_station_module_slot_8^STALKER_station = 7 }
				NOT = { check_variable = { global.antarctica_station_module_slot_8_blizzard_damaged^STALKER_station > 0 } }
			}
			AND = {
				check_variable = { global.antarctica_station_module_slot_9^STALKER_station = 7 }
				NOT = { check_variable = { global.antarctica_station_module_slot_9_blizzard_damaged^STALKER_station > 0 } }
			}
		}
		OR = {
			check_variable = { global.antarctica_station_output_fuel_usage^STALKER_station < 1 }
			check_variable = { global.antarctica_station_output_storage^STALKER_station > 0 }
		}
		check_variable = { var = global.antarctica_station_output_power^STALKER_station value = 0 compare = greater_than_or_equals }
	}
}
