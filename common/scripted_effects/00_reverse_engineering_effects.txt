# reverse_engineering_studied_index: 0-based position of the studied hull in reverse_engineering_techs, -1 for none.
reverse_engineering_grant_bonuses = {
	if = { limit = { has_idea = edu_05 } set_temp_variable = { reverse_engineering_education = 1 } }
	else_if = { limit = { has_idea = edu_04 } set_temp_variable = { reverse_engineering_education = 0.8 } }
	else_if = { limit = { has_idea = edu_03 } set_temp_variable = { reverse_engineering_education = 0.6 } }
	else_if = { limit = { has_idea = edu_02 } set_temp_variable = { reverse_engineering_education = 0.4 } }
	else = { set_temp_variable = { reverse_engineering_education = 0.2 } }
	set_temp_variable = { reverse_engineering_head_start = { value = 0.5 multiply = reverse_engineering_education } }
	if = {
		limit = { check_variable = { reverse_engineering_studied_index > -1 } }
		for_each_loop = {
			array = reverse_engineering_techs
			value = reverse_engineering_tech
			index = reverse_engineering_index
			set_temp_variable = { reverse_engineering_generations_back = { value = reverse_engineering_studied_index subtract = reverse_engineering_index } }
			if = {
				limit = { check_variable = { reverse_engineering_generations_back > 1 } }
				set_temp_variable = { reverse_engineering_bonus = { value = reverse_engineering_two_back_bonus multiply = reverse_engineering_education } }
				reverse_engineering_grant_tech_bonus = yes
			}
			else_if = {
				limit = { check_variable = { reverse_engineering_generations_back = 1 } }
				set_temp_variable = { reverse_engineering_bonus = { value = reverse_engineering_one_back_bonus multiply = reverse_engineering_education } }
				reverse_engineering_grant_tech_bonus = yes
			}
			else_if = {
				limit = { check_variable = { reverse_engineering_generations_back = 0 } }
				set_temp_variable = { reverse_engineering_bonus = { value = reverse_engineering_same_generation_bonus multiply = reverse_engineering_education } }
				reverse_engineering_grant_tech_bonus = yes
			}
		}
	}
	clear_temp_array = reverse_engineering_techs
}

reverse_engineering_grant_tech_bonus = {
	meta_effect = {
		text = {
			if = {
				limit = {
					NOT = { has_tech = [TECH] }
					NOT = { is_researching_technology = [TECH] }
					NOT = { has_country_flag = reverse_engineered_[TECH] }
				}
				set_country_flag = reverse_engineered_[TECH]
				add_tech_bonus = { name = reverse_engineering_tech_bonus bonus = [BONUS] uses = 1 technology = [TECH] }
				reverse_engineering_notify = yes
			}
		}
		TECH = "[?reverse_engineering_tech.GetTokenKey]"
		BONUS = "[?reverse_engineering_bonus|.2]"
	}
}

reverse_engineering_notify = {
	if = {
		limit = {
			is_ai = no
			NOT = { has_country_flag = reverse_engineering_report_pending }
		}
		set_country_flag = { flag = reverse_engineering_report_pending days = 1 value = 1 }
		country_event = reverse_engineering.1
	}
}

# Gen 5+ airframes (index 2 and up) also need stealth research.
reverse_engineering_aircraft_head_start = {
	if = {
		limit = {
			NOT = { is_special_project_completed = sp:sp_aircraft_project }
			NOT = { has_country_flag = reverse_engineered_sp_aircraft_project }
		}
		set_country_flag = reverse_engineered_sp_aircraft_project
		sp:sp_aircraft_project = { add_project_progress_ratio = var:reverse_engineering_head_start }
		reverse_engineering_notify = yes
	}
	if = {
		limit = {
			has_dlc = "By Blood Alone"
			check_variable = { reverse_engineering_studied_index > 1 }
			NOT = { is_special_project_completed = sp:sp_stealth_technology }
			NOT = { has_country_flag = reverse_engineered_sp_stealth_technology }
		}
		set_country_flag = reverse_engineered_sp_stealth_technology
		sp:sp_stealth_technology = { add_project_progress_ratio = var:reverse_engineering_head_start }
		reverse_engineering_notify = yes
	}
}
