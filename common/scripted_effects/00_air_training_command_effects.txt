MD_atc_upgrade_academy = {
	if = {
		limit = { check_variable = { MD_atc_academy_level < 1 } }
		air_experience = @MD_atc_upgrade_cost_1_neg
	}
	else_if = {
		limit = { check_variable = { MD_atc_academy_level = 1 } }
		air_experience = @MD_atc_upgrade_cost_2_neg
		if = {
			limit = { has_idea = MD_atc_academy_1 }
			swap_ideas = { remove_idea = MD_atc_academy_1 add_idea = MD_atc_academy_2 }
		}
	}
	else = {
		air_experience = @MD_atc_upgrade_cost_3_neg
		if = {
			limit = { has_idea = MD_atc_academy_2 }
			swap_ideas = { remove_idea = MD_atc_academy_2 add_idea = MD_atc_academy_3 }
		}
	}
	hidden_effect = {
		add_to_variable = { MD_atc_academy_level = 1 }
	}
}

MD_atc_toggle_window = {
	if = {
		limit = { NOT = { has_country_flag = MD_atc_open } }
		set_country_flag = MD_atc_open
	}
	else = {
		clr_country_flag = MD_atc_open
	}
	update_MD_atc_dirty_variable = yes
}

update_MD_atc_dirty_variable = {
	if = {
		limit = { check_variable = { global.MD_atc_dirty_update_var < 10000 } }
		add_to_variable = { global.MD_atc_dirty_update_var = 1 }
	}
	else = {
		set_variable = { global.MD_atc_dirty_update_var = 1 }
	}
}
