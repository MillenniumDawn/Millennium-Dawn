# Linux-system rule gates read the game rule directly because startup runs before
# cached game-rule flags are available.

linux_system_full_enabled = {
	NOT = { has_game_rule = { rule = rule_linux_ecosystem option = outcomes_only } }
	NOT = { has_game_rule = { rule = rule_linux_ecosystem option = off } }
}

linux_system_outcomes_only_enabled = {
	has_game_rule = { rule = rule_linux_ecosystem option = outcomes_only }
}

linux_system_enabled = {
	OR = {
		linux_system_full_enabled = yes
		linux_system_outcomes_only_enabled = yes
	}
}

linux_system_historical_upstream_route = {
	OR = {
		original_tag = ENG
		original_tag = GER
		original_tag = BRA
	}
}

linux_system_historical_national_route = {
	OR = {
		original_tag = FRA
		original_tag = RAJ
		original_tag = SOV
		original_tag = CHI
	}
}

linux_system_is_participant = {
	OR = {
		original_tag = BRA
		original_tag = CHI
		original_tag = ENG
		original_tag = FRA
		original_tag = GER
		original_tag = POL
		original_tag = RAJ
		original_tag = SOV
		original_tag = USA
		original_tag = VEN
	}
}
linux_system_country_needs_reconstruction = {
	OR = {
		AND = { date > 2001.1.3 NOT = { has_country_flag = linux_system_event_1_resolved } }
		AND = { date > 2008.5.31 NOT = { has_country_flag = linux_system_event_2_resolved } }
		AND = { date > 2015.12.9 NOT = { has_country_flag = linux_system_event_3_resolved } }
		AND = { date > 2019.7.8 NOT = { has_country_flag = linux_system_event_4_resolved } }
		AND = { date > 2024.4.29 NOT = { has_country_flag = linux_system_event_5_resolved } }
	}
}

linux_system_has_active_program = {
	OR = {
		has_idea = linux_system_upstream_maintenance_program
		has_idea = linux_system_enterprise_support_program
		has_idea = linux_system_lifecycle_hardening_program
		has_idea = linux_system_public_procurement_program
	}
}

	OR = {
		has_country_flag = linux_system_initialized
		has_country_flag = linux_system_dirty
		has_decision = linux_system_fund_upstream_maintenance
		has_decision = linux_system_contract_enterprise_support
		has_decision = linux_system_harden_lifecycle
		has_decision = linux_system_public_procurement
		has_country_flag = linux_system_event_1_expected
		has_country_flag = linux_system_event_1_pending
		has_country_flag = linux_system_event_2_expected
		has_country_flag = linux_system_event_2_pending

		has_country_flag = linux_system_event_3_pending
		has_country_flag = linux_system_event_4_expected
		has_country_flag = linux_system_event_4_pending
		has_country_flag = linux_system_event_5_expected
		has_country_flag = linux_system_event_5_pending
		has_idea = linux_system_experimental_adoption

		has_idea = linux_system_broad_economic_adoption

linux_system_program_slot_available = {
	linux_system_full_enabled = yes
	linux_system_is_participant = yes
	has_country_flag = linux_system_initialized
	NOT = { has_country_flag = collapsed_nation }
	NOT = { linux_system_has_active_program = yes }
}

linux_system_can_pay_gdp_0_1_percent = {
	NOT = { has_active_mission = bankruptcy_incoming_collapse }
	check_expr = { value = gdp_total multiply = 0.001 less_than = { value = treasury } }
}

linux_system_can_pay_gdp_0_2_percent = {
	NOT = { has_active_mission = bankruptcy_incoming_collapse }
	check_expr = { value = gdp_total multiply = 0.002 less_than = { value = treasury } }
}
