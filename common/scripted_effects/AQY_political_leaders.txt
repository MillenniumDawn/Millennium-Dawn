set_leader_AQY = {
	if = {
		limit = { TOP_enabled = yes }
		set_temp_variable = { TOP_target = global.TOP_group_leader^2 }
		TOP_apply_office_successor = yes
	}
	else = {
		if = { limit = { check_variable = { ruling_party = 11 } }

			if = { limit = { check_variable = { Caliphate_leader = 0 } }
				add_to_variable = { Caliphate_leader = 1 }
				if = {
					limit = { OR = { TOP_enabled = no check_variable = { global.TOP_status^13 < 2 } } }
					hidden_effect = { kill_country_leader = yes }

					create_country_leader = {
						name = "Qasim al-Raymi"
						picture = "AQY_qashim.dds"
						ideology = Caliphate
						traits = {
							salafist_Caliphate
						}
					}
				}

				if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { Caliphate_leader = 1 } }
				set_temp_variable = { b = 1 }
			}
		}
	}
}
