VEN_ai_alt_path = {
	OR = {
		has_global_flag = VEN_WORLD_REVOLUTION_FOCUS_PATH
		has_global_flag = VEN_DEMOCRATIC_OPPOSITION_FOCUS_PATH
		has_global_flag = VEN_CORPORATE_STATE_FOCUS_PATH
		has_global_flag = VEN_FREE_MARKET_FOCUS_PATH
		has_global_flag = VEN_MILITARY_JUNTA_FOCUS_PATH
	}
}

VEN_ai_historical_path = {
	OR = {
		has_global_flag = VEN_HISTORICAL_FOCUS_PATH
		AND = {
			is_historical_focus_on = yes
			NOT = { VEN_ai_alt_path = yes }
		}
	}
}

VEN_ai_not_historical_path = {
	VEN_ai_alt_path = yes
}

VEN_ai_chavista_path = {
	OR = {
		VEN_ai_historical_path = yes
		has_global_flag = VEN_WORLD_REVOLUTION_FOCUS_PATH
	}
}
