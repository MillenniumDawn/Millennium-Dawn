# Full-screen warning on saves started before the current save version.
# global.md_save_version is stamped in on_startup (00_on_actions.txt).
# on_startup does not fire on load, so old saves read 0.
scripted_gui = {
	MD_old_save_warning = {
		window_name = "MD_old_save_warning_window"
		context_type = player_context

		visible = {
			check_variable = { global.md_save_version < 2 }
		}
	}
}
