JAP_zombie_case_prosecuted = {
	add_to_variable = { JAP_zombie_case_files = 2 tooltip = JAP_zombie_case_files_up_2_tt }
	add_to_variable = { JAP_zombie_obstruction = 8 tooltip = JAP_zombie_obstruction_up_8_tt }
	add_to_variable = { JAP_corporate_restructuring_progress = 3 tooltip = JAP_corporate_restructuring_progress_tt }
	clamp_variable = { var = JAP_zombie_case_files min = 0 max = 14 }
	add_stability = -0.03
	set_temp_variable = { temp_opinion = -12 }
	change_industrial_conglomerates_opinion = yes
	set_temp_variable = { temp_opinion = 5 }
	change_labour_unions_opinion = yes
	set_temp_variable = { temp_opinion = 5 }
	change_small_medium_business_owners_opinion = yes
}

JAP_zombie_case_rescued = {
	add_to_variable = { JAP_zombie_case_files = 1 tooltip = JAP_zombie_case_files_up_1_tt }
	add_to_variable = { JAP_zombie_obstruction = 2 tooltip = JAP_zombie_obstruction_up_2_tt }
	add_to_variable = { JAP_corporate_restructuring_progress = 2 tooltip = JAP_corporate_restructuring_progress_tt }
	clamp_variable = { var = JAP_zombie_case_files min = 0 max = 14 }
	add_stability = 0.01
}

JAP_zombie_case_ignored = {
	subtract_from_variable = { JAP_zombie_obstruction = 3 tooltip = JAP_zombie_obstruction_down_3_tt }
	clamp_variable = { var = JAP_zombie_obstruction min = 0 max = 100 }
	add_stability = 0.02
	set_temp_variable = { temp_opinion = 5 }
	change_industrial_conglomerates_opinion = yes
	set_temp_variable = { temp_opinion = -4 }
	change_labour_unions_opinion = yes
}

JAP_escalate_territorial_disputes = {
	if = {
		limit = { has_idea = JAP_territorial_disputes1 }
		swap_ideas = {
			remove_idea = JAP_territorial_disputes1
			add_idea = JAP_territorial_disputes2
		}
	}
	else_if = {
		limit = { has_idea = JAP_territorial_disputes2 }
		swap_ideas = {
			remove_idea = JAP_territorial_disputes2
			add_idea = JAP_territorial_disputes3
		}
	}
	else_if = {
		limit = { has_idea = JAP_territorial_disputes3 }
		swap_ideas = {
			remove_idea = JAP_territorial_disputes3
			add_idea = JAP_territorial_disputes4
		}
	}
	else = {
		add_ideas = JAP_territorial_disputes1
	}
}
