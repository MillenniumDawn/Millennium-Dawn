TAJ_emomali_family_in_charge = {
	custom_trigger_tooltip = {
		tooltip = TAJ_has_rahmon_family_TT
		OR = {
			has_country_leader = { name = "Emomali Rahmonov" ruling_only = yes }
			has_country_leader = { name = "Emomali Rahmon" ruling_only = yes }
			has_country_leader = { name = "Rustam Emomali" ruling_only = yes }
		}
	}
}

TAJ_ai_alt_path = {
	OR = {
		has_global_flag = TAJ_DEMOCRATIC_FOCUS_PATH
		has_global_flag = TAJ_ISLAMIST_FOCUS_PATH
		has_global_flag = TAJ_COMMUNIST_FOCUS_PATH
	}
}

TAJ_ai_historical_path = {
	OR = {
		has_global_flag = TAJ_HISTORICAL_FOCUS_PATH
		AND = {
			is_historical_focus_on = yes
			NOT = { TAJ_ai_alt_path = yes }
		}
	}
}

TAJ_ai_not_historical_path = {
	TAJ_ai_alt_path = yes
}

TAJ_ai_not_opposition_path = {
	NOT = { TAJ_ai_alt_path = yes }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = TAJ_HISTORICAL_FOCUS_PATH
	}
}

TAJ_ai_not_democratic_path = {
	NOT = { has_global_flag = TAJ_DEMOCRATIC_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = TAJ_HISTORICAL_FOCUS_PATH
		has_global_flag = TAJ_ISLAMIST_FOCUS_PATH
		has_global_flag = TAJ_COMMUNIST_FOCUS_PATH
	}
}
