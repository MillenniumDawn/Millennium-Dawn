state_demolition_allowed = {
	is_owned_by = ROOT
	is_fully_controlled_by = ROOT
	ROOT = { is_ai = no }
}

state_demolition_selection_present = {
	OR = {
		AND = {
			check_variable = { ROOT.state_demolition_building = token:fossil_powerplant }
			fossil_powerplant > 0
		}
		AND = {
			check_variable = { ROOT.state_demolition_building = token:nuclear_reactor }
			nuclear_reactor > 0
		}
		AND = {
			check_variable = { ROOT.state_demolition_building = token:renewable_energy_infra }
			renewable_energy_infra > 0
		}
		AND = {
			check_variable = { ROOT.state_demolition_building = token:air_base }
			air_base > 0
		}
		AND = {
			check_variable = { ROOT.state_demolition_building = token:radar_station }
			radar_station > 0
		}
		AND = {
			check_variable = { ROOT.state_demolition_building = token:anti_air_building }
			anti_air_building > 0
		}
		AND = {
			check_variable = { ROOT.state_demolition_building = token:fuel_silo }
			fuel_silo > 0
		}
	}
}
