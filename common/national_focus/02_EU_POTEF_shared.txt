### Alpine Doctrine ###
shared_focus = {
	id = POTEF100
	icon = POTEF100

	x = -3
	y = 1
	relative_position_id = POTEF001

	cost = 10

	prerequisite = { focus = POTEF001 }

	search_filters = { FOCUS_FILTER_MILITARY_LAWS }

	available = {
		has_idea = EU_president
		EU_member_is_alpine = yes
	}

	bypass = { any_of_scopes = { array = global.EU_member has_completed_focus = POTEF100 } }

	completion_reward = {
		log = "[GetDateText]: [Root.GetName]: Focus POTEF100"
		if = {
			limit = {
				has_idea = EU_member
				NOT = { has_dynamic_modifier = { modifier = POTEF_dynmod_army } }
			}
			add_dynamic_modifier = { modifier = POTEF_dynmod_army }
		}
		if = {
			limit = { NOT = { POTEF_is_alpine = yes } }
			for_each_scope_loop = {
				array = global.EU_member
				tooltip = TT_ALL_EU_MEMBER_NATIONS_GAIN
				add_to_variable = { POTEF_army_infantry_defence_factor = 0.20 tooltip = army_infantry_defence_factor_tt }
				add_to_variable = { POTEF_terrain_penalty_reduction = 0.10 tooltip = terrain_penalty_reduction_tt }
			}
		}
		else = {
			for_each_scope_loop = {
				array = global.EU_member
				tooltip = TT_ALL_EU_MEMBER_NATIONS_GAIN
				add_to_variable = { POTEF_army_infantry_defence_factor = 0.30 tooltip = army_infantry_defence_factor_tt }
				add_to_variable = { POTEF_terrain_penalty_reduction = 0.20 tooltip = terrain_penalty_reduction_tt }
			}
		}
	}

	ai_will_do = { base = 1 }
}

	completion_reward = {
		log = "[GetDateText]: [Root.GetName]: Focus POTEF350"

			limit = {
				has_idea = EU_member

			}
			add_dynamic_modifier = { modifier = POTEF_dynmod_navy }
		}
		if = {
			limit = { NOT = { POTEF_is_british = yes } }
			for_each_scope_loop = {
				array = global.EU_member
				tooltip = TT_ALL_EU_MEMBER_NATIONS_GAIN
				add_to_variable = { POTEF_experience_gain_navy = 0.05 tooltip = experience_gain_navy_tt }
				add_to_variable = { POTEF_experience_gain_navy_factor = 0.10 tooltip = experience_gain_navy_factor_tt }
				add_to_variable = { POTEF_navy_org_factor = 0.10 tooltip = navy_org_factor_tt }
			}
		}
		else = {
			for_each_scope_loop = {
				array = global.EU_member
				tooltip = TT_ALL_EU_MEMBER_NATIONS_GAIN
				add_to_variable = { POTEF_experience_gain_navy = 0.10 tooltip = experience_gain_navy_tt }
