# THIS is the target, FROM is the country spreading influence
ai_influence_target_scorer = {
	targets = {
		target_array = influence_targets
		target_array = neighbors
		target_array = faction_members
		target_array = subjects
		target_array = global.PR_regional_or_greater_powers

		score = {
			base = 100

			modifier = {
				OR = {
					exists = no
					tag = FROM
					has_war_with = FROM
					has_idea = non_state_actor
					AND = {
						is_subject = yes
						NOT = { is_subject_of = FROM }
					}
				}
				factor = 0
			}

			# Historical mode: only historical partners and our own subjects
			modifier = {
				ai_influence_historical_mode = yes
				NOT = { is_in_array = { FROM.influence_targets = this } }
				NOT = { is_subject_of = FROM }
				factor = 0
			}

			# Reach by power rank. Superpowers reach everywhere.
			modifier = {
				FROM = { has_idea = great_power }
				NOT = { is_in_array = { FROM.influence_targets = this } }
				NOT = { is_in_array = { FROM.neighbors = this } }
				NOT = { is_ally_with = FROM }
				NOT = { is_on_same_continent_as = FROM }
				factor = 0
			}
			modifier = {
				FROM = {
					OR = {
						has_idea = large_power
						has_idea = regional_power
					}
				}
				NOT = { is_in_array = { FROM.influence_targets = this } }
				NOT = { is_in_array = { FROM.neighbors = this } }
				NOT = {
					OR = {
						AND = {
							has_country_flag = european_nation_flag
							FROM = { has_country_flag = european_nation_flag }
						}
						AND = {
							has_country_flag = sub_saharan_nation_flag
							FROM = { has_country_flag = sub_saharan_nation_flag }
						}
						AND = {
							has_country_flag = middle_east_nation_flag
							FROM = { has_country_flag = middle_east_nation_flag }
						}
						AND = {
							has_country_flag = asian_nation_flag
							FROM = { has_country_flag = asian_nation_flag }
						}
						AND = {
							has_country_flag = american_nation_flag
							FROM = { has_country_flag = american_nation_flag }
						}
					}
				}
				factor = 0
			}
			modifier = {
				FROM = {
					NOT = {
						OR = {
							has_idea = super_power
							has_idea = great_power
							has_idea = large_power
							has_idea = regional_power
						}
					}
				}
				NOT = { is_in_array = { FROM.influence_targets = this } }
				NOT = { is_in_array = { FROM.neighbors = this } }
				factor = 0
			}

			# Strength: a weaker power rarely sways a stronger one
			modifier = {
				OR = {
					AND = {
						has_idea = super_power
						FROM = { NOT = { has_idea = super_power } }
					}
					AND = {
						has_idea = great_power
						FROM = {
							NOT = {
								OR = {
									has_idea = super_power
									has_idea = great_power
								}
							}
						}
					}
					AND = {
						has_idea = large_power
						FROM = {
							NOT = {
								OR = {
									has_idea = super_power
									has_idea = great_power
									has_idea = large_power
								}
							}
						}
					}
					AND = {
						has_idea = regional_power
						FROM = {
							NOT = {
								OR = {
									has_idea = super_power
									has_idea = great_power
									has_idea = large_power
									has_idea = regional_power
								}
							}
						}
					}
				}
				factor = 0.05
			}
			modifier = {
				OR = {
					AND = {
						has_idea = super_power
						FROM = { has_idea = super_power }
					}
					AND = {
						has_idea = great_power
						FROM = { has_idea = great_power }
					}
					AND = {
						has_idea = large_power
						FROM = { has_idea = large_power }
					}
					AND = {
						has_idea = regional_power
						FROM = { has_idea = regional_power }
					}
				}
				factor = 0.5
			}

			# Formal allies are only worth the spend when someone else is pulling them away
			modifier = {
				OR = {
					is_ally_with = FROM
					ai_has_same_organization = yes
				}
				factor = 0.3
			}
			modifier = {
				OR = {
					is_ally_with = FROM
					ai_has_same_organization = yes
				}
				OR = {
					AND = {
						check_variable = { influence_array^0 > 0 }
						NOT = { check_variable = { influence_array^0 = FROM } }
						var:influence_array^0 = { NOT = { is_ally_with = FROM } }
					}
					AND = {
						check_variable = { domestic_influence_amount < 60 }
						NOT = { check_variable = { influence_array^0 = FROM } }
						NOT = { check_variable = { influence_array^1 = FROM } }
						NOT = { check_variable = { influence_array^2 = FROM } }
					}
					has_opinion = { target = FROM value < 25 }
				}
				# Undoes the 0.3 above and nets 2.5
				factor = 8.3
			}

			modifier = {
				is_in_array = { FROM.influence_targets = this }
				factor = 2
			}

			modifier = {
				OR = {
					check_variable = { FROM.ai_strategy_influence@THIS > 0 }
					check_variable = { FROM.ai_strategy_befriend@THIS > 0 }
					check_variable = { FROM.ai_strategy_protect@THIS > 0 }
					check_variable = { FROM.ai_strategy_support@THIS > 0 }
				}
				factor = 1.5
			}

			modifier = {
				OR = {
					check_variable = { FROM.ai_strategy_antagonize@THIS > 0 }
					check_variable = { FROM.ai_strategy_contain@THIS > 0 }
				}
				factor = 0.5
			}

			modifier = {
				has_government = FROM
				factor = 1.2
			}

			modifier = {
				has_opinion = { target = FROM value > 50 }
				factor = 1.2
			}

			modifier = {
				has_opinion = { target = FROM value < -50 }
				factor = 0.6
			}

			modifier = {
				OR = {
					check_variable = { influence_array^0 = FROM }
					check_variable = { influence_array^1 = FROM }
					check_variable = { influence_array^2 = FROM }
				}
				factor = 1.3
			}

			# Diminishing returns once we hold 80% and have nothing left to do with it
			modifier = {
				check_variable = { influence_array^0 = FROM }
				check_variable = { influence_array_val^0 > 79.999 }
				OR = {
					NOT = { has_government = FROM }
					is_subject_of = FROM
					ai_influence_historical_mode = yes
				}
				factor = 0.2
			}

			# A rival government leads here: worth contesting
			modifier = {
				check_variable = { influence_array^0 > 0 }
				NOT = { check_variable = { influence_array^0 = FROM } }
				var:influence_array^0 = { NOT = { has_government = FROM } }
				factor = 1.3
			}

			modifier = {
				check_variable = { total_resource_amount > 9.999 }
				factor = 1.2
			}

			modifier = {
				has_war = yes
				NOT = { is_ally_with = FROM }
				factor = 0.5
			}

			modifier = {
				has_dlc = "La Resistance"
				FROM = { check_variable = { civilian_intel@PREV > 50 } }
				factor = 1.2
			}

			modifier = {
				has_idea = christian
				FROM = { has_idea = HLS_idea_holy_see }
				factor = 1.3
			}

			modifier = {
				is_ai = no
				OR = {
					difficulty > 2
					has_game_rule = {
						rule = rule_god_of_war
						option = yes
					}
				}
				factor = 1.3
			}

			# Keep what we are already working on
			modifier = {
				FROM = { is_in_array = { auto_influence_array = PREV.id } }
				factor = 2
			}
		}
	}
}
