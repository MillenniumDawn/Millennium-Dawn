# MD copy of vanilla's decryption target scorer. THIS is the target, FROM is the decrypting country.
# Rivalries come from the antagonize, contain, and conquer strategies, not hardcoded tags.

ai_decryption_target_scorer = {
	targets = {
		target_array = potential_and_current_enemies
		target_array = enemies_of_allies
		target_array = neighbors
		target_array = global.majors

		score = {
			base = 100

			modifier = {
				has_capitulated = yes
				factor = 0
			}

			modifier = {
				add = FROM.ai_strategy_decrypt_target@THIS
			}

			# Keep going on a decryption already underway
			modifier = {
				FROM = { is_decrypting = PREV }
				add = 20
			}

			modifier = {
				FROM = {
					is_decrypting = PREV
					check_variable = { decryption_progress@PREV > 0.25 }
				}
				add = 25
			}

			modifier = {
				FROM = {
					is_decrypting = PREV
					check_variable = { decryption_progress@PREV > 0.5 }
				}
				add = 25
			}

			modifier = {
				FROM = {
					is_decrypting = PREV
					check_variable = { decryption_progress@PREV > 0.75 }
				}
				add = 25
			}

			modifier = {
				has_war_with = FROM
				factor = 2.0
			}

			modifier = {
				OR = {
					is_in_array = { FROM.potential_and_current_enemies = this }
					is_in_array = { FROM.enemies_of_allies = this }
				}
				factor = 1.5
			}

			modifier = {
				NOT = { has_war_with = FROM }
				OR = {
					check_variable = { FROM.ai_strategy_conquer@THIS > 0 }
					check_variable = { FROM.ai_strategy_antagonize@THIS > 0 }
					check_variable = { FROM.ai_strategy_declare_war@THIS > 0 }
					check_variable = { FROM.ai_strategy_prepare_for_war@THIS > 0 }
					check_variable = { FROM.ai_strategy_contain@THIS > 0 }
				}
				factor = 1.3
			}

			modifier = {
				is_in_array = { FROM.neighbors = this }
				factor = 1.2
			}

			modifier = {
				OR = {
					has_idea = super_power
					has_idea = great_power
				}
				factor = 1.2
			}

			modifier = {
				OR = {
					has_idea = large_power
					has_idea = regional_power
				}
				factor = 1.1
			}

			modifier = {
				NOT = { has_war_with = FROM }
				ai_has_same_organization = yes
				factor = 0.5
			}

			modifier = {
				has_government = FROM
				NOT = { has_war_with = FROM }
				factor = 0.6
			}

			modifier = {
				check_variable = { num_owned_states > 3 }
				factor = 1.2
			}
		}
	}
}
