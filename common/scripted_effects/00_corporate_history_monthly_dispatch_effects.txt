corporate_history_monthly_dispatch = {
	if = {
		limit = {
			corporate_history_enabled = yes
			NOT = { has_country_flag = collapsed_nation }
			OR = {
				original_tag = USA
				original_tag = CHI
				original_tag = JAP
				original_tag = CAN
				original_tag = FIN
				original_tag = GER
				original_tag = SWE
				original_tag = TAI
				original_tag = POL
				original_tag = FRA
				original_tag = ENG
				original_tag = UKR
				original_tag = SOV
			}
		}

		if = {
			limit = { NOT = { has_country_flag = corporate_history_country_bootstrapped } }
			corporate_history_country_bootstrap = yes
			corporate_history_initialize_midyear_recovery = yes
			set_country_flag = corporate_history_country_bootstrapped
		}

		if = {
			limit = { corporate_history_full_enabled = yes }
			if = {
				limit = {
					date > 1999.12.31
					date < 2000.2.1
					check_variable = { corporate_history_last_dispatched_year < 2000 }
				}
				set_variable = { corporate_history_last_dispatched_year = 2000 }
				corporate_history_dispatch_year_2000 = yes
			}
			if = {
				limit = {
					date > 2000.12.31
					date < 2001.2.1
					check_variable = { corporate_history_last_dispatched_year < 2001 }
				}
				set_variable = { corporate_history_last_dispatched_year = 2001 }
				corporate_history_dispatch_year_2001 = yes
