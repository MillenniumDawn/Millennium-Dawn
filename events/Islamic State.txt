					NUS = { country_event = { days = 30 id = isis.3 } }
				}
			}
			30 = {
				custom_effect_tooltip = isis.2.a_tt

			}
		}
	}

	# Avoid Provocations
	option = {
		name = isis.2.b
		log = "[GetDateText]: [This.GetName]: isis.2.b executed"
		ai_chance = {
			base = 0
		}
	}
}
