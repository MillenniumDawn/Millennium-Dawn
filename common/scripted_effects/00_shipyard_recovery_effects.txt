	set_variable = { shipyard_growth_start = shipyard_crisis_level }
	set_variable = { shipyard_cap_output = 0.30 }
	set_variable = { shipyard_cap_repair = 0.50 }

POL_shipyard_sync_destroyers = {
	if = {
		limit = {
			NOT = { has_completed_focus = POL_destroyer_development }
			check_variable = { POL_shipyard_destroyer_penalty = 0 }
		}
		add_equipment_bonus = {
			name = POL_shipyard_destroyer_equipment
			bonus = {
				destroyer = { instant = yes build_cost_ic = 1.0 }
				stealth_destroyer = { instant = yes build_cost_ic = 1.0 }
			}
		}
		set_variable = { POL_shipyard_destroyer_penalty = 1 }
	}
	else_if = {
		limit = {
			has_completed_focus = POL_destroyer_development
			check_variable = { POL_shipyard_destroyer_penalty = 1 }
		}
		add_equipment_bonus = {
			name = POL_shipyard_destroyer_equipment
			bonus = {
				destroyer = { instant = yes build_cost_ic = -1.0 }
				stealth_destroyer = { instant = yes build_cost_ic = -1.0 }
			}
		}
		set_variable = { POL_shipyard_destroyer_penalty = 0 }
	}
}
