# Registry and capability
# The race owns only AI_RACE/ai_race state. National capability is rebuilt from AI technology
# and never written back.

ai_race_clear_country_state = {
	clear_variable = ai_race_capability_stock
	clear_variable = ai_race_capability_external
	clear_variable = ai_race_capability_funded_usable
	clear_variable = ai_race_capability
	clear_variable = ai_race_frontier_gap
	clear_variable = ai_race_rank
	clr_country_flag = AI_RACE_state_initialized
	ai_race_clear_progression_state = yes
	ai_race_clear_capacity_state = yes
	ai_race_clear_finance_state = yes
	clear_variable = ai_race_effective_capability
	clear_variable = ai_race_gui_dirty
	clear_variable = ai_race_gui_hint_y
}

ai_race_repair_country_state = {
	if = {
		limit = {
			ai_race_enabled = yes
			ai_race_country_initialized = yes
		}
		if = { limit = { has_variable = ai_race_capability_stock } clamp_variable = { var = ai_race_capability_stock min = 0 max = 100 } }
		if = { limit = { has_variable = ai_race_capability_external } clamp_variable = { var = ai_race_capability_external min = 0 max = 100 } }
		if = { limit = { has_variable = ai_race_capability } clamp_variable = { var = ai_race_capability min = 0 max = 100 } }
		if = { limit = { has_variable = ai_race_frontier_gap } clamp_variable = { var = ai_race_frontier_gap min = -100 max = 0 } }
		if = { limit = { has_variable = ai_race_rank } clamp_variable = { var = ai_race_rank min = 0 max = global.ai_race_all_initialized^num } }
		if = { limit = { has_variable = ai_race_stage } clamp_variable = { var = ai_race_stage min = 0 max = 4 } }
		if = { limit = { has_variable = ai_race_stage_months } clamp_variable = { var = ai_race_stage_months min = 0 } }
	}
}

ai_race_initialize_country = {
	if = {
		limit = { ai_race_eligible_participant = yes }
		if = {
			limit = { NOT = { is_in_array = { global.ai_race_all_initialized = THIS } } }
			add_to_array = { global.ai_race_all_initialized = THIS }
			unlock_decision_category_tooltip = AI_RACE_category
		}
		set_country_flag = AI_RACE_state_initialized
		if = {
			limit = { NOT = { is_in_array = { global.ai_race_participants = THIS } } }
			add_to_array = { global.ai_race_participants = THIS }
		}
		ai_race_repair_country_state = yes
	}
}

ai_race_rebuild_country_metrics = {
	if = {
		limit = { ai_race_active = yes }
		# Each AI technology tier adds 5 national capability.
		ai_race_count_ai_technology = yes
		set_variable = { ai_race_capability_external = ai_race_generic_technology }
		set_variable = { ai_race_capability = { value = ai_race_capability_stock add = ai_race_capability_external } }
		ai_race_repair_country_state = yes
	}
}

ai_race_refresh_participants = {
	clear_array = global.ai_race_participants
	for_each_loop = {
		array = global.ai_race_all_initialized
		value = ai_race_participant
		var:ai_race_participant = {
			clear_variable = ai_race_rank
			if = {
				limit = { ai_race_active = yes }
				add_to_array = { global.ai_race_participants = THIS }
			}
		}
	}
}

ai_race_rebuild_frontier_and_ranking = {
	clear_array = global.ai_race_ranked_participants
	clear_variable = global.ai_race_frontier_capability
	set_temp_variable = { ai_race_sort_for_purchase = 0 }
	ai_race_sort_participants = yes
	set_temp_variable = { ai_race_next_rank = 1 }
	for_each_loop = {
		array = global.ai_race_sorted_work
		value = ai_race_ranked_country
		var:ai_race_ranked_country = {
			add_to_array = { global.ai_race_ranked_participants = THIS }
			set_variable = { ai_race_rank = ai_race_next_rank }
			if = {
				limit = { check_variable = { ai_race_effective_capability > global.ai_race_frontier_capability } }
				set_variable = { global.ai_race_frontier_capability = ai_race_effective_capability }
			}
			add_to_temp_variable = { ai_race_next_rank = 1 }
		}
	}
	clear_array = global.ai_race_sorted_work
	for_each_loop = {
		array = global.ai_race_participants
		value = ai_race_participant
		var:ai_race_participant = {
			set_variable = { ai_race_frontier_gap = ai_race_effective_capability }
			subtract_from_variable = { ai_race_frontier_gap = global.ai_race_frontier_capability }
			ai_race_repair_country_state = yes
			ai_race_refresh_gui = yes
		}
	}
	ai_race_bump_view = yes
}

ai_race_repair_global_state = {
	if = {
		limit = {
			ai_race_enabled = yes
			has_global_flag = GLOBAL_ai_race_initialized
		}
		if = { limit = { has_variable = global.ai_race_frontier_capability } clamp_variable = { var = global.ai_race_frontier_capability min = 0 max = 100 } }
		if = { limit = { has_variable = global.ai_race_epoch } clamp_variable = { var = global.ai_race_epoch min = 0 } }
	}
}

ai_race_initialize_global = {
	if = {
		limit = {
			ai_race_enabled = yes
			NOT = { has_global_flag = GLOBAL_ai_race_initialized }
		}
		clear_array = global.ai_race_participants
		clear_array = global.ai_race_ranked_participants
		clear_variable = global.ai_race_frontier_capability
		clear_variable = global.ai_race_epoch
		set_global_flag = GLOBAL_ai_race_initialized
		# Publish a current projection on the first eligible pulse without advancing scheduled state.
		ai_race_rebuild_derived_state = yes
	}
}

ai_race_rebuild_derived_state = {
	if = {
		limit = {
			ai_race_enabled = yes
			has_global_flag = GLOBAL_ai_race_initialized
		}
		ai_race_refresh_mode_state = yes
		ai_race_refresh_participants = yes
		for_each_loop = {
			array = global.ai_race_participants
			value = ai_race_participant
			var:ai_race_participant = {
				ai_race_rebuild_country_metrics = yes
				ai_race_refresh_operating_state = yes

				ai_race_refresh_effective_capability = yes
			}
		}
		ai_race_rebuild_frontier_and_ranking = yes
		ai_race_repair_global_state = yes
	}
}

	clear_array = global.ai_race_ranked_participants
	clear_array = global.ai_race_sorted_work
	clear_array = global.ai_race_sort_input

	clear_variable = global.ai_race_last_processed_month
	clear_variable = global.ai_race_last_discovery_year
	clr_global_flag = GLOBAL_ai_race_initialized
}

ai_race_monthly_dispatch = {
	if = {
		limit = { ai_race_enabled = yes }
		ai_race_initialize_global = yes
		ai_race_discover_participants = yes
		ai_race_ai_discover_candidates = yes
		ai_race_get_month_token = yes
		if = {
			limit = { NOT = { check_variable = { global.ai_race_last_processed_month = ai_race_current_month } } }
			ai_race_refresh_mode_state = yes
			ai_race_refresh_participants = yes
			ai_race_run_monthly_work = yes
			set_variable = { global.ai_race_last_processed_month = ai_race_current_month }
		}
		if = {
			limit = {
				OR = {
					check_variable = { global.month = 1 }
					check_variable = { global.month = 4 }
					check_variable = { global.month = 7 }
					check_variable = { global.month = 10 }
				}
			}
			set_temp_variable = { ai_race_current_quarter = global.year }
			multiply_temp_variable = { ai_race_current_quarter = 4 }
			if = {
				limit = { check_variable = { global.month = 1 } }
				add_to_temp_variable = { ai_race_current_quarter = 1 }
			}
			else_if = {
				limit = { check_variable = { global.month = 4 } }
				add_to_temp_variable = { ai_race_current_quarter = 2 }
			}
			else_if = {
				limit = { check_variable = { global.month = 7 } }
				add_to_temp_variable = { ai_race_current_quarter = 3 }
			}
			else = {
				add_to_temp_variable = { ai_race_current_quarter = 4 }
			}
			if = {
				limit = {
					NOT = { check_variable = { global.ai_race_last_processed_quarter = ai_race_current_quarter } }
				}
				ai_race_quarterly_reconcile = yes
				set_variable = { global.ai_race_last_processed_quarter = ai_race_current_quarter }
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				has_global_flag = GLOBAL_ai_race_initialized
				check_variable = { global.ai_race_all_initialized^num > 0 }
				check_variable = { global.ai_race_ai_countries^num > 0 }
			}
		}
		ai_race_teardown_global = yes
	}
}

# Apply readiness to funded stock before the final cap, preserving the external base.
ai_race_refresh_effective_capability = {
	set_variable = { ai_race_effective_capability = ai_race_capability }
	set_variable = { ai_race_capability_funded_usable = 0 }
	if = {
		limit = {
			ai_race_full_mode = yes
			check_variable = { ai_race_stage > 0 }
		}
		set_variable = { ai_race_effective_capability = ai_race_capability_external }
		set_variable = { ai_race_capability_funded_usable = { value = ai_race_capability_stock multiply = ai_race_current_readiness } }

	}

}

ai_race_discover_participants = {
	if = {
		limit = {
			date > 2015.12.31
			OR = {
				NOT = { has_variable = global.ai_race_last_discovery_year }
				AND = {
					check_variable = { global.month = 1 }
					NOT = { check_variable = { global.ai_race_last_discovery_year = global.year } }

			value = ai_race_prospective_country

		}

	}
}

ai_race_sort_participants = {
	clear_array = global.ai_race_sorted_work
	clear_array = global.ai_race_sort_input
	for_each_loop = {
		array = global.ai_race_participants
		value = ai_race_sort_country
		var:ai_race_sort_country = {
			if = {
				limit = {
					OR = {
						check_variable = { ai_race_sort_for_purchase = 0 }
						AND = {
							is_ai = yes
							ai_race_can_review = yes
						}
					}
				}
				add_to_array = { global.ai_race_sort_input = THIS }
			}
		}
	}
	while_loop_effect = {
		limit = { check_variable = { global.ai_race_sort_input^num > 0 } }
		set_temp_variable = { ai_race_candidate_found = 0 }
		for_each_loop = {
			array = global.ai_race_sort_input
			value = ai_race_sort_country
			var:ai_race_sort_country = {
				set_temp_variable = { ai_race_this_id = THIS.id }
				if = {
					limit = { ai_race_candidate_is_better = yes }
					set_temp_variable = { ai_race_candidate_found = 1 }
					set_temp_variable = { ai_race_candidate_id = THIS.id }
					set_temp_variable = { ai_race_candidate_stage = ai_race_stage }
					set_temp_variable = { ai_race_candidate_months = ai_race_stage_months }
					set_temp_variable = { ai_race_candidate_capability = ai_race_effective_capability }
				}
			}
		}
		var:ai_race_candidate_id = {
			add_to_array = { global.ai_race_sorted_work = THIS }
			remove_from_array = { global.ai_race_sort_input = THIS }
		}
	}
}

ai_race_refresh_capacity_sample = {
	set_variable = {
		ai_race_civs_available = {
			value = 0
			every_collection = {
				named_collection = controlled_states_collection
				add = non_damaged_building_level@industrial_complex
			}
		}
	}
	set_variable = {
		ai_race_network_available = {
			value = 0
			every_collection = {
				named_collection = controlled_states_collection
				add = non_damaged_building_level@internet_station
			}
		}
	}
	set_variable = { ai_race_capacity_settled = 0 }
	if = {
		limit = {
			has_country_flag = energy_state_bases_initialized
			has_variable = ai_race_power_multiplier
			OR = {
				NOT = { has_variable = ai_race_stage }
				check_variable = { ai_race_stage = 0 }
				AND = {
					check_variable = { var = global.num_days value = ai_race_demand_settle_day compare = greater_than_or_equals }
					check_variable = { modifier@ai_race_applied_power = ai_race_demand_power }
					check_variable = { modifier@ai_race_applied_microchips = ai_race_demand_microchips }
					check_variable = { modifier@ai_race_applied_composites = ai_race_demand_composites }
					check_variable = { modifier@ai_race_applied_oil = ai_race_demand_oil }
				}
			}
		}
		set_variable = { ai_race_capacity_settled = 1 }
		set_variable = { ai_race_accounted_microchips = modifier@ai_race_applied_microchips }
		set_variable = { ai_race_accounted_composites = modifier@ai_race_applied_composites }
		set_variable = { ai_race_accounted_oil = modifier@ai_race_applied_oil }
	}
	else = {
		# An unsettled modifier never supplies an optimistic add-back after a purchase or trade loss.
		set_variable = { ai_race_accounted_microchips = 0 }
		set_variable = { ai_race_accounted_composites = 0 }
		set_variable = { ai_race_accounted_oil = 0 }
	}
	set_variable = { ai_race_microchips_available = { value = resource@microchips add = ai_race_accounted_microchips max = 0 } }
	set_variable = { ai_race_composites_available = { value = resource@composites add = ai_race_accounted_composites max = 0 } }
	set_variable = { ai_race_oil_available = { value = resource@oil add = ai_race_accounted_oil max = 0 } }
}

# Outcomes Only reads no capacity state, so only Full mode samples it.
ai_race_refresh_operating_state = {
	if = {
		limit = { ai_race_full_mode = yes }
		if = {
			limit = { has_country_flag = energy_state_bases_initialized }

# Freeze the charge immediately before the actual weekly treasury accrual.
ai_race_prepare_weekly_bill = {
	set_temp_variable = { ai_race_bill_prepared = 0 }
	set_temp_variable = { ai_race_bill_total = 0 }
	if = {
		limit = {
			ai_race_full_mode = yes
			check_variable = { ai_race_finance_outstanding > 0 }
			OR = {
				NOT = { has_variable = ai_race_finance_next_bill_day }
				check_variable = { var = global.num_days value = ai_race_finance_next_bill_day compare = greater_than_or_equals }
			}
		}
		ai_race_refresh_finance_totals = yes
		set_temp_variable = { ai_race_bill_prepared = 1 }
		set_temp_variable = { ai_race_bill_1 = ai_race_finance_due_1 }
		set_temp_variable = { ai_race_bill_2 = ai_race_finance_due_2 }
		set_temp_variable = { ai_race_bill_3 = ai_race_finance_due_3 }
		set_temp_variable = { ai_race_bill_4 = ai_race_finance_due_4 }
		set_temp_variable = { ai_race_bill_total = { value = ai_race_bill_1 add = ai_race_bill_2 add = ai_race_bill_3 add = ai_race_bill_4 } }
	}
	# Correct only the race component, including a repeated callback after the final installment.
	set_temp_variable = { treasury_rate_gain = { value = treasury_rate_gain add = ai_race_finance_rate_component subtract = ai_race_bill_total } }
}

ai_race_commit_weekly_bill = {
	if = {
		limit = { check_variable = { ai_race_bill_prepared = 1 } }
		if = {
			limit = { check_variable = { ai_race_finance_installments_1 > 0 } }
			subtract_from_variable = { ai_race_finance_remaining_1 = ai_race_bill_1 }
			subtract_from_variable = { ai_race_finance_installments_1 = 1 }
			if = {
				limit = { check_variable = { ai_race_finance_installments_1 = 0 } }
				set_variable = { ai_race_finance_remaining_1 = 0 }
			}
		}
		if = {
			limit = { check_variable = { ai_race_finance_installments_2 > 0 } }
			subtract_from_variable = { ai_race_finance_remaining_2 = ai_race_bill_2 }
			subtract_from_variable = { ai_race_finance_installments_2 = 1 }
			if = {
				limit = { check_variable = { ai_race_finance_installments_2 = 0 } }
				set_variable = { ai_race_finance_remaining_2 = 0 }
			}
		}
		if = {
			limit = { check_variable = { ai_race_finance_installments_3 > 0 } }
			subtract_from_variable = { ai_race_finance_remaining_3 = ai_race_bill_3 }
			subtract_from_variable = { ai_race_finance_installments_3 = 1 }
			if = {
				limit = { check_variable = { ai_race_finance_installments_3 = 0 } }
				set_variable = { ai_race_finance_remaining_3 = 0 }
			}
		}
		if = {
			limit = { check_variable = { ai_race_finance_installments_4 > 0 } }
			subtract_from_variable = { ai_race_finance_remaining_4 = ai_race_bill_4 }
			subtract_from_variable = { ai_race_finance_installments_4 = 1 }
			if = {
				limit = { check_variable = { ai_race_finance_installments_4 = 0 } }
				set_variable = { ai_race_finance_remaining_4 = 0 }
			}
		}
		set_variable = { ai_race_finance_next_bill_day = { value = global.num_days add = 7 } }
		ai_race_refresh_finance_totals = yes
		ai_race_refresh_gui = yes
		set_temp_variable = { ai_race_bill_prepared = 0 }
	}
}

				}
				AND = {
					check_variable = { ai_race_requested_financing = 1 }
					ai_race_can_finance = yes
				}
			}
		}
		set_temp_variable = { treasury_change = ai_race_quote_cost }
		multiply_temp_variable = { treasury_change = -1 }
		modify_treasury_effect = yes
		set_variable = { ai_race_stage = ai_race_quote_stage }

		clear_variable = ai_race_stage_months
		ai_race_get_month_token = yes
		set_variable = { ai_race_stage_entry_month = ai_race_current_month }
		set_variable = { ai_race_last_progress_month = ai_race_current_month }
		if = {
			limit = { check_variable = { ai_race_requested_financing = 1 } }
			ai_race_create_financing = yes
		}
		if = {
			limit = { check_variable = { ai_race_stage = 4 } }
			if = {
				limit = { NOT = { has_variable = global.ai_race_first_finisher_id } }
				set_variable = { global.ai_race_first_finisher_id = THIS.id }
				ai_race_bump_view = yes
				news_event = ai_race_news.1
			}
			if = {
				limit = { is_ai = no }
				country_event = ai_race.10
			}
		}
		ai_race_defer_review = yes
		ai_race_refresh_operating_state = yes
		ai_race_rebuild_country_metrics = yes
		ai_race_refresh_offer = yes
		ai_race_refresh_effective_capability = yes
	}
	else = {
		ai_race_defer_review = yes
		if = {
			limit = { is_ai = no }
			country_event = ai_race.11
		}
	}
}

					check_variable = { ai_race_ai_old_target = ai_race_stage }

						AND = {

							check_variable = { ai_race_ai_power_ratio < 1.1 }
						}
						AND = {
							check_variable = { ai_race_ai_microchips_latched = 1 }
							check_variable = { ai_race_ai_microchips_ratio < 1.1 }
						}
						AND = {
							check_variable = { ai_race_ai_composites_latched = 1 }
							check_variable = { ai_race_ai_composites_ratio < 1.1 }
						}
					}
				}
			}
		}
		set_variable = { ai_race_ai_recovery = 1 }
	}
	else_if = {
		limit = { check_variable = { ai_race_stage < 4 } }
		set_variable = { ai_race_ai_earliest_entry_month = 24193 }

				max = ai_race_ai_finish_month
			}

				}
			}

		}
	}
	if = {
		limit = { NOT = { check_variable = { ai_race_ai_target_stage = ai_race_ai_old_target } } }

ai_race_ai_calculate_savings = {
	set_variable = { ai_race_ai_savings_target = 0 }
	if = {
		limit = {
			ai_race_ai_expansion_permitted = yes
			check_variable = { ai_race_ai_target_stage > ai_race_stage }
			check_variable = { var = ai_race_ai_readiness value = 0.75 compare = greater_than_or_equals }
		}
		set_variable = {
			ai_race_ai_savings_target = {
				value = display_expense
				max = 0
				multiply = 52
				add = ai_race_ai_target_cost
			}
		}
		if = {
			limit = { check_variable = { ai_race_ai_readiness < 1 } }
			set_temp_variable = {
				ai_race_ai_possible_payment = {
					value = 1
					subtract = ai_race_ai_readiness
					multiply = 0.4
					add = 0.45
					multiply = gdp_total
					divide = 520
				}
			}
			set_temp_variable = { ai_race_ai_projected_balance = { value = treasury_rate subtract = ai_race_ai_possible_payment } }
			set_temp_variable = { ai_race_ai_debt_limit = { value = gdp_total multiply = 0.6 } }
			if = {
				limit = {
					check_variable = { debt < ai_race_ai_debt_limit }
					check_variable = { var = ai_race_ai_projected_balance value = 0 compare = greater_than_or_equals }
				}
				add_to_variable = { ai_race_ai_savings_target = { value = ai_race_ai_possible_payment multiply = 52 } }
			}
		}
	}
}

		ai_race_ai_civs_total = {
			value = 0
			every_collection = {
				named_collection = controlled_states_collection
				add = building_level@industrial_complex
			}
		}
	}
	set_variable = {
		ai_race_ai_network_total = {
			value = 0
			every_collection = {
				named_collection = controlled_states_collection
				add = building_level@internet_station
			}
		}
	}
	set_variable = {
		ai_race_ai_facilities = {
			value = 0
			every_collection = {
				named_collection = controlled_states_collection
				add = building_level@nuclear_facility
			}
		}
	}
	set_variable = { ai_race_ai_slots_civs = 0 }
	set_variable = { ai_race_ai_slots_network = 0 }
	set_variable = { ai_race_ai_slots_fossil = 0 }
	set_variable = { ai_race_ai_slots_renewable = 0 }
	set_variable = { ai_race_ai_slots_nuclear = 0 }
	set_variable = { ai_race_ai_slots_microchips = 0 }
	set_variable = { ai_race_ai_slots_composites = 0 }
	set_variable = { ai_race_ai_slots_refinery = 0 }
	set_variable = { ai_race_ai_slots_facility = 0 }
	for_each_scope_loop = {
		array = owned_controlled_states
		if = {
			limit = { free_building_slots = { building = industrial_complex size > 0 } }
			set_variable = { PREV.ai_race_ai_slots_civs = 1 }
		}
		if = {
			limit = { free_building_slots = { building = internet_station size > 0 } }
			set_variable = { PREV.ai_race_ai_slots_network = 1 }
		}
		if = {
			limit = { free_building_slots = { building = fossil_powerplant size > 0 } }
			set_variable = { PREV.ai_race_ai_slots_fossil = 1 }
		}
		if = {
			limit = { free_building_slots = { building = renewable_energy_infra size > 0 } }
			set_variable = { PREV.ai_race_ai_slots_renewable = 1 }
		}
		if = {
			limit = { free_building_slots = { building = nuclear_reactor size > 0 } }
			set_variable = { PREV.ai_race_ai_slots_nuclear = 1 }
		}
		if = {
			limit = { free_building_slots = { building = microchip_plant size > 0 } }
			set_variable = { PREV.ai_race_ai_slots_microchips = 1 }
		}
		if = {
			limit = { free_building_slots = { building = composite_plant size > 0 } }
			set_variable = { PREV.ai_race_ai_slots_composites = 1 }
		}
		if = {
			limit = { free_building_slots = { building = synthetic_refinery size > 0 } }
			set_variable = { PREV.ai_race_ai_slots_refinery = 1 }
		}
	}
	# The native facility query requires the planning country as ROOT.
	country_event = ai_race.12
	set_variable = { ai_race_ai_network_workers = { value = 12 multiply = { value = 1 add = modifier@buildings_worker_requirement_modifier add = modifier@infrastructure_workforce_requirement_modifier max = 0.1 } } }
	set_variable = { ai_race_ai_fossil_workers = { value = 7.5 multiply = { value = 1 add = modifier@buildings_worker_requirement_modifier add = modifier@infrastructure_workforce_requirement_modifier max = 0.1 } } }
	set_variable = { ai_race_ai_renewable_workers = { value = 22.5 multiply = { value = 1 add = modifier@buildings_worker_requirement_modifier add = modifier@infrastructure_workforce_requirement_modifier max = 0.1 } } }
	set_variable = { ai_race_ai_nuclear_workers = { value = 115 multiply = { value = 1 add = modifier@buildings_worker_requirement_modifier max = 0.1 } } }
	set_variable = { ai_race_ai_microchip_plant_power = { value = 1 add = modifier@energy_use_modifier_microchip_plant max = 0 multiply = 0.75 multiply = ai_race_power_multiplier } }
	set_variable = { ai_race_ai_composite_plant_power = { value = 1 add = modifier@energy_use_modifier_composite_plant max = 0 multiply = 0.8 multiply = ai_race_power_multiplier } }
	set_variable = { ai_race_ai_civilian_factory_power = { value = 1 add = modifier@energy_use_modifier_civs max = 0 multiply = 0.5 multiply = ai_race_power_multiplier } }
	set_variable = { ai_race_ai_refinery_power = { value = 1 add = modifier@energy_use_modifier_synthetic_refinery max = 0 multiply = 0.2 multiply = ai_race_power_multiplier } }
	set_variable = { ai_race_ai_fossil_fuel_reserve = { value = 1 add = modifier@fossil_pp_fuel_consumption_modifier max = 0 multiply = 20 subtract = fossil_pp_fuel_consumption multiply = 24 multiply = 28 multiply = 0.001 } }
	set_variable = { ai_race_ai_new_reactor_fuel = { value = 1 add = modifier@nuclear_fuel_consumption_modifier max = 0.0001 multiply = 40 } }
	set_variable = { ai_race_ai_reactor_fuel_reserve = { value = nuclear_fuel_consumption add = ai_race_ai_new_reactor_fuel multiply = 52 } }

	# Preparing one real supporting plant may require more power than the race's own demand.
	set_temp_variable = { ai_race_ai_added_plant_power = 0 }
	if = {
		limit = {
			check_variable = { ai_race_ai_civs_ratio < 1 }
			ai_race_ai_expansion_permitted = yes
			check_variable = { ai_race_ai_civs_total < ai_race_ai_civs_required }
			check_variable = { ai_race_ai_slots_civs > 0 }
			can_staff_an_industrial_complex = yes
		}
		set_temp_variable = { ai_race_ai_added_plant_power = { value = ai_race_ai_added_plant_power max = ai_race_ai_civilian_factory_power } }
	}
	if = {
		limit = {
			check_variable = { ai_race_ai_microchips_ratio < 1 }
			check_variable = { ai_race_ai_slots_microchips > 0 }
			has_tech = microchip_production_1
			can_staff_an_microchip_plant = yes
			check_variable = { var = resource@tungsten value = 3 compare = greater_than_or_equals }
			check_variable = { var = resource@chromium value = 2 compare = greater_than_or_equals }
		}
		set_temp_variable = { ai_race_ai_added_plant_power = { value = ai_race_ai_added_plant_power max = ai_race_ai_microchip_plant_power } }
	}
	if = {
		limit = {
			check_variable = { ai_race_ai_composites_ratio < 1 }
			check_variable = { ai_race_ai_slots_composites > 0 }
			has_tech = composite_production_1
			can_staff_an_composite_plant = yes
			check_variable = { var = resource@rubber value = 1 compare = greater_than_or_equals }
			check_variable = { var = resource@chromium value = 1 compare = greater_than_or_equals }
			check_variable = { var = resource@oil value = 1 compare = greater_than_or_equals }
		}
		set_temp_variable = { ai_race_ai_added_plant_power = { value = ai_race_ai_added_plant_power max = ai_race_ai_composite_plant_power } }
	}
	if = {
		limit = {
			ai_race_ai_rubber_feedstock_needed = yes
			ai_race_ai_refinery_base_capacity = yes
			has_tech = fuel_silos
		}
		set_temp_variable = { ai_race_ai_added_plant_power = { value = ai_race_ai_added_plant_power max = ai_race_ai_refinery_power } }
	}
	set_variable = { ai_race_ai_power_build_required = { value = ai_race_ai_paid_power_required add = ai_race_ai_added_plant_power max = ai_race_ai_power_required } }
	set_variable = { ai_race_ai_power_build_ratio = 1 }
	if = {
		limit = { check_variable = { ai_race_ai_power_build_required > 0 } }
		set_variable = { ai_race_ai_power_build_ratio = { value = ai_race_power_available divide = ai_race_ai_power_build_required clamp = { min = 0 max = 2 } } }
	}
	ai_race_ai_choose_power_source = yes
}

ai_race_ai_choose_power_source = {
	set_variable = { ai_race_ai_power_source = 0 }
	set_temp_variable = { ai_race_ai_best_power_score = 0 }
	if = {
		limit = { ai_race_ai_can_add_fossil_power = yes }
		set_temp_variable = { ai_race_ai_power_score = { value = 1 add = modifier@fossil_pp_energy_generation_modifier max = 0 multiply = 2 multiply = { value = 1 add = modifier@production_speed_fossil_powerplant_factor max = 0 } multiply = 100000 divide = 9500 } }
		set_temp_variable = { ai_race_ai_best_power_score = ai_race_ai_power_score }
		set_variable = { ai_race_ai_power_source = 1 }
	}
	if = {
		limit = { ai_race_ai_can_add_renewable_power = yes }
		set_temp_variable = { ai_race_ai_power_score = { value = 1 add = modifier@renewable_energy_gain_multiplier max = 0 multiply = 0.25 multiply = { value = 1 add = modifier@production_speed_renewable_energy_infra_factor max = 0 } multiply = 100000 divide = 50000 } }
		if = {
			limit = { check_variable = { ai_race_ai_power_score > ai_race_ai_best_power_score } }
			set_temp_variable = { ai_race_ai_best_power_score = ai_race_ai_power_score }
			set_variable = { ai_race_ai_power_source = 2 }
		}
	}
	if = {
		limit = { ai_race_ai_can_add_nuclear_power = yes }
		set_temp_variable = { ai_race_ai_power_score = { value = 1 add = modifier@nuclear_energy_generation_modifier max = 0 multiply = 4 multiply = { value = 1 add = modifier@production_speed_nuclear_reactor_factor max = 0 } multiply = 100000 divide = 50000 } }
		if = {
			limit = { check_variable = { ai_race_ai_power_score > ai_race_ai_best_power_score } }
			set_variable = { ai_race_ai_power_source = 3 }
		}
	}
}

ai_race_ai_choose_construction = {
	set_variable = { ai_race_ai_priority = 0 }
	set_temp_variable = { ai_race_ai_best_shortfall = -1 }
	if = {
		limit = { check_variable = { ai_race_ai_power_build_ratio < 1 } }
		set_variable = { ai_race_ai_power_latched = 1 }
	}
	else_if = {
		limit = { check_variable = { var = ai_race_ai_power_build_ratio value = 1.1 compare = greater_than_or_equals } }
		set_variable = { ai_race_ai_power_latched = 0 }
	}
	if = {
		limit = { check_variable = { ai_race_ai_microchips_ratio < 1 } }
		set_variable = { ai_race_ai_microchips_latched = 1 }
	}
	else_if = {
		limit = { check_variable = { var = ai_race_ai_microchips_ratio value = 1.1 compare = greater_than_or_equals } }
		set_variable = { ai_race_ai_microchips_latched = 0 }
	}
	if = {
		limit = { check_variable = { ai_race_ai_composites_ratio < 1 } }
		set_variable = { ai_race_ai_composites_latched = 1 }
	}
	else_if = {
		limit = { check_variable = { var = ai_race_ai_composites_ratio value = 1.1 compare = greater_than_or_equals } }
		set_variable = { ai_race_ai_composites_latched = 0 }
	}
	if = {
		limit = {
			check_variable = { ai_race_ai_civs_ratio < 1 }
			ai_race_ai_can_add_civs = yes
		}
		set_temp_variable = { ai_race_ai_shortfall = { value = 1 subtract = ai_race_ai_civs_ratio max = 0 } }
		if = {
			limit = { check_variable = { ai_race_ai_shortfall > ai_race_ai_best_shortfall } }
			set_temp_variable = { ai_race_ai_best_shortfall = ai_race_ai_shortfall }
			set_variable = { ai_race_ai_priority = 1 }
		}
	}
	if = {
		limit = {
			check_variable = { ai_race_ai_network_ratio < 1 }
			ai_race_ai_can_add_network = yes
		}
		set_temp_variable = { ai_race_ai_shortfall = { value = 1 subtract = ai_race_ai_network_ratio max = 0 } }
		if = {
			limit = { check_variable = { ai_race_ai_shortfall > ai_race_ai_best_shortfall } }
			set_temp_variable = { ai_race_ai_best_shortfall = ai_race_ai_shortfall }
			set_variable = { ai_race_ai_priority = 2 }
		}
	}
	if = {
		limit = {
			check_variable = { ai_race_ai_power_latched = 1 }
			OR = {
				check_variable = { ai_race_ai_power_source > 0 }
				AND = { ai_race_ai_can_add_refinery = yes ai_race_ai_fossil_fuel_support_needed = yes }
			}
		}
		set_temp_variable = { ai_race_ai_shortfall = { value = 1 subtract = ai_race_ai_power_build_ratio max = 0 } }
		if = {
			limit = { check_variable = { ai_race_ai_shortfall > ai_race_ai_best_shortfall } }
			set_temp_variable = { ai_race_ai_best_shortfall = ai_race_ai_shortfall }
			set_variable = { ai_race_ai_priority = 3 }
		}
	}
	if = {
		limit = {
			check_variable = { ai_race_ai_microchips_latched = 1 }
			OR = {
				ai_race_ai_can_add_microchips = yes
				AND = { ai_race_ai_facility_needed = yes ai_race_ai_microchip_project_needed = yes }
			}
		}
		set_temp_variable = { ai_race_ai_shortfall = { value = 1 subtract = ai_race_ai_microchips_ratio max = 0 } }
		if = {
			limit = { check_variable = { ai_race_ai_shortfall > ai_race_ai_best_shortfall } }
			set_temp_variable = { ai_race_ai_best_shortfall = ai_race_ai_shortfall }
			set_variable = { ai_race_ai_priority = 4 }
		}
	}
	if = {
		limit = {
			check_variable = { ai_race_ai_composites_latched = 1 }
			OR = {
				ai_race_ai_can_add_composites = yes
				AND = { ai_race_ai_facility_needed = yes ai_race_ai_composite_project_needed = yes }
				AND = { ai_race_ai_can_add_refinery = yes ai_race_ai_rubber_feedstock_needed = yes }
			}
		}
		set_temp_variable = { ai_race_ai_shortfall = { value = 1 subtract = ai_race_ai_composites_ratio max = 0 } }
		if = {
			limit = { check_variable = { ai_race_ai_shortfall > ai_race_ai_best_shortfall } }
			set_temp_variable = { ai_race_ai_best_shortfall = ai_race_ai_shortfall }
			set_variable = { ai_race_ai_priority = 5 }
		}
	}
}


# Research tokens need runtime substitution; numeric selectors remain temporary inputs.
ai_race_ai_consider_support_tech = {
	if = {
		limit = { check_variable = { ai_race_ai_support_checked = 0 } }
		meta_effect = {
			text = {
				if = {
					limit = { NOT = { has_tech = [TECH] } }
					set_temp_variable = { ai_race_ai_support_checked = 1 }
					if = {
						limit = {
							can_research = [TECH]
							check_variable = { var = global.year value = ai_race_ai_considered_year compare = greater_than_or_equals }
						}
						set_variable = { ai_race_ai_support_tech = ai_race_ai_considered_code }
					}
				}
			}
			TECH = "[?ai_race_ai_considered_tech.GetTokenKey]"
		}
	}
}

}

ai_race_ai_refresh_readiness = {
	set_variable = { ai_race_ai_civs_ratio = 1 }
	if = {
		limit = { check_variable = { ai_race_ai_civs_required > 0 } }
		set_variable = { ai_race_ai_civs_ratio = { value = ai_race_civs_available divide = ai_race_ai_civs_required clamp = { min = 0 max = 2 } } }
	}
	set_variable = { ai_race_ai_network_ratio = 1 }
	if = {
		limit = { check_variable = { ai_race_ai_network_required > 0 } }
		set_variable = { ai_race_ai_network_ratio = { value = ai_race_network_available divide = ai_race_ai_network_required clamp = { min = 0 max = 2 } } }
	}
	set_variable = { ai_race_ai_power_ratio = 1 }
	if = {
		limit = { check_variable = { ai_race_ai_power_required > 0 } }
		set_variable = { ai_race_ai_power_ratio = { value = ai_race_power_available divide = ai_race_ai_power_required clamp = { min = 0 max = 2 } } }
	}
	set_variable = { ai_race_ai_microchips_ratio = 1 }
	if = {
		limit = { check_variable = { ai_race_ai_microchips_required > 0 } }
		set_variable = { ai_race_ai_microchips_ratio = { value = ai_race_microchips_available divide = ai_race_ai_microchips_required clamp = { min = 0 max = 2 } } }
	}
	set_variable = { ai_race_ai_composites_ratio = 1 }
	if = {
		limit = { check_variable = { ai_race_ai_composites_required > 0 } }
		set_variable = { ai_race_ai_composites_ratio = { value = ai_race_composites_available divide = ai_race_ai_composites_required clamp = { min = 0 max = 2 } } }
	}
	set_variable = { ai_race_ai_oil_ratio = 1 }
	if = {
		limit = { check_variable = { ai_race_ai_oil_required > 0 } }
		set_variable = { ai_race_ai_oil_ratio = { value = ai_race_oil_available divide = ai_race_ai_oil_required clamp = { min = 0 max = 2 } } }
	}
	if = {
		limit = { check_variable = { ai_race_ai_target_stage > 0 } NOT = { has_variable = ai_race_power_multiplier } }
		set_variable = { ai_race_ai_power_ratio = 0 }
	}
	set_variable = { ai_race_ai_readiness = { value = 1 min = ai_race_ai_civs_ratio min = ai_race_ai_network_ratio min = ai_race_ai_power_ratio min = ai_race_ai_microchips_ratio min = ai_race_ai_composites_ratio min = ai_race_ai_oil_ratio } }
}
