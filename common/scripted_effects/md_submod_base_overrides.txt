# common/scripted_effects/md_submod_base_overrides.txt
# Overrides base Millennium Dawn's buggy 10-year array shift and smoothing logic.
# Synchronizes all 10 graphs (GDP, GDP/C, Prod, Debt, Pop, Budget, Energy, Defense, Trade, Inflation).

# -------------------------------------------------------------
# 1. OVERRIDE BASE MD's 10-YEAR ARRAY SHIFT
# -------------------------------------------------------------
gdp_graph_shift_array = {
	# Ensure month 119 data is safely recorded before shifting
	set_variable = { debt_array^119 = debt }
	set_variable = { inflation_array^119 = current_infl_pct }
	set_variable = { pop_array^119 = population_total_m }
	set_variable = { workforce_array^119 = workforce_total_display }
	set_variable = { employment_array^119 = md_current_employment_rate }
	set_variable = { revenue_array^119 = md_current_monthly_revenue }
	set_variable = { expense_array^119 = md_current_monthly_expense }
	set_variable = { budget_balance_array^119 = md_current_monthly_balance }
	set_variable = { energy_gen_array^119 = md_current_energy_gen }
	set_variable = { energy_con_array^119 = md_current_energy_con }
	set_variable = { energy_bal_array^119 = md_current_energy_bal }
	set_variable = { defense_spend_array^119 = md_current_defense_spending }
	set_variable = { defense_pct_array^119 = md_current_defense_pct }
	set_variable = { trade_exports_array^119 = md_current_monthly_exports }
	set_variable = { trade_imports_array^119 = md_current_monthly_imports }
	set_variable = { trade_balance_array^119 = md_current_monthly_trade_bal }

	# Shift all 10 datasets left by exactly 12 months (1 full year)
	# Indices 12..119 (Years 2-10) move to 0..107 (Years 1-9)
	set_temp_variable = { new_index = 12 }
	for_loop_effect = {
		start = 0
		end = 108

		# Base MD Arrays
		set_variable = { gdp_array^v = gdp_array^new_index }
		set_variable = { gdpc_array^v = gdpc_array^new_index }
		set_variable = { prod_array^v = prod_array^new_index }

		# Submod Custom Arrays
		set_variable = { debt_array^v = debt_array^new_index }
		set_variable = { inflation_array^v = inflation_array^new_index }
		set_variable = { pop_array^v = pop_array^new_index }
		set_variable = { workforce_array^v = workforce_array^new_index }
		set_variable = { employment_array^v = employment_array^new_index }
		set_variable = { revenue_array^v = revenue_array^new_index }
		set_variable = { expense_array^v = expense_array^new_index }
		set_variable = { budget_balance_array^v = budget_balance_array^new_index }
		set_variable = { energy_gen_array^v = energy_gen_array^new_index }
		set_variable = { energy_con_array^v = energy_con_array^new_index }
		set_variable = { energy_bal_array^v = energy_bal_array^new_index }
		set_variable = { defense_spend_array^v = defense_spend_array^new_index }
		set_variable = { defense_pct_array^v = defense_pct_array^new_index }
		set_variable = { trade_exports_array^v = trade_exports_array^new_index }
		set_variable = { trade_imports_array^v = trade_imports_array^new_index }
		set_variable = { trade_balance_array^v = trade_balance_array^new_index }

		add_to_temp_variable = { new_index = 1 }
	}

	# Pad future unused slots (108..119) with the latest valid value (index 107).
	# This prevents base MD's min/max search from glitching while keeping active slots completely untouched.
	for_loop_effect = {
		start = 108
		end = 120

		set_variable = { gdp_array^v = gdp_array^107 }
		set_variable = { gdpc_array^v = gdpc_array^107 }
		set_variable = { prod_array^v = prod_array^107 }

		set_variable = { debt_array^v = debt_array^107 }
		set_variable = { inflation_array^v = inflation_array^107 }
		set_variable = { pop_array^v = pop_array^107 }
		set_variable = { workforce_array^v = workforce_array^107 }
		set_variable = { employment_array^v = employment_array^107 }
		set_variable = { revenue_array^v = revenue_array^107 }
		set_variable = { expense_array^v = expense_array^107 }
		set_variable = { budget_balance_array^v = budget_balance_array^107 }
		set_variable = { energy_gen_array^v = energy_gen_array^107 }
		set_variable = { energy_con_array^v = energy_con_array^107 }
		set_variable = { energy_bal_array^v = energy_bal_array^107 }
		set_variable = { defense_spend_array^v = defense_spend_array^107 }
		set_variable = { defense_pct_array^v = defense_pct_array^107 }
		set_variable = { trade_exports_array^v = trade_exports_array^107 }
		set_variable = { trade_imports_array^v = trade_imports_array^107 }
		set_variable = { trade_balance_array^v = trade_balance_array^107 }
	}

	# Set current end to 107. When the next monthly cycle arrives, it will increment to 108 and write the new month.
	set_variable = { current_gdp_graph_end = 107 }

	# Advance X-axis date scale by 1 year (e.g. 2000-2010 becomes 2001-2011)
	set_temp_variable = { tmp = date_scale^0 }
	add_to_temp_variable = { tmp = 1 }
	for_loop_effect = {
		start = 0
		end = 12

		set_variable = { date_scale^v = tmp }
		add_to_temp_variable = { tmp = 1 }
	}

	gdp_graph_update_dirty_var = yes
}

# -------------------------------------------------------------
# 2. OVERRIDE BASE MD's BUGGY SMOOTHING EFFECT
# Base MD's correct_gdp_graph erroneously overwrote current_gdp_graph_end
# whenever monthly changes exceeded 5%. We neutralize it here.
# -------------------------------------------------------------
correct_gdp_graph = {
	# Neutralized by MD_Graphs to prevent economic data distortion
	set_temp_variable = { _neutralized = 1 }
}

# -------------------------------------------------------------
# 3. RETROACTIVE SELF-HEALING (Fixes corrupted slot 108 in existing saves)
# -------------------------------------------------------------
md_submod_heal_corrupted_arrays = {
	if = {
		limit = {
			check_variable = { current_gdp_graph_end > 107 }
		}

		# Heal GDP
		if = {
			limit = {
				check_variable = { gdp_array^107 > 0 }
			}
			set_temp_variable = { gdp_ratio = gdp_array^108 }
			divide_temp_variable = { gdp_ratio = gdp_array^107 }
			if = {
				limit = {
					check_variable = { gdp_ratio < 0.7 }
				}
				if = {
					limit = {
						check_variable = { current_gdp_graph_end > 108 }
					}
					set_temp_variable = { healed_gdp = gdp_array^107 }
					add_to_temp_variable = { healed_gdp = gdp_array^109 }
					divide_temp_variable = { healed_gdp = 2 }
					set_variable = { gdp_array^108 = healed_gdp }
				}
				else = {
					set_variable = { gdp_array^108 = gdp_array^107 }
				}
			}
		}

		# Heal GDP/C
		if = {
			limit = {
				check_variable = { gdpc_array^107 > 0 }
			}
			set_temp_variable = { gdpc_ratio = gdpc_array^108 }
			divide_temp_variable = { gdpc_ratio = gdpc_array^107 }
			if = {
				limit = {
					check_variable = { gdpc_ratio < 0.7 }
				}
				if = {
					limit = {
						check_variable = { current_gdp_graph_end > 108 }
					}
					set_temp_variable = { healed_gdpc = gdpc_array^107 }
					add_to_temp_variable = { healed_gdpc = gdpc_array^109 }
					divide_temp_variable = { healed_gdpc = 2 }
					set_variable = { gdpc_array^108 = healed_gdpc }
				}
				else = {
					set_variable = { gdpc_array^108 = gdpc_array^107 }
				}
			}
		}

		# Heal Productivity
		if = {
			limit = {
				check_variable = { prod_array^107 > 0 }
			}
			set_temp_variable = { prod_ratio = prod_array^108 }
			divide_temp_variable = { prod_ratio = prod_array^107 }
			if = {
				limit = {
					check_variable = { prod_ratio < 0.7 }
				}
				if = {
					limit = {
						check_variable = { current_gdp_graph_end > 108 }
					}
					set_temp_variable = { healed_prod = prod_array^107 }
					add_to_temp_variable = { healed_prod = prod_array^109 }
					divide_temp_variable = { healed_prod = 2 }
					set_variable = { prod_array^108 = healed_prod }
				}
				else = {
					set_variable = { prod_array^108 = prod_array^107 }
				}
			}
		}
	}
}
