POL_industrial_sovereignty_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = POL_industrial_sovereignty_state_initialized } }
		set_variable = { POL_industrial_sovereignty_domestic_control = 2 }
		set_variable = { POL_industrial_sovereignty_technological_depth = 2 }
		set_variable = { POL_industrial_sovereignty_export_capacity = 2 }
		set_variable = { POL_industrial_sovereignty_systems_integration = 2 }
		set_variable = { POL_industrial_sovereignty_supply_resilience = 2 }
		set_country_flag = POL_industrial_sovereignty_state_initialized
	}
	POL_industrial_sovereignty_clamp_state = yes
}

POL_industrial_sovereignty_clamp_state = {
	set_temp_variable = { corp_value = POL_industrial_sovereignty_domestic_control }
	corporate_history_clamp_value = yes
	set_variable = { POL_industrial_sovereignty_domestic_control = corp_value }
	set_temp_variable = { corp_value = POL_industrial_sovereignty_technological_depth }
	corporate_history_clamp_value = yes
	set_variable = { POL_industrial_sovereignty_technological_depth = corp_value }
	set_temp_variable = { corp_value = POL_industrial_sovereignty_export_capacity }
	corporate_history_clamp_value = yes
	set_variable = { POL_industrial_sovereignty_export_capacity = corp_value }
	set_temp_variable = { corp_value = POL_industrial_sovereignty_systems_integration }
	corporate_history_clamp_value = yes
	set_variable = { POL_industrial_sovereignty_systems_integration = corp_value }
	set_temp_variable = { corp_value = POL_industrial_sovereignty_supply_resilience }
	corporate_history_clamp_value = yes
	set_variable = { POL_industrial_sovereignty_supply_resilience = corp_value }
}

# Deterministic repair/resolution path for old saves and console-assisted coverage.
POL_industrial_sovereignty_resolve_terminal = {
	POL_industrial_sovereignty_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = POL_industrial_sovereignty_terminal_resolved } }
		if = {
			limit = { POL_industrial_sovereignty_systems_integrator_qualified = yes }
			POL_industrial_sovereignty_apply_central_european_systems_integrator = yes
		}
		else_if = {
			limit = { POL_industrial_sovereignty_production_hub_qualified = yes }
			POL_industrial_sovereignty_apply_european_production_hub = yes
		}
		else_if = {
			limit = { POL_industrial_sovereignty_sovereign_republic_qualified = yes }
			POL_industrial_sovereignty_apply_sovereign_industrial_republic = yes
		}
		else_if = {
			limit = { POL_industrial_sovereignty_defense_digital_arsenal_qualified = yes }
			POL_industrial_sovereignty_apply_defense_digital_arsenal = yes
		}
		else = { POL_industrial_sovereignty_apply_fragmented_assembly_economy = yes }
	}
}
