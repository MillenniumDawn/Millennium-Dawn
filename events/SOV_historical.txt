add_namespace = sov_historical

country_event = {
	id = sov_historical.1
	title = sov_historical.1.t
	desc = sov_historical.1.d
	picture = GFX_SOV_generic
	is_triggered_only = yes

	trigger = { SOV_historical_government = yes }

	immediate = {
		set_variable = {
			SOV_debt_repayment_offer = {
				value = debt
				multiply = 0.05
				clamp = { min = 0 max = 50 }
				clamp = { max = treasury }
			}
		}
	}

	option = {
		name = sov_historical.1.a
		trigger = {
			SOV_historical_government = yes
			NOT = { has_country_flag = SOV_kasyanov_repayment_made }
			check_variable = { SOV_debt_repayment_offer > 0 }
			check_variable = { var = treasury value = SOV_debt_repayment_offer compare = greater_than_or_equals }
			check_variable = { var = debt value = SOV_debt_repayment_offer compare = greater_than_or_equals }
		}
		log = "[GetDateText]: [Root.GetName]: sov_historical.1.a executed"
		if = {
			limit = {
				SOV_historical_government = yes
				NOT = { has_country_flag = SOV_kasyanov_repayment_made }
				check_variable = { SOV_debt_repayment_offer > 0 }
				check_variable = { var = treasury value = SOV_debt_repayment_offer compare = greater_than_or_equals }
				check_variable = { var = debt value = SOV_debt_repayment_offer compare = greater_than_or_equals }
			}
			set_temp_variable = { treasury_change = { value = SOV_debt_repayment_offer multiply = -1 } }
			set_temp_variable = { debt_change = treasury_change }
			modify_treasury_effect = yes
			modify_debt_effect = yes
			set_country_flag = SOV_kasyanov_repayment_made
		}
		clear_variable = SOV_debt_repayment_offer
		ai_chance = {
			base = 100
			modifier = { factor = 0 ai_has_high_deficit = yes }
			modifier = { factor = 0 has_active_mission = bankruptcy_incoming_collapse }
		}
	}

	option = {
		name = sov_historical.1.b
		log = "[GetDateText]: [Root.GetName]: sov_historical.1.b executed"
		clear_variable = SOV_debt_repayment_offer
		ai_chance = { base = 1 }
	}
}

	is_triggered_only = yes
	fire_only_once = yes

	trigger = { SOV_historical_government = yes }

	option = {
		name = sov_historical.2.a
		trigger = { has_idea = SOV_stab_fund_idea SOV_historical_government = yes }
		log = "[GetDateText]: [Root.GetName]: sov_historical.2.a executed"
		if = {
			limit = { has_idea = SOV_stab_fund_idea SOV_historical_government = yes }
			remove_ideas = SOV_stab_fund_idea
			set_temp_variable = { treasury_change = 20 }
			modify_treasury_effect = yes
			increase_economic_growth = yes
		}
		ai_chance = { base = 100 }
	}

	option = {
		name = sov_historical.2.b
		log = "[GetDateText]: [Root.GetName]: sov_historical.2.b executed"
		if = {
			limit = { SOV_historical_government = yes }
			add_timed_idea = { idea = SOV_crisis_austerity days = 180 }
		}
		ai_chance = { base = 1 }
	}
}
