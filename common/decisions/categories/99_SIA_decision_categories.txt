# "Thai Strategic and Economic Decisions" — the consolidated economic, strategic and
# diplomatic hub (former SIA_asean_diplomacy_category, SIA_economy_wave3_category and
# SIA_realm_diplomacy_category merged in). The OR covers every unlock gate across the
# former categories.
SIA_economy_category = {
	allowed = { original_tag = SIA }
	icon = GFX_decision_category_generic_economy
	priority = 80

	# Deliberately open from 2001 with every decision inside visible-but-locked: the
	# unlock conditions live on the decisions' own available blocks. Hiding the category
	# meant a player could walk past the focuses that unlock it without ever knowing the
	# decisions existed.
	visible = {
		date > 2001.1.1
	}
}

# / — the Kra Canal across its whole lifecycle (former SIA_kra_category folded
# in). Shows the phased construction missions while the feasibility study is open and
# the canal is unbuilt, then the post-canal Strait Patrol lever once the strait is open.
# The two windows never overlap, so one category spans both.
SIA_kra_construction = {
	allowed = { original_tag = SIA }
	icon = GFX_decision_category_generic_industry
	priority = 73

	visible = {
		OR = {
			AND = {
				has_country_flag = SIA_kra_feasibility_open
				NOT = { has_completed_focus = SIA_two_oceans_one_kingdom }
			}
			has_completed_focus = SIA_gateway_of_asia
		}
	}
}
