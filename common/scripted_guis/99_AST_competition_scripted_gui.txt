# The tracker panel for the Australian Pacific influence competitions. It is
# attached to the AST_pacific_competitions decision category, which carries no
# decisions of its own and is marked visible_when_empty so the panel is drawn
# on the strength of the registry arrays alone. Nothing here is specific to any
# one contest - it lists whatever the framework has registered, so a future
# Pacific competition appears in it without touching this file.
scripted_gui = {
	AST_competition_panel_gui = {
		window_name = "AST_competition_panel"
		context_type = decision_category
		dirty = global.AST_competition_dirty

		ai_enabled = { always = no }

		visible = {
			original_tag = AST
			OR = {
				check_variable = { global.AST_active_competitions^num > 0 }
				check_variable = { global.AST_resolved_competitions^num > 0 }
			}
		}

		dynamic_lists = {
			AST_competition_active_gridbox = {
				array = global.AST_active_competitions
				change_scope = yes
				entry_container = AST_competition_entry
			}
			AST_competition_resolved_gridbox = {
				array = global.AST_resolved_competitions
				change_scope = yes
				entry_container = AST_competition_entry
			}
		}

		triggers = {
			AST_competition_resolved_header_visible = {
				check_variable = { global.AST_resolved_competitions^num > 0 }
			}
		}
	}
}
