# Effect: BRA_calculate_amazon_system
# Purpose: Calculates the penalties in the Amazon states and at the Brazilian country level.
# BRA_acres_current - The total Acreage in Brazil
# BRA_acres_conserved - The total amount of Acreage in Brazil that is conserved
# BRA_acres_wild - Total amount of Acreage in Brazil that is wild (untamed)
# BRA_wild_acreage_ratio - The ratio of wild acreage
# BRA_conserved_acreage_ratio - The ratio of conserved acreage
# BRA_current_target_ratio - The ratio of total acreage we want
# BRA_current_target - The total number of acres we want to have for the amazon
# BRA_acres_conserved_change - The rate at which wild acres are converted to conserved acres
# BRA_acre_rate_loss - The number of wild acres lost each month
# BRA_acre_rate_gain - The number of wild acres gained each month
# BRA_net_acreage - The net change in wild acreage each month, before conversion
# BRA_amazon_stability_factor - Stability penalty from -0.10 to 0, zero at or above the target
# BRA_amazon_political_power_factor - Political power penalty from -0.15 to 0, zero at or above the target
# BRA_amazon_research_speed_factor - Research bonus from 0 to 0.05, rises with the conserved ratio
# BRA_amazon_neutral_drift - Neutrality drift from 0 to 0.05, rises as the total falls below the target
BRA_calculate_amazon_system = {
	if = {
		limit = { original_tag = BRA }
		set_variable = { BRA_acres_current = { value = BRA_acres_conserved add = BRA_acres_wild } }

		if = {
			limit = { check_variable = { BRA_acres_current > 0 } }
			set_variable = { BRA_wild_acreage_ratio = { value = BRA_acres_wild divide = BRA_acres_current } }
			set_variable = { BRA_conserved_acreage_ratio = { value = BRA_acres_conserved divide = BRA_acres_current } }
		}
		else = {
			set_variable = { BRA_wild_acreage_ratio = 0 }
			set_variable = { BRA_conserved_acreage_ratio = 0 }
		}

		if = {
			limit = { check_variable = { BRA_acres_target > 0 } }
			set_variable = { BRA_current_target_ratio = { value = BRA_acres_current divide = BRA_acres_target } }
		}
		else = {
			set_variable = { BRA_current_target_ratio = 0 }
		}

		# Falling short of the target hurts the country, a larger conserved share helps research
		set_variable = { BRA_amazon_stability_factor = { value = BRA_current_target_ratio subtract = 1 multiply = 0.2 clamp = { min = -0.1 max = 0 } } }
		set_variable = { BRA_amazon_political_power_factor = { value = BRA_current_target_ratio subtract = 1 multiply = 0.3 clamp = { min = -0.15 max = 0 } } }
		set_variable = { BRA_amazon_research_speed_factor = { value = BRA_conserved_acreage_ratio multiply = 0.05 } }
		set_variable = { BRA_amazon_neutral_drift = { value = 1 subtract = BRA_current_target_ratio multiply = 0.1 clamp = { min = 0 max = 0.05 } } }

		# Wild acreage drives the state penalties. Clamp before the local relief so the relief still shows.
		set_temp_variable = { temp_ratio_penalties = { value = BRA_wild_acreage_ratio multiply = -1 } }
		set_temp_variable = { temp_state_penalties = { value = temp_ratio_penalties clamp = { min = -0.5 max = 0 } } }
		for_each_scope_loop = {
			array = BRA.amazon_states
			if = {
				limit = { has_state_flag = improve_local_conservation_efforts }
				set_variable = { BRA_amazon_penalties = { value = temp_state_penalties multiply = 0.75 } }
				set_variable = { BRA_building_slots_factor = { value = temp_ratio_penalties multiply = 0.75 } }
			}
			else = {
				set_variable = { BRA_amazon_penalties = temp_state_penalties }
				set_variable = { BRA_building_slots_factor = temp_ratio_penalties }
			}
		}
	}
}

# Effect: BRA_monthly_amazon_acreage_change
# Purpose: Applies the monthly wild acreage change, then converts wild acreage to conserved acreage
BRA_monthly_amazon_acreage_change = {
	set_variable = { BRA_net_acreage = { value = BRA_acre_rate_gain subtract = BRA_acre_rate_loss } }
	set_variable = { BRA_acres_wild = { value = BRA_acres_wild add = BRA_net_acreage max = 0 } }

	# Convert no more wild acreage than exists
	set_temp_variable = { acres_converted = { value = BRA_acres_conserved_change min = BRA_acres_wild } }
	subtract_from_variable = { BRA_acres_wild = acres_converted }
	add_to_variable = { BRA_acres_conserved = acres_converted }

	BRA_calculate_amazon_system = yes
}

# Favelas keep the cartels alive. Strength and influence stay at 5 or more while one remains, which holds off cartel_event.52
BRA_favela_cartel_floor_effect = {
	if = {
		limit = { has_dynamic_modifier = { modifier = narco_state_bonuses } }
		BRA_favela_narco_takeover_effect = yes
	}
	else_if = {
		limit = { has_cartels_penalties = yes }
		set_temp_variable = { favela_full_count = 0 }
		set_temp_variable = { favela_improving_count = 0 }
		every_owned_state = {
			limit = { BRA_state_has_favelas = yes }
			if = {
				limit = { has_dynamic_modifier = { modifier = BRA_favela_improving_modifier } }
				add_to_temp_variable = { favela_improving_count = 1 }
			}
			else = {
				add_to_temp_variable = { favela_full_count = 1 }
			}
		}
		set_variable = { BRA_favela_strength_floor = 0 }
		set_variable = { BRA_favela_influence_floor = 0 }
		if = {
			limit = {
				OR = {
					check_variable = { favela_full_count > 0 }
					check_variable = { favela_improving_count > 0 }
				}
			}
			set_temp_variable = {
				favela_strength_floor = {
					value = favela_full_count
					multiply = 5
					add = { value = favela_improving_count multiply = 2 }
					max = 5
				}
			}
			set_temp_variable = {
				favela_influence_floor = {
					value = favela_full_count
					multiply = 2
					add = favela_improving_count
					max = 5
				}
			}
			set_variable = { BRA_favela_strength_floor = favela_strength_floor }
			set_variable = { BRA_favela_influence_floor = favela_influence_floor }
			set_temp_variable = { cart_strength_change = { value = favela_strength_floor subtract = strength_of_cartels max = 0 } }
			set_temp_variable = { cart_influence_change = { value = favela_influence_floor subtract = cartel_political_influence max = 0 } }
			if = {
				limit = {
					OR = {
						check_variable = { cart_strength_change > 0 }
						check_variable = { cart_influence_change > 0 }
					}
				}
				modify_cartel_variables_effect = yes
			}
		}
	}
}

# Conceding to the narcos hands every remaining favela to the gangs, including favela states won back later
BRA_favela_narco_takeover_effect = {
	set_temp_variable = { gang_favela_change = 0 }
	every_owned_state = {
		limit = {
			OR = {
				has_dynamic_modifier = { modifier = BRA_favela_modifier }
				has_dynamic_modifier = { modifier = BRA_favela_improving_modifier }
			}
		}
		if = {
			limit = { has_dynamic_modifier = { modifier = BRA_favela_modifier } }
			remove_dynamic_modifier = { modifier = BRA_favela_modifier }
		}
		if = {
			limit = { has_dynamic_modifier = { modifier = BRA_favela_improving_modifier } }
			remove_dynamic_modifier = { modifier = BRA_favela_improving_modifier }
		}
		add_dynamic_modifier = { modifier = BRA_favela_gang_ruled_modifier }
		add_to_temp_variable = { gang_favela_change = 1 }
	}
	if = {
		limit = { check_variable = { gang_favela_change > 0 } }
		BRA_favela_narco_feed_effect = yes
		if = {
			limit = { NOT = { has_country_flag = BRA_favelas_gang_ruled } }
			set_country_flag = BRA_favelas_gang_ruled
			country_event = brazil.192
		}
	}
}

# Effect: BRA_favela_narco_feed_effect
# Purpose: Moves a narco state's cartel strength and influence by 5 and 3 for each gang-ruled favela gained or retaken
# Parameters:
# - gang_favela_change: gang-ruled favelas gained, negative for those retaken
BRA_favela_narco_feed_effect = {
	if = {
		limit = { has_dynamic_modifier = { modifier = narco_state_bonuses } }
		set_variable = { strength_of_cartels = { value = gang_favela_change multiply = 5 add = strength_of_cartels max = 0 } }
		set_variable = { cartel_political_influence = { value = gang_favela_change multiply = 3 add = cartel_political_influence max = 0 } }
		calculate_cartel_bonuses = yes
	}
}
