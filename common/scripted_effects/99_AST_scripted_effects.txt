bushfires_tick_up = {
	if = {
		limit = { check_variable = { bushfire_ticker < 6 } }
		add_to_variable = { bushfire_ticker = 1 }
	}
}

bushfires_tick_down = {
	if = {
		limit = { check_variable = { bushfire_ticker > 1 } }
		subtract_from_variable = { bushfire_ticker = 1 }
	}
}

AST_fire_bushfire_event = {
	if = {
		limit = { date < 2004.1.1 }
		AST = { news_event = { id = generic_bushfires_ast.2 days = 1 random_days = 360 } }
	}
	else_if = {
		limit = {
			date > 2008.1.1
			date < 2011.1.1
			NOT = {
				has_country_flag = black_saturday_1_done
			}
		}
		AST = { news_event = { id = black_saturday_ast.1 days = 1 random_days = 360 } }
	}
	else = {
		AST = { country_event = { id = generic_bushfires_ast.1 days = 1 random_days = 360 } }
	}
}

AST_increase_indigenous_laws = {
	if = {
		limit = { has_idea = reform_despair_ast }
		swap_ideas = {
			remove_idea = reform_despair_ast
			add_idea = reconciliation_discontent
		}
	}
	else_if = {
		limit = { has_idea = reconciliation_discontent }
		swap_ideas = {
			remove_idea = reconciliation_discontent
			add_idea = rekindled_hope_ast
		}
	}
	else_if = {
		limit = { has_idea = rekindled_hope_ast }
		swap_ideas = {
			remove_idea = rekindled_hope_ast
			add_idea = a_new_era_ast
		}
	}
	else_if = {
		limit = { has_idea = a_new_era_ast }
		swap_ideas = {
			remove_idea = a_new_era_ast
			add_idea = radical_protections_ast
		}
	}
	else = {
		add_stability = 0.03
	}
}
