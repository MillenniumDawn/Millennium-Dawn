ASEAN_resync_member_array = {
	for_each_scope_loop = {
		array = global.ASEAN_Member
		if = {
			limit = {
				NOT = { has_idea = ASEAN_Member }
				NOT = { has_idea = ASEAN_Member_drifting }
				NOT = { has_idea = ASEAN_Member_detached }
				NOT = { has_idea = ASEAN_Member_estranged }
			}
			remove_from_array = { global.ASEAN_Member = THIS }
		}
	}
	if = {
		limit = {
			OR = {
				has_idea = ASEAN_Member
				has_idea = ASEAN_Member_drifting
				has_idea = ASEAN_Member_detached
				has_idea = ASEAN_Member_estranged
			}
			NOT = { is_in_array = { global.ASEAN_Member = THIS } }
		}
		add_to_array = { global.ASEAN_Member = THIS }
	}
}

ASEAN_check_breach = {
	set_variable = { THIS.ASEAN_breach = 0 }

	if = {
		limit = {
			any_of_scopes = {
				array = global.ASEAN_Member
				has_war_with = PREV
				NOT = { original_tag = PREV.original_tag }
			}
		}
		add_to_variable = { THIS.ASEAN_breach = 3 }
	}

	if = {
		limit = { has_elections = no }
		add_to_variable = { THIS.ASEAN_breach = 2 }
	}

	# Ideology is not a breach: ASEAN doctrine is non-interference in how members govern
	# themselves. What the Association tracks is whether a member can govern at all, on a
	# -2..+2 axis running from settled peace to open internal conflict.
	if = {
		limit = { has_civil_war = yes }
		add_to_variable = { THIS.ASEAN_breach = 2 }
	}
	else_if = {
		limit = { has_stability < 0.35 }
		add_to_variable = { THIS.ASEAN_breach = 1 }
	}
	else_if = {
		limit = {
			has_war = no
			has_stability > 0.7
		}
		add_to_variable = { THIS.ASEAN_breach = -2 }
	}
	else_if = {
		limit = {
			has_war = no
			has_stability > 0.5
		}
		add_to_variable = { THIS.ASEAN_breach = -1 }
	}

	if = {
		limit = { has_country_flag = ASEAN_recent_coup }
		add_to_variable = { THIS.ASEAN_breach = 2 }
	}

	if = {
		limit = {
			OR = {
				has_country_flag = ASEAN_aligned_china
				has_country_flag = ASEAN_aligned_america
			}
		}
		add_to_variable = { THIS.ASEAN_breach = 2 }
	}

	if = {
		limit = { has_country_flag = ASEAN_suspended_flag }
		add_to_variable = { THIS.ASEAN_breach = 2 }
	}

	if = {
		limit = { has_country_flag = ASEAN_objection_recent }
		add_to_variable = { THIS.ASEAN_breach = 1 }
	}

	if = {
		limit = { has_country_flag = ASEAN_condemned_recent }
		add_to_variable = { THIS.ASEAN_breach = 1 }
	}

	if = {
		limit = {
			OR = {
				has_idea = corruption_level_06
				has_idea = corruption_level_07
				has_idea = corruption_level_08
				has_idea = corruption_level_09
				has_idea = corruption_level_10
			}
		}
		add_to_variable = { THIS.ASEAN_breach = 1 }
	}
}

# swap_ideas fires on_remove before on_add, so without this guard a band change would
# splice the country out of global.ASEAN_Member and re-append it mid-iteration.
ASEAN_apply_band = {
	set_country_flag = ASEAN_band_swap_in_progress
	if = {
		limit = { check_variable = { THIS.ASEAN_breach < 2 } }
		if = {
			limit = {
				NOT = { has_idea = ASEAN_Member }
				OR = {
					has_idea = ASEAN_Member_drifting
					has_idea = ASEAN_Member_detached
					has_idea = ASEAN_Member_estranged
				}
			}
			if = { limit = { has_idea = ASEAN_Member_drifting }
				swap_ideas = { remove_idea = ASEAN_Member_drifting add_idea = ASEAN_Member }
			}
			else_if = { limit = { has_idea = ASEAN_Member_detached }
				swap_ideas = { remove_idea = ASEAN_Member_detached add_idea = ASEAN_Member }
			}
			else_if = { limit = { has_idea = ASEAN_Member_estranged }
				swap_ideas = { remove_idea = ASEAN_Member_estranged add_idea = ASEAN_Member }
			}
		}
	}
	else_if = {
		limit = { check_variable = { THIS.ASEAN_breach < 4 } }
		if = {
			limit = {
				NOT = { has_idea = ASEAN_Member_drifting }
				OR = {
					has_idea = ASEAN_Member
					has_idea = ASEAN_Member_detached
					has_idea = ASEAN_Member_estranged
				}
			}
			if = { limit = { has_idea = ASEAN_Member }
				swap_ideas = { remove_idea = ASEAN_Member add_idea = ASEAN_Member_drifting }
			}
			else_if = { limit = { has_idea = ASEAN_Member_detached }
				swap_ideas = { remove_idea = ASEAN_Member_detached add_idea = ASEAN_Member_drifting }
			}
			else_if = { limit = { has_idea = ASEAN_Member_estranged }
				swap_ideas = { remove_idea = ASEAN_Member_estranged add_idea = ASEAN_Member_drifting }
			}
		}
	}
	else_if = {
		limit = { check_variable = { THIS.ASEAN_breach < 6 } }
		if = {
			limit = {
				NOT = { has_idea = ASEAN_Member_detached }
				OR = {
					has_idea = ASEAN_Member
					has_idea = ASEAN_Member_drifting
					has_idea = ASEAN_Member_estranged
				}
			}
			if = { limit = { has_idea = ASEAN_Member }
				swap_ideas = { remove_idea = ASEAN_Member add_idea = ASEAN_Member_detached }
			}
			else_if = { limit = { has_idea = ASEAN_Member_drifting }
				swap_ideas = { remove_idea = ASEAN_Member_drifting add_idea = ASEAN_Member_detached }
			}
			else_if = { limit = { has_idea = ASEAN_Member_estranged }
				swap_ideas = { remove_idea = ASEAN_Member_estranged add_idea = ASEAN_Member_detached }
			}
		}
	}
	else = {
		if = {
			limit = {
				NOT = { has_idea = ASEAN_Member_estranged }
				OR = {
					has_idea = ASEAN_Member
					has_idea = ASEAN_Member_drifting
					has_idea = ASEAN_Member_detached
				}
			}
			if = { limit = { has_idea = ASEAN_Member }
				swap_ideas = { remove_idea = ASEAN_Member add_idea = ASEAN_Member_estranged }
			}
			else_if = { limit = { has_idea = ASEAN_Member_drifting }
				swap_ideas = { remove_idea = ASEAN_Member_drifting add_idea = ASEAN_Member_estranged }
			}
			else_if = { limit = { has_idea = ASEAN_Member_detached }
				swap_ideas = { remove_idea = ASEAN_Member_detached add_idea = ASEAN_Member_estranged }
			}
		}
	}
	clr_country_flag = ASEAN_band_swap_in_progress
}

ASEAN_recount_bands = {
	set_temp_variable = { ASEAN_committed_count_tmp = 0 }
	set_temp_variable = { ASEAN_drifting_count_tmp = 0 }
	set_temp_variable = { ASEAN_detached_count_tmp = 0 }
	set_temp_variable = { ASEAN_estranged_count_tmp = 0 }
	set_temp_variable = { ASEAN_suspend_candidates_tmp = 0 }
	for_each_scope_loop = {
		array = global.ASEAN_Member
		if = { limit = { has_idea = ASEAN_Member }
			add_to_temp_variable = { ASEAN_committed_count_tmp = 1 }
		}
		else_if = { limit = { has_idea = ASEAN_Member_drifting }
			add_to_temp_variable = { ASEAN_drifting_count_tmp = 1 }
		}
		else_if = { limit = { has_idea = ASEAN_Member_detached }
			add_to_temp_variable = { ASEAN_detached_count_tmp = 1 }
		}
		else_if = { limit = { has_idea = ASEAN_Member_estranged }
			add_to_temp_variable = { ASEAN_estranged_count_tmp = 1 }
		}
		if = {
			limit = {
				NOT = { has_country_flag = ASEAN_suspended_flag }
				OR = {
					check_variable = { ASEAN_breach > 3 }
					has_country_flag = ASEAN_recent_coup
				}
			}
			add_to_temp_variable = { ASEAN_suspend_candidates_tmp = 1 }
		}
	}
	set_variable = { global.ASEAN_committed_count = ASEAN_committed_count_tmp }
	set_variable = { global.ASEAN_drifting_count = ASEAN_drifting_count_tmp }
	set_variable = { global.ASEAN_detached_count = ASEAN_detached_count_tmp }
	set_variable = { global.ASEAN_estranged_count = ASEAN_estranged_count_tmp }
	set_variable = { global.ASEAN_active_count = ASEAN_committed_count_tmp }
	add_to_variable = { global.ASEAN_active_count = ASEAN_drifting_count_tmp }
	set_variable = { global.ASEAN_suspend_candidates = ASEAN_suspend_candidates_tmp }
}

ASEAN_recount_founding_committed = {
	set_temp_variable = { ASEAN_founding_committed_tmp = 0 }
	if = { limit = { IND = { has_idea = ASEAN_Member } }
		add_to_temp_variable = { ASEAN_founding_committed_tmp = 1 }
	}
	if = { limit = { MAY = { has_idea = ASEAN_Member } }
		add_to_temp_variable = { ASEAN_founding_committed_tmp = 1 }
	}
	if = { limit = { PHI = { has_idea = ASEAN_Member } }
		add_to_temp_variable = { ASEAN_founding_committed_tmp = 1 }
	}
	if = { limit = { SIA = { has_idea = ASEAN_Member } }
		add_to_temp_variable = { ASEAN_founding_committed_tmp = 1 }
	}
	if = { limit = { SIN = { has_idea = ASEAN_Member } }
		add_to_temp_variable = { ASEAN_founding_committed_tmp = 1 }
	}
	if = { limit = { BRU = { has_idea = ASEAN_Member } }
		add_to_temp_variable = { ASEAN_founding_committed_tmp = 1 }
	}
	set_variable = { global.ASEAN_founding_committed = ASEAN_founding_committed_tmp }
}

ASEAN_fracture_watch = {
	if = {
		limit = { check_variable = { global.ASEAN_founding_committed < 3 } }
		if = {
			limit = { NOT = { has_global_flag = ASEAN_fracture_watch_active } }
			set_global_flag = ASEAN_fracture_watch_active
			set_variable = { global.ASEAN_fracture_watch_months = 0 }
		}
		add_to_variable = { global.ASEAN_fracture_watch_months = 1 }
	}
	else = {
		if = {
			limit = { has_global_flag = ASEAN_fracture_watch_active }
			clr_global_flag = ASEAN_fracture_watch_active
			set_variable = { global.ASEAN_fracture_watch_months = 0 }
		}
	}
	if = {
		limit = {
			check_variable = { global.ASEAN_fracture_watch_months > 11 }
			NOT = { has_global_flag = ASEAN_emergency_summit_called }
		}
		set_global_flag = ASEAN_emergency_summit_called
		for_each_scope_loop = {
			array = global.ASEAN_Member
			country_event = { id = asean_crisis.10 days = 3 }
		}
	}
}

# The IAI grant is a timed flag, so the bonus is applied once on arrival and withdrawn once on lapse. A plain monthly re-add would compound forever.
ASEAN_iai_recipient_upkeep = {
	if = {
		limit = {
			has_country_flag = ASEAN_iai_recipient_flag
			NOT = { has_country_flag = ASEAN_iai_bonus_applied }
		}
		set_country_flag = ASEAN_iai_bonus_applied
		set_temp_variable = { ASEAN_production_speed_infrastructure_factor_temp = 0.05 }
		add_to_variable = { ASEAN_production_speed_infrastructure_factor = ASEAN_production_speed_infrastructure_factor_temp tooltip = production_speed_infrastructure_factor_tt }
	}
	else_if = {
		limit = {
			NOT = { has_country_flag = ASEAN_iai_recipient_flag }
			has_country_flag = ASEAN_iai_bonus_applied
		}
		clr_country_flag = ASEAN_iai_bonus_applied
		set_temp_variable = { ASEAN_production_speed_infrastructure_factor_temp = -0.05 }
		add_to_variable = { ASEAN_production_speed_infrastructure_factor = ASEAN_production_speed_infrastructure_factor_temp tooltip = production_speed_infrastructure_factor_tt }
	}
}

# A member who leaves ASEAN mid-window drops out of global.ASEAN_Member, so the monthly
# loop above never visits it again to run the withdraw branch. Called instead from each
# band idea's on_remove once that on_remove has confirmed a real departure, so the +0.05
# is withdrawn on the way out no matter which band the country left from.
ASEAN_iai_recipient_withdraw_on_departure = {
	if = {
		limit = { has_country_flag = ASEAN_iai_bonus_applied }
		clr_country_flag = ASEAN_iai_bonus_applied
		set_temp_variable = { ASEAN_production_speed_infrastructure_factor_temp = -0.05 }
		add_to_variable = { ASEAN_production_speed_infrastructure_factor = ASEAN_production_speed_infrastructure_factor_temp tooltip = production_speed_infrastructure_factor_tt }
	}
}

# The four band ideas carry a flat -1000 ai_get_ally_desire_factor so AI members do not
# drift into unrelated factions. That same veto leaves SEATO with no willing members once
# it exists, whichever tree founded it. Rather than weaken the band ideas' own modifier for
# every member regardless of context, grant a self-healing offset idea once the pact
# actually exists so the veto lifts only then.
ASEAN_seato_alignment_upkeep = {
	if = {
		limit = {
			has_global_flag = ASEAN_seato_founded
			is_asean_member = yes
			NOT = { has_idea = ASEAN_seato_alignment }
		}
		add_ideas = ASEAN_seato_alignment
	}
}

ASEAN_monthly_update = {
	for_each_scope_loop = {
		array = global.ASEAN_Member
		ASEAN_check_breach = yes
		ASEAN_apply_band = yes
		ASEAN_iai_recipient_upkeep = yes
		ASEAN_seato_alignment_upkeep = yes
	}
	ASEAN_recount_bands = yes
	ASEAN_recount_founding_committed = yes
	ASEAN_fracture_watch = yes
	ASEAN_defense_pact_watch = yes
}

# The alliance capstone stays shut until an outside major power is already shooting at a member: the pact is a reaction, never a plan.
ASEAN_defense_pact_watch = {
	if = {
		limit = {
			NOT = { has_global_flag = asean_defense_pact_gate }
			any_of_scopes = {
				array = global.ASEAN_Member
				has_war = yes
				any_enemy_country = {
					NOT = { is_asean_member = yes }
					OR = {
						has_idea = large_power
						has_idea = great_power
						has_idea = super_power
					}
				}
			}
		}
		set_global_flag = asean_defense_pact_gate
	}
}

ASEAN_apply_suspension = {
	if = { limit = { NOT = { has_country_flag = ASEAN_suspended_flag } }
		set_country_flag = ASEAN_suspended_flag
		if = { limit = { NOT = { has_idea = ASEAN_suspended } }
			add_ideas = ASEAN_suspended
		}
		# The flag is already set, so the loop's own check excludes the suspended member
		# from its own bulletin without depending on how PREV binds inside the loop.
		hidden_effect = {
			for_each_scope_loop = {
				array = global.ASEAN_Member
				if = {
					limit = { NOT = { has_country_flag = ASEAN_suspended_flag } }
					news_event = { id = asean_crisis.22 days = 2 }
				}
			}
		}
	}
}

ASEAN_lift_suspension = {
	if = { limit = { has_country_flag = ASEAN_suspended_flag }
		clr_country_flag = ASEAN_suspended_flag
		if = { limit = { has_idea = ASEAN_suspended }
			remove_ideas = ASEAN_suspended
		}
	}
}

ASEAN_record_objection = {
	set_country_flag = { flag = ASEAN_objection_recent days = 180 value = 1 }
}
