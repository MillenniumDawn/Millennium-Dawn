# common/on_actions/md_submod_on_actions.txt

on_actions = {
	# 1. INITIALIZE THE ARRAYS AT GAME START
	on_startup = {
		effect = {
			if = {
				limit = { is_ai = no }
				md_submod_heal_corrupted_arrays = yes
				md_submod_debt_graph_init = yes
				md_submod_inflation_graph_init = yes
				md_submod_pop_graph_init = yes
				md_submod_budget_graph_init = yes
				md_submod_energy_graph_init = yes
				md_submod_defense_graph_init = yes
				md_submod_trade_graph_init = yes
			}
		}
	}

	# 2. THE MONTHLY SNAPSHOTS & CALCULATIONS (Fires exactly on the 1st!)
	on_monthly = {
		effect = {
			if = {
				limit = { is_ai = no }

				# Heal corrupted slots from base MD if present
				md_submod_heal_corrupted_arrays = yes

				# Take the snapshot on the 1st!
				md_submod_debt_graph_on_monthly = yes
				md_submod_inflation_graph_on_monthly = yes
				md_submod_pop_graph_on_monthly = yes
				md_submod_budget_graph_on_monthly = yes
				md_submod_energy_graph_on_monthly = yes
				md_submod_defense_graph_on_monthly = yes
				md_submod_trade_graph_on_monthly = yes

				# Update the graphs on the 1st!
				update_md_flaxbeard_gdp_graph = yes
				update_md_flaxbeard_gdpc_graph = yes
				update_md_flaxbeard_prod_graph = yes
				update_md_flaxbeard_debt_graph = yes
				update_md_flaxbeard_inflation_graph = yes
				update_md_flaxbeard_pop_graph = yes
				update_md_flaxbeard_budget_graph = yes
				update_md_flaxbeard_energy_graph = yes
				update_md_flaxbeard_defense_graph = yes
				update_md_flaxbeard_trade_graph = yes
			}
		}
	}
}
