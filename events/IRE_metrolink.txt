# Railway Order hearing, fired when planning finishes
country_event = {
	id = IRE_metrolink.1
	title = IRE_metrolink.1.t
	desc = IRE_metrolink.1.d
	picture = GFX_IRE_railway_order_hearing
	is_triggered_only = yes

	immediate = { set_variable = { IRE_metrolink_stage = 2 } }

	option = {
		name = IRE_metrolink.1.a
		log = "[GetDateText]: [This.GetName]: IRE_metrolink.1.a executed"
		set_temp_variable = { treasury_change = -0.3 }
		modify_treasury_effect = yes
		unlock_decision_tooltip = IRE_metrolink_construction
		ai_chance = {
			base = 5
			modifier = { factor = 0.5 ai_has_high_deficit = yes }
			modifier = { factor = 0 has_active_mission = bankruptcy_incoming_collapse }
		}
	}

	option = {
		name = IRE_metrolink.1.b
		log = "[GetDateText]: [This.GetName]: IRE_metrolink.1.b executed"
		set_country_flag = { flag = IRE_metrolink_hearing_delay days = 180 value = 1 }
		custom_effect_tooltip = IRE_metrolink_hearing_delay_effect_tt
		unlock_decision_tooltip = IRE_metrolink_construction
		ai_chance = { base = 10 }
	}

	option = {
		name = IRE_metrolink.1.c
		log = "[GetDateText]: [This.GetName]: IRE_metrolink.1.c executed"
		set_country_flag = IRE_metrolink_cut_scope
		custom_effect_tooltip = IRE_metrolink_cut_scope_tt
		unlock_decision_tooltip = IRE_metrolink_construction
		ai_chance = {
			base = 1
			modifier = { add = 4 ai_is_fiscally_stressed = yes }
		}
	}
}

# Fiscal crisis deferral, scheduled by planning and construction
country_event = {
	id = IRE_metrolink.2
	title = IRE_metrolink.2.t
	desc = IRE_metrolink.2.d
	picture = GFX_IRE_metro_deferred
	is_triggered_only = yes

	trigger = {
		NOT = { has_country_flag = IRE_metrolink_deferral_done }
		OR = {
			check_variable = { IRE_metrolink_stage = 1 }
			check_variable = { IRE_metrolink_stage = 3 }
		}
		OR = {
			ai_is_fiscally_stressed = yes
			has_active_mission = bankruptcy_incoming_collapse
		}
	}

	immediate = { set_country_flag = IRE_metrolink_deferral_done }

	option = {
		name = IRE_metrolink.2.a
		log = "[GetDateText]: [This.GetName]: IRE_metrolink.2.a executed"
		if = {
			limit = { check_variable = { IRE_metrolink_stage = 3 } }
			set_variable = { IRE_metrolink_resume_stage = 2 }
			set_temp_variable = { treasury_change = 1.5 }
		}
		else = {
			set_variable = { IRE_metrolink_resume_stage = 0 }
			set_temp_variable = { treasury_change = 0.25 }
		}
		modify_treasury_effect = yes
		set_variable = { IRE_metrolink_stage = 0 }
		set_country_flag = IRE_metrolink_deferred
		set_country_flag = { flag = IRE_metrolink_deferral_cooldown days = 365 value = 1 }
		custom_effect_tooltip = IRE_metrolink_deferred_tt
		unlock_decision_tooltip = IRE_metrolink_revival
		ai_chance = {
			base = 5
			modifier = { add = 10 has_active_mission = bankruptcy_incoming_collapse }
		}
	}

	option = {
		name = IRE_metrolink.2.b
		log = "[GetDateText]: [This.GetName]: IRE_metrolink.2.b executed"
		add_stability = -0.03
		set_temp_variable = { treasury_change = -0.5 }
		modify_treasury_effect = yes
		ai_chance = {
			base = 5
			modifier = { factor = 0 has_stability < 0.35 }
			modifier = { factor = 0 has_active_mission = bankruptcy_incoming_collapse }
		}
	}
}
