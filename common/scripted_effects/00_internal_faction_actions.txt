# Internal factions v3 player and AI actions, see .claude/docs/internal-factions-reference.md
@internal_faction_grant_opinion = 10
@internal_faction_revoke_opinion = -15
@internal_faction_revoke_stability = -0.03
@internal_faction_crackdown_opinion = -20
@internal_faction_crackdown_influence = -20
@internal_faction_crackdown_stability = -0.05
@internal_faction_lift_opinion = 5
@internal_faction_appease_opinion = 10
@internal_faction_appease_rival_opinion = -3
@internal_faction_ai_coup_plot_threshold = 50
@internal_faction_ai_coup_pp = 150
@internal_faction_ai_privilege_pp = 200

# Parameters:
# - internal_faction_id: faction id 1-23; economic factions (category 1) pay a GDP share from the treasury, the rest pay political power
internal_faction_pay_privilege_cost = {
	if = {
		limit = { check_variable = { global.internal_faction_category^internal_faction_id = 1 } }
		set_temp_variable = { treasury_change = gdp_total }
		multiply_temp_variable = { treasury_change = global.internal_faction_cost_grant_gdp_share }
		multiply_temp_variable = { treasury_change = -1 }
		modify_treasury_effect = yes
	}
	else = {
		set_temp_variable = { internal_faction_pp = global.internal_faction_cost_grant_pp }
		multiply_temp_variable = { internal_faction_pp = -1 }
		add_political_power = internal_faction_pp
	}
}

# Parameters:
# - internal_faction_p: privilege id (faction id - 1) * 4 + k, k 1-3
internal_faction_grant_privilege = {
	if = {
		limit = { internal_faction_can_grant_privilege = yes }
		set_temp_variable = { internal_faction_id = global.internal_faction_policy_faction^internal_faction_p }
		add_to_array = { internal_faction_policies = internal_faction_p }
		set_variable = { internal_faction_policy_cooldown^internal_faction_id = global.internal_faction_policy_cooldown_months }
		set_temp_variable = { temp_opinion = @internal_faction_grant_opinion }
		internal_faction_change_opinion = yes
		log = "[GetDateText]: [Root.GetName]: granted privilege [?internal_faction_p] to faction [?internal_faction_id]"
		internal_faction_pay_privilege_cost = yes
	}
}

# Parameters:
# - internal_faction_p: privilege id (faction id - 1) * 4 + k, k 1-3
internal_faction_revoke_privilege = {
	if = {
		limit = { internal_faction_can_revoke_privilege = yes }
		set_temp_variable = { internal_faction_id = global.internal_faction_policy_faction^internal_faction_p }
		remove_from_array = { array = internal_faction_policies value = internal_faction_p }
		set_variable = { internal_faction_policy_cooldown^internal_faction_id = global.internal_faction_policy_cooldown_months }
		set_temp_variable = { temp_opinion = @internal_faction_revoke_opinion }
		internal_faction_change_opinion = yes
		add_stability = @internal_faction_revoke_stability
		log = "[GetDateText]: [Root.GetName]: revoked privilege [?internal_faction_p] from faction [?internal_faction_id]"
		set_temp_variable = { internal_faction_pp = global.internal_faction_cost_revoke }
		multiply_temp_variable = { internal_faction_pp = -1 }
		add_political_power = internal_faction_pp
	}
}

# Parameters:
# - internal_faction_id: faction id 1-23
internal_faction_enact_crackdown = {
	if = {
		limit = { internal_faction_can_enact_crackdown = yes }
		set_temp_variable = { internal_faction_cp = internal_faction_id }
		multiply_temp_variable = { internal_faction_cp = 4 }
		add_to_array = { internal_faction_policies = internal_faction_cp }
		subtract_from_temp_variable = { internal_faction_cp = 1 }
		remove_from_array = { array = internal_faction_policies value = internal_faction_cp }
		subtract_from_temp_variable = { internal_faction_cp = 1 }
		remove_from_array = { array = internal_faction_policies value = internal_faction_cp }
		subtract_from_temp_variable = { internal_faction_cp = 1 }
		remove_from_array = { array = internal_faction_policies value = internal_faction_cp }
		set_variable = { internal_faction_policy_cooldown^internal_faction_id = global.internal_faction_policy_cooldown_months }
		set_temp_variable = { temp_opinion = @internal_faction_crackdown_opinion }
		internal_faction_change_opinion = yes
		set_temp_variable = { temp_influence = @internal_faction_crackdown_influence }
		internal_faction_change_influence = yes
		add_stability = @internal_faction_crackdown_stability
		log = "[GetDateText]: [Root.GetName]: enacted crackdown on faction [?internal_faction_id]"
		set_temp_variable = { internal_faction_pp = global.internal_faction_cost_crackdown }
		multiply_temp_variable = { internal_faction_pp = -1 }
		add_political_power = internal_faction_pp
	}
}

# Parameters:
# - internal_faction_id: faction id 1-23; other active factions of the same category lose opinion
internal_faction_appease = {
	if = {
		limit = { internal_faction_can_appease = yes }
		set_temp_variable = { internal_faction_appease_id = internal_faction_id }
		set_temp_variable = { temp_opinion = @internal_faction_appease_opinion }
		internal_faction_change_opinion = yes
		for_each_loop = {
			array = internal_faction_active
			value = internal_faction_v
			if = {
				limit = {
					NOT = { check_variable = { internal_faction_v = internal_faction_appease_id } }
					check_variable = { global.internal_faction_category^internal_faction_v = global.internal_faction_category^internal_faction_appease_id }
				}
				set_temp_variable = { internal_faction_id = internal_faction_v }
				set_temp_variable = { temp_opinion = @internal_faction_appease_rival_opinion }
				internal_faction_change_opinion = yes
			}
		}
		set_temp_variable = { internal_faction_id = internal_faction_appease_id }
		log = "[GetDateText]: [Root.GetName]: appeased faction [?internal_faction_id]"
		set_temp_variable = { internal_faction_pp = global.internal_faction_cost_appease }
		multiply_temp_variable = { internal_faction_pp = -1 }
		add_political_power = internal_faction_pp
	}
}

# Parameters:
# - internal_faction_id: active faction to replace; fills internal_faction_swap_candidates with every available inactive faction
internal_faction_open_swap_window = {
	set_variable = { internal_faction_swap_from = internal_faction_id }
	clear_array = internal_faction_swap_candidates
	for_loop_effect = {
		start = 1
		end = 24
		value = internal_faction_id
		if = {
			limit = {
				NOT = { is_in_array = { array = internal_faction_active value = internal_faction_id } }
				internal_faction_is_available = yes
			}
			add_to_array = { internal_faction_swap_candidates = internal_faction_id }
		}
	}
	set_country_flag = internal_faction_swap_open
	update_internal_faction_dirty_variable = yes
}

internal_faction_close_swap_window = {
	clr_country_flag = internal_faction_swap_open
	clear_array = internal_faction_swap_candidates
	update_internal_faction_dirty_variable = yes
}

# Parameters:
# - internal_faction_new_id: replacement faction id; the outgoing faction is internal_faction_swap_from
internal_faction_swap_faction = {
	set_temp_variable = { internal_faction_id = internal_faction_swap_from }
	if = {
		limit = {
			internal_faction_can_swap_out = yes
			NOT = { is_in_array = { array = internal_faction_active value = internal_faction_new_id } }
		}
		internal_faction_remove_faction = yes
		set_temp_variable = { internal_faction_id = internal_faction_new_id }
		internal_faction_add_faction = yes
		set_variable = { internal_faction_swap_cooldown^internal_faction_id = global.internal_faction_swap_cooldown_months }
		log = "[GetDateText]: [Root.GetName]: swapped faction [?internal_faction_swap_from] for [?internal_faction_new_id]"
		set_temp_variable = { internal_faction_pp = global.internal_faction_cost_swap }
		multiply_temp_variable = { internal_faction_pp = -1 }
		add_political_power = internal_faction_pp
	}
}

# Parameters:
# - internal_faction_p: privilege id; revokes when active, grants otherwise
internal_faction_toggle_privilege = {
	if = {
		limit = { is_in_array = { array = internal_faction_policies value = internal_faction_p } }
		internal_faction_revoke_privilege = yes
	}
	else = {
		internal_faction_grant_privilege = yes
	}
}

# Parameters:
# - internal_faction_id: faction id; lifts when its crackdown is active, enacts otherwise
internal_faction_toggle_crackdown = {
	set_temp_variable = { internal_faction_cp = internal_faction_id }
	multiply_temp_variable = { internal_faction_cp = 4 }
	if = {
		limit = { is_in_array = { array = internal_faction_policies value = internal_faction_cp } }
		internal_faction_lift_crackdown = yes
	}
	else = {
		internal_faction_enact_crackdown = yes
	}
}

# One action per month for AI countries; called at the end of internal_faction_monthly_tick
internal_faction_ai_monthly = {
	if = {
		limit = {
			check_variable = { internal_faction_coup_plot > @internal_faction_ai_coup_plot_threshold }
			has_political_power > @internal_faction_ai_coup_pp
			is_in_array = { internal_faction_active = 7 }
		}
		set_temp_variable = { internal_faction_id = 7 }
		if = {
			limit = {
				is_in_array = { internal_faction_active = 9 }
				check_variable = { var = internal_faction_opinion^9 value = 60 compare = greater_than_or_equals }
				internal_faction_can_enact_crackdown = yes
			}
			log = "[GetDateText]: [Root.GetName]: AI coup plot [?internal_faction_coup_plot|0], enacting political commissars"
			internal_faction_enact_crackdown = yes
		}
		else_if = {
			limit = { internal_faction_can_appease = yes }
			log = "[GetDateText]: [Root.GetName]: AI coup plot [?internal_faction_coup_plot|0], appeasing the military"
			internal_faction_appease = yes
		}
	}
	else_if = {
		limit = { has_political_power > @internal_faction_ai_privilege_pp }
		set_temp_variable = { internal_faction_ai_done = 0 }
		for_each_loop = {
			array = internal_faction_active
			value = internal_faction_v
			if = {
				limit = {
					check_variable = { internal_faction_ai_done = 0 }
					set_temp_variable = { internal_faction_id = internal_faction_v }
					internal_faction_influence_powerful = yes
					internal_faction_tier_hostile = yes
				}
				for_loop_effect = {
					start = 0
					end = 3
					value = internal_faction_k
					if = {
						limit = { check_variable = { internal_faction_ai_done = 0 } }
						set_temp_variable = { internal_faction_p = internal_faction_v }
						subtract_from_temp_variable = { internal_faction_p = 1 }
						multiply_temp_variable = { internal_faction_p = 4 }
						add_to_temp_variable = { internal_faction_p = 1 }
						add_to_temp_variable = { internal_faction_p = internal_faction_k }
						if = {
							limit = { internal_faction_can_grant_privilege = yes }
							log = "[GetDateText]: [Root.GetName]: AI granting privilege [?internal_faction_p] to powerful hostile faction [?internal_faction_v]"
							internal_faction_grant_privilege = yes
							set_temp_variable = { internal_faction_ai_done = 1 }
						}
					}
				}
			}
		}
	}
}
