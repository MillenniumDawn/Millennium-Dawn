# Sets country group flags at startup for O(1) lookups in runtime triggers.
# Called from on_startup. The corresponding triggers in MD_Country_Groups_Triggers.txt
# check has_country_flag first (fast path) and fall back to original_tag OR-lists
# for allowed blocks that may evaluate before on_startup.

set_country_group_flags = {
	if = { limit = { is_european_nation = yes }
		set_country_flag = md_european_nation
	}
	if = { limit = { is_sub_saharan_nation = yes }
		set_country_flag = md_sub_saharan_nation
	}
	if = { limit = { is_middle_eastern_nation = yes }
		set_country_flag = md_middle_eastern_nation
	}
	if = { limit = { is_asian_nation = yes }
		set_country_flag = md_asian_nation
	}
	if = { limit = { is_american_nation = yes }
		set_country_flag = md_american_nation
	}
	if = { limit = { is_western_nation = yes }
		set_country_flag = md_western_nation
	}
	if = { limit = { is_african_nation = yes }
		set_country_flag = md_african_nation
	}
	if = { limit = { is_arabic_nation = yes }
		set_country_flag = md_arabic_nation
	}
}
