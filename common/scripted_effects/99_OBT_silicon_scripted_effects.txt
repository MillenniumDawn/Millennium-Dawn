# Operation Blue Tiger: Silicon Island microchip plants
# A plant is built at once if Taiwan can staff and power it and is not under a full blockade;
# otherwise it is deferred and a decision in the Economic Policy category builds it later.

TAI_science_park_fab_hsinchu = {
	if = {
		limit = {
			TAI_can_build_microchip_plant = yes
			1155 = { is_controlled_by = ROOT }
		}
		1155 = {
			add_extra_state_shared_building_slots = 1
			add_building_construction = { type = microchip_plant level = 1 instant_build = yes }
		}
		set_temp_variable = { treasury_change = -10.5 }
		modify_treasury_effect = yes
	}
	else = {
		set_country_flag = TAI_deferred_fab_hsinchu
		custom_effect_tooltip = TAI_deferred_fab_hsinchu_tt
	}
}
