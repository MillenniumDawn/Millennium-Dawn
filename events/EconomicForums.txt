	id = econ_forum.1
	title = econ_forum.1.t
	desc = econ_forum.1.d
	picture = GFX_economic_forum_aze
	is_triggered_only = yes

	option = {
		name = econ_forum.1.a

		unlock_decision_category_tooltip = econ_forum_category
		ai_chance = { base = 1 }
	}
}

# An invitation to another forum's summit. econ_forum_invited_to holds the forum id.
country_event = {
	id = econ_forum.2
	title = econ_forum.2.t
	desc = econ_forum.2.d
	picture = GFX_economic_forum_aze
	is_triggered_only = yes

	trigger = {
		check_variable = { econ_forum_status^econ_forum_invited_to > 0 }
		check_variable = { econ_forum_status^econ_forum_invited_to < 3 }
	}

	option = {
		name = econ_forum.2.a
		trigger = { has_political_power > 19 }
		log = "[GetDateText]: [This.GetName]: econ_forum.2.a executed"
		add_political_power = -20
		custom_effect_tooltip = econ_forum_send_head_tt
		set_variable = { econ_forum_status^econ_forum_invited_to = 4 }
		ai_chance = { base = 1 }
	}

	option = {
		name = econ_forum.2.b
		trigger = { has_political_power > 4 }
		log = "[GetDateText]: [This.GetName]: econ_forum.2.b executed"
		add_political_power = -5
		custom_effect_tooltip = econ_forum_send_ministers_tt
		set_variable = { econ_forum_status^econ_forum_invited_to = 3 }
		ai_chance = { base = 2 }
	}

	option = {
		name = econ_forum.2.c
		log = "[GetDateText]: [This.GetName]: econ_forum.2.c executed"
		set_variable = { econ_forum_status^econ_forum_invited_to = -1 }
		ai_chance = { base = 1 }
	}
}

}
