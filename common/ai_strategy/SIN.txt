SIN_hold_new_wars_while_losing = {
	allowed = { original_tag = SIN }
	enable = {
		has_war = yes
		surrender_progress > 0.15
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = declare_war id = "MAY" value = -4000 }
	ai_strategy = { type = declare_war id = "IND" value = -4000 }
}

SIN_avoid_unready_war_with_may = {
	allowed = { original_tag = SIN }
	enable = {
		has_wargoal_against = MAY
		NOT = { has_war_with = MAY }
		OR = {
			strength_ratio = { tag = MAY ratio < 1 }
			MAY = { is_in_faction = yes }
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = conquer id = "MAY" value = -4000 }
	ai_strategy = { type = declare_war id = "MAY" value = -4000 }
}
