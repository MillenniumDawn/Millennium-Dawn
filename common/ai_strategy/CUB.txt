CUB_hate_america_while_embargoed = {
	allowed = { original_tag = CUB }
	enable = {
		has_opinion_modifier = CUB_USA_Embargo
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = antagonize id = "USA" value = 200 }
}

CUB_avoid_starting_wars = {
	allowed = { original_tag = CUB }
	enable = {
		has_war = yes
		enemies_strength_ratio > 1.2
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = avoid_starting_wars value = -200 }
}
