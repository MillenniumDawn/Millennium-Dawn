# China Naval Procurement Contracts — on_actions (Hong Kong)
# Weekly: deadline countdown for every active contract, save migration,
# capstone watchdog. Monthly: AI-PRC contract offer cadence.

on_actions = {
	on_weekly = {
		effect = {
			if = {
				limit = {
					original_tag = HKG
					has_country_flag = HKG_contract_active
				}

				# Upgrade a save made before the multi-contract rebuild.
				if = {
					limit = { NOT = { check_variable = { HKG_contract_version = 2 } } }
					HKG_contract_migrate_v2 = yes
				}

				# Count every active contract's deadline down; expired ones are missed.
				HKG_contract_weekly_tick = yes

				HKG_contract_check_capstone = yes
			}
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = {
					original_tag = HKG
					has_country_flag = HKG_contract_active
				}

				# AI-controlled Beijing offers a contract on a throttled cadence when
				# HKG has a free slot and the shipyards can build. Player-controlled
				# CHI uses the Commission Hong Kong Warships decision instead.
				if = {
					limit = {
						HKG_contract_can_solicit = yes
						HKG_contract_can_build_ships = yes
						NOT = { has_country_flag = HKG_contract_offer_cooldown }
						country_exists = CHI
						CHI = { is_ai = yes }
						NOT = { has_war_with = CHI }
					}
					country_event = { id = HKG_contract.1 days = 1 }
				}
			}
		}
	}
}
