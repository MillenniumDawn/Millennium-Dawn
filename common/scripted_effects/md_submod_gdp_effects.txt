# common/scripted_effects/md_submod_gdp_effects.txt

update_md_flaxbeard_gdp_graph = {
	clear_array = md_flax_offsets_green
	clear_array = md_flax_offsets_red
	clear_array = md_flax_slopes

	# --- YOY (JANUARY-ONLY SNAPSHOT) ---
	set_temp_variable = { months_since_jan = global.month }
	subtract_from_temp_variable = { months_since_jan = 1 }

	set_temp_variable = { recent_jan_idx = current_gdp_graph_end }
	subtract_from_temp_variable = { recent_jan_idx = months_since_jan }
	if = { limit = { check_variable = { recent_jan_idx < 0 } } set_temp_variable = { recent_jan_idx = 0 } }

	set_temp_variable = { prev_jan_idx = recent_jan_idx }
	subtract_from_temp_variable = { prev_jan_idx = 12 }
	if = { limit = { check_variable = { prev_jan_idx < 0 } } set_temp_variable = { prev_jan_idx = 0 } }

	set_temp_variable = { recent_jan_gdp = gdp_array^recent_jan_idx }
	set_temp_variable = { prev_jan_gdp = gdp_array^prev_jan_idx }
	if = { limit = { check_variable = { prev_jan_gdp < 0.001 } } set_temp_variable = { prev_jan_gdp = 0.001 } }

	# ADDED: Calculate the absolute change in Billions!
	set_variable = { md_yoy_gdp_change_billions = recent_jan_gdp }
	subtract_from_variable = { md_yoy_gdp_change_billions = prev_jan_gdp }

	# Calculate % Change using the billions we just calculated
	set_variable = { md_yoy_gdp_change = md_yoy_gdp_change_billions }
	divide_variable = { md_yoy_gdp_change = prev_jan_gdp }
	multiply_variable = { md_yoy_gdp_change = 100 }
	# ---------------------------------------------

	set_temp_variable = { true_max = gdp_array^0 }
	set_temp_variable = { true_min = gdp_array^0 }
	set_temp_variable = { max_jump = 0 }

	# 1. FIND BOUNDS & MAXIMUM SPIKE
	for_each_loop = {
		array = gdp_array
		value = v
		index = i

		if = {
			limit = { NOT = { check_variable = { i > current_gdp_graph_end } } }
			if = { limit = { check_variable = { v > true_max } } set_temp_variable = { true_max = v } }
			if = { limit = { check_variable = { v < true_min } } set_temp_variable = { true_min = v } }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }
			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { jump = gdp_array^next_i }
				subtract_from_temp_variable = { jump = v }
				if = { limit = { check_variable = { jump < 0 } } multiply_temp_variable = { jump = -1 } }
				if = { limit = { check_variable = { jump > max_jump } } set_temp_variable = { max_jump = jump } }
			}
		}
	}

	set_temp_variable = { data_diff = true_max }
	subtract_from_temp_variable = { data_diff = true_min }

	# 2. STANDARD PADDING
	if = {
		limit = { check_variable = { data_diff < 1 } }
		set_temp_variable = { high = true_max }
		multiply_temp_variable = { high = 1.2 }
		set_temp_variable = { low = true_min }
		multiply_temp_variable = { low = 0.8 }
	}
	else = {
		set_temp_variable = { pad = data_diff }
		multiply_temp_variable = { pad = 0.1 }
		set_temp_variable = { high = true_max }
		add_to_temp_variable = { high = pad }
		set_temp_variable = { low = true_min }
		subtract_from_temp_variable = { low = pad }
	}

	# 3. DYNAMIC ANTI-TEAR ZOOM & STEP
	md_apply_axis_zoom_and_step = yes

	# 4. Y-AXIS LABELS
	set_temp_variable = { buf = low }
	for_loop_effect = {
		start = 0 end = 11
		set_variable = { md_y_axis_label^v = buf }
		add_to_temp_variable = { buf = step }
	}

	# 5. DRAW THE LINE
	for_each_loop = {
		array = gdp_array
		value = v
		index = i

		if = {
			limit = { check_variable = { i > current_gdp_graph_end } }
			add_to_array = { md_flax_offsets_green = 1000 }
			add_to_array = { md_flax_offsets_red = 1000 }
			add_to_array = { md_flax_slopes = 50 }
		}
		else = {
			set_temp_variable = { y1_val = v }
			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }

			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { y2_val = gdp_array^next_i }
			}
			else = {
				set_temp_variable = { y2_val = v }
			}

			md_calc_segment_math = yes

			if = {
				limit = { check_variable = { is_falling = 1 } }
				add_to_array = { md_flax_offsets_red = offset }
				add_to_array = { md_flax_offsets_green = 1000 }
			}
			else = {
				add_to_array = { md_flax_offsets_green = offset }
				add_to_array = { md_flax_offsets_red = 1000 }
			}

			add_to_array = { md_flax_slopes = slope }
		}
	}

	gdp_graph_update_dirty_var = yes
}
