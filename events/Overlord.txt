# The follow-up fires from the former overlord. FROM.FROM keeps the original sender.
country_event = {
	id = subject_independence.2
	title = subject_independence.2.t
	desc = subject_independence.2.d
	picture = GFX_politics_negotiations
	is_triggered_only = yes

	option = {
		name = subject_independence.2.a
		ai_chance = { base = 85 }
	}

	option = {
		name = subject_independence.2.b
		trigger = {
			is_subject = no
			FROM.FROM = {
				exists = yes
				is_subject = no
				NOT = {
					is_in_faction_with = ROOT
					has_war_with = ROOT
					has_truce_with = ROOT
					has_non_aggression_pact_with = ROOT
				}
			}
			NOT = { has_wargoal_against = FROM.FROM }
		}
		log = "[GetDateText]: [This.GetName]: subject_independence.2.b executed"
		set_temp_variable = { wargoal_on = FROM.FROM.id }
		add_threat_from_wargoal_effect = yes
		create_wargoal = { type = puppet_wargoal_focus target = FROM.FROM expire = 120 }
		ai_chance = {
			base = 15
			modifier = {
				add = 20
				FROM.FROM = {
					is_neighbor_of = ROOT
					ROOT = { strength_ratio = { tag = PREV ratio > 3 } }
				}
			}
			modifier = {
				add = 10
				FROM.FROM = { ROOT = { has_opinion = { target = PREV value < -100 } } }
			}
			modifier = { factor = 0.25 FROM.FROM = { is_in_faction = yes } }
			modifier = { factor = 0.5 has_war_support < 0.35 }
			modifier = {
				factor = 0
				OR = {
					has_war = yes
					has_active_mission = bankruptcy_incoming_collapse
					ai_has_high_deficit = yes
					NOT = {
						FROM.FROM = {
							NOT = { is_top3_thirty_influencer = yes }
							ROOT = {
								strength_ratio = { tag = PREV ratio > 1.5 }
								has_opinion = { target = PREV value < -50 }
							}
						}
					}
				}
			}
		}
	}
}
