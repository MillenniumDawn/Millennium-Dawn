SOO_ai_alt_path = {
	OR = {
		has_global_flag = SOO_COMMUNIST_FOCUS_PATH
		has_global_flag = SOO_NATIONALIST_FOCUS_PATH
	}
}

SOO_ai_historical_path = {
	OR = {
		has_global_flag = SOO_HISTORICAL_FOCUS_PATH
		AND = {
			is_historical_focus_on = yes
			NOT = { SOO_ai_alt_path = yes }
		}
	}
}

SOO_ai_not_historical_path = {
	SOO_ai_alt_path = yes
}

SOO_ai_not_communist_path = {
	NOT = { has_global_flag = SOO_COMMUNIST_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = SOO_HISTORICAL_FOCUS_PATH
		has_global_flag = SOO_NATIONALIST_FOCUS_PATH
	}
}

SOO_ai_not_nationalist_path = {
	NOT = { has_global_flag = SOO_NATIONALIST_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = SOO_HISTORICAL_FOCUS_PATH
		has_global_flag = SOO_COMMUNIST_FOCUS_PATH
	}
}

SOO_ai_patriotic_path = {
	OR = {
		SOO_ai_historical_path = yes
		has_global_flag = SOO_NATIONALIST_FOCUS_PATH
	}
}

SOO_ai_not_patriotic_path = {
	has_global_flag = SOO_COMMUNIST_FOCUS_PATH
}
