AUS_toll_system_category = {
	AUS_increase_toll_rate = {
		fire_only_once = no
		visible = { NOT = { has_active_mission = AUS_increase_toll_rate_mission } }

		available = {
			check_variable = { toll_cost < 10 }
			NOT = { has_active_mission = AUS_increase_toll_rate_mission }
			NOT = { has_active_mission = AUS_decrease_toll_rate_mission }
		}

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision AUS_increase_toll_rate"
			activate_mission = AUS_increase_toll_rate_mission
		}

		ai_will_do = { base = 0 }
	}

	AUS_decrease_toll_rate = {
		fire_only_once = no
		visible = { NOT = { has_active_mission = AUS_decrease_toll_rate_mission } }

		available = {
			check_variable = { toll_cost > 0 }
			NOT = { has_active_mission = AUS_increase_toll_rate_mission }
			NOT = { has_active_mission = AUS_decrease_toll_rate_mission }
		}

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision AUS_decrease_toll_rate"
			activate_mission = AUS_decrease_toll_rate_mission
		}

		ai_will_do = { base = 0 }
	} #need more positive effects of tollcosts are low and more negative effects if tollcosts are high to make this more balanced
	AUS_increase_toll_rate_mission = {
		available = { always = no }
		activation = { always = no }
		days_mission_timeout = 30
		fire_only_once = no
		is_good = yes

		timeout_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision AUS_increase_toll_rate_mission"
			add_to_variable = { toll_cost = 2.5 }
			clamp_variable = { var = toll_cost min = 0 max = 10 }
			calc_toll_cost = yes
		}
	}

	AUS_decrease_toll_rate_mission = {
		available = { always = no }
		activation = { always = no }
		days_mission_timeout = 30
		fire_only_once = no
		is_good = yes

		timeout_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision AUS_decrease_toll_rate_mission"
			subtract_from_variable = { toll_cost = 2.5 }
			clamp_variable = { var = toll_cost min = 0 max = 10 }
			calc_toll_cost = yes
		}
	}
}
