	ASEAN_call_suspension_vote = {

		icon = GFX_decision_gulf_expel_button

		allowed = { is_asean_nation = yes }

		target_trigger = {
			FROM = {
				is_asean_member = yes
				NOT = { original_tag = ROOT.original_tag }
				NOT = { has_country_flag = ASEAN_suspended_flag }
				OR = {
					check_variable = { FROM.ASEAN_breach > 3 }
					has_country_flag = ASEAN_recent_coup
					any_of_scopes = {
						array = global.ASEAN_Member
						has_war_with = FROM
						NOT = { original_tag = FROM.original_tag }
					}
				}
			}
		}

		visible = {
			is_asean_member_committed = yes
			has_global_flag = ASEAN_asean_charter_completed
			check_variable = { global.ASEAN_suspend_candidates > 0 }
		}

		available = {
			has_political_power > 150
			custom_trigger_tooltip = {
				tooltip = ASEAN_suspension_vote_already_open_tt
				NOT = { has_global_flag = ASEAN_suspension_vote_open }
			}
		}

		cost = 150

		days_remove = 30
		days_re_enable = 365

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision ASEAN_call_suspension_vote"
			add_political_power = -75
			set_global_flag = { flag = ASEAN_suspension_vote_open days = 30 value = 1 }
			# The proposer's own ballot is cast with the motion, as the decision text says.
			set_variable = { global.ASEAN_suspension_vote_yes = 1 }
			set_variable = { global.ASEAN_suspension_vote_no = 0 }
			set_variable = { global.ASEAN_suspension_target = FROM.id }
			# Sent from the accused so asean_crisis.21 reads the motion's target in FROM.
			# The accused does not sit on the ballot for its own suspension.
			FROM = {
				for_each_scope_loop = {
					array = global.ASEAN_Member
					if = {
						limit = {
							NOT = { original_tag = ROOT.original_tag }
							NOT = { check_variable = { global.ASEAN_suspension_target = THIS.id } }
						}
						country_event = { id = asean_crisis.21 days = 5 }
					}
				}
			}
		}

		remove_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision ASEAN_call_suspension_vote resolved"
			if = {
				limit = {
					FROM = { exists = yes }
					check_variable = { global.ASEAN_suspension_vote_yes > global.ASEAN_suspension_vote_no }
					NOT = { has_global_flag = ASEAN_suspension_vetoed }
				}
				FROM = { ASEAN_apply_suspension = yes }
			}
			if = {
				limit = { has_global_flag = ASEAN_suspension_vetoed }
				clr_global_flag = ASEAN_suspension_vetoed
			}
			clear_variable = global.ASEAN_suspension_target
		}

		ai_will_do = {
			base = 0
			modifier = {
				add = 20
				FROM = { has_country_flag = ASEAN_recent_coup }
			}
		}
	}
