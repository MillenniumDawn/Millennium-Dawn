# The American side of the Solomon Islands contest.
#
# Two distinct postures, decided by the configuration the competition is
# running in rather than by anything the player clicks here:
#
#   Mode 2 - Washington is bankrolling Canberra. The joint decisions below cost
#            American money and score in Australia's column, because the point
#            of the alliance is that the ally is the one holding the ground.
#   Mode 3 - Canberra has changed sides. Washington negotiates with Honiara
#            directly, on the Compact of Free Association model that has held
#            Micronesia, the Marshalls and Palau since 1986. That takes four
#            beats, real money, and it costs the Australian relationship.
#
# The Compact arc tracks its progress in USA_SOL_compact_stage on the United
# States:
#   0  nothing opened
#   1  exploratory talks held
#   2  assistance package delivered
#   3  strategic denial clause agreed
#   4  Compact ratified

	# Australian crews. Washington pays and Canberra gets the credit, which is
	# the entire point of having an ally in the near region.

			country_exists = SOL

				limit = { country_exists = AST }

					navy_experience = 15

						target = ROOT
						modifier = USA_pacific_burden_sharing
					}

					target = AST
					modifier = USA_pacific_burden_sharing
				}

					add_opinion_modifier = {
						target = AST
						modifier = recently_invest

				}
			}
		}

			base = 15

				has_active_mission = bankruptcy_incoming_collapse

			}

			log = "[GetDateText]: [Root.GetName]: Decision remove USA_SOL_compact_ratification"
			set_variable = { USA_SOL_compact_stage = 4 }
			set_country_flag = USA_SOL_compact_signed
			set_temp_variable = { comp_points = 8 }
			SOL = { SOL_award_USA_points = yes }
			set_temp_variable = { percent_change = 8 }
			set_temp_variable = { tag_index = USA }
			set_temp_variable = { influence_target = SOL.id }
			change_influence_percentage = yes
			# Honiara's own side of the Compact standing is applied by the event
			# below, on the option Honiara actually takes. Applying it here as
			# well would stack the modifier twice.
			SOL = {
				country_event = { id = usa_pacific.15 days = 2 }
			}
			add_opinion_modifier = {
				target = SOL
				modifier = USA_pacific_compact_partner
			}
			# Beijing's standing in Honiara does not survive an exclusive
			# strategic denial clause.
			if = {
				limit = { country_exists = CHI }
				set_temp_variable = { percent_change = -8 }
				set_temp_variable = { tag_index = CHI }
				set_temp_variable = { influence_target = SOL.id }
				change_influence_percentage = yes
			}
			# And neither does Canberra's, which is the price of the whole arc.
			if = {
				limit = {
					country_exists = AST
					NOT = { has_war_with = AST }
				}
				set_temp_variable = { percent_change = -6 }
				set_temp_variable = { tag_index = AST }
				set_temp_variable = { influence_target = SOL.id }
				change_influence_percentage = yes
				# What this costs the Australian relationship depends on how
				# Canberra answers it, so the standing penalty is applied on
				# the option taken in usa_pacific.21 rather than twice over.
				AST = { country_event = { id = usa_pacific.21 days = 4 } }
			}
			news_event = { id = usa_pacific_news.3 days = 3 }
			# A resolution that stood itself down to let this negotiation finish
			# has nothing left to wait for now that the Compact is ratified, so
			# the contest settles rather than idling out the deferral cap.
			if = {
				limit = { SOL = { has_country_flag = SOL_comp_resolution_deferred } }
				SOL = { country_event = { id = solomon_competition.39 days = 14 } }
			}
		}

			base = 30
			modifier = {
				has_active_mission = bankruptcy_incoming_collapse
				factor = 0
			}
			modifier = {
				SOL_competition_caller_leads = yes
				add = 15
			}
			modifier = {
				is_historical_focus_on = yes
				add = 10
			}

	}

AST_solomon_engagement_category = {

	# The Australian half of the alliance path. Canberra runs the programme and
	# Washington underwrites it, which is cheaper for Australia than doing it
	# alone and cheaper for the United States than doing it itself. Only exists
	# while Australia is still the Western horse in the race.
	AST_SOL_joint_pacific_step_up = {
		icon = GFX_decision_hol_exchange_intelligence_data

		allowed = {
			original_tag = AST
		}

		available = {
			country_exists = SOL
			country_exists = USA
			SOL_competition_active = yes
			SOL_competition_is_usa_backing_ast = yes
			NOT = { has_war_with = SOL }
			NOT = { has_war_with = USA }
			AST_is_us_aligned_in_pacific = yes
			USA = {
				OR = {
					western_conservatism_are_in_power = yes
					western_liberals_are_in_power = yes
				}
			}
			check_variable = { treasury > 8 }
			# Three quarters of this programme is drawn on Washington's budget,
			# not Canberra's. An ally that cannot cover its own share is not
			# underwriting anything, so the American line is checked here as
			# well rather than quietly overdrawing a partner's treasury.
			custom_trigger_tooltip = {
				tooltip = AST_SOL_joint_step_up_usa_can_pay_tt

			NOT = { has_active_mission = bankruptcy_incoming_collapse }

		visible = {
			country_exists = SOL
			country_exists = USA
			SOL_competition_active = yes
			SOL_competition_is_usa_backing_ast = yes
		}

		fire_only_once = no
		cost = 35
		days_remove = 120
		days_re_enable = 545

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision AST_SOL_joint_pacific_step_up"

			log = "[GetDateText]: [Root.GetName]: Decision remove AST_SOL_joint_pacific_step_up"

			SOL = {
				add_stability = 0.02
				add_opinion_modifier = {
					target = AST
					modifier = recently_invest
				}
			}
			USA = {
				add_opinion_modifier = {
					target = ROOT
					modifier = USA_pacific_burden_sharing
				}
			}
			add_opinion_modifier = {
				target = USA
				modifier = USA_pacific_burden_sharing
			}
		}

		# rather than the American one the rest of this file uses.

			base = 14
			modifier = {
				factor = 0
				has_active_mission = bankruptcy_incoming_collapse
			}

			}
		}
	}
