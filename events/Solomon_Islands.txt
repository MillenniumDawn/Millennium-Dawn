#2003 intervention
country_event = {
	id = Solomon_Islands.5
	title = Solomon_Islands.5.t
	desc = Solomon_Islands.5.d
	picture = GFX_SOL_collapse
	is_triggered_only = yes

	trigger = {
		country_exists = SOL
		SOL = {
			controls_state = 1225
			controls_state = 533
			has_idea = SOL_aftermath_of_the_tensions
			is_puppet = no
		}
	}

	immediate = {
		hidden_effect = {
			add_stability = -0.3
			add_threat = 4
			533 = {
				if = {
				limit = { infrastructure > 0 }
				damage_building = {
					type = infrastructure
					damage = 1
				}
				}
			}
		}
	}

	#request intervention from AST
	option = {
		name = Solomon_Islands.5.a
		log = "[GetDateText]: [This.GetName]: Solomon_Islands.5.a executed"
		swap_ideas = {
			remove_idea = SOL_aftermath_of_the_tensions
			add_idea = SOL_collapsing_state
		}
		decrease_two_level_economic_growth = yes
		newline = yes
		AST = {
			country_event = solomon_islands_intervention.2
		}
		ai_chance = {
			base = 1
		}
	}

	#civil war
	option = {
		name = Solomon_Islands.5.b
		log = "[GetDateText]: [This.GetName]: Solomon_Islands.5.b executed"
		custom_effect_tooltip = SOL_Lead_to_hostilities_TT
		newline = yes
		swap_ideas = {
			remove_idea = SOL_aftermath_of_the_tensions
			add_idea = SOL_collapsing_state
		}
		decrease_two_level_economic_growth = yes
		hidden_effect = {
			country_event = {
				id = Solomon_Islands.9
				days = 8 
			}
		}
		ai_chance = {
			base = 0
		}
	}
}
