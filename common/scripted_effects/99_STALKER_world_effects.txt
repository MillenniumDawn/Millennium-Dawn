# How the rest of the world reacts to the Zones: diplomacy before Kaimanov, casualties among foreign teams.
STALKER_monthly_world_pulse = {
	STALKER_refresh_unsc_expected = yes
	STALKER_finish_security_council = yes
	if = {
		limit = {
			NOT = { has_global_flag = STALKER_kaimanov_outcome }
			NOT = { has_global_flag = STALKER_ipsf_formed }
			NOT = { has_global_flag = STALKER_great_power_talks_cooldown }
			any_of_scopes = {
				array = global.STALKER_active_zone_anchors
				check_variable = { STALKER_zone_id = 1 }
				check_variable = { STALKER_zone_containment < 40 }
			}
		}
		random = {
			chance = 10
			set_global_flag = { flag = STALKER_great_power_talks_cooldown value = 1 days = 730 }
			random_scope_in_array = {
				array = global.STALKER_active_zone_anchors
				limit = {
					check_variable = { STALKER_zone_id = 1 }
					controller = { exists = yes }
				}
				controller = { save_global_event_target_as = STALKER_zone_host_target set_variable = { global.STALKER_zone_host_target = THIS } }
			}
			if = {
				limit = { country_exists = USA }
				USA = { country_event = STALKER.132 }
			}
			if = {
				limit = { country_exists = SOV }
				SOV = { country_event = STALKER.133 }
			}
		}
	}
	if = {
		limit = { NOT = { has_global_flag = STALKER_expedition_casualty_cooldown } }
		random = {
			chance = 3
			random_country = {
				limit = {
					has_country_flag = STALKER_zone_research_access
					NOT = { STALKER_controls_active_zone = yes }
				}
				set_global_flag = { flag = STALKER_expedition_casualty_cooldown value = 1 days = 365 }
				save_global_event_target_as = STALKER_casualty_country
				set_variable = { global.STALKER_casualty_country = THIS }
				random_scope_in_array = {
					array = global.STALKER_active_zone_anchors
					limit = { controller = { exists = yes } }
					controller = { save_event_target_as = STALKER_zone_controller_target }
				}
				country_event = STALKER.134
				news_event = STALKER.135
			}
		}
	}
}

# One scripted Council session after the worldwide Kaymanov news. Not a UN resolution type.
# Ballots are sent to each permanent member, and to the original Zone's controller when that country is not already seated.
STALKER_schedule_security_council = {
	if = {
		limit = {
			STALKER_scenario_enabled = yes
			NOT = { has_global_flag = STALKER_unsc_scheduled }
		}
		set_global_flag = STALKER_unsc_scheduled
		set_variable = { global.STALKER_unsc_ballots = 0 }
		set_variable = { global.STALKER_unsc_containment = 0 }
		set_variable = { global.STALKER_unsc_access = 0 }
		set_variable = { global.STALKER_unsc_adjourn = 0 }
		set_variable = { global.STALKER_unsc_expected = 0 }
		set_variable = { global.STALKER_unsc_result = 0 }
		for_each_scope_loop = {
			array = global.p5_sc_members
			if = {
				limit = {
					exists = yes
					NOT = { has_country_flag = STALKER_unsc_seat }
				}
				set_country_flag = STALKER_unsc_seat
				country_event = { id = STALKER.137 days = 50 }
				add_to_variable = { global.STALKER_unsc_expected = 1 }
			}
		}
		698 = {
			if = {
				limit = {
					controller = {
						exists = yes
						NOT = { has_country_flag = STALKER_unsc_seat }
					}
				}
				controller = {
					set_country_flag = STALKER_unsc_seat
					country_event = { id = STALKER.137 days = 50 }
				}
				add_to_variable = { global.STALKER_unsc_expected = 1 }
			}
		}
	}
}

STALKER_refresh_unsc_expected = {
	if = {
		limit = {
			has_global_flag = STALKER_unsc_scheduled
			NOT = { has_global_flag = STALKER_unsc_resolved }
		}
		set_variable = { global.STALKER_unsc_expected = global.STALKER_unsc_ballots }
		every_country = {
			limit = {
				has_country_flag = STALKER_unsc_seat
				NOT = { has_country_flag = STALKER_unsc_voted }
			}
			add_to_variable = { global.STALKER_unsc_expected = 1 }
		}
	}
}

STALKER_finish_security_council = {
	STALKER_refresh_unsc_expected = yes
	if = {
		limit = {
			has_global_flag = STALKER_unsc_scheduled
			NOT = { has_global_flag = STALKER_unsc_resolved }
			check_variable = { global.STALKER_unsc_ballots = global.STALKER_unsc_expected }
		}
		set_global_flag = STALKER_unsc_resolved
		set_variable = { global.STALKER_unsc_result = 3 }
		if = {
			limit = {
				check_variable = { global.STALKER_unsc_containment > global.STALKER_unsc_access }
				check_variable = { global.STALKER_unsc_containment > global.STALKER_unsc_adjourn }
			}
			set_variable = { global.STALKER_unsc_result = 1 }
		}
		else_if = {
			limit = {
				check_variable = { global.STALKER_unsc_access > global.STALKER_unsc_containment }
				check_variable = { global.STALKER_unsc_access > global.STALKER_unsc_adjourn }
			}
			set_variable = { global.STALKER_unsc_result = 2 }
			every_country = {
				limit = { has_country_flag = STALKER_unsc_voted_access }
				STALKER_grant_zone_research_access = yes
			}
		}
		news_event = STALKER.138
	}
}

# After Kaimanov, the registered reference-map Zones awaken one at a time rather than all at once.
STALKER_monthly_proliferation_pulse = {
	if = {
		limit = {
			STALKER_scenario_enabled = yes
			has_global_flag = STALKER_global_proliferation
		}
		random = {
			chance = 50
			random_scope_in_array = {
				array = global.STALKER_zone_anchor_states
				limit = {
					STALKER_is_zone_location = yes
					NOT = { check_variable = { STALKER_zone_id = 1 } }
					NOT = { STALKER_is_active_zone = yes }
					NOT = { has_state_flag = STALKER_zone_policy_pending }
				}
				STALKER_awaken_reference_zone = yes
			}
		}
	}
}

STALKER_awaken_reference_zone = {
	log = "[GetDateText]: STALKER reference Zone [?STALKER_zone_id] awakens in [THIS.GetName]"
	# Expires so a lost policy answer cannot freeze the Zone; the event times out long before.
	set_state_flag = { flag = STALKER_zone_policy_pending value = 1 days = 60 }
	if = {
		limit = { controller = { exists = yes } }
		STALKER_target_zone = yes
		controller = { country_event = STALKER.1 }
		if = {
			limit = { controller = { is_major = yes } }
			controller = {
				set_variable = { STALKER_reaction_zone = PREV }
				country_event = { id = STALKER.136 days = 30 }
			}
		}
	}
	else = { STALKER_activate_zone = yes }
}

# ---- 99_STALKER_footprint_effects.txt (consolidated) ----
# A barely contained, highly active Zone spills into neighbouring states; a well-held one pulls back. A
# reference Zone at its full footprint can split, and one between the two extremes drifts.
STALKER_monthly_footprint_pulse = {
	for_each_scope_loop = {
		array = global.STALKER_active_zone_anchors
		if = {
			limit = {
				STALKER_is_active_zone_anchor = yes
				NOT = { has_state_flag = STALKER_zone_policy_pending }
				NOT = { has_state_flag = STALKER_footprint_cooldown }
				controller = { exists = yes }
			}
			if = {
				limit = {
					check_variable = { STALKER_zone_id > 1 }
					check_variable = { STALKER_zone_id < 100 }
					NOT = { has_state_flag = STALKER_zone_split }
					check_variable = { STALKER_zone_activity > 89 }
					check_variable = { STALKER_zone_containment < 25 }
					check_variable = { STALKER_zone_footprint^num > 2 }
				}
				random = {
					chance = 10
					STALKER_split_zone = yes
				}
			}
			else_if = {
				limit = {
					check_variable = { STALKER_zone_activity > 74 }
					check_variable = { STALKER_zone_containment < 40 }
					NOT = { check_variable = { STALKER_zone_footprint^num > 2 } }
				}
				random = {
					chance = 10
					STALKER_spread_zone_footprint = yes
				}
			}
			else_if = {
				limit = {
					check_variable = { STALKER_zone_containment > 69 }
					check_variable = { STALKER_zone_activity < 40 }
					check_variable = { STALKER_zone_footprint^num > 0 }
				}
				random = {
					chance = 10
					STALKER_recede_zone_footprint = yes
				}
			}
			else_if = {
				limit = {
					check_variable = { STALKER_zone_activity > 49 }
					check_variable = { STALKER_zone_containment < 70 }
					check_variable = { STALKER_zone_footprint^num > 0 }
				}
				random = {
					chance = 5
					STALKER_drift_zone_footprint = yes
				}
			}
		}
	}
}

STALKER_spread_zone_footprint = {
	controller = { save_event_target_as = STALKER_footprint_controller }
	random_neighbor_state = {
		limit = {
			is_controlled_by = event_target:STALKER_footprint_controller
			NOT = { STALKER_is_zone_location = yes }
			NOT = { has_state_flag = STALKER_zone_footprint }
		}
		set_state_flag = STALKER_zone_footprint
		set_variable = { STALKER_footprint_anchor = PREV }
		add_dynamic_modifier = { modifier = STALKER_zone_presence }
		save_event_target_as = STALKER_footprint_event_target
		PREV = {
			add_to_array = { STALKER_zone_footprint = PREV }
			set_state_flag = { flag = STALKER_footprint_cooldown value = 1 days = 180 }
		}
		log = "[GetDateText]: STALKER Zone [?PREV.STALKER_zone_id] spreads into [THIS.GetName]"
		PREV = { STALKER_target_zone = yes }
		event_target:STALKER_footprint_controller = {
			STALKER_refresh_zone_administration = yes
			country_event = STALKER.110
		}
	}
}

# The newest footprint state is released first.
STALKER_recede_zone_footprint = {
	set_state_flag = { flag = STALKER_footprint_cooldown value = 1 days = 180 }
	set_temp_variable = { STALKER_footprint_last = { value = STALKER_zone_footprint^num subtract = 1 } }
	set_temp_variable = { STALKER_footprint_state = STALKER_zone_footprint^STALKER_footprint_last }
	remove_from_array = { array = STALKER_zone_footprint index = STALKER_footprint_last }
	STALKER_target_zone = yes
	var:STALKER_footprint_state = {
		STALKER_clear_zone_footprint_state = yes
		save_event_target_as = STALKER_footprint_event_target
		log = "[GetDateText]: STALKER Zone recedes from [THIS.GetName]"
	}
	if = {
		limit = { controller = { exists = yes } }
		controller = {
			STALKER_refresh_zone_administration = yes
			country_event = STALKER.111
		}
	}
}

# The oldest footprint state is released and a new neighbour taken, so the Zone's edge moves.
STALKER_drift_zone_footprint = {
	controller = { save_event_target_as = STALKER_footprint_controller }
	set_temp_variable = { STALKER_footprint_count = STALKER_zone_footprint^num }
	random_neighbor_state = {
		limit = {
			is_controlled_by = event_target:STALKER_footprint_controller
			NOT = { STALKER_is_zone_location = yes }
			NOT = { has_state_flag = STALKER_zone_footprint }
		}
		set_state_flag = STALKER_zone_footprint
		set_variable = { STALKER_footprint_anchor = PREV }
		add_dynamic_modifier = { modifier = STALKER_zone_presence }
		save_event_target_as = STALKER_footprint_event_target
		PREV = { add_to_array = { STALKER_zone_footprint = PREV } }
	}
	if = {
		limit = { check_variable = { STALKER_zone_footprint^num > STALKER_footprint_count } }
		set_state_flag = { flag = STALKER_footprint_cooldown value = 1 days = 180 }
		set_temp_variable = { STALKER_footprint_state = STALKER_zone_footprint^0 }
		remove_from_array = { array = STALKER_zone_footprint index = 0 }
		var:STALKER_footprint_state = {
			STALKER_clear_zone_footprint_state = yes
			save_event_target_as = STALKER_footprint_released_target
		}
		log = "[GetDateText]: STALKER Zone [?STALKER_zone_id] drifts into a new neighbouring state"
		STALKER_target_zone = yes
		event_target:STALKER_footprint_controller = {
			STALKER_refresh_zone_administration = yes
			country_event = STALKER.110
		}
	}
}

# The oldest footprint state breaks away as a new Zone whose ID is this Zone's plus 100.
STALKER_split_zone = {
	set_state_flag = STALKER_zone_split
	set_state_flag = { flag = STALKER_footprint_cooldown value = 1 days = 180 }
	add_to_variable = { STALKER_zone_activity = -15 }
	set_temp_variable = { STALKER_footprint_state = STALKER_zone_footprint^0 }
	remove_from_array = { array = STALKER_zone_footprint index = 0 }
	var:STALKER_footprint_state = {
		STALKER_clear_zone_footprint_state = yes
		set_temp_variable = { STALKER_new_zone_id = { value = PREV.STALKER_zone_id add = 100 } }
		set_temp_variable = { STALKER_new_zone_profile = PREV.STALKER_zone_profile }
		STALKER_register_zone_anchor = yes
		log = "[GetDateText]: STALKER Zone [?PREV.STALKER_zone_id] splits into [THIS.GetName]"
		STALKER_awaken_reference_zone = yes
	}
	if = {
		limit = { controller = { exists = yes } }
		controller = { STALKER_refresh_zone_administration = yes }
	}
}

STALKER_clear_zone_footprint_state = {
	clr_state_flag = STALKER_zone_footprint
	clear_variable = STALKER_footprint_anchor
	if = {
		limit = { has_dynamic_modifier = { modifier = STALKER_zone_presence } }
		remove_dynamic_modifier = { modifier = STALKER_zone_presence }
	}
}

# Smuggled artifacts pool globally; every five can surface as a foreign customs seizure. The seizure is traced
# to the Zone whose black market leaked the most last month.
STALKER_roll_artifact_seizure = {
	if = {
		limit = {
			check_variable = { global.STALKER_smuggled_artifacts > 4.99 }
			NOT = { has_global_flag = STALKER_artifact_seizure_cooldown }
		}
		random = {
			chance = 25
			set_temp_variable = { STALKER_seizure_state = 0 }
			set_temp_variable = { STALKER_seizure_best = 0 }
			for_each_scope_loop = {
				array = global.STALKER_active_zone_anchors
				if = {
					limit = {
						controller = { exists = yes }
						check_variable = { STALKER_zone_black_market > STALKER_seizure_best }
					}
					set_temp_variable = { STALKER_seizure_best = STALKER_zone_black_market }
					set_temp_variable = { STALKER_seizure_state = THIS }
				}
			}
			if = {
				limit = { NOT = { check_variable = { STALKER_seizure_state = 0 } } }
				var:STALKER_seizure_state = {
					controller = { save_event_target_as = STALKER_zone_controller_target }
				}
			}
			random_scope_in_array = {
				array = global.majors
				limit = {
					exists = yes
					NOT = { STALKER_controls_active_zone = yes }
				}
				set_global_flag = { flag = STALKER_artifact_seizure_cooldown value = 1 days = 180 }
				add_to_variable = { global.STALKER_smuggled_artifacts = -5 }
				country_event = STALKER.20
			}
		}
	}
}

			NOT = { has_country_flag = STALKER_zone_research_access }

		}

		event_target:STALKER_zone_controller_target = { country_event = STALKER.21 }
	}
}

	if = {

			}

# Success scales with Polesie security; a high roll exposes officials on the smugglers' payroll.
STALKER_run_kgb_investigation = {
	set_temp_variable = { STALKER_kgb_roll = 0 }
	randomize_temp_variable = { var = STALKER_kgb_roll min = 1 max = 100 distribution = uniform }
	set_temp_variable = { STALKER_kgb_success = { value = STALKER_polesie_security divide = 2 add = 25 } }
	if = {
		limit = { check_variable = { STALKER_kgb_roll < STALKER_kgb_success } }
		country_event = STALKER.61
	}
	else_if = {
		limit = { check_variable = { STALKER_kgb_roll > 80 } }
		country_event = STALKER.62
	}
	else = {
		country_event = STALKER.64
	}
}

# Treaty observers who reach a station during a polar artifact study seize the samples and report them.
# The cut-short flag outlives the 180-day result event by a day.
STALKER_polar_study_detected = {

	}

}
