gpu_development_monthly_driver = {
	if = {
		limit = {
			corporate_history_enabled = yes
			NOT = { has_country_flag = collapsed_nation }
			OR = { original_tag = USA original_tag = CAN original_tag = TAI original_tag = KOR original_tag = CHI original_tag = JAP }
		}
		if = {
			limit = { NOT = { has_country_flag = gpu_development_country_bootstrapped } }
			gpu_development_reconstruct_history = yes
			set_country_flag = gpu_development_country_bootstrapped
		}
		if = {
			limit = { corporate_history_full_enabled = yes }
			gpu_development_schedule_year_events = yes
			gpu_development_recover_due_events = yes
		}
		else_if = {
			limit = { corporate_history_outcomes_only_enabled = yes }
			gpu_development_reconstruct_history = yes
		}
	}
}
