# SEATO founded through the shared ASEAN tree can be led by any member, so the cohesion
# weight SIA_seato_member_cohesion gives the Thai path needs one block per plausible leader
# other than Thailand, which SIA_seato_member_cohesion already covers.
# Without it the AI joins and then re-evaluates the alliance on its own terms and leaves,
# which is the failure documented on SIA_seato_member_cohesion in SIA.txt.

ASEAN_seato_cohesion_IND = {
	allowed = {
		is_asean_nation = yes
		NOT = { original_tag = IND }
	}
	enable = {
		has_global_flag = ASEAN_seato_founded
		is_in_faction_with = IND
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = alliance id = "IND" value = 200 }
	ai_strategy = { type = befriend id = "IND" value = 100 }
}
