MDRC_offer_ceasefires = {
	if = {
		limit = { country_exists = RWA has_war_with = RWA }
		RWA = {
			country_event = { id = mdrc.1 days = 1 }
		}
	}
	if = {
		limit = { country_exists = UGA has_war_with = UGA }
		UGA = {
			country_event = { id = mdrc.1 days = 1 }
		}
	}
	if = {
		limit = { country_exists = RCD has_war_with = RCD }
		RCD = {
			country_event = { id = mdrc.1 days = 1 }
		}
	}
	if = {
		limit = { country_exists = MLC has_war_with = MLC }
		MLC = {
			country_event = { id = mdrc.1 days = 1 }
		}
	}
}

MDRC_offer_integration = {
	if = {
		limit = {
			country_exists = RCD
			NOT = { has_war_with = RCD }
			RCD = {
				OR = { is_subject = no is_subject_of = DRC }
			}
		}
		RCD = {
			country_event = { id = mdrc.2 days = 1 }
		}
	}
	if = {
		limit = {
			country_exists = MLC
			NOT = { has_war_with = MLC }
			MLC = {
				OR = { is_subject = no is_subject_of = DRC }
			}
		}
		MLC = {
			country_event = { id = mdrc.2 days = 1 }
		}
	}
}

MDRC_rebel_wargoals = {
	if = {
		limit = {
			country_exists = RCD
			NOT = { has_war_with = RCD }
			NOT = { is_in_faction_with = RCD }
			RCD = { is_subject = no }
		}
		create_wargoal = { type = annex_everything target = RCD }
	}
	if = {
		limit = {
			country_exists = MLC
			NOT = { has_war_with = MLC }
			NOT = { is_in_faction_with = MLC }
			MLC = { is_subject = no }
		}
		create_wargoal = { type = annex_everything target = MLC }
	}
}
