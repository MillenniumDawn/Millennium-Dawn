# Regional profiles adjust a Zone's starting values once it awakens.
STALKER_apply_zone_profile = {
	if = {
		limit = { check_variable = { STALKER_zone_profile = 1 } }
		add_to_variable = { STALKER_zone_activity = 10 }
		add_to_variable = { STALKER_zone_containment = 10 }
		add_to_variable = { STALKER_zone_stalkers = -10 }
		add_to_variable = { STALKER_zone_bandits = -15 }
	}
	else_if = {
		limit = { check_variable = { STALKER_zone_profile = 2 } }
		add_to_variable = { STALKER_zone_containment = -10 }
		add_to_variable = { STALKER_zone_stalkers = 10 }
		add_to_variable = { STALKER_zone_bandits = 10 }
	}
	else_if = {
		limit = { check_variable = { STALKER_zone_profile = 3 } }
		add_to_variable = { STALKER_zone_activity = 10 }
		add_to_variable = { STALKER_zone_containment = -10 }
		add_to_variable = { STALKER_zone_bandits = 15 }
	}
	else_if = {
		limit = { check_variable = { STALKER_zone_profile = 4 } }
		add_to_variable = { STALKER_zone_containment = 15 }
		add_to_variable = { STALKER_zone_exploitation = 10 }
		if = {
			limit = { check_variable = { STALKER_zone_id = 1 } }
			add_to_variable = { STALKER_zone_duty = 10 }
		}
		else = { add_to_variable = { STALKER_zone_military = 10 } }
	}
	clamp_variable = { var = STALKER_zone_activity min = 0 max = 100 }
	STALKER_clamp_zone_society = yes
}

STALKER_roll_regional_event = {
	if = {
		limit = {
			check_variable = { STALKER_zone_profile > 0 }
			NOT = { has_state_flag = STALKER_regional_event_cooldown }
			controller = { exists = yes }
		}
		random = {
			chance = 5
			set_state_flag = { flag = STALKER_regional_event_cooldown value = 1 days = 365 }
			STALKER_target_zone = yes
			if = {
				limit = { check_variable = { STALKER_zone_profile = 1 } }
				if = {
					limit = {
						OR = {
							check_variable = { STALKER_zone_id = 2 }
							check_variable = { STALKER_zone_id = 6 }
							check_variable = { STALKER_zone_id = 7 }
							check_variable = { STALKER_zone_id = 19 }
							check_variable = { STALKER_zone_id = 20 }
							check_variable = { STALKER_zone_id = 21 }
							check_variable = { STALKER_zone_id = 22 }
						}
					}
					random_list = {
						50 = { controller = { country_event = STALKER.41 } }
						50 = { controller = { country_event = STALKER.46 } }
					}
				}
				else = { controller = { country_event = STALKER.41 } }
			}
			else_if = {
				limit = { check_variable = { STALKER_zone_profile = 2 } }
				if = {
					limit = {
						OR = {
							check_variable = { STALKER_zone_id = 13 }
							check_variable = { STALKER_zone_id = 23 }
						}
					}
					random_list = {
						50 = { controller = { country_event = STALKER.42 } }
						50 = { controller = { country_event = STALKER.47 } }
					}
				}
				else = { controller = { country_event = STALKER.42 } }
			}
			else_if = {
				limit = { check_variable = { STALKER_zone_profile = 3 } }
				if = {
					limit = {
						OR = {
							check_variable = { STALKER_zone_id = 9 }
							check_variable = { STALKER_zone_id = 10 }
							check_variable = { STALKER_zone_id = 11 }
							check_variable = { STALKER_zone_id = 25 }
							check_variable = { STALKER_zone_id = 30 }
							check_variable = { STALKER_zone_id = 36 }
							check_variable = { STALKER_zone_id = 37 }
						}
					}
					random_list = {
						50 = { controller = { country_event = STALKER.43 } }
						50 = { controller = { country_event = STALKER.48 } }
					}
				}
				else = { controller = { country_event = STALKER.43 } }
			}
			else = {
				if = {
					limit = { check_variable = { STALKER_zone_id = 17 } }
					random_list = {
						50 = { controller = { country_event = STALKER.44 } }
						50 = { controller = { country_event = STALKER.45 } }
					}
				}
				else = { controller = { country_event = STALKER.44 } }
			}
		}
	}
}
