#################################
### Voting Decision triggers ####
#################################

is_socialist_block = {
	OR = {
		western_social_democrats_are_in_power = yes
		neutrality_neutral_social_are_in_power = yes
	}
}

EU_voting_decision_result_trigger = {
	check_variable = { global.EU_pop_ratio_yes > 0.65 }
	check_expr = {
		value = global.EU_active_vote_supporters^num
		divide = global.EU_member^num
		greater_than = { value = 0.55 }
	}
}

EU_eurozone_reforms_passed = {
	is_in_array = { array = global.EU_passed_votes value = 202 }
	is_in_array = { array = global.EU_passed_votes value = 203 }
	is_in_array = { array = global.EU_passed_votes value = 205 }

}
