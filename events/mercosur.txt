			limit = { has_idea = BRA_idea_associate_states }

		}
		if = {
			limit = { FROM = { has_completed_focus = BRA_education_integration } }
			add_ideas = BRA_idea_mercosur_member_state_trade_agreement_after_education
		}
		else_if = {
			limit = { FROM = { has_completed_focus = BRA_trade_agreement } }
			add_ideas = BRA_idea_mercosur_member_state_trade_agreement_before_education
		}
		else = {
			add_ideas = BRA_idea_mercosur_member_state
		}

		add_opinion_modifier = { target = FROM modifier = mercosur_focus_tt }

		FROM = {
			add_opinion_modifier = { target = ROOT modifier = mercosur_focus_tt }
			country_event = { id = mercosur.07 days = 0 }
		}

		ai_chance = {
			base = 8

			modifier = {
				add = 25
				has_idea = BRA_idea_associate_states
			}
			modifier = {
				add = 20

				add = 60

			}

				add = 15
				FROM = { has_completed_focus = BRA_education_integration }

				add = -30

		}
	}

	option = {

			add_political_power = -25
			add_opinion_modifier = { target = ROOT modifier = mercosur_rejected }
			country_event = { id = mercosur.08 days = 0 }
