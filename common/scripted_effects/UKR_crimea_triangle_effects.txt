# Recalculate diplomatic percentages from the MD parallel influence arrays.
# One unit of raw influence is shared by all foreign contributors and domestic independence.
UKR_ct_foreign_influence_refresh = {
 set_variable = { UKR_ct_ukr_raw = 0 }
 set_variable = { UKR_ct_sov_raw = 0 }
 set_temp_variable = { ct_total = domestic_influence_amount }
 for_each_loop = { array = influence_array_val value = ct_entry
  add_to_temp_variable = { ct_total = ct_entry }
 }
 for_each_loop = { array = influence_array value = ct_actor index = ct_idx
  if = { limit = { check_variable = { var = ct_actor value = UKR.id } }
   set_variable = { UKR_ct_ukr_raw = influence_array_val^ct_idx }
  }
  if = { limit = { check_variable = { var = ct_actor value = SOV.id } }
   set_variable = { UKR_ct_sov_raw = influence_array_val^ct_idx }
  }
 }
 set_variable = { UKR_ct_ukr_share = 0 }
 if = { limit = { check_variable = { var = ct_total value = 0 compare = greater_than } }
  set_variable = { UKR_ct_ukr_share = UKR_ct_ukr_raw }
  divide_variable = { var = UKR_ct_ukr_share value = ct_total }
  multiply_variable = { UKR_ct_ukr_share = 100 }
 }
 set_variable = { UKR_ct_russian_triple_limit = UKR_ct_ukr_raw }
 multiply_variable = { UKR_ct_russian_triple_limit = 3 }
}

# v0.7: The only action permitted to raise Ukrainian influence over the standard 65-point ceiling.

# Idempotent across the existing Sevastopol event and on_monthly synchronization.

 if = {
  limit = {
   has_country_flag = UKR_ct_initialized_flag
   has_country_flag = UKR_ct_sevastopol_was_foreign_flag
   NOT = { has_country_flag = UKR_ct_sevastopol_bonus_applied_flag }
   OR = {
    669 = { is_owned_by = UKR }
    AND = { 669 = { is_owned_by = CRM } CRM = { is_subject_of = UKR } }
   }
   1112 = { is_owned_by = UKR }
  }
  set_temp_variable = { ct_v1_bay_u_before = UKR_ct_u }
  set_temp_variable = { ct_v1_bay_t_before = UKR_ct_t }
  set_temp_variable = { ct_v1_bay_r_before = UKR_ct_r }
  set_country_flag = UKR_ct_sevastopol_bonus_applied_flag
  set_temp_variable = { ct_bay_bonus_remaining = 15 }
  set_temp_variable = { ct_bay_room = 100 }
  subtract_from_temp_variable = { ct_bay_room = UKR_ct_u }
  if = {
   limit = { check_variable = { var = ct_bay_bonus_remaining value = ct_bay_room compare = greater_than } }
   set_temp_variable = { ct_bay_bonus_remaining = ct_bay_room }
  }
  # Primary donor is SOV / pro-Russian bloc.
