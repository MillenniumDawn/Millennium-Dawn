internal_investment_purchase_action = {
	set_temp_variable = { internal_investment_state = id }
	internal_investment_set_action_cost = yes
	calculate_internal_investment_costs = yes
	effect_tooltip = { internal_investment_apply_benefit = yes }
	ROOT = { custom_effect_tooltip = internal_investment_cost_tt }
	if = { limit = { NOT = { check_for_investment_project_limits = yes } } custom_effect_tooltip = internal_investment_no_slot_tt }
	if = { limit = { ROOT = { has_active_mission = bankruptcy_incoming_collapse } } custom_effect_tooltip = internal_investment_bankruptcy_tt }
	if = { limit = { internal_investment_action_on_cooldown = yes } custom_effect_tooltip = internal_investment_active_tt }
	if = {
		limit = { NOT = { internal_investment_action_eligible = yes } }
		if = { limit = { check_variable = { internal_investment_action = 3 } } custom_effect_tooltip = internal_investment_capacity_limit_tt }
		else_if = { limit = { check_variable = { internal_investment_action = 11 } } custom_effect_tooltip = internal_investment_coastal_required_tt }
		else_if = { limit = { check_variable = { internal_investment_action = 13 } } custom_effect_tooltip = internal_investment_amazon_required_tt }
	}
	if = { limit = { NOT = { internal_investment_can_pay = yes } } custom_effect_tooltip = internal_investment_insufficient_funds_tt }
	hidden_effect = {
		if = {
			limit = {
				automated_internal_investment_state_available = yes
				internal_investment_action_eligible = yes
				NOT = { internal_investment_action_on_cooldown = yes }
				internal_investment_can_pay = yes
			}
			internal_investment_apply_benefit = yes
			internal_investment_charge_costs = yes
		}
	}
}

internal_investment_toggle_request = {
	if = {
		limit = { internal_investment_repeat_toggle_allowed = yes }
		ROOT = {
			if = {
				limit = { is_in_array = { array = automated_internal_investment_requests@var:internal_investment_state value = internal_investment_action } }
				remove_from_array = { array = automated_internal_investment_requests@var:internal_investment_state value = internal_investment_action }
			}
			else = {
				add_to_array = { array = automated_internal_investment_requests@var:internal_investment_state value = internal_investment_action }
			}
			if = {
				limit = { check_variable = { automated_internal_investment_requests@var:internal_investment_state^num > 0 } }
				if = {
					limit = { NOT = { is_in_array = { array = automated_internal_investment_states value = PREV } } }
					add_to_array = { array = automated_internal_investment_states value = PREV }
				}
			}
			else = { remove_from_array = { array = automated_internal_investment_states value = PREV } }
		}
		refresh_investment_gui = yes
	}
}

# Build the next queue without changing the array being traversed.
automation_internal_investments_weekly = {
	clear_temp_array = automated_internal_investment_waiting
	clear_temp_array = automated_internal_investment_rotate
	set_temp_variable = { automation_internal_investment_attempt = 1 }
	for_each_scope_loop = {
		array = automated_internal_investment_states
		set_temp_variable = { internal_investment_state = id }
		if = {
			limit = { NOT = { OR = { is_owned_by = ROOT is_controlled_by = ROOT } } }
			ROOT = { clear_array = automated_internal_investment_requests@var:internal_investment_state }
		}
		else_if = {
			limit = { has_automated_internal_investment_request = yes }
			set_temp_variable = { automation_internal_investment_started = 0 }
			if = {
				limit = { automated_internal_investment_state_available = yes }
				set_temp_variable = { automation_internal_investment_available_pp = { value = ROOT.political_power subtract = ROOT.automation_pp_reserve } }
				for_loop_effect = {
					start = 1
					end = 14
					value = internal_investment_action
					break = automation_internal_investment_started
					if = {
						limit = {
							internal_investment_repeat_selected = yes
							internal_investment_action_eligible = yes
							NOT = { internal_investment_action_on_cooldown = yes }
						}
						internal_investment_purchase_action = yes
					}
				}
			}
			if = {
				limit = { has_automated_internal_investment_request = yes }
				ROOT = {
					if = {
						limit = { check_variable = { automation_internal_investment_started = 1 } }
						add_to_temp_array = { automated_internal_investment_rotate = PREV }
					}
					else = { add_to_temp_array = { automated_internal_investment_waiting = PREV } }
				}
			}
		}
	}
	set_temp_variable = { automation_internal_investment_attempt = 0 }
	clear_array = automated_internal_investment_states
	for_each_scope_loop = {
		array = automated_internal_investment_waiting
		ROOT = { add_to_array = { array = automated_internal_investment_states value = PREV } }
	}
	for_each_scope_loop = {
		array = automated_internal_investment_rotate
		ROOT = { add_to_array = { array = automated_internal_investment_states value = PREV } }
	}
	clear_temp_array = automated_internal_investment_waiting
	clear_temp_array = automated_internal_investment_rotate
}

automation_internal_investments_daily = {
	set_temp_variable = { automation_internal_investment_available_pp = { value = political_power subtract = automation_pp_reserve } }
	if = {
		limit = { check_variable = { var = automation_internal_investment_available_pp value = 0.15 compare = greater_than_or_equals } }
		add_political_power = -0.15
		set_country_flag = { flag = automation_internal_investment_upkeep_paid value = 1 days = 2 }
	}
}
