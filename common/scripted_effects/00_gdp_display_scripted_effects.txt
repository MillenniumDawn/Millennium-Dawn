# common/scripted_effects/00_gdp_display_scripted_effects.txt
# Submod override of base Millennium Dawn's GDP display script.
# Overhauls the 10-year shift mechanism to maintain a clean 108-month rolling window (Years 1-9 on Pointers 0-8).
# Synchronizes all 10 graphs (GDP, GDP/C, Prod, Debt, Pop, Budget, Energy, Defense, Trade, Inflation).

graph_initial_setup = {
	resize_array = {
		array = gdp_graphx
		value = 0
		size = 120
	}

	resize_array = {
		array = gdp_graph
		value = 0
		size = 120
	}

	resize_array = {
		array = gdp_array
		value = gdp_total
		size = 120
	}

	resize_array = {
		array = gdpc_array
		value = gdp_per_capita
		size = 120
	}

	resize_array = {
		array = prod_array
		value = overall_productivity
		size = 120
	}

	set_temp_variable = { tmp = global.year }
	for_loop_effect = {
		start = 0
		end = 10

		set_variable = { date_scale^v = tmp }
		add_to_temp_variable = { tmp = 1 }
	}

	set_variable = { current_gdp_graph_end = 0 }
	set_variable = { gdp_graph_dirty_var = 0 }

	set_variable = { gdp_array^current_gdp_graph_end = gdp_total }
	set_variable = { gdpc_array^current_gdp_graph_end = gdp_per_capita }
	set_variable = { prod_array^current_gdp_graph_end = overall_productivity }
}

initialize_x_axis = {
	set_temp_variable = { x_coord = 0 }

	for_loop_effect = {
		start = 0
		end = current_gdp_graph_end
		compare = less_than_or_equals

		set_variable = { gdp_graphx^v = x_coord }
		add_to_temp_variable = { x_coord = 7.05 }
	}
}

recalculate_gdp_display = {
	update_md_flaxbeard_gdp_graph = yes
}

recalculate_gdp_c_display = {
	update_md_flaxbeard_gdpc_graph = yes
}

recalculate_prod_display = {
	update_md_flaxbeard_prod_graph = yes
}

# -------------------------------------------------------------
# SHIFT ALL 10 DATASETS BY EXACTLY 12 MONTHS (1 FULL YEAR)
# -------------------------------------------------------------
gdp_graph_shift_array = {
	# Shift indices 12..107 down to 0..95
	set_temp_variable = { new_index = 12 }
	for_loop_effect = {
		start = 0
		end = 96

		# Base MD Arrays
		set_variable = { gdp_array^v = gdp_array^new_index }
		set_variable = { gdpc_array^v = gdpc_array^new_index }
		set_variable = { prod_array^v = prod_array^new_index }

		# Submod Custom Arrays
		set_variable = { debt_array^v = debt_array^new_index }
		set_variable = { inflation_array^v = inflation_array^new_index }
		set_variable = { pop_array^v = pop_array^new_index }
		set_variable = { workforce_array^v = workforce_array^new_index }
		set_variable = { employment_array^v = employment_array^new_index }
		set_variable = { revenue_array^v = revenue_array^new_index }
		set_variable = { expense_array^v = expense_array^new_index }
		set_variable = { budget_balance_array^v = budget_balance_array^new_index }
		set_variable = { energy_gen_array^v = energy_gen_array^new_index }
		set_variable = { energy_con_array^v = energy_con_array^new_index }
		set_variable = { energy_bal_array^v = energy_bal_array^new_index }
		set_variable = { defense_spend_array^v = defense_spend_array^new_index }
		set_variable = { defense_pct_array^v = defense_pct_array^new_index }
		set_variable = { trade_exports_array^v = trade_exports_array^new_index }
		set_variable = { trade_imports_array^v = trade_imports_array^new_index }
		set_variable = { trade_balance_array^v = trade_balance_array^new_index }

		add_to_temp_variable = { new_index = 1 }
	}

	# Pad future unused slots (97..119) with latest valid value (index 95)
	for_loop_effect = {
		start = 97
		end = 120

		set_variable = { gdp_array^v = gdp_array^95 }
		set_variable = { gdpc_array^v = gdpc_array^95 }
		set_variable = { prod_array^v = prod_array^95 }

		set_variable = { debt_array^v = debt_array^95 }
		set_variable = { inflation_array^v = inflation_array^95 }
		set_variable = { pop_array^v = pop_array^95 }
		set_variable = { workforce_array^v = workforce_array^95 }
		set_variable = { employment_array^v = employment_array^95 }
		set_variable = { revenue_array^v = revenue_array^95 }
		set_variable = { expense_array^v = expense_array^95 }
		set_variable = { budget_balance_array^v = budget_balance_array^95 }
		set_variable = { energy_gen_array^v = energy_gen_array^95 }
		set_variable = { energy_con_array^v = energy_con_array^95 }
		set_variable = { energy_bal_array^v = energy_bal_array^95 }
		set_variable = { defense_spend_array^v = defense_spend_array^95 }
		set_variable = { defense_pct_array^v = defense_pct_array^95 }
		set_variable = { trade_exports_array^v = trade_exports_array^95 }
		set_variable = { trade_imports_array^v = trade_imports_array^95 }
		set_variable = { trade_balance_array^v = trade_balance_array^95 }
	}

	# Set current end to 96 so the current January writes to index 96 (Pointer 8)
	set_variable = { current_gdp_graph_end = 96 }

	# Advance X-axis date scale by 1 year
	set_temp_variable = { tmp = date_scale^0 }
	add_to_temp_variable = { tmp = 1 }
	for_loop_effect = {
		start = 0
		end = 10

		set_variable = { date_scale^v = tmp }
		add_to_temp_variable = { tmp = 1 }
	}

	gdp_graph_update_dirty_var = yes
}

# Neutralize base MD's 5% artificial flattening/smoothing
correct_gdp_graph = { }

# Base MD monthly hook called from MD_on_actions
gdp_graph_on_monthly = {
	# Advance month index
	add_to_variable = { current_gdp_graph_end = 1 }
	set_variable = { gdp_array^current_gdp_graph_end = gdp_total }
	set_variable = { gdpc_array^current_gdp_graph_end = gdp_per_capita }
	set_variable = { prod_array^current_gdp_graph_end = overall_productivity }

	# If we reach month 108 (Pointer 9, Year 10 Jan), shift immediately!
	if = {
		limit = { check_variable = { current_gdp_graph_end > 107 } }
		gdp_graph_shift_array = yes
	}
}

gdp_graph_update_dirty_var = {
	if = {
		limit = { check_variable = { gdp_graph_dirty_var > 1000 } }
		set_variable = { gdp_graph_dirty_var = 0 }
	}
	else = {
		add_to_variable = { gdp_graph_dirty_var = 1 }
	}
}
