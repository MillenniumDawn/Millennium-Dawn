	custom_effect_tooltip = yugo_sends_invites_tt

		news_event = news.85
		if = {
			limit = {
				country_exists = SER
				NOT = { original_tag = SER }
			}
			SER = { country_event = yugoslavia.1 }
		}
		if = {
			limit = {
				country_exists = CRO
				NOT = { original_tag = CRO }
			}
			CRO = { country_event = yugoslavia.1 }
		}
		if = {
			limit = {
				country_exists = SLV
				NOT = { original_tag = SLV }
			}
			SLV = { country_event = yugoslavia.1 }
		}
		if = {
			limit = {
				country_exists = BOS
				NOT = { original_tag = BOS }
			}
			BOS = { country_event = yugoslavia.1 }
		}
		if = {
			limit = {
				country_exists = FYR
				NOT = { original_tag = FYR }
			}
			FYR = { country_event = yugoslavia.1 }
		}
		if = {
			limit = {
				country_exists = MNT
				NOT = { original_tag = MNT }
			}
			MNT = { country_event = yugoslavia.1 }
		}
		if = {
			limit = {
				country_exists = KOS
				NOT = { original_tag = KOS }
			}
			KOS = { country_event = yugoslavia.1 }
		}
		set_global_flag = GLOBAL_yugoslavia_restored
		add_named_threat = {
			threat = 1.4
			name = formed_yugoslavia
		}
		add_ideas = formed_yugoslavia
		set_cosmetic_tag = yugoslavia_restored
		set_capital = {
			state = 131
			remember_old_capital = no
		}
		133 = { set_demilitarized_zone = no }
	}
	custom_effect_tooltip = TT_IF_THEY_ACCEPT
	effect_tooltip = { custom_effect_tooltip = yugo_invite_accept_tt }
	custom_effect_tooltip = TT_IF_THEY_REJECT
	effect_tooltip = { custom_effect_tooltip = yugo_invite_reject_tt }
}
