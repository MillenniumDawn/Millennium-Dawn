add_namespace = STALKER

country_event = {
	id = STALKER.80
	title = STALKER.80.t
	desc = STALKER.80.d
	picture = GFX_crime_fighting
	is_triggered_only = yes

	trigger = {
		STALKER_scenario_enabled = yes
		has_variable = STALKER_mdst_penetration
	}

	immediate = {
		log = "[GetDateText]: [This.GetName]: STALKER.80 immediate executed"
		set_temp_variable = { STALKER_mdst_stolen = STALKER_artifact_stockpile }
		clamp_temp_variable = { var = STALKER_mdst_stolen min = 0 max = 5 }
		subtract_from_variable = { STALKER_artifact_stockpile = STALKER_mdst_stolen }
		add_to_variable = { global.STALKER_smuggled_artifacts = STALKER_mdst_stolen }
		if = {
			limit = { check_variable = { STALKER_xlab_progress > 0 } }
			add_to_variable = { STALKER_xlab_progress = -20 }
			clamp_variable = { var = STALKER_xlab_progress min = 0 max = 100 }
		}
	}

	option = {
		name = STALKER.80.a
		log = "[GetDateText]: [This.GetName]: STALKER.80.a executed"
		add_political_power = -25
		custom_effect_tooltip = STALKER.80.a_tt
		hidden_effect = {
			add_to_variable = { STALKER_mdst_penetration = -10 }
			clamp_variable = { var = STALKER_mdst_penetration min = 0 max = 80 }
		}
		ai_chance = {
			base = 60
			modifier = {
				factor = 0
				has_political_power < 25
			}
		}
	}

	option = {
		name = STALKER.80.b
		log = "[GetDateText]: [This.GetName]: STALKER.80.b executed"
		custom_effect_tooltip = STALKER.80.b_tt
		ai_chance = { base = 40 }
	}
}
