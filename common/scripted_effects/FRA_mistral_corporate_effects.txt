FRA_mistral_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = FRA_mistral_state_initialized } }
		set_variable = { FRA_mistral_model_capability = 3 }
		set_variable = { FRA_mistral_open_weights = 6 }
		set_variable = { FRA_mistral_compute_sovereignty = 2 }
		set_variable = { FRA_mistral_commercial_scale = 1 }
		set_variable = { FRA_mistral_strategic_independence = 6 }
		set_country_flag = FRA_mistral_state_initialized
	}
	FRA_mistral_clamp_state = yes
}

FRA_mistral_clamp_state = {
	clamp_variable = { var = FRA_mistral_model_capability min = 0 max = 10 }
	clamp_variable = { var = FRA_mistral_open_weights min = 0 max = 10 }

FRA_mistral_resolve_capstone = {
	FRA_mistral_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = FRA_mistral_capstone_resolved } }
		if = {
			limit = { check_variable = { FRA_mistral_capstone_choice = 1 } }
			FRA_mistral_apply_european_champion = yes
		}
		else_if = {
			limit = { check_variable = { FRA_mistral_capstone_choice = 2 } }
			FRA_mistral_apply_open_weights_standard = yes
		}
		else_if = {
			limit = { check_variable = { FRA_mistral_capstone_choice = 3 } }
			FRA_mistral_apply_sovereign_compute_stack = yes
		}
		else_if = {
			limit = { check_variable = { FRA_mistral_capstone_choice = 4 } }
			FRA_mistral_apply_hyperscaler_annex = yes
		}
		else = { FRA_mistral_apply_credible_challenger = yes }
	}
}
