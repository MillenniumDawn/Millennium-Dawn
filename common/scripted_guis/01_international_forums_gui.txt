scripted_gui = {
	MD_forums_system_gui = {
		window_name = MD_forums_system_window
		context_type = player_context
		parent_window_name = MD_countrymissilesview

		dirty = global.international_systems_updated

		visible = {
			check_variable = { var_open_MD_forums_gui = 2 }
			has_country_flag = open_MD_countrymissilesview
		}

		triggers = {
			forums_no_forum_visible = { NOT = { has_variable = econ_forum_hosted } }
			forums_preparation_visible = { has_variable = econ_forum_hosted }
			forums_program_visible = { has_variable = econ_forum_hosted }
			econ_forum_row_sponsor_visible = { econ_forum_guest_row = yes }
			econ_forum_row_sponsor_click_enabled = { econ_forum_can_sponsor = yes }
			econ_forum_row_speak_visible = { econ_forum_guest_row = yes }
			econ_forum_row_speak_click_enabled = { econ_forum_can_bid = yes }
			econ_forum_row_request_visible = { econ_forum_guest_row = yes }
			econ_forum_row_request_click_enabled = { econ_forum_can_request = yes }
			econ_forum_row_boycott_visible = { econ_forum_guest_row = yes }
			econ_forum_row_boycott_click_enabled = {
				OR = {
					check_variable = { econ_forum_boycott^v = 1 }
					econ_forum_can_boycott = yes
				}
			}
		}

		effects = {
			econ_forum_row_sponsor_click = { econ_forum_sponsor_session = yes }
			econ_forum_row_speak_click = { econ_forum_bid_to_speak = yes }
			econ_forum_row_request_click = { econ_forum_request_invitation = yes }
			econ_forum_row_boycott_click = { econ_forum_toggle_boycott = yes }
		}

		dynamic_lists = {
			forums_rows_grid = {
				array = econ_forum_screen_rows
				entry_container = econ_forum_row
			}
			forums_leaders_grid = {
				array = econ_forum_screen_tracks
				entry_container = econ_forum_leader_row
			}
		}

		ai_enabled = { always = no }
	}
}
