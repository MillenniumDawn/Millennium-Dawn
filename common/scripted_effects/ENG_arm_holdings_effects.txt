# Arm Holdings Corporate History: bounded state, route ownership, and terminal resolution.

ENG_arm_holdings_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = ENG_arm_holdings_state_initialized } }
		set_variable = { ENG_arm_holdings_ecosystem_strength = 3 }
		set_variable = { ENG_arm_holdings_domestic_value_capture = 3 }
		set_variable = { ENG_arm_holdings_strategic_control = 1 }
		set_country_flag = ENG_arm_holdings_state_initialized
	}
	ENG_arm_holdings_clamp_state = yes
}

ENG_arm_holdings_clamp_state = {
	set_temp_variable = { corp_value = ENG_arm_holdings_ecosystem_strength }
	corporate_history_clamp_value = yes
	set_variable = { ENG_arm_holdings_ecosystem_strength = corp_value }
	set_temp_variable = { corp_value = ENG_arm_holdings_domestic_value_capture }
	corporate_history_clamp_value = yes
	set_variable = { ENG_arm_holdings_domestic_value_capture = corp_value }
	set_temp_variable = { corp_value = ENG_arm_holdings_strategic_control }
	corporate_history_clamp_value = yes
	set_variable = { ENG_arm_holdings_strategic_control = corp_value }
}

# Each external ecosystem can contribute at most one point, and none is a route gate.
ENG_arm_holdings_apply_external_ecosystem_signals = {
	if = {
		limit = {
			NOT = { has_country_flag = ENG_arm_holdings_apple_signal_integrated }
			country_exists = USA
			USA = {
				OR = {
					has_country_flag = USA_apple_carrier_iphone
					has_country_flag = USA_apple_open_mobile_platform
				}
			}
		}
		add_to_variable = { ENG_arm_holdings_ecosystem_strength = 1 }
		set_country_flag = ENG_arm_holdings_apple_signal_integrated
	}
	if = {
		limit = {
			NOT = { has_country_flag = ENG_arm_holdings_nokia_signal_integrated }
			country_exists = FIN
			FIN = {
				OR = {
					has_country_flag = FIN_nokia_device_scale_priority
					has_country_flag = FIN_nokia_mass_market_scale
				}
			}
		}
		add_to_variable = { ENG_arm_holdings_ecosystem_strength = 1 }
		set_country_flag = ENG_arm_holdings_nokia_signal_integrated
	}
	if = {
		limit = {
			NOT = { has_country_flag = ENG_arm_holdings_tsmc_signal_integrated }
			country_exists = TAI
			TAI = {
				OR = {
					has_country_flag = TAI_tsmc_pure_play_foundry
					has_country_flag = TAI_tsmc_open_innovation_platform
					has_country_flag = TAI_tsmc_mobile_anchor_strategy
				}
			}
		}
		add_to_variable = { ENG_arm_holdings_ecosystem_strength = 1 }
		set_country_flag = ENG_arm_holdings_tsmc_signal_integrated
	}
	ENG_arm_holdings_clamp_state = yes
}

ENG_arm_holdings_resolve_terminal = {
	ENG_arm_holdings_initialize_state = yes
	if = {
		limit = { NOT = { ENG_arm_holdings_has_terminal_outcome = yes } }
		if = {
			limit = { ENG_arm_holdings_can_resolve_nvidia_control = yes }
			ENG_arm_holdings_apply_nvidia_control_capstone = yes
		}
		else_if = {
			limit = { ENG_arm_holdings_can_resolve_golden_share = yes }
			ENG_arm_holdings_apply_golden_share_capstone = yes
		}
		else_if = {
			limit = { ENG_arm_holdings_can_resolve_sovereign_champion = yes }
			ENG_arm_holdings_apply_sovereign_champion_capstone = yes
		}
		else = { ENG_arm_holdings_apply_british_founded_capstone = yes }
	}
}

		set_country_flag = ENG_arm_holdings_event_04_resolved

		clr_country_flag = ENG_arm_holdings_event_04_delivery_expected
		add_to_variable = { ENG_arm_holdings_ecosystem_strength = 1 }
		add_to_variable = { ENG_arm_holdings_domestic_value_capture = 2 }
	}

			date > 2016.7.18
			NOT = { has_country_flag = ENG_arm_holdings_event_05_resolved }

		ENG_arm_holdings_clear_ownership_route = yes

		}

		clr_country_flag = ENG_arm_holdings_event_06_pending
		clr_country_flag = ENG_arm_holdings_event_06_delivery_expected

		add_to_variable = { ENG_arm_holdings_strategic_control = 1 }
		ENG_arm_holdings_apply_external_ecosystem_signals = yes
	}
	if = {
		limit = {
			date > 2020.9.13
			NOT = { has_country_flag = ENG_arm_holdings_event_07_resolved }
		}
		ENG_arm_holdings_clear_nvidia_proposal_route = yes
		set_country_flag = ENG_arm_holdings_nvidia_full_review
		set_country_flag = ENG_arm_holdings_event_07_resolved
		clr_country_flag = ENG_arm_holdings_event_07_pending
		clr_country_flag = ENG_arm_holdings_event_07_delivery_expected
		add_to_variable = { ENG_arm_holdings_strategic_control = 2 }
	}
	if = {
		limit = {
			date > 2021.4.19
			NOT = { has_country_flag = ENG_arm_holdings_event_08_resolved }
		}
		set_country_flag = ENG_arm_holdings_event_08_resolved
		clr_country_flag = ENG_arm_holdings_event_08_pending
		clr_country_flag = ENG_arm_holdings_event_08_delivery_expected
		add_to_variable = { ENG_arm_holdings_strategic_control = 2 }
	}

		limit = {
			date > 2022.2.8
			NOT = { has_country_flag = ENG_arm_holdings_event_09_resolved }
		}
		ENG_arm_holdings_clear_nvidia_resolution = yes
		set_country_flag = ENG_arm_holdings_nvidia_abandoned

		clr_country_flag = ENG_arm_holdings_event_09_delivery_expected

	}
	if = {
		limit = {
			date > 2023.9.14
			NOT = { has_country_flag = ENG_arm_holdings_event_10_resolved }
			NOT = { has_country_flag = ENG_arm_holdings_terminal_resolved }
		}
		set_country_flag = ENG_arm_holdings_nasdaq_softbank_control
		set_country_flag = ENG_arm_holdings_event_10_resolved
		clr_country_flag = ENG_arm_holdings_event_10_pending
		clr_country_flag = ENG_arm_holdings_event_10_delivery_expected
		add_to_variable = { ENG_arm_holdings_ecosystem_strength = 1 }
		add_to_variable = { ENG_arm_holdings_domestic_value_capture = 1 }
		add_to_variable = { ENG_arm_holdings_strategic_control = -1 }
		ENG_arm_holdings_apply_british_founded_capstone = yes
	}
	if = {
		limit = {
			date > 2023.9.14
			has_country_flag = ENG_arm_holdings_event_10_resolved
			NOT = { ENG_arm_holdings_has_terminal_outcome = yes }
		}
		ENG_arm_holdings_resolve_terminal = yes
	}
	if = {
		limit = {
			date > 2023.9.14
			ENG_arm_holdings_has_terminal_outcome = yes
			NOT = { has_country_flag = ENG_arm_holdings_reconstruct_complete }
		}
		set_country_flag = ENG_arm_holdings_reconstruct_complete
	}
	ENG_arm_holdings_clamp_state = yes
}
