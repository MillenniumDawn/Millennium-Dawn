		original_tag = BOS

		original_tag = MNT
		original_tag = KOS
	}
}

yugoslav_consensus_trigger = {
	custom_trigger_tooltip = {
		tooltip = yugoslavia_not_restored_tt
		NOT = { has_global_flag = GLOBAL_yugoslavia_restored }
	}
	custom_trigger_tooltip = {
		tooltip = yugo_consensus_tt
		all_of_scopes = {
			array = global.yugo_members
			OR = {
				NOT = { exists = yes }
				check_variable = { yugo_nostalgia > 74 }
			}
		}
	}
}

is_yugo_dominant_trigger = {
	custom_trigger_tooltip = {
		tooltip = yugo_dominant_tt
		check_variable = { global.yugo_dominant = ROOT.id }
	}
}
