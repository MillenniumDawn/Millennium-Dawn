TAI_foxconn_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = TAI_foxconn_state_initialized } }

		set_variable = { TAI_foxconn_geographic_resilience = 2 }
		set_variable = { TAI_foxconn_customer_diversification = 4 }
		set_variable = { TAI_foxconn_technology_depth = 4 }
		set_variable = { TAI_foxconn_labor_governance = 2 }
		set_country_flag = TAI_foxconn_state_initialized
	}
	if = {
		limit = {
			NOT = { has_country_flag = TAI_foxconn_capstone_resolved }
			NOT = { has_idea = TAI_foxconn_factory_behind_brand }
		}
		add_ideas = TAI_foxconn_factory_behind_brand
	}
	TAI_foxconn_clamp_state = yes
}

TAI_foxconn_clamp_state = {
	clamp_variable = { var = TAI_foxconn_geographic_resilience min = 0 max = 10 }
	clamp_variable = { var = TAI_foxconn_customer_diversification min = 0 max = 10 }
	clamp_variable = { var = TAI_foxconn_technology_depth min = 0 max = 10 }
	clamp_variable = { var = TAI_foxconn_labor_governance min = 0 max = 10 }
}

TAI_foxconn_apply_wisconsin_reconstruction = {
	if = {
		limit = {
			country_exists = USA
			USA = { has_country_flag = USA_foxconn_wisconsin_rejected }
		}
		set_country_flag = TAI_foxconn_wisconsin_declined
		set_country_flag = TAI_foxconn_asia_reinvestment
		add_to_variable = { TAI_foxconn_geographic_resilience = 2 }
		add_to_variable = { TAI_foxconn_customer_diversification = 1 }
	}
	else_if = {
		limit = {
			country_exists = USA
			NOT = { has_war_with = USA }
			USA = {
				NOT = { has_country_flag = collapsed_nation }
				779 = { is_owned_by = USA }
			}
		}
		set_country_flag = TAI_foxconn_wisconsin_maximum_proposed
		set_country_flag = TAI_foxconn_wisconsin_committed
		add_to_variable = { TAI_foxconn_geographic_resilience = 1 }
		add_to_variable = { TAI_foxconn_technology_depth = 1 }
		add_ideas = TAI_foxconn_wisconsin_obligation
		USA = {
			if = {
				limit = {
					NOT = { has_country_flag = USA_foxconn_wisconsin_full_support }
					NOT = { has_country_flag = USA_foxconn_wisconsin_conditional_support }
					NOT = { has_country_flag = USA_foxconn_wisconsin_rejected }
				}
				set_country_flag = USA_foxconn_wisconsin_full_support
			}
		}
	}
	else = {
		set_country_flag = TAI_foxconn_wisconsin_declined
		set_country_flag = TAI_foxconn_asia_reinvestment
		add_to_variable = { TAI_foxconn_geographic_resilience = 2 }
		add_to_variable = { TAI_foxconn_customer_diversification = 1 }
	}
}
