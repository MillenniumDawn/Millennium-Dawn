# Second Economic Committee — foreign arms contract gating.
# "Class active" triggers report whether any of the 3 contract slots holds an
# unfulfilled contract of a given class (1 = infantry arms, 2 = artillery,
# 3 = armor, 4 = ballistic missiles). Slots are unrolled (triggers cannot loop).

NKO_contract_inf_active = {
	OR = {
		AND = { check_variable = { NKO_arms_contract_occupied_array^0 = 1 } check_variable = { NKO_arms_contract_class_array^0 = 1 } check_variable = { NKO_arms_contract_quota_array^0 > 0 } }
		AND = { check_variable = { NKO_arms_contract_occupied_array^1 = 1 } check_variable = { NKO_arms_contract_class_array^1 = 1 } check_variable = { NKO_arms_contract_quota_array^1 > 0 } }
		AND = { check_variable = { NKO_arms_contract_occupied_array^2 = 1 } check_variable = { NKO_arms_contract_class_array^2 = 1 } check_variable = { NKO_arms_contract_quota_array^2 > 0 } }
	}
}

NKO_contract_arty_active = {
	OR = {
		AND = { check_variable = { NKO_arms_contract_occupied_array^0 = 1 } check_variable = { NKO_arms_contract_class_array^0 = 2 } check_variable = { NKO_arms_contract_quota_array^0 > 0 } }
		AND = { check_variable = { NKO_arms_contract_occupied_array^1 = 1 } check_variable = { NKO_arms_contract_class_array^1 = 2 } check_variable = { NKO_arms_contract_quota_array^1 > 0 } }
		AND = { check_variable = { NKO_arms_contract_occupied_array^2 = 1 } check_variable = { NKO_arms_contract_class_array^2 = 2 } check_variable = { NKO_arms_contract_quota_array^2 > 0 } }
	}
}

NKO_contract_armor_active = {
	OR = {
		AND = { check_variable = { NKO_arms_contract_occupied_array^0 = 1 } check_variable = { NKO_arms_contract_class_array^0 = 3 } check_variable = { NKO_arms_contract_quota_array^0 > 0 } }
		AND = { check_variable = { NKO_arms_contract_occupied_array^1 = 1 } check_variable = { NKO_arms_contract_class_array^1 = 3 } check_variable = { NKO_arms_contract_quota_array^1 > 0 } }
		AND = { check_variable = { NKO_arms_contract_occupied_array^2 = 1 } check_variable = { NKO_arms_contract_class_array^2 = 3 } check_variable = { NKO_arms_contract_quota_array^2 > 0 } }
	}
}

NKO_contract_missile_active = {
	OR = {
		AND = { check_variable = { NKO_arms_contract_occupied_array^0 = 1 } check_variable = { NKO_arms_contract_class_array^0 = 4 } check_variable = { NKO_arms_contract_quota_array^0 > 0 } }
		AND = { check_variable = { NKO_arms_contract_occupied_array^1 = 1 } check_variable = { NKO_arms_contract_class_array^1 = 4 } check_variable = { NKO_arms_contract_quota_array^1 > 0 } }
		AND = { check_variable = { NKO_arms_contract_occupied_array^2 = 1 } check_variable = { NKO_arms_contract_class_array^2 = 4 } check_variable = { NKO_arms_contract_quota_array^2 > 0 } }
	}
}

NKO_arms_contract_can_solicit = {
	has_country_flag = NKO_arms_export_active
	NOT = { has_country_flag = NKO_arms_contract_offer_pending }
	check_variable = { NKO_arms_contract_count < NKO_arms_contract_max_slots }
}
