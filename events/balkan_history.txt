		SER = { has_country_flag = SER_icty_cooperation_karadzic }

	option = {
		name = balkan_news.1.b
		trigger = { original_tag = SER }
		log = "[GetDateText]: [This.GetName]: balkan_news.1.b executed"

		set_temp_variable = { party_popularity_increase = 0.02 }
		add_relative_party_popularity = yes
		custom_effect_tooltip = SER_nat_populism_gain_tt
	}

	option = {
		name = balkan_news.1.c
		trigger = {
			NOT = { original_tag = BOS }
			NOT = { original_tag = SER }
		}
	}
