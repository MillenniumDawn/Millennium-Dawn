	icon = south_america

	x = 0
	y = 8

	cost = 5

	search_filters = { FOCUS_FILTER_DIPLOMACY FOCUS_FILTER_MERCOSUR FOCUS_FILTER_UNASUL }

	allow_branch = { has_country_flag = mercosur_joint_tree_member }

	offset = {
		x = -34
		y = -8
		trigger = { original_tag = BRA }
	}

	offset = {
		x = 50
		y = 6
		trigger = { original_tag = BOL }
	}

	offset = {
		x = 47
		y = 2
		trigger = { original_tag = VEN }
	}

	available = {
		is_subject = no
		has_at_least_regional_power_status = yes
		has_dynamic_modifier = mercosur_member_state_modifier
	}

	completion_reward = {
		log = "[GetDateText]: [THIS.GetName]: Focus BRA_americas_unity"

		for_each_scope_loop = {
			array = global.mercosur_potential
			if = {
				limit = {
					exists = yes
					NOT = { tag = ROOT }
					NOT = { tag = MEX }
					NOT = { tag = PAN }
				}
				add_opinion_modifier = { target = ROOT modifier = medium_increase }
				reverse_add_opinion_modifier = { target = ROOT modifier = medium_increase }
			}
		}

		if = {
			limit = { NOT = { has_global_flag = mercosur.30_broadcasted } }
			set_global_flag = mercosur.30_broadcasted

			for_each_scope_loop = {
				array = global.mercosur_potential
				if = {
					limit = {
						exists = yes
						NOT = { tag = MEX }
						NOT = { tag = PAN }
						NOT = { has_country_flag = mercosur.30_received }
					}
					set_country_flag = mercosur.30_received
					country_event = { id = mercosur.30 }
				}
			}
		}

		custom_effect_tooltip = BRA_americas_unity_TT
	}

	ai_will_do = {
		base = 45
		modifier = {
			factor = 2
			is_historical_focus_on = yes
		}
	}
}
