MD_ftc_start_exercise_capital = {
	navy_experience = @MD_ftc_exercise_cost_neg
	add_timed_idea = { idea = MD_ftc_exercise_capital days = @MD_ftc_exercise_days }
	if = {
		limit = { check_variable = { MD_ftc_exercise_level < 1 } }
		add_daily_mastery = { name = MD_ftc_exercise_capital track = capital_ships days = @MD_ftc_exercise_days amount = @MD_ftc_mastery_level_0 }
	}
	else_if = {
		limit = { check_variable = { MD_ftc_exercise_level = 1 } }
		add_daily_mastery = { name = MD_ftc_exercise_capital track = capital_ships days = @MD_ftc_exercise_days amount = @MD_ftc_mastery_level_1 }
	}
	else_if = {
		limit = { check_variable = { MD_ftc_exercise_level = 2 } }
		add_daily_mastery = { name = MD_ftc_exercise_capital track = capital_ships days = @MD_ftc_exercise_days amount = @MD_ftc_mastery_level_2 }
	}
	else = {
		add_daily_mastery = { name = MD_ftc_exercise_capital track = capital_ships days = @MD_ftc_exercise_days amount = @MD_ftc_mastery_level_3 }
	}
}

MD_ftc_upgrade_exercise_program = {
	if = {
		limit = { check_variable = { MD_ftc_exercise_level < 1 } }
		navy_experience = @MD_ftc_upgrade_cost_1_neg
	}
	else_if = {
		limit = { check_variable = { MD_ftc_exercise_level = 1 } }
		navy_experience = @MD_ftc_upgrade_cost_2_neg
	}
	else = {
		navy_experience = @MD_ftc_upgrade_cost_3_neg
	}
	add_to_variable = { MD_ftc_exercise_level = 1 }
}

MD_ftc_upgrade_doctrine_center = {
	if = {
		limit = { check_variable = { MD_ftc_doctrine_center_level < 1 } }
		navy_experience = @MD_ftc_upgrade_cost_1_neg
		add_ideas = MD_ftc_doctrine_center_1
	}
	else_if = {
		limit = { check_variable = { MD_ftc_doctrine_center_level = 1 } }
		navy_experience = @MD_ftc_upgrade_cost_2_neg
		swap_ideas = { remove_idea = MD_ftc_doctrine_center_1 add_idea = MD_ftc_doctrine_center_2 }
	}
	else = {
		navy_experience = @MD_ftc_upgrade_cost_3_neg
		swap_ideas = { remove_idea = MD_ftc_doctrine_center_2 add_idea = MD_ftc_doctrine_center_3 }
	}
	add_to_variable = { MD_ftc_doctrine_center_level = 1 }
}

MD_ftc_run_staff_course = {
	navy_experience = @MD_ftc_staff_course_cost_neg
	every_navy_leader = { gain_xp = @MD_ftc_staff_course_xp }
	set_country_flag = { flag = MD_ftc_staff_course_cooldown days = @MD_ftc_staff_course_cooldown_days value = 1 }
}

MD_ftc_run_flag_officer_course = {
	navy_experience = @MD_ftc_flag_officer_cost_neg
	random_navy_leader = {
		limit = { skill < @MD_ftc_flag_officer_skill_cap }
		add_skill_level = 1
	}
	set_country_flag = { flag = MD_ftc_flag_officer_course_cooldown days = @MD_ftc_flag_officer_cooldown_days value = 1 }
}
