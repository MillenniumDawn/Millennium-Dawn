#---------# Chechnya On Actions: #---------#
#------------------------------------------#
on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = {
					tag = CHE
				}
				CK_init_supply_yield_effect = yes
			}
		}
	}

	on_weekly_CHE = {
		effect = {
			if = {
				limit = {
					country_exists = CHE
				}
				CHE_contraband_effect_on_action = yes
			}
		}
	}
}
