set_leader_MAC = {

	if = { limit = { check_variable = { ruling_party = 6 } }

		if = { limit = { check_variable = { reactionary_leader = 0 } }
			add_to_variable = { reactionary_leader = 1 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Fernando Chui"
				picture = "fernando_chui.dds"
				ideology = reactionary
				traits = {
					emerging_reactionary
					pro_china
				}
			}

			if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { reactionary_leader = 1 } }
			if = { limit = { date < 2016.1.2 } set_temp_variable = { b = 1 } } #skip if 2017
		}
		if = { limit = { check_variable = { reactionary_leader = 1 } NOT = { check_variable = { b = 1 } } }
			add_to_variable = { reactionary_leader = 1 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Ho lat seng"
				picture = "ho_lat_seng.dds"
				ideology = reactionary
				traits = {
					emerging_reactionary
					pro_china
				}
			}

			if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { reactionary_leader = 1 } }
			if = { limit = { date < 2019.12.20 } set_temp_variable = { b = 1 } } #skip if 2017
		}
		if = { limit = { check_variable = { reactionary_leader = 2 } NOT = { check_variable = { b = 1 } } }
			add_to_variable = { reactionary_leader = 1 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Sam Hou Fai"
				picture = "sam_hou_fai.dds"
				ideology = reactionary
				traits = {
					emerging_reactionary
					pro_china
				}
			}

			if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { reactionary_leader = 1 } }
			if = { limit = { date < 2024.12.20 } set_temp_variable = { b = 1 } } #skip if 2017
		}
	}
}
