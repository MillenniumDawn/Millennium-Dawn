# SIA_trade_diplomacy_network
# Thailand's trade-diplomacy soft-power lever, granted by the SIA_regional_trade_network
# focus. Its foreign_influence_modifier is variable-backed: SIA_recalculate_trade_diplomacy
# reads the generic MD signed_trade_agreements counter, scales it, and clamps it, so a
# wider Thai trade network buys more reach when spreading influence.
#
# Deliberately weaker than Singapore's SIN_singaporean_trade_strength: half the per-agreement
# rate (0.01 vs 0.02) and a third of the ceiling (+0.10 vs +0.30). Refreshed monthly in
# 99_SIA_on_actions.txt and on focus completion, then force_update_dynamic_modifier = yes.
SIA_trade_diplomacy_network = {
	enable = { always = yes }
	icon = "GFX_idea_trade"

	foreign_influence_modifier = SIA_trade_influence_strength
	custom_modifier_tooltip = SIA_trade_diplomacy_network_tt
}
