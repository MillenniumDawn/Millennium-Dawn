# A barely contained, highly active Zone spills into neighbouring states; a well-held one pulls back. A
# reference Zone at its full footprint can split, and one between the two extremes drifts.
STALKER_monthly_footprint_pulse = {
	for_each_scope_loop = {
		array = global.STALKER_active_zone_anchors
		if = {
			limit = {
				STALKER_is_active_zone_anchor = yes
				NOT = { has_state_flag = STALKER_zone_policy_pending }
				NOT = { has_state_flag = STALKER_footprint_cooldown }
				controller = { exists = yes }
			}
			if = {
				limit = {
					check_variable = { STALKER_zone_id > 1 }
					check_variable = { STALKER_zone_id < 100 }
					NOT = { has_state_flag = STALKER_zone_split }
					check_variable = { STALKER_zone_activity > 89 }
					check_variable = { STALKER_zone_containment < 25 }
					check_variable = { STALKER_zone_footprint^num > 2 }
				}
				random = {
					chance = 10
					STALKER_split_zone = yes
				}
			}
			else_if = {
				limit = {
					check_variable = { STALKER_zone_activity > 74 }
					check_variable = { STALKER_zone_containment < 40 }
					NOT = { check_variable = { STALKER_zone_footprint^num > 2 } }
				}
				random = {
					chance = 10
					STALKER_spread_zone_footprint = yes
				}
			}
			else_if = {
				limit = {
					check_variable = { STALKER_zone_containment > 69 }
					check_variable = { STALKER_zone_activity < 40 }
					check_variable = { STALKER_zone_footprint^num > 0 }
				}
				random = {
					chance = 10
					STALKER_recede_zone_footprint = yes
				}
			}
			else_if = {
				limit = {
					check_variable = { STALKER_zone_activity > 49 }
					check_variable = { STALKER_zone_containment < 70 }
					check_variable = { STALKER_zone_footprint^num > 0 }
				}
				random = {
					chance = 5
					STALKER_drift_zone_footprint = yes
				}
			}
		}
	}
}

STALKER_spread_zone_footprint = {
	controller = { save_event_target_as = STALKER_footprint_controller }
	random_neighbor_state = {
		limit = {
			is_controlled_by = event_target:STALKER_footprint_controller
			NOT = { STALKER_is_zone_location = yes }
			NOT = { has_state_flag = STALKER_zone_footprint }
		}
		set_state_flag = STALKER_zone_footprint
		set_variable = { STALKER_footprint_anchor = PREV }
		add_dynamic_modifier = { modifier = STALKER_zone_presence }
		save_event_target_as = STALKER_footprint_event_target
		PREV = {
			add_to_array = { STALKER_zone_footprint = PREV }
			set_state_flag = { flag = STALKER_footprint_cooldown value = 1 days = 180 }
		}
		log = "[GetDateText]: STALKER Zone [?PREV.STALKER_zone_id] spreads into [THIS.GetName]"
		PREV = { STALKER_target_zone = yes }
		event_target:STALKER_footprint_controller = {
			STALKER_refresh_zone_administration = yes
			country_event = STALKER.110
		}
	}
}

# The newest footprint state is released first.
STALKER_recede_zone_footprint = {
	set_state_flag = { flag = STALKER_footprint_cooldown value = 1 days = 180 }
	set_temp_variable = { STALKER_footprint_last = { value = STALKER_zone_footprint^num subtract = 1 } }
	set_temp_variable = { STALKER_footprint_state = STALKER_zone_footprint^STALKER_footprint_last }
	remove_from_array = { array = STALKER_zone_footprint index = STALKER_footprint_last }
	STALKER_target_zone = yes
	var:STALKER_footprint_state = {
		STALKER_clear_zone_footprint_state = yes
		save_event_target_as = STALKER_footprint_event_target
		log = "[GetDateText]: STALKER Zone recedes from [THIS.GetName]"
	}
	if = {
		limit = { controller = { exists = yes } }
		controller = {
			STALKER_refresh_zone_administration = yes
			country_event = STALKER.111
		}
	}
}

# The oldest footprint state is released and a new neighbour taken, so the Zone's edge moves.
STALKER_drift_zone_footprint = {
	controller = { save_event_target_as = STALKER_footprint_controller }
	set_temp_variable = { STALKER_footprint_count = STALKER_zone_footprint^num }
	random_neighbor_state = {
		limit = {
			is_controlled_by = event_target:STALKER_footprint_controller
			NOT = { STALKER_is_zone_location = yes }
			NOT = { has_state_flag = STALKER_zone_footprint }
		}
		set_state_flag = STALKER_zone_footprint
		set_variable = { STALKER_footprint_anchor = PREV }
		add_dynamic_modifier = { modifier = STALKER_zone_presence }
		save_event_target_as = STALKER_footprint_event_target
		PREV = { add_to_array = { STALKER_zone_footprint = PREV } }
	}
	if = {
		limit = { check_variable = { STALKER_zone_footprint^num > STALKER_footprint_count } }
		set_state_flag = { flag = STALKER_footprint_cooldown value = 1 days = 180 }
		set_temp_variable = { STALKER_footprint_state = STALKER_zone_footprint^0 }
		remove_from_array = { array = STALKER_zone_footprint index = 0 }
		var:STALKER_footprint_state = {
			STALKER_clear_zone_footprint_state = yes
			save_event_target_as = STALKER_footprint_released_target
		}
		log = "[GetDateText]: STALKER Zone [?STALKER_zone_id] drifts into a new neighbouring state"
		STALKER_target_zone = yes
		event_target:STALKER_footprint_controller = {
			STALKER_refresh_zone_administration = yes
			country_event = STALKER.110
		}
	}
}

# The oldest footprint state breaks away as a new Zone whose ID is this Zone's plus 100.
STALKER_split_zone = {
	set_state_flag = STALKER_zone_split
	set_state_flag = { flag = STALKER_footprint_cooldown value = 1 days = 180 }
	add_to_variable = { STALKER_zone_activity = -15 }
	set_temp_variable = { STALKER_footprint_state = STALKER_zone_footprint^0 }
	remove_from_array = { array = STALKER_zone_footprint index = 0 }
	var:STALKER_footprint_state = {
		STALKER_clear_zone_footprint_state = yes
		set_temp_variable = { STALKER_new_zone_id = { value = PREV.STALKER_zone_id add = 100 } }
		set_temp_variable = { STALKER_new_zone_profile = PREV.STALKER_zone_profile }
		STALKER_register_zone_anchor = yes
		log = "[GetDateText]: STALKER Zone [?PREV.STALKER_zone_id] splits into [THIS.GetName]"
		STALKER_awaken_reference_zone = yes
	}
	if = {
		limit = { controller = { exists = yes } }
		controller = { STALKER_refresh_zone_administration = yes }
	}
}

STALKER_clear_zone_footprint_state = {
	clr_state_flag = STALKER_zone_footprint
	clear_variable = STALKER_footprint_anchor
	if = {
		limit = { has_dynamic_modifier = { modifier = STALKER_zone_presence } }
		remove_dynamic_modifier = { modifier = STALKER_zone_presence }
	}
}
