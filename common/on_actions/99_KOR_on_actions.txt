on_actions = {
	on_weekly_KOR = {
		random_events = {
			100 = 0

			1000 = korea.16
			1000 = korea.17
			1000 = korea.22
		}
	}

	on_weekly_NKO = {
		random_events = {
			100 = 0

			1000 = korea.22
		}
	}
	on_annex = {
		effect = {
			if = {
				limit = {
					OR = {
						AND = {
							ROOT = { tag = NKO }
							FROM = { tag = KOR }
						}
						AND = {
							ROOT = { tag = KOR }
							FROM = { tag = NKO }
						}
					}
				}
				if = {
					limit = { country_exists = KOR }
					KOR = {
						set_country_flag = korea_peninsula_reunited
						add_ideas = KOR_reunification_shocks
						add_ideas = KOR_divided_military
						add_ideas = KOR_mass_migration
						add_ideas = KOR_fragmented_governance
						add_ideas = KOR_uneven_economy1
						news_event = { id = korea.50 days = 1 }
					}
				}
				else_if = {
					limit = { country_exists = NKO }
					NKO = {
						set_country_flag = korea_peninsula_reunited
						add_ideas = KOR_reunification_shocks
						add_ideas = KOR_divided_military
						add_ideas = KOR_mass_migration
						add_ideas = KOR_fragmented_governance
						add_ideas = KOR_uneven_economy1
						news_event = { id = korea.50 days = 1 }
					}
					every_country = {
						limit = {
							OR = {
								tag = JAP
								tag = TAI
								tag = USA
								tag = PHI
							}
						}
						if = {
							limit = { NKO = { has_country_flag = NKO_reunified_by_agreement } }
							add_opinion_modifier = { target = NKO modifier = NKO_negotiated_reunification }
						}
						else = {
							add_opinion_modifier = { target = NKO modifier = NKO_forced_reunification }
						}
					}
				}
			}
		}
	}
}
