TOP_retire_SAU_mohammed_bin_salman = {
	if = {
		limit = {
			OR = {
				has_country_leader = {
					name = "Mohammed bin Salman Al-Saud"
					ruling_only = yes
				}
				check_variable = { SAU_current_king = 5 }
			}
		}
		set_variable = { SAU_current_king = 4 }
		set_variable = { Kingdom_leader = 4 }
		if = {
			limit = {
				check_variable = { ruling_party = 10 }
				OR = {
					has_country_leader = {
						name = "Mohammed bin Salman Al-Saud"
						ruling_only = yes
					}
					NOT = { has_dynamic_modifier = { modifier = SAU_Military_Rule } }
				}
			}
			set_temp_variable = { b = 0 }
			set_leader_SAU = yes
		}
	}
	if = {
		limit = {
			OR = {
				check_variable = { SAU_crown_prince = 5 }
				has_idea = SAU_crown_prince_mbs
			}
		}
		set_variable = { SAU_crown_prince = 3 }
		set_variable = { SAU_heir_choice = 2 }
		set_variable = { SAU_mbs_points = 1 }
		clamp_variable = { var = SAU_mbn_points min = 2 }
		if = {
			limit = { NOT = { check_variable = { SAU_current_king = 4 } } }
			set_country_flag = SAU_crown_prince_named
		}
		if = {
			limit = { has_idea = SAU_crown_prince_mbs }
			remove_ideas = SAU_crown_prince_mbs
			if = {
				limit = { has_dlc = "La Resistance" }
				add_ideas = SAU_crown_prince_mbn_lar
			}
			else = { add_ideas = SAU_crown_prince_mbn }
		}
		set_variable = { mbs = 0 }
	}
}
