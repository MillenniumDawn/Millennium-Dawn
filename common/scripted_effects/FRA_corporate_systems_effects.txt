		limit = { NOT = { FRA_corporate_systems_has_capstone_outcome = yes } }

			limit = { FRA_corporate_systems_qualifies_european_strategic_autonomy = yes }
			FRA_corporate_systems_apply_european_strategic_autonomy = yes

			FRA_corporate_systems_apply_french_national_champions = yes
		}
		else_if = {
			limit = { FRA_corporate_systems_qualifies_china_connected_infrastructure = yes }
			FRA_corporate_systems_apply_china_connected_infrastructure = yes
		}
		else_if = {
			limit = { FRA_corporate_systems_qualifies_globalized_technology_market = yes }
			FRA_corporate_systems_apply_globalized_technology_market = yes
		}
		else = { FRA_corporate_systems_apply_resilient_multivendor_network = yes }
	}
}
