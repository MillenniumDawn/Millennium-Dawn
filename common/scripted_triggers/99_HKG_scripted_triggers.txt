# Hong Kong scripted triggers
# LegCo motion activation triggers — each tests for the corresponding country flag

# Standard policy motions
legco_increase_administration_motion = {
	has_country_flag = legco_increase_bureau
}

legco_decrease_administration_motion = {
	has_country_flag = legco_decrease_bureau
}

legco_increase_policing_motion = {
	has_country_flag = legco_increase_police
}

legco_decrease_policing_motion = {
	has_country_flag = legco_decrease_police
}

legco_increase_education_motion = {
	has_country_flag = legco_increase_education
}

legco_decrease_education_motion = {
	has_country_flag = legco_decrease_education
}

legco_increase_health_motion = {
	has_country_flag = legco_increase_health
}

legco_decrease_health_motion = {
	has_country_flag = legco_decrease_health
}

legco_increase_social_motion = {
	has_country_flag = legco_increase_social
}

legco_decrease_social_motion = {
	has_country_flag = legco_decrease_social
}

# Active-motion umbrella triggers
legco_not_has_a_active_motion = {
	NOT = { has_country_flag = legco_increase_bureau }
	NOT = { has_country_flag = legco_decrease_bureau }
	NOT = { has_country_flag = legco_increase_police }
	NOT = { has_country_flag = legco_decrease_police }
	NOT = { has_country_flag = legco_increase_education }
	NOT = { has_country_flag = legco_decrease_education }
	NOT = { has_country_flag = legco_increase_health }
	NOT = { has_country_flag = legco_decrease_health }
	NOT = { has_country_flag = legco_increase_social }
	NOT = { has_country_flag = legco_decrease_social }
	NOT = { has_country_flag = HKG_electoral_reform_done }
	NOT = { has_country_flag = HKG_electoral_reform_mainland_a }
	NOT = { has_country_flag = HKG_electoral_reform_mainland_b }
	NOT = { has_country_flag = HKG_electoral_reform_mainland_c }
}

legco_has_a_special_active_motion = {
	OR = {
		has_country_flag = HKG_electoral_reform_done
		has_country_flag = HKG_electoral_reform_mainland_a
		has_country_flag = HKG_electoral_reform_mainland_b
		has_country_flag = HKG_electoral_reform_mainland_c
	}
}
