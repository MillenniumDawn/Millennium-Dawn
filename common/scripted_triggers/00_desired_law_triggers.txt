# Administrative spending frames
generate_expected_adm_spending_frame = {
	if = {
		limit = {
			check_variable = {
				var = bureau_law
				value = index
				compare = greater_than_or_equals
			}
			check_variable = { index > expected_adm_spending }
		}

		meta_trigger = {
			text = {
				set_temp_variable = { adm_spending_frame_[INDEX] = 3 }
			}
			INDEX = "[?index]"
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = bureau_law
				value = index
				compare = greater_than_or_equals
			}
		}

		meta_trigger = {
			text = {
				set_temp_variable = { adm_spending_frame_[INDEX] = 2 }
			}
			INDEX = "[?index]"
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = expected_adm_spending
				value = index
				compare = greater_than_or_equals
			}
		}

		meta_trigger = {
			text = {
				set_temp_variable = { adm_spending_frame_[INDEX] = 4 }
			}
			INDEX = "[?index]"
		}
	}
	else = {
		meta_trigger = {
			text = {
				set_temp_variable = { adm_spending_frame_[INDEX] = 1 }
			}
			INDEX = "[?index]"
		}
	}
}
