AFG_ai_alt_path = {
	OR = {
		has_global_flag = AFG_JAMIAT_FOCUS_PATH
		has_global_flag = AFG_HAZARA_FOCUS_PATH
		has_global_flag = AFG_MONARCHY_FOCUS_PATH
		has_global_flag = AFG_EMIRATE_FOCUS_PATH
		has_global_flag = AFG_JUNBISH_FOCUS_PATH
	}
}

AFG_ai_any_path = {
	OR = {
		has_global_flag = AFG_HISTORICAL_FOCUS_PATH
		AFG_ai_alt_path = yes
	}
}

AFG_ai_historical_path = {
	OR = {
		has_global_flag = AFG_HISTORICAL_FOCUS_PATH
		AND = {
			is_historical_focus_on = yes
			NOT = { AFG_ai_alt_path = yes }
		}
	}
}

AFG_ai_not_historical_path = {
	AFG_ai_alt_path = yes
}
