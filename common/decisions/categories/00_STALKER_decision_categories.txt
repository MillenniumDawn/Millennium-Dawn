STALKER_zone_management_category = {
	allowed = { always = yes }

	icon = GFX_decision_category_generic_propaganda
	priority = 110

	visible = {
		STALKER_scenario_enabled = yes
		check_variable = { global.STALKER_active_zone_anchors^num > 0 }
		OR = {
			STALKER_controls_active_zone = yes
			AND = {
				is_ai = no
				NOT = { STALKER_controls_active_zone = yes }
				NOT = { has_country_flag = STALKER_zone_research_access }
			}
			AND = {
				original_tag = BLR
				NOT = { has_global_flag = STALKER_kaimanov_outcome }
				698 = { STALKER_is_active_zone_anchor = yes }
			}
		}
	}
}
