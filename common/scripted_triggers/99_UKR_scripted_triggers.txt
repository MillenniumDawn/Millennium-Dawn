# Trigger: UKR_storage_upkeep_can_pay
# Purpose: Checks the whole-million treasury payment due for a depot service order.
# Parameters:
# - ukr_storage_upkeep_bill: The current service price in thousands of dollars.

	check_variable = { ukr_storage_upkeep_bill > 0 }
	set_temp_variable = { ukr_storage_upkeep_balance = { value = UKR_storage_upkeep_due add = ukr_storage_upkeep_bill } }
	set_temp_variable = { ukr_storage_upkeep_remainder = ukr_storage_upkeep_balance }
	modulo_temp_variable = { var = ukr_storage_upkeep_remainder value = 1000 }
	set_temp_variable = { ukr_storage_upkeep_charge = { value = ukr_storage_upkeep_balance subtract = ukr_storage_upkeep_remainder divide = 1000000 } }
	check_variable = { var = treasury value = ukr_storage_upkeep_charge compare = greater_than_or_equals }
}
