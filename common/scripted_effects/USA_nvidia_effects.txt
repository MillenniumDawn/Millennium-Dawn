USA_nvidia_resolve_capstone = {
	USA_nvidia_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = USA_nvidia_capstone_resolved } }
		if = {
			limit = { has_country_flag = USA_nvidia_fortress_route }
			USA_nvidia_apply_cuda_fortress = yes
		}
		else_if = {
			limit = { has_country_flag = USA_nvidia_open_route }
			USA_nvidia_apply_open_commonwealth = yes
		}
		else_if = {
			limit = { has_country_flag = USA_nvidia_sovereign_route }
			USA_nvidia_apply_sovereign_stack = yes
		}
		else_if = {
			limit = { has_country_flag = USA_nvidia_hybrid_route }
			USA_nvidia_apply_hybrid_champion = yes
		}
		else = { USA_nvidia_apply_commodity_decline = yes }

		USA_corporate_systems_update_economic_bridge = yes
	}
}
