# One entry point per milestone, called from the yearly pulse. Full presents the
# choice; Outcomes Only applies the historical persistent route silently without
# the live event's one-time costs or rewards.
TAI_foxconn_milestone_2001 = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		country_event = { id = taiwan.300 days = 120 }
	}
	else_if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
		set_country_flag = TAI_foxconn_contract_scale
	}
}
