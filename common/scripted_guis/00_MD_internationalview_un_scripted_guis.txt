	MD_UN_states_to_recognize_gui = {
		window_name = UN_states_to_recognize_menu
		context_type = player_context
		parent_window_name = MD_UN_system_sub_window

		dirty = global.international_systems_updated

		visible = {
			check_variable = { var_open_MD_state_recog = 1 }
			check_variable = { var_open_MD_UN_gui = 2 }
			has_country_flag = open_MD_countrymissilesview
			NOT = { has_idea = lacks_international_recognition }
		}

		effects = {
			grant_recog_button_click = {
				add_political_power = -25

				if = {
					limit = {
						var:open_state = {
							NOT = { is_in_array = { original_owners = ROOT } }
							NOT = { is_in_array = { original_owner_supporters = ROOT } }
						}
					}
					custom_effect_tooltip = TT_ALL_UN_MEMBERS
					every_collection_element = {
						input = { input = collection:un_general_assembly_members_collection }
						if = {
							limit = {
								var:ROOT.open_state = {
									is_in_array = { original_owner_supporters = PREV }
									NOT = { is_in_array = { recognised_by = PREV } }
								}
							}
							add_opinion_modifier = { target = ROOT modifier = recognises_them }
						} else_if = {
							limit = { var:ROOT.open_state = { is_in_array = { recognised_by = PREV } } }
							add_opinion_modifier = { target = ROOT modifier = supporting_recognized_ally }
						}
					}

					var:open_state = {
						random_scope_in_array = {
							array = original_owners
							limit = {
								exists = yes
								NOT = { is_in_array = { PREV.recognised_by = THIS } }
							}
							add_opinion_modifier = { target = ROOT modifier = recognises_them }
							country_event = recognition.1
						}
					}
				}
				var:open_state = {
					add_opinion_modifier = {
						target = ROOT
						modifier = recognises_us
					}
					recognition_grant_increment = yes
				}
				international_systems_update = yes
			}
			pressure_countries_recog_button_click = {
				every_collection_element = {
					input = { input = collection:un_general_assembly_members_collection }
					if = {
						limit = {
							is_influencer = yes
							var:ROOT.open_state = {
								NOT = { is_in_array = { recognised_by = PREV } }
								NOT = { has_country_flag = pressured_to_recognize_@PREV }
							}
							NOT = { tag = ROOT }
							is_full_state = yes
						}
						var:ROOT.open_state = { set_country_flag = pressured_to_recognize_@PREV }
						country_event = recognition.8
					}
				}
				set_variable = { recognition_pressure_end@var:open_state = global.num_days }
				add_to_variable = { recognition_pressure_end@var:open_state = 180 }
				add_political_power = -100
				international_systems_update = yes
			}
			rescind_recog_button_click = {
				var:open_state = {
					remove_from_array = { recognised_by = ROOT }
					random_scope_in_array = {
						array = original_owners
						limit = {
							exists = yes
							NOT = { is_in_array = { PREV.recognised_by = THIS } }
						}
						add_opinion_modifier = {
							target = ROOT
							modifier = withdrew_recognition_them
						}
						country_event = recognition.4
					}
					add_opinion_modifier = { modifier = withdrew_recognition target = ROOT }
					country_event = recognition.3
				}
				international_systems_update = yes
			}
			name_of_state_button_click = {
				if = {
					limit = { NOT = { open_state = 0 } }

					set_variable = { UN_recog_buttons_frames^open_state_index = 2 }
				}

				set_variable = { UN_recog_buttons_frames^i = 1 }

				set_variable = { open_state = v }
				set_variable = { open_state_index = i }
				if = { limit = { is_ai = yes }
					set_country_flag = {
						flag = state_swap_slow_flag
						days = 120 value = 1
					}
				}
				international_systems_update = yes
			}
		}

		triggers = {
			pressure_countries_recog_button_click_enabled = {
				var:open_state = { has_idea = lacks_international_recognition }
				country_exists = var:open_state
				is_full_state = yes
				custom_trigger_tooltip = {
					tooltip = has_recog_tt
					var:open_state = { is_in_array = { recognised_by = ROOT } }
				}
				custom_trigger_tooltip = {
					tooltip = recent_pressure_tt
					NOT = { check_variable = { recognition_pressure_end@var:open_state > global.num_days } }
				}
				if = {
					limit = { original_tag = SWI }
					has_dynamic_modifier = { modifier = united_nations_member }
				}

				has_political_power > 100
			}
			grant_recog_button_click_enabled = {
				var:open_state = { has_idea = lacks_international_recognition }
				country_exists = var:open_state
				is_full_state = yes
				if = {
					limit = { original_tag = SWI }
					has_dynamic_modifier = { modifier = united_nations_member }
				}
				custom_trigger_tooltip = {
					tooltip = not_recog_tt
					NOT = { var:open_state = { is_in_array = { recognised_by = ROOT } } }
				}

				has_political_power > 25
			}
			rescind_recog_button_click_enabled = {
				var:open_state = { has_idea = lacks_international_recognition }
				country_exists = var:open_state
				is_full_state = yes
				if = {
					limit = { original_tag = SWI }
					has_dynamic_modifier = { modifier = united_nations_member }
				}
				custom_trigger_tooltip = {
					tooltip = has_recog_tt
					var:open_state = { is_in_array = { recognised_by = ROOT } }
				}
				has_political_power > 25
			}
			name_of_state_button_visible = {
				var:v = { has_idea = lacks_international_recognition }
			}
			name_of_state_button_text_visible = {
				var:v = { has_idea = lacks_international_recognition }
			}
			name_of_state_button_click_enabled = {
				NOT = { tag = var:v }
				NOT = { has_war_with = var:v }
				has_civil_war = no
			}
		}

			list_of_unrecognized_states = {
