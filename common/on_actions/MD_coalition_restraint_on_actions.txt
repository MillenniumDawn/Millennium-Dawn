on_actions = {
	# Monthly upkeep of the coalition restraint (MD_coalition_restraint_effects.txt).
	# Script wargoals get it at creation through add_threat_from_wargoal_effect.
	on_monthly = {
		effect = {
			if = {
				limit = { is_ai = yes }
				set_temp_variable = { coalition_attacker = ROOT }
				every_country = {
					limit = {
						ROOT = {
							OR = {
								has_wargoal_against = PREV
								has_country_flag = coalition_restraint_strong@PREV
							}
						}
					}
					coalition_restraint_update_effect = yes
				}
			}
		}
	}
}
