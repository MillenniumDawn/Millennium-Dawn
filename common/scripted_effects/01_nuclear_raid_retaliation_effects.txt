record_nuclear_strike = {
	var:victim_country = {
		var:PREV.actor_country = {
			if = {
				limit = {
					OR = {
						has_country_flag = nuclear_tactical_strike_against@PREV
						has_country_flag = nuclear_state_strike_against@PREV
						has_country_flag = nuclear_strategic_strike_against@PREV
					}
				}
				set_country_flag = nuclear_repeated_strikes_against@PREV
			}
			PREV = {
				clr_country_flag = nuclear_tactical_strike_against@PREV
				clr_country_flag = nuclear_state_strike_against@PREV
				clr_country_flag = nuclear_strategic_strike_against@PREV
				clr_country_flag = nuclear_repeated_strikes_against@PREV
			}
		}
	}
}
