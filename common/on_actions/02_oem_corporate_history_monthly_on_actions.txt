# OEM-only monthly hosts for tags without an existing native host. Tags that already
# own an on_monthly_TAG block keep their Corporate History callback in that file.

on_actions = {
	on_monthly_FIN = {
		effect = {
			FIN_corporate_history_monthly_outcomes = yes
		}
	}

	on_monthly_TAI = {
		effect = {
			gpu_development_monthly_driver = yes
			TAI_corporate_history_monthly_outcomes = yes
		}
	}
}
