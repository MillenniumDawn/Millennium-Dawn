# BlackBerry corporate history events
country_event = {
	id = blackberry_events.1
	title = blackberry_events.1.t
	desc = blackberry_events.1.d
	picture = GFX_CAN_generic
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		corporate_history_full_enabled = yes
		NOT = { has_country_flag = CAN_corporate_history_midyear_blackberry_events_1_resolved }
		original_tag = CAN
		NOT = { has_country_flag = collapsed_nation }
	}

	immediate = { set_country_flag = CAN_corporate_history_midyear_blackberry_events_1_resolved }

	option = {
		name = blackberry_events.1.a
		log = "[GetDateText]: [This.GetName]: blackberry_events.1.a executed"
		custom_effect_tooltip = blackberry_events.1.a_tt
		add_political_power = 10
		set_country_flag = CAN_blackberry_private_expansion
		ai_chance = {
			base = 90
			modifier = {
				add = 10
				is_historical_focus_on = yes
			}
		}
	}

	option = {
		name = blackberry_events.1.b
		log = "[GetDateText]: [This.GetName]: blackberry_events.1.b executed"
		custom_effect_tooltip = blackberry_events.1.b_tt
		set_temp_variable = { treasury_change = -2.00 }
		modify_treasury_effect = yes
		add_tech_bonus = {
			name = blackberry_events.1.research_bonus
			bonus = 0.10
			uses = 1
			category = CAT_computing_tech
		}
		set_country_flag = CAN_blackberry_waterloo_support
		ai_chance = {
			base = 10
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
		}
	}
}
