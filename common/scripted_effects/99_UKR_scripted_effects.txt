			set_temp_variable = { tension_change = -15 }
			UKR_add_political_tension = yes
		}
		if = {
			limit = { has_country_flag = UKR_seuar_reintegrated }
			set_temp_variable = { tension_change = -20 }
			UKR_add_political_tension = yes
		}
		if = {
			limit = { has_country_flag = UKR_crimea_2004_restrained }
			set_temp_variable = { tension_change = -15 }
			UKR_add_political_tension = yes
		}
		if = {
			limit = { has_country_flag = UKR_crimea_2004_departed }
			set_temp_variable = { tension_change = 20 }
			UKR_add_political_tension = yes
		}
		if = {
			limit = { has_country_flag = UKR_crimea_2004_release_failed }
			set_temp_variable = { tension_change = 5 }
			UKR_add_political_tension = yes
		}
		if = {
			limit = { has_country_flag = UKR_seuar_defeated }
			set_temp_variable = { tension_change = 10 }
			UKR_add_political_tension = yes
		}
		set_country_flag = UKR_euromaidan_severity_setup
	}
	clr_country_flag = UKR_euromaidan_mild
	clr_country_flag = UKR_euromaidan_severe
	if = {
		limit = {
			check_variable = { UKR_political_tension < 40 }
			is_historical_focus_on = no
		}
		set_country_flag = UKR_euromaidan_mild
	}
	else_if = {
		limit = { check_variable = { UKR_political_tension > 69 } }
		set_country_flag = UKR_euromaidan_severe
	}
}
UKR_start_euromaidan = {
	log = "[GetDateText]: [Root.GetName]: Effect UKR_start_euromaidan"
	if = {
		limit = {
			tag = UKR
			date > 2013.10.31
			NOT = { has_country_flag = ukr_revoult_bilo }
		}
		UKR_setup_euromaidan_severity = yes
		if = {
			limit = { has_country_flag = UKR_euromaidan_severe }

						is_ai = yes

				}

		}

UKR_prepare_DPR_wave = {
	if = {
		limit = {
			OR = {
				NOT = { country_exists = DPR }
				DPR = { UKR_novorossiya_participant_eligible = yes }
			}
		}
		693 = { add_core_of = DPR }
		1030 = { add_core_of = DPR }
		if = {
			limit = {
				NOT = { country_exists = DPR }
				OR = {
					owns_state = 693
					owns_state = 1030
				}
			}
			release = DPR
		}
		if = {
			limit = { country_exists = DPR }
			DPR = {
				set_country_flag = UKR_subject_agree
				if = {
					limit = { NOT = { has_country_flag = UKR_provisional_flavor_first_republic_queued } }
					set_country_flag = UKR_provisional_flavor_first_republic_queued
					country_event = { id = ukraine_revolution.52 days = 1 }
				}
				if = { limit = { NOT = { has_idea = UKR_republic_ukr } } add_ideas = UKR_republic_ukr }
				if = { limit = { UKR = { owns_state = 693 } } transfer_state = 693 }
				if = { limit = { UKR = { owns_state = 1030 } } transfer_state = 1030 }
				if = {
					limit = { NOT = { has_country_flag = UKR_republic_army_initialized } }
					DPR_maidan_create_army = yes
					set_country_flag = UKR_republic_army_initialized
				}
				if = {
					limit = { is_ai = yes }
					give_AI_templates = yes
				}
			}
			if = {
				limit = { DPR = { UKR_novorossiya_participant_eligible = yes } }
				set_autonomy = {
					target = DPR
					autonomy_state = autonomy_republic_ukr
				}
			}
		}
	}
}

UKR_do_math_boi = { # Calculates influence of others on you to determine election winner
	# Rank weight for influence_array slots 0-6, most influential first
	for_loop_effect = {
		start = 0
		end = 7
		if = {
			limit = { UKR = { check_variable = { influence_array^v = SOV } } }
			add_to_variable = {
				var = UKR_soviet_influence
				value = { value = 10 subtract = v }
			}
		}
		if = {
			limit = { UKR = { check_variable = { influence_array^v = USA } } }
			add_to_variable = {
				var = UKR_american_influence
				value = { value = 10 subtract = v }
			}
		}
	}
	# Neither the US nor Russia holds each of the top 3 slots: national influence gains ground
	for_loop_effect = {
		start = 0
		end = 3
		if = {
			limit = { NOT = { UKR = { check_variable = { influence_array^v = USA } } } }
			add_to_variable = { var = UKR_national_influence value = 2 }
		}
		if = {
			limit = { NOT = { UKR = { check_variable = { influence_array^v = SOV } } } }
			add_to_variable = { var = UKR_national_influence value = 2 }
		}
	}
}

			id = ukraine_parliament.300

		}

	if = {
		limit = {
			check_variable = {
				UKR_american_influence > UKR_national_influence

			check_variable = {
				UKR_american_influence > UKR_soviet_influence
			}

UKR_start_workforce_setup = {
	set_variable = { UKR_start_workforce_agriculture = 1 }
	set_variable = { UKR_start_workforce_civ_facs = 0.25 }
	set_variable = { UKR_start_workforce_mil_facs = -0.1 }
	set_variable = { UKR_start_workforce_resources = 0.75 }
	set_variable = { UKR_start_workforce_offices = -0.1 }
	add_dynamic_modifier = { modifier = UKR_start_workforce_modifier }
}

# A target is the start count plus one building per step that brings the requirement back to 0
UKR_start_workforce_record_counts = {
	# mil_facs_worker_requirement_modifier also sets the workers per dockyard, so both count as one workplace
	set_variable = { UKR_start_workforce_mil_facs_count = { value = military_factory_total add = naval_factory_total } }
	set_variable = { UKR_start_workforce_agriculture_target = { value = UKR_start_workforce_agriculture divide = 0.1 add = agriculture_district_total } }
	set_variable = { UKR_start_workforce_civ_facs_target = { value = UKR_start_workforce_civ_facs divide = 0.025 add = industrial_complex_total } }
	set_variable = { UKR_start_workforce_mil_facs_target = { value = UKR_start_workforce_mil_facs divide = -0.02 add = UKR_start_workforce_mil_facs_count } }
	set_variable = { UKR_start_workforce_offices_target = { value = UKR_start_workforce_offices divide = -0.02 add = office_park_total } }
}

UKR_start_workforce_update = {
	if = {
		limit = { has_dynamic_modifier = { modifier = UKR_start_workforce_modifier } }
		if = {
			limit = { NOT = { has_variable = UKR_start_workforce_civ_facs_target } }
			UKR_start_workforce_record_counts = yes
		}
		set_variable = { UKR_start_workforce_mil_facs_count = { value = military_factory_total add = naval_factory_total } }
		set_temp_variable = { agriculture_was = UKR_start_workforce_agriculture }
		set_temp_variable = { civ_facs_was = UKR_start_workforce_civ_facs }
		set_temp_variable = { mil_facs_was = UKR_start_workforce_mil_facs }
		set_temp_variable = { resources_was = UKR_start_workforce_resources }
		set_temp_variable = { offices_was = UKR_start_workforce_offices }
		set_temp_variable = { population_gain = 0 }

		# Each workplace sits one step from 0 per building it is short of its target, and lost buildings never move it back
		set_temp_variable = { workforce_new = { value = UKR_start_workforce_agriculture_target subtract = agriculture_district_total multiply = 0.1 } }
		clamp_temp_variable = { var = workforce_new min = 0 }
		if = {
			limit = { check_variable = { workforce_new < UKR_start_workforce_agriculture } }
			set_variable = { UKR_start_workforce_agriculture = workforce_new }
			if = {
				limit = { NOT = { check_variable = { workforce_new > 0 } } }
				add_to_temp_variable = { population_gain = 0.1 }
			}
		}
		set_temp_variable = { workforce_new = { value = UKR_start_workforce_civ_facs_target subtract = industrial_complex_total multiply = 0.025 } }
		clamp_temp_variable = { var = workforce_new min = 0 }
		if = {
			limit = { check_variable = { workforce_new < UKR_start_workforce_civ_facs } }
			set_variable = { UKR_start_workforce_civ_facs = workforce_new }
			if = {
				limit = { NOT = { check_variable = { workforce_new > 0 } } }
				add_to_temp_variable = { population_gain = 0.1 }
			}
		}
		set_temp_variable = { workforce_new = { value = UKR_start_workforce_mil_facs_count subtract = UKR_start_workforce_mil_facs_target multiply = 0.02 } }
		clamp_temp_variable = { var = workforce_new max = 0 }
		if = {
			limit = { check_variable = { workforce_new > UKR_start_workforce_mil_facs } }
			set_variable = { UKR_start_workforce_mil_facs = workforce_new }
			if = {
				limit = { NOT = { check_variable = { workforce_new < 0 } } }
				add_to_temp_variable = { population_gain = 0.05 }
			}
		}
		set_temp_variable = { workforce_new = { value = office_park_total subtract = UKR_start_workforce_offices_target multiply = 0.02 } }
		clamp_temp_variable = { var = workforce_new max = 0 }
		if = {
			limit = { check_variable = { workforce_new > UKR_start_workforce_offices } }
			set_variable = { UKR_start_workforce_offices = workforce_new }
			if = {
				limit = { NOT = { check_variable = { workforce_new < 0 } } }
				add_to_temp_variable = { population_gain = 0.05 }
			}
		}

		# Every workplace back to normal staffing raises population growth in Ukrainian Economic Development for good
		if = {
			limit = { check_variable = { population_gain > 0 } }
			add_to_variable = { UKR_econ_mod_MONTHLY_POPULATION = population_gain tooltip = MONTHLY_POPULATION_tt }
		}

		# Every step the other four workplaces take cuts resource extraction by 2.5%: their 30 steps clear its 75%
		set_temp_variable = {
			workforce_steps = {
				value = agriculture_was
				subtract = UKR_start_workforce_agriculture
				divide = 0.1
				add = { value = civ_facs_was subtract = UKR_start_workforce_civ_facs divide = 0.025 }
				add = { value = UKR_start_workforce_mil_facs subtract = mil_facs_was divide = 0.02 }
				add = { value = UKR_start_workforce_offices subtract = offices_was divide = 0.02 }
			}
		}
		subtract_from_variable = { UKR_start_workforce_resources = { value = workforce_steps multiply = 0.025 } }
		clamp_variable = { var = UKR_start_workforce_resources min = 0 }

		if = {
			limit = {
				NOT = {
					OR = {
						check_variable = { UKR_start_workforce_agriculture > 0 }
						check_variable = { UKR_start_workforce_civ_facs > 0 }
						check_variable = { UKR_start_workforce_mil_facs < 0 }
						check_variable = { UKR_start_workforce_resources > 0 }
						check_variable = { UKR_start_workforce_offices < 0 }
					}
				}
			}
			remove_dynamic_modifier = { modifier = UKR_start_workforce_modifier }
		}

		if = {
			limit = {
				is_ai = no
				check_variable = { workforce_steps > 0 }
			}
			set_variable = { UKR_start_workforce_agriculture_change = { value = UKR_start_workforce_agriculture subtract = agriculture_was } }
			set_variable = { UKR_start_workforce_civ_facs_change = { value = UKR_start_workforce_civ_facs subtract = civ_facs_was } }
			set_variable = { UKR_start_workforce_mil_facs_change = { value = UKR_start_workforce_mil_facs subtract = mil_facs_was } }
			set_variable = { UKR_start_workforce_resources_change = { value = UKR_start_workforce_resources subtract = resources_was } }
			set_variable = { UKR_start_workforce_offices_change = { value = UKR_start_workforce_offices subtract = offices_was } }
			set_variable = { UKR_start_workforce_population_change = population_gain }
			country_event = ukr_start_workforce.1
		}
	}
}

# Reads ukraine_bul_btr60_offer_amount (20 or 40) and ukraine_bul_btr60_offer_mode (0 = in service as they are, 1 = overhaul into the BTR-60D)
UKR_bul_btr60_offer_accepted = {
	set_temp_variable = { ukr_btr60_amount = ukraine_bul_btr60_offer_amount }
	if = {
		limit = { check_variable = { ukraine_bul_btr60_offer_mode = 0 } }
		UKR_sov_btr60_bulgaria_send = yes
		add_to_variable = { ukraine_btr_60d_batches = 1 }
		if = {
			limit = { check_variable = { ukraine_btr_60d_batches < 3 } }
			hidden_effect = { country_event = { id = ukraine_modernize_sov_equipment_btr_60_modernization.8 days = 60 random_days = 30 } }
		}
	}
	else = {
		UKR_sov_btr60_bulgaria_remove_stock = yes
		if = {
			limit = { check_variable = { ukraine_bul_btr60_offer_amount = 20 } }
			set_temp_variable = { treasury_change = -0.07 }
			modify_treasury_effect = yes
			hidden_effect = { country_event = { id = ukraine_modernize_sov_equipment_btr_60_modernization.7 days = 120 random_days = 15 } }
		}
		else_if = {
			limit = { has_country_flag = ukraine_modernize_sov_equipment_btr_60_modernization_btr_60d_flag }
			set_temp_variable = { treasury_change = -0.03 }
			modify_treasury_effect = yes
			hidden_effect = { country_event = { id = ukraine_modernize_sov_equipment_btr_60_modernization.9 days = 90 random_days = 10 } }
		}
		else = {
			set_temp_variable = { treasury_change = -0.07 }
			modify_treasury_effect = yes
			hidden_effect = { country_event = { id = ukraine_modernize_sov_equipment_btr_60_modernization.9 days = 120 random_days = 15 } }
		}
	}
}

# Storage list in the UKR_reactivate_sov_category GUI (UKR_reactivate_sov_storage_gui).
# Row ids: 1-5 are section headers, 11 and up are vehicles, the tens digit is their section.
# Sections and vehicles with nothing in storage are hidden. Only the section in UKR_storage_open_section
# shows its vehicles (0 = all closed). UKR_storage_refresh rebuilds the rows.
UKR_storage_rebuild_rows = {
	clear_array = UKR_storage_rows
	if = {
		limit = { check_variable = { UKR_storage_total_1 > 0 } }
		add_to_array = { UKR_storage_rows = 1 }
		if = {
			limit = { check_variable = { UKR_storage_open_section = 1 } }
			if = { limit = { check_variable = { ukraine_has_t55_stockpile > 0 } } add_to_array = { UKR_storage_rows = 11 } }
			if = { limit = { check_variable = { ukraine_has_t62_stockpile > 0 } } add_to_array = { UKR_storage_rows = 12 } }
			if = { limit = { check_variable = { ukraine_has_pt76_stockpile > 0 } } add_to_array = { UKR_storage_rows = 13 } }
		}
	}
	if = {
		limit = { check_variable = { UKR_storage_total_2 > 0 } }
		add_to_array = { UKR_storage_rows = 2 }
		if = {
			limit = { check_variable = { UKR_storage_open_section = 2 } }
			if = { limit = { check_variable = { ukraine_has_btr50_stockpile > 0 } } add_to_array = { UKR_storage_rows = 21 } }
			if = { limit = { check_variable = { ukraine_has_gtmu_stockpile > 0 } } add_to_array = { UKR_storage_rows = 22 } }
		}
	}
	if = {
		limit = { check_variable = { UKR_storage_total_3 > 0 } }
		add_to_array = { UKR_storage_rows = 3 }
		if = {
			limit = { check_variable = { UKR_storage_open_section = 3 } }
			if = { limit = { check_variable = { ukraine_has_2k11_stockpile > 0 } } add_to_array = { UKR_storage_rows = 31 } }
			if = { limit = { check_variable = { ukraine_has_9k31_stockpile > 0 } } add_to_array = { UKR_storage_rows = 32 } }
		}
	}
	if = {
		limit = { check_variable = { UKR_storage_total_4 > 0 } }
		add_to_array = { UKR_storage_rows = 4 }
		if = {
			limit = { check_variable = { UKR_storage_open_section = 4 } }
			if = { limit = { check_variable = { ukraine_has_9k11_stockpile > 0 } } add_to_array = { UKR_storage_rows = 41 } }
		}
	}
	if = {
		limit = { check_variable = { UKR_storage_total_5 > 0 } }
		add_to_array = { UKR_storage_rows = 5 }
		if = {
			limit = { check_variable = { UKR_storage_open_section = 5 } }
			if = { limit = { check_variable = { ukraine_has_tu16k_stockpile > 0 } } add_to_array = { UKR_storage_rows = 51 } }
			if = { limit = { check_variable = { ukraine_has_tu22kd_stockpile > 0 } } add_to_array = { UKR_storage_rows = 52 } }
			if = { limit = { check_variable = { ukraine_has_tu22rd_stockpile > 0 } } add_to_array = { UKR_storage_rows = 53 } }
		}
	}
}

# Header click in the storage list: open the section in temp variable storage_section, or close it when it is already open
UKR_storage_toggle_section = {
	if = {
		limit = { check_variable = { UKR_storage_open_section = storage_section } }
		set_variable = { UKR_storage_open_section = 0 }
	}
	else = {
		set_variable = { UKR_storage_open_section = storage_section }
	}
	UKR_storage_refresh = yes
}

# Call after every change of a ukraine_has_*_stockpile variable: section totals, rows and a GUI refresh
UKR_storage_refresh = {
	if = {
		limit = { NOT = { has_country_flag = UKR_storage_initialized_flag } }
		set_country_flag = UKR_storage_initialized_flag
		set_variable = { UKR_storage_open_section = 1 }
	}
	set_variable = { UKR_storage_total_1 = ukraine_has_t55_stockpile }
	add_to_variable = { UKR_storage_total_1 = ukraine_has_t62_stockpile }
	add_to_variable = { UKR_storage_total_1 = ukraine_has_pt76_stockpile }
	set_variable = { UKR_storage_total_2 = ukraine_has_btr50_stockpile }
	add_to_variable = { UKR_storage_total_2 = ukraine_has_gtmu_stockpile }
	set_variable = { UKR_storage_total_3 = ukraine_has_2k11_stockpile }
	add_to_variable = { UKR_storage_total_3 = ukraine_has_9k31_stockpile }
	set_variable = { UKR_storage_total_4 = ukraine_has_9k11_stockpile }
	set_variable = { UKR_storage_total_5 = ukraine_has_tu16k_stockpile }
	add_to_variable = { UKR_storage_total_5 = ukraine_has_tu22kd_stockpile }
	add_to_variable = { UKR_storage_total_5 = ukraine_has_tu22rd_stockpile }
	UKR_storage_rebuild_rows = yes
	if = {
		limit = { check_variable = { UKR_storage_dirty_var < 10000 } }
		add_to_variable = { UKR_storage_dirty_var = 1 }
	}
	else = {
		set_variable = { UKR_storage_dirty_var = 1 }
	}
}

# In a Central Asian buyer's scope: does Russia talk it out of the Krug deal? Sets ROOT's ukraine_2k11_russia_objects (0/1).
# The chance is twice the gap between Russian and Ukrainian influence there, in percent.
UKR_2k11_russian_pressure = {
	set_temp_variable = { ukr_infl = 0 }
	set_temp_variable = { sov_infl = 0 }
	for_each_loop = {
		array = influence_array
		value = v
		index = i
		if = {
			limit = { var:v = { tag = ROOT } }
			set_temp_variable = { ukr_infl = influence_array_val^i }
		}
		if = {
			limit = { var:v = { tag = SOV } }
			set_temp_variable = { sov_infl = influence_array_val^i }
		}
	}
	set_temp_variable = { ukr_russia_pressure = sov_infl }
	subtract_from_temp_variable = { ukr_russia_pressure = ukr_infl }
	multiply_temp_variable = { ukr_russia_pressure = 2 }
	set_temp_variable_to_random = { var = ukr_russia_roll min = 0 max = 100 }
	set_variable = { ROOT.ukraine_2k11_russia_objects = 0 }
	if = {
		limit = {
			country_exists = SOV
			check_variable = { ukr_russia_roll < ukr_russia_pressure }
		}
		set_variable = { ROOT.ukraine_2k11_russia_objects = 1 }
	}
}
