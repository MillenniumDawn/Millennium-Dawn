HLS_ai_alt_path = {
	OR = {
		has_global_flag = HLS_CRUSADER_FOCUS_PATH
		has_global_flag = HLS_CALIPHATE_FOCUS_PATH
	}
}

HLS_ai_historical_path = {
	OR = {
		has_global_flag = HLS_HISTORICAL_FOCUS_PATH
		AND = {
			is_historical_focus_on = yes
			NOT = { HLS_ai_alt_path = yes }
		}
	}
}

HLS_ai_crusader_path = {
	has_global_flag = HLS_CRUSADER_FOCUS_PATH
}

HLS_ai_caliphate_path = {
	has_global_flag = HLS_CALIPHATE_FOCUS_PATH
}

HLS_ai_not_historical_path = {
	HLS_ai_alt_path = yes
}

HLS_ai_not_crusader_path = {
	NOT = { has_global_flag = HLS_CRUSADER_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = HLS_HISTORICAL_FOCUS_PATH
		has_global_flag = HLS_CALIPHATE_FOCUS_PATH
	}
}

HLS_ai_not_caliphate_path = {
	NOT = { has_global_flag = HLS_CALIPHATE_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = HLS_HISTORICAL_FOCUS_PATH
		has_global_flag = HLS_CRUSADER_FOCUS_PATH
	}
}
