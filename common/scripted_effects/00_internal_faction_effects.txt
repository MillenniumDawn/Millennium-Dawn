# Generic election funding calculator
# Maps opinion tiers to funding: <20 = -2, 20-39 = -1, 40-59 = 0, 60-79 = +1, >=80 = +2
# Usage: set temp_faction_opinion before calling
calculate_election_funding_from_opinion = {
	if = { limit = { check_variable = { temp_faction_opinion < 20 } }
		add_to_variable = { campaign_funding_count = -2 }
	}
	else_if = { limit = { check_variable = { temp_faction_opinion < 40 } }
		add_to_variable = { campaign_funding_count = -1 }
	}
	else_if = { limit = { check_variable = { temp_faction_opinion > 79 } }
		add_to_variable = { campaign_funding_count = 2 }
	}
	else_if = { limit = { check_variable = { temp_faction_opinion > 59 } }
		add_to_variable = { campaign_funding_count = 1 }
	}
	# 40-59 adds 0, no action needed
}

		set_temp_variable = { max_opinion = 100 }

		if = {

#  - nation_to_copy_from - accepts all tags as a variable

		if = { limit = { has_idea = small_medium_business_owners }
			PREV = { add_ideas = small_medium_business_owners }
