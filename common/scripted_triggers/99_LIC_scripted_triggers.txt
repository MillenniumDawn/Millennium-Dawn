LIC_ai_alt_path = {
	OR = {
		has_global_flag = LIC_REPUBLIC_FOCUS_PATH
		has_global_flag = LIC_PRINCELY_FOCUS_PATH
	}
}

LIC_ai_historical_path = {
	OR = {
		has_global_flag = LIC_HISTORICAL_FOCUS_PATH
		AND = {
			is_historical_focus_on = yes
			NOT = { LIC_ai_alt_path = yes }
		}
	}
}

LIC_ai_monarchy_path = {
	OR = {
		LIC_ai_historical_path = yes
		has_global_flag = LIC_PRINCELY_FOCUS_PATH
	}
}

LIC_ai_not_monarchy_path = {
	has_global_flag = LIC_REPUBLIC_FOCUS_PATH
}

LIC_ai_open_path = {
	OR = {
		LIC_ai_historical_path = yes
		has_global_flag = LIC_REPUBLIC_FOCUS_PATH
	}
}

LIC_ai_not_open_path = {
	has_global_flag = LIC_PRINCELY_FOCUS_PATH
}

LIC_ai_not_republic_path = {
	NOT = { has_global_flag = LIC_REPUBLIC_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = LIC_HISTORICAL_FOCUS_PATH
		has_global_flag = LIC_PRINCELY_FOCUS_PATH
	}
}

LIC_ai_not_princely_path = {
	NOT = { has_global_flag = LIC_PRINCELY_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = LIC_HISTORICAL_FOCUS_PATH
		has_global_flag = LIC_REPUBLIC_FOCUS_PATH
	}
}

LIC_ai_not_armed_path = {
	NOT = { has_global_flag = LIC_PRINCELY_FOCUS_PATH }
	LIC_ai_historical_path = yes
}
