corruption_reform_challenge_category = {
	corruption_reform_challenge = {
		icon = GFX_decision_generic_political_reform
		activation = { always = no }
		available = { always = no }
		selectable_mission = no
		days_mission_timeout = 365
		is_good = no
		fire_only_once = no
		fixed_random_seed = no

		cancel_trigger = {
			check_variable = { modifier@political_manipulation_corruption_factor > 0 }
			check_variable = { modifier@political_manipulation_corruption_factor < corruption_reform_start_factor }
		}

		cancel_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision corruption_reform_challenge"
			clear_variable = corruption_reform_start_factor
			grant_random_corruption_offer_buff = yes
		}

		timeout_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision corruption_reform_challenge"
			clear_variable = corruption_reform_start_factor
			increase_corruption = yes
			add_timed_idea = {
				idea = corruption_reform_resistance
				days = 365
			}
		}
	}
}
