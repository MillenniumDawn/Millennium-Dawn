on_actions = {
	on_monthly_DEN = {
		effect = {
			# About 1 point of anger fades each quarter
			subtract_from_variable = { DEN_farmer_anger = 0.34 }
			clamp_variable = { var = DEN_farmer_anger min = 1 max = 100 }
			subtract_from_variable = { DEN_environmentalist_anger = 0.34 }
			clamp_variable = { var = DEN_environmentalist_anger min = 1 max = 100 }
			if = {
				limit = {
					has_capitulated = no
					has_completed_focus = DEN_agriculture
				}
				if = {
					limit = {
						NOT = { has_country_flag = DEN_agrar_event_pending }
						NOT = { has_country_flag = DEN_agrar_event_cooldown }
					}
					DEN_agrar_resolve = yes
				}
				if = {
					limit = {
						NOT = { has_country_flag = DEN_agrar_event_pending_II }
						NOT = { has_country_flag = DEN_agrar_event_cooldown_II }
					}
					DEN_agrar_resolve_II = yes
				}
			}
		}
	}
}
