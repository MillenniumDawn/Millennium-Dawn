# Border War Mechanic
# Shared scripted triggers for the RAJ-PAK and RAJ-CHI border conflict systems.

# Eligible = at least one contested border exists and the country hasn't
# capitulated. The "at war with the rival" guard lives on each category so a
# war with one neighbour doesn't freeze the conflict with the other.
is_border_war_eligible = {
	has_capitulated = no
	OR = {
		AND = {
			original_tag = RAJ
			OR = {
				country_exists = PAK
				country_exists = CHI
			}
			OR = {
				PAK = { is_subject = no }
				CHI = { is_subject = no }
			}
		}
		AND = {
			original_tag = PAK
			country_exists = RAJ
			RAJ = { is_subject = no }
			NOT = { is_in_faction_with = RAJ }
		}
		AND = {
			original_tag = CHI
			country_exists = RAJ
			RAJ = { is_subject = no }
			NOT = { is_in_faction_with = RAJ }
		}
	}
}

# AI gate for the prepare_/training decisions. Suppresses spam from
# nations that lack motive (no threat, not aligned with the target's
# ideology, or at peace with a third party that would object).
border_war_ai_should_prepare = {
	threat > 0.30
	has_war = no
	has_capitulated = no
	OR = {
		AND = {
			original_tag = RAJ
			OR = {
				has_opinion = { target = PAK value < 0 }
				has_opinion = { target = CHI value < 0 }
			}
		}
		AND = {
			original_tag = PAK
			has_opinion = { target = RAJ value < 0 }
		}
		AND = {
			original_tag = CHI
			has_opinion = { target = RAJ value < 0 }
		}
	}
}

# AI gate for the attack decisions. Stricter than prepare:
# target country must exist, hold the contested state, not be in a
# defensive pact, and ROOT must actually have troops in the source state.
border_war_ai_should_attack = {
	threat > 0.50
	has_war = no
	has_capitulated = no
	NOT = { has_country_flag = border_war_attacking_pak_raj }
	NOT = { has_country_flag = border_war_attacking_chi_raj }
	check_variable = { border_readiness > 60 }
	check_variable = { global.border_tension > 80 }
	has_country_flag = border_war_recently_active
}

# Reusable gates for the attack decisions: shared trigger so every
# `attack_X_from_Y` decision stays short.
border_war_attack_available = {
	custom_trigger_tooltip = {
		tooltip = border_war_not_attacking_TT
		NOT = { has_country_flag = border_war_attacking_pak_raj }
		NOT = { has_country_flag = border_war_attacking_chi_raj }
	}
	custom_trigger_tooltip = {
		tooltip = border_war_tension_TT
		check_variable = { global.border_tension > 79 }
	}
	custom_trigger_tooltip = {
		tooltip = border_war_readiness_TT
		check_variable = { border_readiness > 60 }
	}
}

# A captured state on either side is held only by the captor; the
# grab decision becomes available to the captor with progress > 79.
border_war_grab_available = {
	custom_trigger_tooltip = {
		tooltip = border_war_progress_TT
		check_variable = { border_war_progress > 79 }
	}
}
