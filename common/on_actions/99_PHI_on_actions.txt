### Philippines corruption random event pulse
### Fires PHI_random_corruption_event_pulse monthly if PHI_locked_corruption is active.

on_actions = {
	on_monthly_PHI = {
		effect = {
			PHI_random_corruption_event_pulse = yes
		}
	}
	on_startup = {
		effect = {
			PHI = {
				country_event = {
                	id = PHI.1
                	days = 274
            }
		}
	}
}
	on_daily = {
		effect = {
			if = {
				limit = { original_tag = PHI }
				PHI_update_war_flag_effect = yes
			}
		}
	}
}
