	has_global_flag = GLOBAL_STALKER_zone_registry_initialized
}

	STALKER_scenario_enabled = yes
	NOT = { has_global_flag = STALKER_2012_resolved }
	698 = {
		STALKER_is_active_zone_anchor = yes
		check_variable = { STALKER_zone_id = 1 }
		is_controlled_by = ROOT
	}
}

}

}

STALKER_zone_in_crisis = {
	STALKER_is_active_zone_anchor = yes
	OR = {
		has_state_flag = STALKER_perimeter_breach_active
		has_dynamic_modifier = { modifier = STALKER_emission_disruption }
		check_variable = { STALKER_zone_containment < 30 }
		check_variable = { STALKER_zone_activity > 79 }
	}
}
