# Scope: target. coalition_attacker = AI with the wargoal; coalition_new_wargoal = 1 while it is created.
# add_ai_strategy entries add up, so the flag pairs each -1000 with one later +1000.
coalition_restraint_update_effect = {
	set_temp_variable = { coalition_restraint_on = 0 }
	if = {
		limit = {
			OR = {
				check_variable = { coalition_new_wargoal = 1 }
				var:coalition_attacker = {
					OR = {
						has_wargoal_against = PREV
						is_justifying_wargoal_against = PREV
					}
				}
			}
			NOT = { has_war_with = var:coalition_attacker }
		}
		set_temp_variable = { coalition_them = num_divisions }
		every_country = {
			limit = {
				NOT = { tag = PREV }
				NOT = { tag = var:coalition_attacker }
				NOT = { is_in_faction_with = var:coalition_attacker }
				OR = {
					is_in_faction_with = PREV
					PREV = { is_guaranteed_by = PREV }
				}
			}
			add_to_temp_variable = { coalition_them = num_divisions }
		}
		var:coalition_attacker = {
			set_temp_variable = { coalition_us = num_divisions }
			every_country = {
				limit = {
					NOT = { tag = PREV }
					is_in_faction_with = PREV
				}
				add_to_temp_variable = { coalition_us = num_divisions }
			}
		}
		set_temp_variable = { coalition_us_x2 = { value = coalition_us multiply = 2 } }
		if = {
			limit = { check_variable = { coalition_us_x2 < coalition_them } }
			set_temp_variable = { coalition_restraint_on = 1 }
		}
	}
	if = {
		limit = {
			check_variable = { coalition_restraint_on = 0 }
			var:coalition_attacker = { has_country_flag = coalition_restraint_strong@PREV }
		}
		var:coalition_attacker = {
			add_ai_strategy = { type = declare_war id = PREV value = 1000 }
			clr_country_flag = coalition_restraint_strong@PREV
		}
	}
	else_if = {
		limit = {
			check_variable = { coalition_restraint_on = 1 }
			var:coalition_attacker = { NOT = { has_country_flag = coalition_restraint_strong@PREV } }
		}
		var:coalition_attacker = {
			add_ai_strategy = { type = declare_war id = PREV value = -1000 }
			set_country_flag = coalition_restraint_strong@PREV
		}
	}
}
