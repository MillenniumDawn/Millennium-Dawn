##Australia reacts to a Chinese security pact in Honiara
AST_honiara_pact_response = {
	allowed = { original_tag = AST }
	enable = {
		has_country_flag = CHI_honiara_pact_signed
		country_exists = CHI
		NOT = { AST_is_china_aligned = yes }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = antagonize id = "CHI" value = 50 }
	ai_strategy = { type = contain id = "CHI" value = 50 }
}

##Australia contests the Solomon Islands against China (default alignment)

	allowed = { original_tag = AST }
	enable = {
		country_exists = SOL
		NOT = { AST_is_china_aligned = yes }
		SOL = { check_variable = { AST_comp_registered = 1 } }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = befriend id = "SOL" value = 60 }
	ai_strategy = { type = influence id = "SOL" value = 50 }
	ai_strategy = { type = contain id = "CHI" value = 30 }
}
