on_actions = {
	on_weekly_ALG = {
		effect = {
			# If protests are ongoing and exhaustion is high, chance they end
			if = {
				limit = { has_country_flag = ALG_protests_ongoing }
				if = {
					limit = {
						check_variable = { ALG_protest_exhaustion > 19 }
					}
					random = {
						chance = 30
						country_event = algerian_events.2025
					}
				}
				# Protests are not tired
				if = { limit = { check_variable = { ALG_protest_exhaustion < 10 } }
					random = {
						chance = 70
						ALG_trigger_protest = yes
					}
				}
					# Protests are starting to tire
				else_if = { limit = { check_variable = { ALG_protest_exhaustion > 9 } }
					random = {
						chance = 40
						ALG_trigger_protest = yes
					}
				}

				# Protests are tired
				else_if = { limit = { check_variable = { ALG_protest_exhaustion > 15 } }
					random = {
						chance = 20
						ALG_trigger_protest = yes
					}
				}
			}
		}
	}

	on_monthly_ALG = {
		effect = {
			random_list = {
				chance_to_kill_poachers = {
					subtract_from_variable = { no_of_poachers = no_of_poachers_to_kill }
					clamp_variable = { var = no_of_poachers min = 0 }
				}
				chance_to_not_kill_poachers = {
					add_to_variable = { no_of_poachers = 15 }
				}
			}
			if = {
				limit = {
					NOT = { has_country_flag = on_yearly_done_ALG }
					date > 2001.1.1
				}
				random_list = {
					sahara_expands = {
						country_event = { id = Algeria_pulse.1 }
					}
					sahara_steady = {
						country_event = { id = Algeria_pulse.2 }
					}
					sahara_retreats = {
						country_event = { id = Algeria_pulse.3 }
					}
				}
				set_country_flag = { flag = on_yearly_done_ALG days = 350 value = 1 }
			}
			if = {
				limit = {
					tag = ALG
					has_country_flag = ALG_ethnic_data_initialized
				}
				ALG_monthly_ethnic_decay = yes
			}
		}
	}
	# Auto-trigger ethnic system - Black Spring 2001 historical trigger
	on_startup = {
		effect = {
			ALG = {
				if = {
					limit = {
						is_subject = no
						NOT = { has_country_flag = ALG_ethnic_data_initialized }
					}
					# Will trigger within first year, simulating post-Civil War ethnic tensions
					country_event = { id = ALG_ethnic_data.0 days = 60 random_days = 300 }
				}
			}
		}
	}
}
