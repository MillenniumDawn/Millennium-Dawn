	OR = {
		JAP = { shipyard_can_host_repairs = yes }
		KOR = { shipyard_can_host_repairs = yes }
		ENG = { shipyard_can_host_repairs = yes }
		FRA = { shipyard_can_host_repairs = yes }
		GER = { shipyard_can_host_repairs = yes }
		ITA = { shipyard_can_host_repairs = yes }
		SPR = { shipyard_can_host_repairs = yes }
		HOL = { shipyard_can_host_repairs = yes }
	}

shipyard_can_host_repairs = {
	exists = yes
	num_of_naval_factories > 10
	NOT = { has_war_with = ROOT }
	OR = {
		is_in_faction_with = ROOT
		is_in_tech_sharing_group = Major_Non_NATO_Ally_Share
	}
}
