EST_ai_any_path = {
	OR = {
		has_global_flag = EST_HISTORICAL_FOCUS_PATH
		has_global_flag = EST_CONSERVATIVE_FOCUS_PATH
		has_global_flag = EST_SOCIAL_DEMOCRATIC_FOCUS_PATH
		has_global_flag = EST_CENTRIST_FOCUS_PATH
		has_global_flag = EST_NATIONALIST_FOCUS_PATH
	}
}

EST_ai_historical_path = {
	OR = {
		is_historical_focus_on = yes
		has_global_flag = EST_HISTORICAL_FOCUS_PATH
	}
}

EST_ai_not_historical_path = {
	NOT = { EST_ai_historical_path = yes }
	EST_ai_any_path = yes
}

EST_ai_western_path = {
	OR = {
		EST_ai_historical_path = yes
		has_global_flag = EST_CONSERVATIVE_FOCUS_PATH
		has_global_flag = EST_SOCIAL_DEMOCRATIC_FOCUS_PATH
	}
}
