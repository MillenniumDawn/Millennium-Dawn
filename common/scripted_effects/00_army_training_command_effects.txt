MD_artc_upgrade_academy = {
	if = {
		limit = { check_variable = { MD_artc_academy_level < 1 } }
		army_experience = @MD_artc_upgrade_cost_1_neg
	}
	else_if = {
		limit = { check_variable = { MD_artc_academy_level = 1 } }
		army_experience = @MD_artc_upgrade_cost_2_neg
		if = {
			limit = { has_idea = MD_artc_academy_1 }
			swap_ideas = { remove_idea = MD_artc_academy_1 add_idea = MD_artc_academy_2 }
		}
	}
	else = {
		army_experience = @MD_artc_upgrade_cost_3_neg
		if = {
			limit = { has_idea = MD_artc_academy_2 }
			swap_ideas = { remove_idea = MD_artc_academy_2 add_idea = MD_artc_academy_3 }
		}
	}
	hidden_effect = {
		add_to_variable = { MD_artc_academy_level = 1 }
	}
}

MD_artc_run_staff_college = {
	army_experience = @MD_artc_staff_college_cost_neg
	add_command_power = @MD_artc_command_power_cost_neg
	every_army_leader = { gain_xp = @MD_artc_staff_college_xp }
	random = {
		chance = 20
		random_army_leader = {
			limit = {
				OR = {
					NOT = { has_trait = organizer }
					NOT = { has_trait = panzer_leader }
					NOT = { has_trait = infantry_leader }
					NOT = { has_trait = armoured_cavalry_leader }
					NOT = { has_trait = air_cavalry_leader }
					NOT = { has_trait = trait_engineer }
					NOT = { has_trait = artillery_leader }
					NOT = { has_trait = trickster }
					NOT = { has_trait = skilled_staffer }
				}
			}
			random_list = {
				1 = {
					modifier = { factor = 0 has_trait = organizer }
					add_unit_leader_trait = organizer
				}
				1 = {
					modifier = { factor = 0 has_trait = panzer_leader }
					add_unit_leader_trait = panzer_leader
				}
				1 = {
					modifier = { factor = 0 has_trait = infantry_leader }
					add_unit_leader_trait = infantry_leader
				}
				1 = {
					modifier = { factor = 0 has_trait = armoured_cavalry_leader }
					add_unit_leader_trait = armoured_cavalry_leader
				}
				1 = {
					modifier = { factor = 0 has_trait = air_cavalry_leader }
					add_unit_leader_trait = air_cavalry_leader
				}
				1 = {
					modifier = { factor = 0 has_trait = trait_engineer }
					add_unit_leader_trait = trait_engineer
				}
				1 = {
					modifier = { factor = 0 has_trait = artillery_leader }
					add_unit_leader_trait = artillery_leader
				}
				1 = {
					modifier = { factor = 0 has_trait = trickster }
					add_unit_leader_trait = trickster
				}
				1 = {
					modifier = { factor = 0 has_trait = skilled_staffer }
					add_unit_leader_trait = skilled_staffer
				}
			}
		}
	}
	add_timed_idea = { idea = MD_artc_staff_college_cooldown days = @MD_artc_staff_college_cooldown_days }
}

MD_artc_run_senior_course = {
	army_experience = @MD_artc_senior_course_cost_neg
	add_command_power = @MD_artc_command_power_cost_neg
	if = {
		limit = {
			any_army_leader = {
				is_field_marshal = yes
				skill < @MD_artc_senior_course_skill_cap
			}
		}
		random_army_leader = {
			limit = {
				is_field_marshal = yes
				skill < @MD_artc_senior_course_skill_cap
			}
			add_skill_level = 1
		}
	}
	else = {
		random_army_leader = {
			limit = { skill < @MD_artc_senior_course_skill_cap }
			add_skill_level = 1
		}
	}
	add_timed_idea = { idea = MD_artc_senior_course_cooldown days = @MD_artc_senior_course_cooldown_days }
}

MD_artc_toggle_window = {
	if = {
		limit = { NOT = { has_country_flag = MD_artc_open } }
		set_country_flag = MD_artc_open
	}
	else = {
		clr_country_flag = MD_artc_open
	}
	update_MD_artc_dirty_variable = yes
}

update_MD_artc_dirty_variable = {
	if = {
		limit = { check_variable = { global.MD_artc_dirty_update_var < 10000 } }
		add_to_variable = { global.MD_artc_dirty_update_var = 1 }
	}
	else = {
		set_variable = { global.MD_artc_dirty_update_var = 1 }
	}
}
