# Kadyrov changes sides
country_event = {
	id = sov_army.1
	title = sov_army.1.t
	desc = sov_army.1.d
	picture = GFX_chech_soldiers_loyal
	is_triggered_only = yes

	option = {
		name = sov_army.1.a
		log = "[GetDateText]: [This.GetName]: sov_army.1.a executed"
		add_stability = 0.02
		if = {
			limit = { NOT = { CHE = { is_subject_of = SOV } } }
			hidden_effect = {
				if = {
					limit = { country_exists = CHE has_war_with = CHE }
					white_peace = CHE
				}
				if = {
					limit = { NOT = { country_exists = CHE } }
					release = CHE
				}
				CHE = {
					set_country_flag = SOV_subject_republic
					transfer_state = 672
					add_state_core = 672
				}
				set_autonomy = {
					target = CHE
					autonomy_state = autonomy_republic_rf
				}
				add_state_core = 672
				CHE = {
					set_temp_variable = { rul_party_temp = 6 }
					set_temp_variable = { change_leader_temp = 1 }
					change_ruling_party_effect = yes
					add_ideas = SUB_small_taxes
				}
				set_temp_variable = { modify_subjectnalogs = 3 }
				modify_subjectnalogs_support = yes
				SUB_separatism_plus_subject = yes
			}
			custom_effect_tooltip = SOV_kadyrov_autonomy_tt
		}
		country_event = { id = sov_army.2 days = 1300 random_days = 200 }
		ai_chance = { base = 1 }
	}
}
