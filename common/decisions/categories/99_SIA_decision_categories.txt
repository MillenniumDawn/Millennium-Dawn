# "Thai Strategic and Economic Decisions" — the consolidated economic, strategic and
# diplomatic hub (former SIA_asean_diplomacy_category, SIA_economy_wave3_category and
# SIA_realm_diplomacy_category merged in). The OR covers every unlock gate across the
# former categories.
SIA_economy_category = {
	allowed = { original_tag = SIA }
	icon = GFX_decision_category_generic_economy
	priority = 80

	visible = {
		OR = {
			has_idea = SIA_tourism_powerhouse
			has_completed_focus = SIA_medical_hub
			has_country_flag = SIA_eec_active
			has_country_flag = SIA_dynasty_holdings_unlocked
			has_country_flag = SIA_royal_projects_unlocked
			has_country_flag = SIA_kra_canal_tolls_unlocked
			has_country_flag = SIA_arms_export_unlocked
			has_country_flag = SIA_defense_exports_active
			has_idea = SIA_bamboo_diplomacy
			has_completed_focus = SIA_eec_take_off
			has_completed_focus = SIA_treaty_ally
			has_country_flag = SIA_realm_decisions_unlocked
			date > 2005.1.1
		}
	}
}

# / CAP-7 — the Kra Canal across its whole lifecycle (former SIA_kra_category folded
# in). Shows the phased construction missions while the feasibility study is open and
# the canal is unbuilt, then the post-canal Strait Patrol lever once the strait is open.
# The two windows never overlap, so one category spans both.
SIA_kra_construction = {
	allowed = { original_tag = SIA }
	icon = GFX_decision_category_generic_industry
	priority = 73

	visible = {
		OR = {
			AND = {
				has_country_flag = SIA_kra_feasibility_open
				NOT = { has_country_flag = SIA_kra_canal_complete }
			}
			has_country_flag = SIA_strait_patrol_unlocked
		}
	}
}
