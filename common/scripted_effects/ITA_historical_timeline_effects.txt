# Effect: ITA_shift_party_popularity
# Purpose: Moves a share of one party slot's popularity to another slot, inside one outlook or across outlooks
# Parameters:
# - shift_from_index: party slot 0-23 that loses popularity
# - shift_to_index: party slot 0-23 that gains popularity
# - shift_fraction: share of the source slot's popularity that moves, 0.5 moves half
ITA_shift_party_popularity = {
	hidden_effect = {
		set_temp_variable = { shift_amount = party_pop_array^shift_from_index }
		multiply_temp_variable = { shift_amount = shift_fraction }
		set_temp_variable = { outlook_party_index = shift_from_index }
		set_outlook_from_party_index = yes
		set_temp_variable = { shift_from_outlook = outlook_id }
		set_temp_variable = { outlook_party_index = shift_to_index }
		set_outlook_from_party_index = yes
		set_temp_variable = { shift_to_outlook = outlook_id }
		#a negative add_popularity hands the freed share to every other outlook in equal parts, so set all five at once
		if = {
			limit = { NOT = { check_variable = { shift_from_outlook = shift_to_outlook } } }
			set_outlook_popularity_array = yes
			add_to_temp_variable = { outlook_popularity_by_id^shift_to_outlook = shift_amount }
			subtract_from_temp_variable = { outlook_popularity_by_id^shift_from_outlook = shift_amount }
			#set_popularities needs the five to sum to 100
			set_temp_variable = {
				shift_remainder = {
					value = 1
					subtract = outlook_popularity_by_id^0
					subtract = outlook_popularity_by_id^1
					subtract = outlook_popularity_by_id^2
					subtract = outlook_popularity_by_id^3
					subtract = outlook_popularity_by_id^4
				}
			}
			add_to_temp_variable = { outlook_popularity_by_id^shift_to_outlook = shift_remainder }
			set_temp_variable = { shift_democratic = { value = outlook_popularity_by_id^0 multiply = 100 } }
			set_temp_variable = { shift_communism = { value = outlook_popularity_by_id^1 multiply = 100 } }
			set_temp_variable = { shift_fascism = { value = outlook_popularity_by_id^2 multiply = 100 } }
			set_temp_variable = { shift_neutrality = { value = outlook_popularity_by_id^3 multiply = 100 } }
			set_temp_variable = { shift_nationalist = { value = outlook_popularity_by_id^4 multiply = 100 } }
			set_popularities = {
				democratic = shift_democratic
				communism = shift_communism
				fascism = shift_fascism
				neutrality = shift_neutrality
				nationalist = shift_nationalist
			}
		}
		add_to_variable = { party_pop_array^shift_to_index = shift_amount }
		multiply_temp_variable = { shift_amount = -1 }
		add_to_variable = { party_pop_array^shift_from_index = shift_amount }
		recalculate_party = yes
	}
}
