PER_ai_path_selected = {
	OR = {
		has_global_flag = PER_HISTORICAL_FOCUS_PATH
		has_global_flag = PER_RESTORATION_FOCUS_PATH
		has_global_flag = PER_IRANIC_FEDERATION_FOCUS_PATH
		has_global_flag = PER_MOJAHEDIN_FOCUS_PATH
		has_global_flag = PER_SECULAR_REPUBLIC_FOCUS_PATH
		has_global_flag = PER_GREEN_PATH_FOCUS_PATH
		has_global_flag = PER_HEIRS_FOCUS_PATH
		has_global_flag = PER_WORKERS_REPUBLIC_FOCUS_PATH
		has_global_flag = PER_MILITARY_JUNTA_FOCUS_PATH
	}
}

PER_ai_historical_path = {
	OR = {
		is_historical_focus_on = yes
		has_global_flag = PER_HISTORICAL_FOCUS_PATH
	}
}

PER_ai_not_historical_path = {
	OR = {
		has_global_flag = PER_RESTORATION_FOCUS_PATH
		has_global_flag = PER_IRANIC_FEDERATION_FOCUS_PATH
		has_global_flag = PER_MOJAHEDIN_FOCUS_PATH
		has_global_flag = PER_SECULAR_REPUBLIC_FOCUS_PATH

		has_global_flag = PER_MILITARY_JUNTA_FOCUS_PATH
	}
}

PER_ai_reform_path = {
	OR = {
		has_global_flag = PER_SECULAR_REPUBLIC_FOCUS_PATH
		has_global_flag = PER_GREEN_PATH_FOCUS_PATH
	}
}

PER_ai_not_reform_path = {
	NOT = { PER_ai_reform_path = yes }
	OR = {
		is_historical_focus_on = yes
		PER_ai_path_selected = yes
	}
}
