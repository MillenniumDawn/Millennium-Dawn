USA_apple_resolve_capstone = {
	USA_apple_initialize_state = yes

	if = {
		limit = { NOT = { has_country_flag = USA_apple_capstone_resolved } }

		if = {
			limit = {
				check_variable = { var = USA_apple_privacy_legitimacy value = 7 compare = greater_than_or_equals }
				check_variable = { var = USA_apple_silicon_autonomy value = 6 compare = greater_than_or_equals }
				OR = {
					has_country_flag = USA_apple_private_cloud_compute
					has_country_flag = USA_apple_on_device_ai
				}
			}
			USA_apple_apply_private_capstone = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_apple_ecosystem_control value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_apple_services_dependence value = 7 compare = greater_than_or_equals }
			}
			USA_apple_apply_integrated_capstone = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_apple_supply_resilience value = 7 compare = greater_than_or_equals }
				check_variable = { var = USA_apple_silicon_autonomy value = 6 compare = greater_than_or_equals }
				check_variable = { var = USA_apple_china_exposure value = 4 compare = less_than_or_equals }
			}
			USA_apple_apply_resilient_capstone = yes
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { var = USA_apple_regulatory_pressure value = 7 compare = greater_than_or_equals }
					AND = {
						check_variable = { var = USA_apple_services_dependence value = 8 compare = greater_than_or_equals }
						check_variable = { var = USA_apple_privacy_legitimacy value = 3 compare = less_than_or_equals }
					}
				}
			}
			USA_apple_apply_regulated_capstone = yes
		}
		else = { USA_apple_apply_premium_capstone = yes }
	}
}
