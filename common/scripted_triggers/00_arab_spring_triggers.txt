arab_spring_eligible = {
	exists = yes
	NOT = { has_country_flag = AS_disabled_arab_spring }
	NOT = { has_country_flag = AS_arab_spring_completed }
}

arab_spring_protests_active = {
	OR = {
		has_country_flag = arab_spring_active
		has_active_mission = arab_spring_protests_2
		has_active_mission = syr_arab_spring_protests_2
	}
}

arab_spring_first_wave_available = {
	NOT = { has_global_flag = global_arab_spring_started }
	NOT = {
		any_of_scopes = {
			array = global.arabic_countries
			has_country_flag = arab_spring_active
		}
	}
}

arab_spring_ben_ali_in_power = {
	tag = TUN
	has_country_leader = { name = "Zine El Abidine Ben Ali" ruling_only = yes }
}

arab_spring_egyptian_handover = {
	tag = EGY
	date > 2011.2.11
	has_country_leader = { name = "Hosni Mubarak" ruling_only = yes }
}

arab_spring_saleh_in_power = {
	tag = YEM
	western_autocrats_are_in_power = yes
	has_country_leader = { name = "Ali Abdullah Saleh" ruling_only = yes }
}

arab_spring_ai_can_start_response = {
	NOT = {
		OR = {
			has_decision = arab_spring_encourage_supporters
			has_decision = arab_spring_request_crackdown
			has_decision = arab_spring_deploy_the_military
			has_decision = arab_spring_return_to_barracks
			has_decision = arab_spring_reshuffle_government
			has_decision = arab_spring_amend_the_constitution
			AND = {
				original_tag = SYR
				SYR_ai_historical_path = yes
			}
			AND = {
				is_historical_focus_on = yes
				date > 2010.12.17
				arab_spring_ben_ali_in_power = yes
			}
			AND = {
				arab_spring_egyptian_handover = yes
				EGY_ai_historical_path = yes
			}
			AND = {
				is_historical_focus_on = yes
				date > 2011.1.27
				arab_spring_saleh_in_power = yes
			}
		}
	}
}
