on_actions = {
	on_startup = {
		effect = {
			if = { limit = { country_exists = UKR } UKR = { UKR_ct_init = yes } }
			if = { limit = { country_exists = UZB } UZB = { UKR_ct_foreign_influence_refresh = yes } }
			if = { limit = { country_exists = KAZ } KAZ = { UKR_ct_foreign_influence_refresh = yes } }
			if = { limit = { country_exists = KYR } KYR = { UKR_ct_foreign_influence_refresh = yes } }
			if = { limit = { country_exists = TAJ } TAJ = { UKR_ct_foreign_influence_refresh = yes } }
			if = { limit = { country_exists = TRK } TRK = { UKR_ct_foreign_influence_refresh = yes } }
		}
	}

	on_daily_UKR = {
		effect = {
			if = {
				limit = { has_country_flag = UKR_ct_initialized_flag }
				UKR_ct_v1_forecast_recalculate = yes
				UKR_ct_v1_publish_marker = yes
			}
		}
	}

	on_daily_UZB = {
		effect = { UKR_ct_foreign_influence_refresh = yes }
	}

	on_daily_KAZ = {
		effect = { UKR_ct_foreign_influence_refresh = yes }
	}

	on_daily_KYR = {
		effect = { UKR_ct_foreign_influence_refresh = yes }
	}

	on_daily_TAJ = {
		effect = { UKR_ct_foreign_influence_refresh = yes }
	}

	on_daily_TRK = {
		effect = { UKR_ct_foreign_influence_refresh = yes }
	}

	on_monthly_UKR = {
		effect = { UKR_ct_monthly = yes }
	}
}
