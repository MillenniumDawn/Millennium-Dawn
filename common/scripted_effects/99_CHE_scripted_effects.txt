#---------# Chechnya Scripted Effects: #---------#
#------------------------------------------------#
#
TST = { 
	set_variable = {
		devastation_percentage = 70
	}
	
	add_to_global_variable = {
		md_gui_dirty = 1
	}
	chechnya_update_devastation_effect = yes
}


CHE_init_devastation_effect = {
	#---------# Startup Effects #---------#

	set_country_flag = chechnya_devastation_mechanic_active

	set_variable = {
		devastation_percentage = 40
	}
	set_variable = {
		devastation_frame = 0
	}
	set_variable = {
		devastation_dirty = 0
	}
	set_variable = { #---------# base drift #---------#
		devastation_weekly_drift = 0.2
	}
	chechnya_update_devastation_effect = yes
}

chechnya_update_devastation_effect = {

	#---------# Clamping the Varible #---------#
	if = {
		limit = {
			check_variable = {
				devastation_percentage > 100
			}
		}
		set_variable = {
			devastation_percentage = 100
		}
	}

	if = {
		limit = {
			check_variable = {
				devastation_percentage < 0
			}
		}
		set_variable = {
			devastation_percentage = 0
		}
	}

	#---------# Math for the Progress Bar #---------#
	set_variable = {
		devastation_frame = devastation_percentage
	}
	multiply_variable = {
		devastation_frame = 10
	}

	if = {
		limit = {
			check_variable = {
				devastation_frame > 1000
			}
		}
		set_variable = {
			devastation_frame = 1000
		}
	}

	#---------# Fliping the dirty flag so the scripted GUI redraws #---------#
	set_temp_variable = {
		devastation_flip = 1
	}
	subtract_from_temp_variable = {
		devastation_flip = devastation_dirty
	}
	set_variable = {
		devastation_dirty = devastation_flip
	}
}

CHE_apply_weekly_devastation_drift_effect = {
	#---------# This is from the Three Toggles #---------# >>>>>>>>>> #---------# TODO #---------#
	if = {
		limit = {
			has_country_flag = chechnya_stance_tier_1
		}
		set_variable = {
			devastation_weekly_drift = 0.6
		}
	}
	else_if = {
		limit = {
			has_country_flag = chechnya_stance_tier_2
		}
		set_variable = {
			devastation_weekly_drift = 1.4
		}
	}
	else_if = {
		limit = {
			has_country_flag = chechnya_stance_tier_3
		}
		set_variable = {
			devastation_weekly_drift = 2.8
		}
	}
	else = {
		set_variable = {
			devastation_weekly_drift = 0.2
		}
	}

	 #---------# Prevents Negative Drift #---------#
	if = {
		limit = {
			check_variable = {
				devastation_weekly_drift < 0
			}
		}
		set_variable = {
			devastation_weekly_drift = 0
		}
	}

	add_to_variable = {
		devastation_percentage = devastation_weekly_drift
	}

	chechnya_update_devastation_effect = yes
}
