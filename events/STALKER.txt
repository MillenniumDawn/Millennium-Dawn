		text = STALKER.1.d
	}
	desc = {
		trigger = { event_target:STALKER_zone_display = { check_variable = { STALKER_zone_id = 27 } } }
		text = STALKER.1.d_lake_mungo
	}
	desc = {
		trigger = {
			event_target:STALKER_zone_display = {
				NOT = { check_variable = { STALKER_zone_id = 1 } }
				NOT = { check_variable = { STALKER_zone_id = 27 } }
			}
		}
		text = STALKER.1.d_global
	}
	picture = GFX_nuclear_power_plant
	is_triggered_only = yes

	trigger = {
		STALKER_scenario_enabled = yes
		has_event_target = STALKER_zone_event_target

			STALKER_is_zone_location = yes
			has_state_flag = STALKER_zone_policy_pending
			is_controlled_by = ROOT
		}
	}

	immediate = { event_target:STALKER_zone_event_target = { STALKER_activate_zone = yes } }

	option = {
		name = STALKER.1.a
		unlock_decision_category_tooltip = STALKER_zone_management_category
		custom_effect_tooltip = STALKER_military_policy_tt
		hidden_effect = {
			if = {
				limit = {
					event_target:STALKER_zone_event_target = {
						has_state_flag = STALKER_zone_policy_pending
						is_controlled_by = ROOT
					}
				}
				log = "[GetDateText]: [This.GetName]: STALKER.1.a executed"
				small_expenditure = yes
				event_target:STALKER_zone_event_target = {
					clr_state_flag = STALKER_zone_policy_pending
					add_to_variable = { STALKER_zone_containment = 30 }
					STALKER_clamp_zone_society = yes
					controller = { STALKER_refresh_zone_administration = yes }
				}
			}
		}

	option = {
