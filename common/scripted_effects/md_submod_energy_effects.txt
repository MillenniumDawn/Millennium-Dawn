# common/scripted_effects/md_submod_energy_effects.txt

md_sample_energy_metrics = {
	# Total generation = active generation + idle fossil powerplants (they still push power to the grid)
	set_variable = { md_current_energy_gen = energy_sum }
	add_to_variable = { md_current_energy_gen = free_fossil_powerplants_power }
	set_variable = { md_current_energy_con = energy_consumption }
	set_variable = { md_current_energy_bal = md_current_energy_gen }
	subtract_from_variable = { md_current_energy_bal = md_current_energy_con }
}

md_submod_energy_graph_init = {
	md_sample_energy_metrics = yes

	resize_array = { array = energy_gen_array value = md_current_energy_gen size = 120 }
	resize_array = { array = energy_con_array value = md_current_energy_con size = 120 }
	resize_array = { array = energy_bal_array value = md_current_energy_bal size = 120 }
}

# 1. THE ENERGY MONTHLY SHIFTER (Self-Healing)
md_submod_energy_graph_on_monthly = {
	md_sample_energy_metrics = yes

	if = {
		limit = { NOT = { check_variable = { energy_gen_array^num > 0 } } }
		md_submod_energy_graph_init = yes
	}

	set_variable = { energy_gen_array^current_gdp_graph_end = md_current_energy_gen }
	set_variable = { energy_con_array^current_gdp_graph_end = md_current_energy_con }
	set_variable = { energy_bal_array^current_gdp_graph_end = md_current_energy_bal }

	if = {
		limit = { check_variable = { current_gdp_graph_end = 1 } }
		set_variable = { energy_gen_array^0 = md_current_energy_gen }
		set_variable = { energy_con_array^0 = md_current_energy_con }
		set_variable = { energy_bal_array^0 = md_current_energy_bal }
	}
}

# 2. THE MATH TRANSLATOR & FAILSAFE
update_md_flaxbeard_energy_graph = {
	md_sample_energy_metrics = yes

	if = {
		limit = { NOT = { check_variable = { energy_gen_array^num > 0 } } }
		md_submod_energy_graph_init = yes
	}

	clear_array = md_flax_offsets_energy_gen_yellow
	clear_array = md_flax_slopes_energy_gen

	clear_array = md_flax_offsets_energy_con_blue
	clear_array = md_flax_slopes_energy_con

	clear_array = md_flax_offsets_energy_bal_green
	clear_array = md_flax_offsets_energy_bal_red
	clear_array = md_flax_slopes_energy_bal

	# BOUNDS: Generation, Consumption, and Net Balance in GW
	set_temp_variable = { true_max = energy_gen_array^0 }
	set_temp_variable = { true_min = energy_bal_array^0 }
	set_temp_variable = { max_jump = 0 }

	for_each_loop = {
		array = energy_gen_array
		value = v
		index = i

		if = {
			limit = { NOT = { check_variable = { i > current_gdp_graph_end } } }
			if = { limit = { check_variable = { v > true_max } } set_temp_variable = { true_max = v } }
			if = { limit = { check_variable = { v < true_min } } set_temp_variable = { true_min = v } }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }
			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { jump = energy_gen_array^next_i }
				subtract_from_temp_variable = { jump = v }
				if = { limit = { check_variable = { jump < 0 } } multiply_temp_variable = { jump = -1 } }
				if = { limit = { check_variable = { jump > max_jump } } set_temp_variable = { max_jump = jump } }
			}
		}
	}

	for_each_loop = {
		array = energy_con_array
		value = v
		index = i

		if = {
			limit = { NOT = { check_variable = { i > current_gdp_graph_end } } }
			if = { limit = { check_variable = { v > true_max } } set_temp_variable = { true_max = v } }
			if = { limit = { check_variable = { v < true_min } } set_temp_variable = { true_min = v } }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }
			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { jump = energy_con_array^next_i }
				subtract_from_temp_variable = { jump = v }
				if = { limit = { check_variable = { jump < 0 } } multiply_temp_variable = { jump = -1 } }
				if = { limit = { check_variable = { jump > max_jump } } set_temp_variable = { max_jump = jump } }
			}
		}
	}

	for_each_loop = {
		array = energy_bal_array
		value = v
		index = i

		if = {
			limit = { NOT = { check_variable = { i > current_gdp_graph_end } } }
			if = { limit = { check_variable = { v > true_max } } set_temp_variable = { true_max = v } }
			if = { limit = { check_variable = { v < true_min } } set_temp_variable = { true_min = v } }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }
			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { jump = energy_bal_array^next_i }
				subtract_from_temp_variable = { jump = v }
				if = { limit = { check_variable = { jump < 0 } } multiply_temp_variable = { jump = -1 } }
				if = { limit = { check_variable = { jump > max_jump } } set_temp_variable = { max_jump = jump } }
			}
		}
	}

	# PADDING, ANTI-TEAR ZOOM & STEP
	md_calc_standard_axis = yes

	# Y-AXIS LABELS (GW)
	set_temp_variable = { buf = low }
	for_loop_effect = {
		start = 0 end = 11
		set_variable = { md_energy_y_axis_label^v = buf }
		add_to_temp_variable = { buf = step }
	}

	# 1. DRAW GENERATION LINE (Yellow)
	for_each_loop = {
		array = energy_gen_array
		value = v
		index = i

		if = {
			limit = { check_variable = { i > current_gdp_graph_end } }
			add_to_array = { md_flax_offsets_energy_gen_yellow = 1000 }
			add_to_array = { md_flax_slopes_energy_gen = 50 }
		}
		else = {
			set_temp_variable = { y1_val = v }
			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }

			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { y2_val = energy_gen_array^next_i }
			}
			else = {
				set_temp_variable = { y2_val = v }
			}

			md_calc_segment_math = yes

			add_to_array = { md_flax_offsets_energy_gen_yellow = offset }
			add_to_array = { md_flax_slopes_energy_gen = slope }
		}
	}

	# 2. DRAW CONSUMPTION LINE (Blue)
	for_each_loop = {
		array = energy_con_array
		value = v
		index = i

		if = {
			limit = { check_variable = { i > current_gdp_graph_end } }
			add_to_array = { md_flax_offsets_energy_con_blue = 1000 }
			add_to_array = { md_flax_slopes_energy_con = 50 }
		}
		else = {
			set_temp_variable = { y1_val = v }
			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }

			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { y2_val = energy_con_array^next_i }
			}
			else = {
				set_temp_variable = { y2_val = v }
			}

			md_calc_segment_math = yes

			add_to_array = { md_flax_offsets_energy_con_blue = offset }
			add_to_array = { md_flax_slopes_energy_con = slope }
		}
	}

	# 3. DRAW POWER BALANCE LINE (Green on Surplus, Red on Deficit)
	for_each_loop = {
		array = energy_bal_array
		value = v
		index = i

		if = {
			limit = { check_variable = { i > current_gdp_graph_end } }
			add_to_array = { md_flax_offsets_energy_bal_green = 1000 }
			add_to_array = { md_flax_offsets_energy_bal_red = 1000 }
			add_to_array = { md_flax_slopes_energy_bal = 50 }
		}
		else = {
			set_temp_variable = { y1_val = v }
			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }

			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { y2_val = energy_bal_array^next_i }
			}
			else = {
				set_temp_variable = { y2_val = v }
			}

			md_calc_segment_math = yes

			if = {
				limit = { check_variable = { v < 0 } } # Deficit = Red
				add_to_array = { md_flax_offsets_energy_bal_red = offset }
				add_to_array = { md_flax_offsets_energy_bal_green = 1000 }
			}
			else = { # Surplus = Green
				add_to_array = { md_flax_offsets_energy_bal_green = offset }
				add_to_array = { md_flax_offsets_energy_bal_red = 1000 }
			}

			add_to_array = { md_flax_slopes_energy_bal = slope }
		}
	}

	gdp_graph_update_dirty_var = yes
}
