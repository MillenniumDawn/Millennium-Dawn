country_event = {
	id = mdrc_ldk.4
	title = mdrc_ldk.4.t
	desc = mdrc_ldk.4.d
	picture = GFX_african_demonstration
	is_triggered_only = yes

	trigger = {
		original_tag = DRC
		NOT = { has_country_flag = MDRC_LDK_diplomacy_chosen }
	}

	immediate = { log = "[GetDateText]: [Root.GetName]: Event mdrc_ldk.4" set_country_flag = MDRC_LDK_diplomacy_chosen }

	option = {
		log = "[GetDateText]: [Root.GetName]: Event mdrc_ldk.4 option mdrc_ldk.4.a"
		name = mdrc_ldk.4.a
		ai_chance = {
			base = 65
			modifier = { factor = 0.25 has_active_mission = bankruptcy_incoming_collapse }
		}
		set_country_flag = MDRC_LDK_negotiating
		add_political_power = -50
		MDRC_offer_ceasefires = yes
		if = {
			limit = {
				country_exists = RWA
				NOT = { has_war_with = RWA }
			}
			add_opinion_modifier = { target = RWA modifier = MDRC_partnership }
			reverse_add_opinion_modifier = { target = RWA modifier = MDRC_partnership }
		}
		if = {
			limit = {
				country_exists = UGA
				NOT = { has_war_with = UGA }
			}
			add_opinion_modifier = { target = UGA modifier = MDRC_partnership }
			reverse_add_opinion_modifier = { target = UGA modifier = MDRC_partnership }
		}
	}

	option = {
		log = "[GetDateText]: [Root.GetName]: Event mdrc_ldk.4 option mdrc_ldk.4.b"
		name = mdrc_ldk.4.b
		ai_chance = { base = 35 }
		set_country_flag = MDRC_LDK_confrontation
		add_ideas = MDRC_LDK_escalation
		add_war_support = 0.03
		if = {
			limit = { country_exists = RWA }
			RWA = {
				add_opinion_modifier = { target = DRC modifier = MDRC_LDK_hostility }
			}
		}
		if = {
			limit = { country_exists = UGA }
			UGA = {
				add_opinion_modifier = { target = DRC modifier = MDRC_LDK_hostility }
			}
		}
	}
}
