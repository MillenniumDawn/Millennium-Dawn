USA_e3_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = USA_e3_state_initialized } }

		set_variable = { USA_e3_institutional_centrality = 8 }
		set_variable = { USA_e3_exhibitor_confidence = 8 }
		set_variable = { USA_e3_consumer_reach = 5 }
		set_variable = { USA_e3_digital_capability = 2 }
		set_variable = { USA_e3_operating_burden = 6 }
		set_variable = { USA_game_showcase_decentralization = 1 }
		set_variable = { USA_esa_executive_convening = 6 }
		set_variable = { USA_e3_lifecycle_stage = 0 }
		set_variable = { USA_showcase_ecosystem_stage = 0 }
		set_variable = { USA_esa_convening_stage = 0 }
		set_country_flag = USA_e3_state_initialized
	}

	USA_e3_clamp_state = yes
}

USA_e3_clamp_state = {
	set_temp_variable = { corp_value = USA_e3_institutional_centrality }
	corporate_history_clamp_value = yes
	set_variable = { USA_e3_institutional_centrality = corp_value }
	set_temp_variable = { corp_value = USA_e3_exhibitor_confidence }
	corporate_history_clamp_value = yes
	set_variable = { USA_e3_exhibitor_confidence = corp_value }
	set_temp_variable = { corp_value = USA_e3_consumer_reach }
	corporate_history_clamp_value = yes
	set_variable = { USA_e3_consumer_reach = corp_value }
	set_temp_variable = { corp_value = USA_e3_digital_capability }
	corporate_history_clamp_value = yes
	set_variable = { USA_e3_digital_capability = corp_value }
	set_temp_variable = { corp_value = USA_e3_operating_burden }
	corporate_history_clamp_value = yes
	set_variable = { USA_e3_operating_burden = corp_value }
	set_temp_variable = { corp_value = USA_game_showcase_decentralization }
	corporate_history_clamp_value = yes
	set_variable = { USA_game_showcase_decentralization = corp_value }
	set_temp_variable = { corp_value = USA_esa_executive_convening }
	corporate_history_clamp_value = yes
	set_variable = { USA_esa_executive_convening = corp_value }

	USA_e3_validate_registers = yes
}

USA_e3_validate_registers = {
	set_temp_variable = { corp_value = USA_e3_lifecycle_stage }
	clamp_temp_variable = { var = corp_value min = 0 max = 7 }
	set_variable = { USA_e3_lifecycle_stage = corp_value }
	set_temp_variable = { corp_value = USA_showcase_ecosystem_stage }
	clamp_temp_variable = { var = corp_value min = 0 max = 3 }
	set_variable = { USA_showcase_ecosystem_stage = corp_value }
	set_temp_variable = { corp_value = USA_esa_convening_stage }
	clamp_temp_variable = { var = corp_value min = 0 max = 3 }
	set_variable = { USA_esa_convening_stage = corp_value }
}

USA_e3_apply_lifecycle_idea = {
	remove_ideas = USA_e3_world_industry_marketplace
	remove_ideas = USA_e3_media_business_summit
	remove_ideas = USA_e3_restored_industry_expo
	remove_ideas = USA_e3_public_hybrid
	remove_ideas = USA_e3_digital_common_stage
	remove_ideas = USA_e3_revival_in_progress
	remove_ideas = USA_e3_retired_legacy

	if = {
		limit = { check_variable = { USA_e3_lifecycle_stage = 0 } }
		add_ideas = USA_e3_world_industry_marketplace
	}
	else_if = {
		limit = { check_variable = { USA_e3_lifecycle_stage = 1 } }
		add_ideas = USA_e3_media_business_summit
	}
	else_if = {
		limit = { check_variable = { USA_e3_lifecycle_stage = 2 } }
		add_ideas = USA_e3_restored_industry_expo
	}
	else_if = {
		limit = { check_variable = { USA_e3_lifecycle_stage = 3 } }
		add_ideas = USA_e3_public_hybrid
	}
	else_if = {
		limit = { check_variable = { USA_e3_lifecycle_stage = 4 } }
		if = {
			limit = { has_country_flag = USA_e3_public_access }
			add_ideas = USA_e3_public_hybrid
		}
		else = { add_ideas = USA_e3_restored_industry_expo }
	}
	else_if = {
		limit = { check_variable = { USA_e3_lifecycle_stage = 5 } }
		add_ideas = USA_e3_digital_common_stage
	}
	else_if = {
		limit = { check_variable = { USA_e3_lifecycle_stage = 6 } }
		add_ideas = USA_e3_revival_in_progress
	}
	else_if = {
		limit = { check_variable = { USA_e3_lifecycle_stage = 7 } }
		add_ideas = USA_e3_retired_legacy
	}
}

USA_e3_apply_capstone_idea = {
	remove_ideas = USA_e3_decentralized_showcase_economy
	remove_ideas = USA_e3_federated_showcase_protocol

	if = {
		limit = { has_country_flag = USA_decentralized_showcase_ecosystem }
		add_ideas = USA_e3_decentralized_showcase_economy
	}
	else_if = {
		limit = { has_country_flag = USA_e3_federated_protocol }
		add_ideas = USA_e3_federated_showcase_protocol
	}
}

USA_e3_apply_iicon_idea = {
	remove_ideas = USA_iicon_executive_network
	if = {
		limit = { has_country_flag = USA_iicon_completed }
		add_ideas = USA_iicon_executive_network
	}
}

USA_e3_sync_presentation = {
	USA_e3_apply_lifecycle_idea = yes
	USA_e3_apply_capstone_idea = yes
	USA_e3_apply_iicon_idea = yes
}

USA_e3_resolve_capstone = {
	if = {
		limit = {
			NOT = { has_country_flag = USA_e3_capstone_resolved }
			has_country_flag = USA_e3_permanently_retired
		}
		if = {
			limit = {
				NOT = { has_country_flag = USA_e3_federated_protocol }
				OR = {
					has_country_flag = USA_decentralized_showcase_ecosystem
					check_variable = { USA_game_showcase_decentralization > 6 }
				}
			}
			set_country_flag = USA_decentralized_showcase_ecosystem
		}
		set_country_flag = USA_e3_capstone_resolved
	}

	USA_e3_apply_capstone_idea = yes
}
