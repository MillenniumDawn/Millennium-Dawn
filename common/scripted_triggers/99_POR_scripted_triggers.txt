	any_owned_state = {

			is_claimed_by = ROOT
			is_core_of = ROOT
		}
	}
	all_owned_state = {
		OR = {
			is_fully_controlled_by = ROOT
			AND = {
				NOT = { is_claimed_by = ROOT }
				NOT = { is_core_of = ROOT }
			}
		}
	}
}
