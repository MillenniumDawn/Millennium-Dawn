MD_artc_can_upgrade_academy = {
	OR = {
		AND = {
			check_variable = { MD_artc_academy_level < 1 }
			has_army_experience > @MD_artc_upgrade_cost_1
		}
		AND = {
			check_variable = { MD_artc_academy_level = 1 }
			has_army_experience > @MD_artc_upgrade_cost_2
		}
		AND = {
			check_variable = { MD_artc_academy_level = 2 }
			has_army_experience > @MD_artc_upgrade_cost_3
		}
	}
}
