# Effective dockyards: one per hull class our yards still keep in service, plus one per
# merger. Four dockyards against two effective ones is the 2000 starting position.
UKR_yard_recalc = {
	set_variable = { shipyard_dockyards = num_of_naval_factories }
	set_variable = { shipyard_effective_dockyards = ukr_yard_tier }
	add_to_variable = { shipyard_effective_dockyards = shipyard_consolidation_stage }

	set_variable = { ukr_idle_dockyards = shipyard_dockyards }
	subtract_from_variable = { ukr_idle_dockyards = shipyard_effective_dockyards }
	clamp_variable = { var = ukr_idle_dockyards min = 0 max = 1000 }

	# An idle dockyard produces nothing, so the penalty is its share of the whole yard.
	set_variable = { ukr_mod_structure = ukr_idle_dockyards }
	if = {
		limit = { check_variable = { shipyard_dockyards > 0 } }
		divide_variable = { ukr_mod_structure = shipyard_dockyards }
	}
	multiply_variable = { ukr_mod_structure = -1 }
	clamp_variable = { var = ukr_mod_structure min = -1 max = 0 }
	set_temp_variable = { ukr_consolidation_bonus = shipyard_consolidation_stage }
	multiply_temp_variable = { ukr_consolidation_bonus = 0.05 }
	add_to_variable = { ukr_mod_structure = ukr_consolidation_bonus }

	set_variable = { ukr_cost_dockyard = 5 }
	multiply_variable = { ukr_cost_dockyard = shipyard_cost_mult }
	set_variable = { ukr_cost_frigate = 8 }
	multiply_variable = { ukr_cost_frigate = shipyard_cost_mult }
	set_variable = { ukr_cost_large = 20 }
	multiply_variable = { ukr_cost_large = shipyard_cost_mult }
	set_variable = { ukr_cost_subsidy = 1.2 }
	multiply_variable = { ukr_cost_subsidy = shipyard_cost_mult }
	set_variable = { ukr_cost_consolidation = 1.5 }
	multiply_variable = { ukr_cost_consolidation = shipyard_cost_mult }
}

UKR_yard_tier_up = {
	add_to_variable = { ukr_yard_tier = 1 }
	if = {
		limit = { check_variable = { ukr_yard_tier = 3 } }
		UKR_yard_frigate_relief = yes
	}
	else_if = {
		limit = { check_variable = { ukr_yard_tier = 4 } }
		UKR_yard_large_relief = yes
	}
	shipyard_recalc = yes
}

UKR_yard_subsidy_complete = {
	add_to_variable = { shipyard_crisis_level = -3 }
	clamp_variable = { var = shipyard_crisis_level min = 0 max = 100 }
	shipyard_recalc = yes
}

UKR_yard_consolidate_complete = {
	add_to_variable = { shipyard_consolidation_stage = 1 }
	add_to_variable = { shipyard_crisis_level = -4 }
	clamp_variable = { var = shipyard_crisis_level min = 0 max = 100 }
	shipyard_recalc = yes
}
