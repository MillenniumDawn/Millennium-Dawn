# NATO AI strategies. Membership is decision-driven (common/decisions/NATO.txt), so the
# ascension strategy keys on the NATO_member idea and NATO_Aspirant flag instead of
# historical accession dates, and allowed mirrors the NATO_decisions category gate.

		NOT = { has_game_rule = { rule = rule_disable_nato option = yes } }

	allowed = {
		NOT = { has_game_rule = { rule = rule_disable_nato option = yes } }
	}
	enable = { has_idea = NATO_member }

	ai_strategy = { type = antagonize id = "SOV" value = 25 }

}

	allowed = {
		NOT = { has_game_rule = { rule = rule_disable_nato option = yes } }
	}
	enable = {
		has_idea = NATO_member

NATO_civil_war_in_Belarus = {

		NOT = { has_game_rule = { rule = rule_disable_nato option = yes } }
		OR = {
			tag = USA
			tag = POL
			tag = FRA
		}
	}
	enable = {
		country_exists = BLG
		BLG = { has_war_with = BLR }
		has_relation_modifier = {
			target = BLG
			modifier = generic_increased_military_support
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = send_volunteers_desire id = BLG value = 500 }
}
