# Own-choice budget law and tax changes add decaying costs; scripted changes set spending_law_scripted_change

@policy_transition_cost_per_level = 0.10
@policy_transition_cost_cap = 0.30
@policy_transition_cost_decay = 0.01
@policy_transition_cut_per_level = 0.05
@policy_transition_cut_cap = -0.20
@policy_transition_tax_cap = -0.15
@policy_transition_penalty_recovery = 0.005
@policy_transition_population_tax_per_point = 0.01
@policy_transition_corporate_tax_per_point = 0.006
@policy_transition_tax_peak_decay = 1
@policy_transition_mandate_factor = 0.5

policy_transition_ensure_modifier = {
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = policy_transition_costs } } }
		add_dynamic_modifier = { modifier = policy_transition_costs }
		custom_effect_tooltip = { localization_key = adds_dynamic_modifier_tt MODIFIER = policy_transition_costs }
	}
	else = {
		custom_effect_tooltip = { localization_key = modifies_dynamic_modifier_tt MODIFIER = policy_transition_costs }
	}
}

policy_transition_apply_mandate = {
	if = {
		limit = { has_idea = policy_reform_mandate }
		multiply_temp_variable = { policy_transition_amount = @policy_transition_mandate_factor }
	}
}

# Charges only the part of a move away from the expected level: > 0 raises above it, < 0 cuts below it
policy_transition_charged_levels = {
	set_temp_variable = { policy_transition_delta = 0 }
	if = {
		limit = { check_variable = { spending_new_level > policy_transition_old } }
		set_temp_variable = {
			policy_transition_delta = {
				value = spending_new_level
				subtract = { value = policy_transition_old max = policy_transition_expected }
				max = 0
			}
		}
	}
	else_if = {
		limit = { check_variable = { spending_new_level < policy_transition_old } }
		set_temp_variable = {
			policy_transition_delta = {
				value = policy_transition_old
				min = policy_transition_expected
				subtract = spending_new_level
				max = 0
				multiply = -1
			}
		}
	}
}

policy_transition_raise_amount = {
	set_temp_variable = { policy_transition_amount = { value = policy_transition_delta multiply = @policy_transition_cost_per_level } }
	policy_transition_apply_mandate = yes
}

policy_transition_cut_amount = {
	set_temp_variable = { policy_transition_amount = { value = policy_transition_delta multiply = @policy_transition_cut_per_level } }
	policy_transition_apply_mandate = yes
}

# Cuts and tax hikes have separate caps; the dynamic modifier shows their sum
policy_transition_sync_stability = {
	set_temp_variable = {
		policy_transition_change = {
			value = policy_transition_cut_stability
			add = policy_transition_tax_stability
			subtract = policy_transition_stability
		}
	}
	if = {
		limit = { check_variable = { policy_transition_change < 0 } }
		policy_transition_ensure_modifier = yes
		add_to_variable = { var = policy_transition_stability value = policy_transition_change tooltip = stability_factor_tt }
	}
}

policy_transition_add_cut_penalty = {
	add_to_variable = { policy_transition_cut_stability = policy_transition_amount }
	clamp_variable = { var = policy_transition_cut_stability min = @policy_transition_cut_cap max = 0 }
	policy_transition_sync_stability = yes
}

policy_transition_add_tax_penalty = {
	add_to_variable = { policy_transition_tax_stability = policy_transition_amount }
	clamp_variable = { var = policy_transition_tax_stability min = @policy_transition_tax_cap max = 0 }
	policy_transition_sync_stability = yes
}

# Called from law on_add with temp spending_new_level, before <cat>_law is updated
bureau_transition_check = {
	if = {
		limit = {
			NOT = { has_country_flag = spending_law_scripted_change }
			has_variable = bureau_law
		}
		set_temp_variable = { policy_transition_old = bureau_law }
		set_temp_variable = { policy_transition_expected = expected_adm_spending }
		policy_transition_charged_levels = yes
		if = {
			limit = { check_variable = { policy_transition_delta > 0 } }
			policy_transition_raise_amount = yes
			policy_transition_ensure_modifier = yes
			add_to_variable = { var = policy_transition_bureau_cost value = policy_transition_amount tooltip = bureaucracy_cost_multiplier_modifier_tt }
			clamp_variable = { var = policy_transition_bureau_cost min = 0 max = @policy_transition_cost_cap }
		}
		else_if = {
			limit = { check_variable = { policy_transition_delta < 0 } }
			policy_transition_cut_amount = yes
			policy_transition_add_cut_penalty = yes
		}
	}
}

# Call after a tax change with temp previous_tax_rate; charges points above the peak, which holds while the penalty is capped
population_tax_transition_check = {
	if = {
		limit = { check_variable = { population_tax_rate > previous_tax_rate } }
		if = {
			limit = {
				OR = {
					NOT = { has_variable = policy_transition_population_tax_peak }
					check_variable = { previous_tax_rate > policy_transition_population_tax_peak }
				}
			}
			set_variable = { policy_transition_population_tax_peak = previous_tax_rate }
		}
		set_temp_variable = { policy_transition_tax_cap_temp = @policy_transition_tax_cap }
		if = {
			limit = {
				check_variable = { population_tax_rate > policy_transition_population_tax_peak }
				check_variable = { policy_transition_tax_stability > policy_transition_tax_cap_temp }
			}
			set_temp_variable = {
				policy_transition_amount = {
					value = policy_transition_population_tax_peak
					subtract = population_tax_rate
					multiply = @policy_transition_population_tax_per_point
				}
			}
			set_variable = { policy_transition_population_tax_peak = population_tax_rate }
			policy_transition_apply_mandate = yes
			policy_transition_add_tax_penalty = yes
		}
	}
}

policy_transition_monthly_decay = {
	hidden_effect = {
		if = {
			limit = { check_variable = { policy_transition_bureau_cost > 0 } }
			set_variable = { policy_transition_bureau_cost = { value = policy_transition_bureau_cost subtract = @policy_transition_cost_decay max = 0 } }
		}
		if = {
			limit = { check_variable = { policy_transition_police_cost > 0 } }
			set_variable = { policy_transition_police_cost = { value = policy_transition_police_cost subtract = @policy_transition_cost_decay max = 0 } }
		}
		if = {
			limit = { check_variable = { policy_transition_education_cost > 0 } }
			set_variable = { policy_transition_education_cost = { value = policy_transition_education_cost subtract = @policy_transition_cost_decay max = 0 } }
		}
		if = {
			limit = { check_variable = { policy_transition_health_cost > 0 } }
			set_variable = { policy_transition_health_cost = { value = policy_transition_health_cost subtract = @policy_transition_cost_decay max = 0 } }
		}
		if = {
			limit = { check_variable = { policy_transition_social_cost > 0 } }
			set_variable = { policy_transition_social_cost = { value = policy_transition_social_cost subtract = @policy_transition_cost_decay max = 0 } }
		}
		if = {
			limit = { check_variable = { policy_transition_military_cost > 0 } }
			set_variable = { policy_transition_military_cost = { value = policy_transition_military_cost subtract = @policy_transition_cost_decay max = 0 } }
		}
		if = {
			limit = { check_variable = { policy_transition_cut_stability < 0 } }
			set_variable = { policy_transition_cut_stability = { value = policy_transition_cut_stability add = @policy_transition_penalty_recovery min = 0 } }
		}
		if = {
			limit = { check_variable = { policy_transition_tax_stability < 0 } }
			set_variable = { policy_transition_tax_stability = { value = policy_transition_tax_stability add = @policy_transition_penalty_recovery min = 0 } }
		}
		set_variable = { policy_transition_stability = { value = policy_transition_cut_stability add = policy_transition_tax_stability } }
		if = {
			limit = { check_variable = { policy_transition_war_support < 0 } }
			set_variable = { policy_transition_war_support = { value = policy_transition_war_support add = @policy_transition_penalty_recovery min = 0 } }
		}
		if = {
			limit = { check_variable = { policy_transition_population_tax_peak > population_tax_rate } }
			set_variable = { policy_transition_population_tax_peak = { value = policy_transition_population_tax_peak subtract = @policy_transition_tax_peak_decay max = population_tax_rate } }
		}
		if = {
			limit = { check_variable = { policy_transition_corporate_tax_peak > corporate_tax_rate } }
			set_variable = { policy_transition_corporate_tax_peak = { value = policy_transition_corporate_tax_peak subtract = @policy_transition_tax_peak_decay max = corporate_tax_rate } }
		}
		if = {
			limit = {
				NOT = { check_variable = { policy_transition_bureau_cost > 0 } }
				NOT = { check_variable = { policy_transition_police_cost > 0 } }
				NOT = { check_variable = { policy_transition_education_cost > 0 } }
				NOT = { check_variable = { policy_transition_health_cost > 0 } }
				NOT = { check_variable = { policy_transition_social_cost > 0 } }
				NOT = { check_variable = { policy_transition_military_cost > 0 } }
				NOT = { check_variable = { policy_transition_stability < 0 } }
				NOT = { check_variable = { policy_transition_war_support < 0 } }
			}
			clear_policy_transition = yes
		}
	}
}

clear_policy_transition = {
	if = {
		limit = { has_dynamic_modifier = { modifier = policy_transition_costs } }
		remove_dynamic_modifier = { modifier = policy_transition_costs }
	}
	clear_variable = policy_transition_bureau_cost
	clear_variable = policy_transition_police_cost
	clear_variable = policy_transition_education_cost
	clear_variable = policy_transition_health_cost
	clear_variable = policy_transition_social_cost
	clear_variable = policy_transition_military_cost
	clear_variable = policy_transition_cut_stability
	clear_variable = policy_transition_tax_stability
	clear_variable = policy_transition_stability
	clear_variable = policy_transition_war_support
	clear_variable = policy_transition_population_tax_peak
	clear_variable = policy_transition_corporate_tax_peak
}

# Re-granting restarts the 365 days instead of doing nothing
grant_reform_mandate = {
	if = {
		limit = { has_idea = policy_reform_mandate }
		hidden_effect = { remove_ideas = policy_reform_mandate }
	}
	add_timed_idea = { idea = policy_reform_mandate days = 365 }
}
