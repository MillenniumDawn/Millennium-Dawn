				add_to_temp_variable = { int_investment_change = BUL.gdp_per_capita }

				multiply_temp_variable = { debt_change = 0.31 }
				modify_debt_effect = yes
				set_temp_variable = { percent_change = 4.56 }
				set_temp_variable = { influence_target = BUL }
				change_influence_percentage = yes
			}
		}
		else_if = {
			limit = {
				original_tag = ROM
			}
			ROM = {
				random_owned_state = {
					add_building_construction = {
						type = industrial_complex
						level = 1
						instant_build = yes
					}
					add_extra_state_shared_building_slots = 1
				}
			}
			AUS = {
				add_to_temp_variable = { int_investment_change = ROM.gdp_per_capita }
				multiply_temp_variable = { int_investment_change = 0.42 }
				modify_international_investment_effect = yes

				change_influence_percentage = yes
			}
		}
		else_if = {
			limit = {
				original_tag = SLO
			}
			SLO = {
				random_owned_state = {
					add_building_construction = {
						type = industrial_complex
						level = 1
						instant_build = yes
					}
					add_building_construction = {
						type = arms_factory
						level = 1
						instant_build = yes
					}
					add_extra_state_shared_building_slots = 2
				}
			}

				set_temp_variable = { percent_change = 6.58 }
				set_temp_variable = { influence_target = SLO }
				change_influence_percentage = yes
			}

			base = 80
		}
