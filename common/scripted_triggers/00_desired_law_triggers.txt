# INPUT
# index (index of the law, starting from 1)
generate_expected_welfare_spending_frame = {
	# welfare spending frames
	if = {
		limit = {
			check_variable = { 
				var = social_law
				value = index
				compare = greater_than_or_equals
			}
			check_variable = { index > expected_welfare_spending }
		}

		meta_trigger = {
			text = {
				set_temp_variable = { welfare_spending_frame_[INDEX] = 3 } 
			}
			INDEX = "[?index]"
		}
	}
	else_if = {
		limit = {
			check_variable = { 
				var = social_law
				value = index
				compare = greater_than_or_equals
			}
		}

		meta_trigger = {
			text = {
				set_temp_variable = { welfare_spending_frame_[INDEX] = 2 } 
			}
			INDEX = "[?index]"
		}
	}
	else_if = {
		limit = {
			check_variable = { 
				var = expected_welfare_spending
				value = index
				compare = greater_than_or_equals
			}
		}

		meta_trigger = {
			text = {
				set_temp_variable = { welfare_spending_frame_[INDEX] = 4 } 
			}
			INDEX = "[?index]"
		}
	}
	else = {
		meta_trigger = {
			text = {
				set_temp_variable = { welfare_spending_frame_[INDEX] = 1 } 
			}
			INDEX = "[?index]"
		}
	}
}
