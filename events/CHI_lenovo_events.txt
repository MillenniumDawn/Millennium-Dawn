country_event = {
	id = CHI_lenovo_events.1
	title = CHI_lenovo_events.1.t
	desc = CHI_lenovo_events.1.d
	picture = GFX_computer
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		corporate_history_full_enabled = yes
		original_tag = CHI
		NOT = { has_country_flag = collapsed_nation }
		country_exists = USA
		USA = {
			NOT = { has_country_flag = collapsed_nation }
			OR = {
				has_country_flag = USA_ibm_pc_divested
				has_country_flag = USA_ibm_pc_retained
				has_country_flag = USA_ibm_pc_spinout
			}
		}
		NOT = { has_war_with = USA }
		NOT = { has_country_flag = CHI_lenovo_pc_path_chosen }
	}

	immediate = {
		set_country_flag = CHI_lenovo_event_1_delivered
		CHI_lenovo_initialize_state = yes
	}

	option = {
		name = CHI_lenovo_events.1.a
		trigger = { USA = { has_country_flag = USA_ibm_pc_divested } }
		log = "[GetDateText]: [Root.GetName]: CHI_lenovo_events.1.a executed"
		custom_effect_tooltip = CHI_lenovo_events.1.a_tt
		set_country_flag = CHI_lenovo_pc_path_chosen
		set_country_flag = CHI_lenovo_ibm_bid_accepted
		CHI_lenovo_schedule_cfius_followup = yes
		ai_chance = {
			base = 90
			modifier = { add = 10 is_historical_focus_on = yes }
		}
	}

	option = {
		name = CHI_lenovo_events.1.b
		trigger = {
			USA = {
				OR = {
					has_country_flag = USA_ibm_pc_divested
					has_country_flag = USA_ibm_pc_spinout
				}
			}
		}
		log = "[GetDateText]: [Root.GetName]: CHI_lenovo_events.1.b executed"
		custom_effect_tooltip = CHI_lenovo_events.1.b_tt
		set_country_flag = CHI_lenovo_pc_path_chosen
		set_country_flag = CHI_lenovo_narrow_ibm_bridge
		if = {
			limit = { USA = { has_country_flag = USA_ibm_pc_divested } }
			set_country_flag = CHI_lenovo_ibm_bid_accepted
			CHI_lenovo_schedule_cfius_followup = yes
		}
		else = {
			set_country_flag = { flag = CHI_lenovo_event_5_pending value = 1 days = 405 }
			country_event = { id = CHI_lenovo_events.5 days = 365 }
			set_country_flag = { flag = CHI_lenovo_event_6_pending value = 1 days = 220 }
			country_event = { id = CHI_lenovo_events.6 days = 180 }
		}
		ai_chance = {
			base = 10
			modifier = {
				factor = 0
				is_historical_focus_on = yes
				USA = { has_country_flag = USA_ibm_pc_divested }
			}
		}
	}

	option = {
		name = CHI_lenovo_events.1.c
		trigger = {
			USA = {
				OR = {
					has_country_flag = USA_ibm_pc_retained
					has_country_flag = USA_ibm_pc_spinout
				}
			}
		}
		log = "[GetDateText]: [Root.GetName]: CHI_lenovo_events.1.c executed"
		custom_effect_tooltip = CHI_lenovo_events.1.c_tt
		set_country_flag = CHI_lenovo_pc_path_chosen
		set_country_flag = CHI_lenovo_organic_globalization
		set_country_flag = { flag = CHI_lenovo_event_5_pending value = 1 days = 405 }
		country_event = { id = CHI_lenovo_events.5 days = 365 }
		ai_chance = { base = 10 }
	}
}
