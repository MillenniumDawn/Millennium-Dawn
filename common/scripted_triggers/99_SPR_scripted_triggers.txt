			OR = {
				western_conservatism_are_in_power_or_coalition = yes
				neutrality_neutral_conservatism_in_power_or_coalition = yes

			neutrality_neutral_conservatism_in_power_or_coalition = yes

	owns_state = 88
	88 = { is_core_of = NAV }
	NOT = {
		OR = {
			AND = { check_variable = { SPR_culture_opinion^0 < 31 } 86 = { is_core_of = CAT } }
			AND = {
				check_variable = { SPR_culture_opinion^2 < 31 }
				OR = { 944 = { is_core_of = GAL } 1055 = { is_core_of = GAL } }
			}

			AND = { check_variable = { SPR_culture_opinion^4 < 31 } 95 = { is_core_of = CNR } }
		}
	}
}
