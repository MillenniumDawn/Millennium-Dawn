opec_can_apply = {
	custom_trigger_tooltip = {
		tooltip = OPEC_NET_EXPORTER_TT
		check_variable = { resource_produced@oil > resource_consumed@oil }
	}
	custom_trigger_tooltip = {
		tooltip = OPEC_MIN_PRODUCTION_TT
		check_variable = { resource_produced@oil > 15 }
	}
	custom_trigger_tooltip = {
		tooltip = OPEC_NOT_LEFT_RECENTLY_TT
		NOT = { has_country_flag = opec_left_recently }
	}
}

opec_can_select_proposal = {
	custom_trigger_tooltip = {
		tooltip = OPEC_IS_MEMBER_TT
		is_in_array = { global.opec_members = THIS }
	}
	custom_trigger_tooltip = {
		tooltip = OPEC_NOT_CONFIRMED_TT
		NOT = { check_variable = { opec_vote > 0 } }
	}
	custom_trigger_tooltip = {
		tooltip = OPEC_NOT_IN_SESSION_TT
		NOT = { has_global_flag = opec_conference_in_session }
	}
}
