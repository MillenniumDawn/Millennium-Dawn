# PARTY BANS
	# SWA - ESWATINI
		SWA_initialize_party_bans = {
			set_variable = { party_banned_array^4 = 1 }
			set_variable = { party_banned_array^5 = 1 }
		}

	# OAU / African Union member opinion modifiers
	oau_relations = {
		if = {
			limit = { has_idea = OAU_member }
			add_opinion_modifier = { target = THIS modifier = oau_relations }
		}
		else = {
			remove_opinion_modifier = { target = THIS modifier = oau_relations }
		}
	}
	african_union_relations = {
		if = {
			limit = { has_idea = AU_member }
			add_opinion_modifier = { target = THIS modifier = african_union_relations }
		}
		else = {
			remove_opinion_modifier = { target = THIS modifier = african_union_relations }
		}
	}
