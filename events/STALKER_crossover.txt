country_event = {
	id = STALKER.100
	title = STALKER.100.t
	desc = STALKER.100.d
	picture = GFX_libya_militia_training
	is_triggered_only = yes

	trigger = {
		STALKER_meme_content_enabled = yes
		has_event_target = STALKER_zone_event_target
		event_target:STALKER_zone_event_target = {
			STALKER_is_active_zone_anchor = yes
			is_controlled_by = ROOT
		}
	}

	option = {
		name = STALKER.100.a
		log = "[GetDateText]: [This.GetName]: STALKER.100.a executed"
		add_political_power = 10
		ai_chance = { base = 1 }
	}

	option = {
		name = STALKER.100.b
		log = "[GetDateText]: [This.GetName]: STALKER.100.b executed"
		custom_effect_tooltip = STALKER.100.b_tt
		hidden_effect = {
			if = {
				limit = {
					event_target:STALKER_zone_event_target = {
						STALKER_is_active_zone_anchor = yes
						is_controlled_by = ROOT
					}
				}
				event_target:STALKER_zone_event_target = {
					add_to_variable = { STALKER_zone_containment = 2 }
					STALKER_clamp_zone_society = yes
				}
			}
		}
		ai_chance = { base = 1 }
	}
}

country_event = {
	id = STALKER.105
	title = STALKER.105.t
	desc = STALKER.105.d
	picture = GFX_crime_fighting
	is_triggered_only = yes

	trigger = {
		STALKER_meme_content_enabled = yes
		NOT = { has_global_flag = STALKER_alice_said_hello }
		has_event_target = STALKER_zone_event_target
		event_target:STALKER_zone_event_target = {
			check_variable = { STALKER_zone_id = 27 }
			STALKER_is_active_zone_anchor = yes
			is_controlled_by = ROOT
		}
	}

	immediate = {
		log = "[GetDateText]: [This.GetName]: STALKER.105 immediate executed"
		set_global_flag = STALKER_alice_said_hello
	}

	option = {
		name = STALKER.105.a
		log = "[GetDateText]: [This.GetName]: STALKER.105.a executed"
		custom_effect_tooltip = STALKER.105.a_tt
		hidden_effect = {
			if = {
				limit = {
					event_target:STALKER_zone_event_target = {
						STALKER_is_active_zone_anchor = yes
						is_controlled_by = ROOT
					}
				}
				set_variable = { global.STALKER_alice_choice = 1 }
				set_global_flag = { flag = STALKER_alice_return_wait value = 1 days = 180 }
			}
			else = { clr_global_flag = STALKER_alice_said_hello }
		}
		ai_chance = { base = 1 }
	}

	option = {
		name = STALKER.105.b
		trigger = { NOT = { has_political_power < 25 } }
		log = "[GetDateText]: [This.GetName]: STALKER.105.b executed"
		custom_effect_tooltip = STALKER.105.b_tt
		hidden_effect = {
			if = {
				limit = {
					event_target:STALKER_zone_event_target = {
						STALKER_is_active_zone_anchor = yes
						is_controlled_by = ROOT
					}
				}
				set_variable = { global.STALKER_alice_choice = 2 }
				set_global_flag = { flag = STALKER_alice_return_wait value = 1 days = 180 }
				add_political_power = -25
				event_target:STALKER_zone_event_target = {
					add_to_variable = { STALKER_zone_exploitation = 5 }
					add_to_variable = { STALKER_zone_activity = 5 }
					STALKER_clamp_zone_society = yes
					clamp_variable = { var = STALKER_zone_activity min = 0 max = 100 }
					ROOT = { STALKER_refresh_zone_administration = yes }
				}
			}
			else = { clr_global_flag = STALKER_alice_said_hello }
		}
		ai_chance = { base = 1 }
	}

	option = {
		name = STALKER.105.c
		log = "[GetDateText]: [This.GetName]: STALKER.105.c executed"
		custom_effect_tooltip = STALKER.105.c_tt
		hidden_effect = {
			if = {
				limit = {
					event_target:STALKER_zone_event_target = {
						STALKER_is_active_zone_anchor = yes
						is_controlled_by = ROOT
					}
				}
				set_variable = { global.STALKER_alice_choice = 3 }
				set_global_flag = { flag = STALKER_alice_return_wait value = 1 days = 180 }
				event_target:STALKER_zone_event_target = {
					add_to_variable = { STALKER_zone_containment = 5 }
					add_to_variable = { STALKER_zone_exploitation = -3 }
					STALKER_clamp_zone_society = yes
					ROOT = { STALKER_refresh_zone_administration = yes }
				}
			}
			else = { clr_global_flag = STALKER_alice_said_hello }
		}
		ai_chance = { base = 1 }
	}
}
