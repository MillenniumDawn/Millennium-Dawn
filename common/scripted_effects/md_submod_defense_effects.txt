md_submod_defense_graph_init = {
	set_temp_variable = { def_ann = defence_gain }
	multiply_temp_variable = { def_ann = 52 }
	set_variable = { md_current_defense_spending = def_ann }

	set_temp_variable = { def_p = md_current_defense_spending }
	set_temp_variable = { gdp_denom = gdp_total }
	if = { limit = { check_variable = { gdp_denom < 0.01 } } set_temp_variable = { gdp_denom = 0.01 } }
	divide_temp_variable = { def_p = gdp_denom }
	multiply_temp_variable = { def_p = 100 }
	set_variable = { md_current_defense_pct = def_p }

	resize_array = { array = defense_spend_array value = md_current_defense_spending size = 120 }
	resize_array = { array = defense_pct_array value = md_current_defense_pct size = 120 }

	set_variable = { defense_spend_array^0 = md_current_defense_spending }
	set_variable = { defense_pct_array^0 = md_current_defense_pct }
}

# 1. THE DEFENSE MONTHLY SHIFTER (Self-Healing)
md_submod_defense_graph_on_monthly = {
	set_temp_variable = { def_ann = defence_gain }
	multiply_temp_variable = { def_ann = 52 }
	set_variable = { md_current_defense_spending = def_ann }

	set_temp_variable = { def_p = md_current_defense_spending }
	set_temp_variable = { gdp_denom = gdp_total }
	if = { limit = { check_variable = { gdp_denom < 0.01 } } set_temp_variable = { gdp_denom = 0.01 } }
	divide_temp_variable = { def_p = gdp_denom }
	multiply_temp_variable = { def_p = 100 }
	set_variable = { md_current_defense_pct = def_p }

	if = {
		limit = { NOT = { check_variable = { defense_spend_array^num > 0 } } }
		md_submod_defense_graph_init = yes
	}

	set_variable = { defense_spend_array^current_gdp_graph_end = md_current_defense_spending }
	set_variable = { defense_pct_array^current_gdp_graph_end = md_current_defense_pct }

	if = {
		limit = { check_variable = { current_gdp_graph_end = 1 } }
		set_variable = { defense_spend_array^0 = md_current_defense_spending }
		set_variable = { defense_pct_array^0 = md_current_defense_pct }
	}
}

# 2. THE MATH TRANSLATOR & FAILSAFE
update_md_flaxbeard_defense_graph = {
	set_temp_variable = { def_ann = defence_gain }
	multiply_temp_variable = { def_ann = 52 }
	set_variable = { md_current_defense_spending = def_ann }

	set_temp_variable = { def_p = md_current_defense_spending }
	set_temp_variable = { gdp_denom = gdp_total }
	if = { limit = { check_variable = { gdp_denom < 0.01 } } set_temp_variable = { gdp_denom = 0.01 } }
	divide_temp_variable = { def_p = gdp_denom }
	multiply_temp_variable = { def_p = 100 }
	set_variable = { md_current_defense_pct = def_p }

	if = {
		limit = { NOT = { check_variable = { defense_spend_array^num > 0 } } }
		md_submod_defense_graph_init = yes
	}

	clear_array = md_flax_offsets_defense_blue
	clear_array = md_flax_offsets_defense_red
	clear_array = md_flax_slopes_defense

	clear_array = md_flax_offsets_defense_pct_green
	clear_array = md_flax_offsets_defense_pct_red
	clear_array = md_flax_slopes_defense_pct

	# --- YOY (JANUARY-ONLY SNAPSHOT) ---
	set_temp_variable = { months_since_jan = global.month }
	subtract_from_temp_variable = { months_since_jan = 1 }

	set_temp_variable = { recent_jan_idx = current_gdp_graph_end }
	subtract_from_temp_variable = { recent_jan_idx = months_since_jan }
	if = { limit = { check_variable = { recent_jan_idx < 0 } } set_temp_variable = { recent_jan_idx = 0 } }

	set_temp_variable = { prev_jan_idx = recent_jan_idx }
	subtract_from_temp_variable = { prev_jan_idx = 12 }
	if = { limit = { check_variable = { prev_jan_idx < 0 } } set_temp_variable = { prev_jan_idx = 0 } }

	set_temp_variable = { recent_jan_def = defense_spend_array^recent_jan_idx }
	set_temp_variable = { prev_jan_def = defense_spend_array^prev_jan_idx }

	set_variable = { md_yoy_defense_change_billions = recent_jan_def }
	subtract_from_variable = { md_yoy_defense_change_billions = prev_jan_def }
	# ---------------------------------------------

	# LEFT AXIS: Defense Spending ($B/yr)
	set_temp_variable = { true_max = defense_spend_array^0 }
	set_temp_variable = { true_min = defense_spend_array^0 } 
	set_temp_variable = { max_jump = 0 }

	for_each_loop = {
		array = defense_spend_array
		value = v
		index = i

		if = {
			limit = { NOT = { check_variable = { i > current_gdp_graph_end } } }
			if = { limit = { check_variable = { v > true_max } } set_temp_variable = { true_max = v } }
			if = { limit = { check_variable = { v < true_min } } set_temp_variable = { true_min = v } }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }
			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { jump = defense_spend_array^next_i }
				subtract_from_temp_variable = { jump = v }
				if = { limit = { check_variable = { jump < 0 } } multiply_temp_variable = { jump = -1 } }
				if = { limit = { check_variable = { jump > max_jump } } set_temp_variable = { max_jump = jump } }
			}
		}
	}

	set_temp_variable = { data_diff = true_max }
	subtract_from_temp_variable = { data_diff = true_min }

	# FAILSAFE PADDING
	if = {
		limit = { check_variable = { data_diff < 1 } }
		set_temp_variable = { high = true_max }
		add_to_temp_variable = { high = 1.0 }
		set_temp_variable = { low = true_min }
		subtract_from_temp_variable = { low = 1.0 }
	}
	else = {
		set_temp_variable = { pad = data_diff }
		multiply_temp_variable = { pad = 0.1 } 
		set_temp_variable = { high = true_max }
		add_to_temp_variable = { high = pad }
		set_temp_variable = { low = true_min }
		subtract_from_temp_variable = { low = pad }
	}

	if = { limit = { check_variable = { low < 0 } } set_temp_variable = { low = 0 } }
	set_temp_variable = { diff = high }
	subtract_from_temp_variable = { diff = low }
	if = { limit = { check_variable = { diff < 0.01 } } set_temp_variable = { diff = 0.01 } }

	# DYNAMIC ANTI-TEAR ZOOM 
	set_temp_variable = { safe_diff = max_jump }
	multiply_temp_variable = { safe_diff = 45 }
	divide_temp_variable = { safe_diff = 10 }

	if = {
		limit = { check_variable = { diff < safe_diff } }
		set_temp_variable = { shortfall = safe_diff }
		subtract_from_temp_variable = { shortfall = diff }
		divide_temp_variable = { shortfall = 2 }

		add_to_temp_variable = { high = shortfall }
		subtract_from_temp_variable = { low = shortfall }
		if = { limit = { check_variable = { low < 0 } } set_temp_variable = { low = 0 } }
		set_temp_variable = { diff = high }
		subtract_from_temp_variable = { diff = low }
	}

	# LEFT Y-AXIS LABELS ($B/yr)
	set_temp_variable = { step = diff }
	multiply_temp_variable = { step = 0.1 } 
	set_temp_variable = { buf = low }
	for_loop_effect = {
		start = 0 end = 11
		set_variable = { md_defense_y_axis_label^v = buf }
		add_to_temp_variable = { buf = step }
	}

	# 1. DRAW DEFENSE SPENDING LINE (Blue on up, Red on down)
	for_each_loop = {
		array = defense_spend_array
		value = v
		index = i

		if = {
			limit = { check_variable = { i > current_gdp_graph_end } }
			add_to_array = { md_flax_offsets_defense_blue = 1000 } 
			add_to_array = { md_flax_offsets_defense_red = 1000 } 
			add_to_array = { md_flax_slopes_defense = 50 }
		}
		else = {
			set_temp_variable = { y1_val = v }
			subtract_from_temp_variable = { y1_val = low }
			divide_temp_variable = { y1_val = diff } 
			set_temp_variable = { y1_px = y1_val }
			multiply_temp_variable = { y1_px = 450 }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }

			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } } 
				set_temp_variable = { y2_val = defense_spend_array^next_i }
			}
			else = {
				set_temp_variable = { y2_val = v } 
			}

			subtract_from_temp_variable = { y2_val = low }
			divide_temp_variable = { y2_val = diff } 
			set_temp_variable = { y2_px = y2_val }
			multiply_temp_variable = { y2_px = 450 } 

			set_temp_variable = { offset = y1_px }
			multiply_temp_variable = { offset = -1 }

			if = {
				limit = { check_variable = { y2_val < y1_val } } # Down = Red
				add_to_array = { md_flax_offsets_defense_red = offset }
				add_to_array = { md_flax_offsets_defense_blue = 1000 }
			}
			else = { # Up / Flat = Blue
				add_to_array = { md_flax_offsets_defense_blue = offset }
				add_to_array = { md_flax_offsets_defense_red = 1000 }
			}

			set_temp_variable = { slope = y2_px }
			subtract_from_temp_variable = { slope = y1_px }
			divide_temp_variable = { slope = 2 }
			add_to_temp_variable = { slope = 50 }

			if = { limit = { check_variable = { slope > 100 } } set_temp_variable = { slope = 100 } }
			if = { limit = { check_variable = { slope < 1 } } set_temp_variable = { slope = 1 } }

			round_variable = slope
			add_to_array = { md_flax_slopes_defense = slope }
		}
	}

	# RIGHT AXIS: Defense % of GDP (Dynamic Scaling)
	set_temp_variable = { true_max_pct = defense_pct_array^0 }
	set_temp_variable = { true_min_pct = defense_pct_array^0 } 
	set_temp_variable = { max_jump_pct = 0 }

	for_each_loop = {
		array = defense_pct_array
		value = v
		index = i

		if = {
			limit = { NOT = { check_variable = { i > current_gdp_graph_end } } }
			if = { limit = { check_variable = { v > true_max_pct } } set_temp_variable = { true_max_pct = v } }
			if = { limit = { check_variable = { v < true_min_pct } } set_temp_variable = { true_min_pct = v } }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }
			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } }
				set_temp_variable = { jump = defense_pct_array^next_i }
				subtract_from_temp_variable = { jump = v }
				if = { limit = { check_variable = { jump < 0 } } multiply_temp_variable = { jump = -1 } }
				if = { limit = { check_variable = { jump > max_jump_pct } } set_temp_variable = { max_jump_pct = jump } }
			}
		}
	}

	set_temp_variable = { data_diff_pct = true_max_pct }
	subtract_from_temp_variable = { data_diff_pct = true_min_pct }

	# FAILSAFE PADDING
	if = {
		limit = { check_variable = { data_diff_pct < 1 } }
		set_temp_variable = { high_pct = true_max_pct }
		add_to_temp_variable = { high_pct = 1.0 }
		set_temp_variable = { low_pct = true_min_pct }
		subtract_from_temp_variable = { low_pct = 1.0 }
	}
	else = {
		set_temp_variable = { pad_pct = data_diff_pct }
		multiply_temp_variable = { pad_pct = 0.1 } 
		set_temp_variable = { high_pct = true_max_pct }
		add_to_temp_variable = { high_pct = pad_pct }
		set_temp_variable = { low_pct = true_min_pct }
		subtract_from_temp_variable = { low_pct = pad_pct }
	}

	if = { limit = { check_variable = { low_pct < 0 } } set_temp_variable = { low_pct = 0 } }
	set_temp_variable = { diff_pct = high_pct }
	subtract_from_temp_variable = { diff_pct = low_pct }
	if = { limit = { check_variable = { diff_pct < 0.01 } } set_temp_variable = { diff_pct = 0.01 } }

	# DYNAMIC ANTI-TEAR ZOOM 
	set_temp_variable = { safe_diff_pct = max_jump_pct }
	multiply_temp_variable = { safe_diff_pct = 45 }
	divide_temp_variable = { safe_diff_pct = 10 }

	if = {
		limit = { check_variable = { diff_pct < safe_diff_pct } }
		set_temp_variable = { shortfall_pct = safe_diff_pct }
		subtract_from_temp_variable = { shortfall_pct = diff_pct }
		divide_temp_variable = { shortfall_pct = 2 }

		add_to_temp_variable = { high_pct = shortfall_pct }
		subtract_from_temp_variable = { low_pct = shortfall_pct }
		if = { limit = { check_variable = { low_pct < 0 } } set_temp_variable = { low_pct = 0 } }
		set_temp_variable = { diff_pct = high_pct }
		subtract_from_temp_variable = { diff_pct = low_pct }
	}

	# RIGHT Y-AXIS LABELS (% of GDP)
	set_temp_variable = { step_pct = diff_pct }
	multiply_temp_variable = { step_pct = 0.1 } 
	set_temp_variable = { buf_pct = low_pct }
	for_loop_effect = {
		start = 0 end = 11
		set_variable = { md_defense_r_y_axis_label^v = buf_pct }
		add_to_temp_variable = { buf_pct = step_pct }
	}

	# 2. DRAW DEFENSE % OF GDP LINE (Green on up, Red on down) — Dynamic Right Axis
	for_each_loop = {
		array = defense_pct_array
		value = v
		index = i

		if = {
			limit = { check_variable = { i > current_gdp_graph_end } }
			add_to_array = { md_flax_offsets_defense_pct_green = 1000 } 
			add_to_array = { md_flax_offsets_defense_pct_red = 1000 } 
			add_to_array = { md_flax_slopes_defense_pct = 50 }
		}
		else = {
			set_temp_variable = { y1_val = v }
			subtract_from_temp_variable = { y1_val = low_pct }
			divide_temp_variable = { y1_val = diff_pct } 
			set_temp_variable = { y1_px = y1_val }
			multiply_temp_variable = { y1_px = 450 }

			set_temp_variable = { next_i = i }
			add_to_temp_variable = { next_i = 1 }

			if = {
				limit = { NOT = { check_variable = { next_i > current_gdp_graph_end } } } 
				set_temp_variable = { y2_val = defense_pct_array^next_i }
			}
			else = {
				set_temp_variable = { y2_val = v } 
			}

			subtract_from_temp_variable = { y2_val = low_pct }
			divide_temp_variable = { y2_val = diff_pct } 
			set_temp_variable = { y2_px = y2_val }
			multiply_temp_variable = { y2_px = 450 } 

			set_temp_variable = { offset = y1_px }
			multiply_temp_variable = { offset = -1 }

			if = {
				limit = { check_variable = { y2_val < y1_val } } 
				add_to_array = { md_flax_offsets_defense_pct_red = offset }
				add_to_array = { md_flax_offsets_defense_pct_green = 1000 }
			}
			else = { 
				add_to_array = { md_flax_offsets_defense_pct_green = offset }
				add_to_array = { md_flax_offsets_defense_pct_red = 1000 }
			}

			set_temp_variable = { slope = y2_px }
			subtract_from_temp_variable = { slope = y1_px }
			divide_temp_variable = { slope = 2 }
			add_to_temp_variable = { slope = 50 }

			if = { limit = { check_variable = { slope > 100 } } set_temp_variable = { slope = 100 } }
			if = { limit = { check_variable = { slope < 1 } } set_temp_variable = { slope = 1 } }

			round_variable = slope
			add_to_array = { md_flax_slopes_defense_pct = slope }
		}
	}
}
