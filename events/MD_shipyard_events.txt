add_namespace = md_shipyard

# raising the crisis level, costs stability and loads the yard states.
country_event = {
	id = md_shipyard.1
	title = md_shipyard.1.t
	desc = md_shipyard.1.d
	picture = GFX_UKR_Reorganize_Navy_Finish_Cruiser_Ukrayina

	is_triggered_only = yes

	option = {
		name = md_shipyard.1.a
		log = "[GetDateText]: [Root.GetName]: Event md_shipyard.1.a"
		ai_chance = { factor = 80 }
		custom_effect_tooltip = md_shipyard_overload_free_tt
	}

	option = {
		name = md_shipyard.1.b
		log = "[GetDateText]: [Root.GetName]: Event md_shipyard.1.b"
		ai_chance = { factor = 20 }
		add_stability = -0.01
		add_to_variable = { shipyard_crisis_level = 1 }
		clamp_variable = { var = shipyard_crisis_level min = 0 max = 100 }
		hidden_effect = { shipyard_recalc = yes }
		custom_effect_tooltip = md_shipyard_overload_push_tt
	}
}
