BOL_avoid_unready_war_with_bra = {
	allowed = { original_tag = BOL }
	enable = {
		has_wargoal_against = BRA
		NOT = { has_war_with = BRA }
		NOT = { strength_ratio = { tag = BRA ratio > 1.2 } }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = conquer id = BRA value = -1000 }
	ai_strategy = { type = declare_war id = BRA value = -1000 }
}
