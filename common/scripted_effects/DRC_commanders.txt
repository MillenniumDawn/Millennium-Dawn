MDRC_LDK_appoint_nkunda = {
	if = {
		limit = {
			NOT = { has_country_flag = MDRC_LDK_laurent_nkunda_appointed }
		}
		for_each_scope_loop = {
			array = global.countries
			if = {
				limit = { has_character = RCD_laurent_nkunda }
				retire_character = RCD_laurent_nkunda
			}
		}
		set_country_flag = MDRC_LDK_laurent_nkunda_appointed
		custom_effect_tooltip = { localization_key = DRC_commander_available_tt COMMANDER = MDRC_LDK_laurent_nkunda_name }
	}
}
