	OR = {
		has_global_flag = SIN_OPPOSITION_VICTORY_FOCUS_PATH

	}

SIN_ai_historical_path = {
	OR = {
		has_global_flag = SIN_HISTORICAL_FOCUS_PATH
		AND = {
			is_historical_focus_on = yes
			NOT = { SIN_ai_alt_path = yes }
		}
	}
}

# Master of the Straits steers the balance of power to Singapore First, the other paths to World Dominance, and No Path holds the middle
SIN_ai_avoid_world_dominance_bop = {
	OR = {
		has_global_flag = SIN_MASTER_OF_THE_STRAITS_FOCUS_PATH
		AND = {
			NOT = { SIN_ai_historical_path = yes }
			NOT = { SIN_ai_alt_path = yes }
			power_balance_value = { id = SIN_power_balance value < -0.1 }
		}
	}
}

SIN_ai_avoid_singapore_first_bop = {
	OR = {
		SIN_ai_not_straits_path = yes
		AND = {
			NOT = { SIN_ai_historical_path = yes }
			NOT = { SIN_ai_alt_path = yes }
			power_balance_value = { id = SIN_power_balance value > 0.1 }
		}
	}
}
