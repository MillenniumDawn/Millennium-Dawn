# The engine's invasion planner zeroes its reach budget for a navy holding no
# capital_ship-type hull, so one capital is the difference between having an invasion
# arm and none. Default ratios make it the LAST thing a small navy builds, and the
# obvious hull is a trap: destroyer equipment is claimed by BOTH the capital destroyer
# and screen_destroyer sub_units, and a BUILT hull resolves to the later definition --
# every AI-built destroyer deploys as a screen (measured: PAN 22 years, PER live).
# cruiser is the only collision-free capital archetype a builder can reach, on the
# same research gate. role_ratio stacks, so this adds to defaults and self-retires.
MD_first_capital_ship_priority = {
	enable = {
		OR = {
			has_war = yes
			ai_is_threatened = yes
		}
		num_of_naval_factories > 0
		has_no_capital_ships = yes
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = role_ratio id = naval_cruiser value = 60 }
}
