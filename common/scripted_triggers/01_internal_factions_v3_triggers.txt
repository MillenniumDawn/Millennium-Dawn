# Opinion tiers on internal_faction_opinion^internal_faction_id
internal_faction_tier_hostile = {
	check_variable = { internal_faction_opinion^internal_faction_id < 20 }
}
internal_faction_tier_negative = {
	check_variable = { var = internal_faction_opinion^internal_faction_id value = 20 compare = greater_than_or_equals }
	check_variable = { internal_faction_opinion^internal_faction_id < 40 }
}
internal_faction_tier_indifferent = {
	check_variable = { var = internal_faction_opinion^internal_faction_id value = 40 compare = greater_than_or_equals }
	check_variable = { internal_faction_opinion^internal_faction_id < 60 }
}
internal_faction_tier_positive = {
	check_variable = { var = internal_faction_opinion^internal_faction_id value = 60 compare = greater_than_or_equals }
	check_variable = { internal_faction_opinion^internal_faction_id < 80 }
}
internal_faction_tier_enthusiastic = {
	check_variable = { var = internal_faction_opinion^internal_faction_id value = 80 compare = greater_than_or_equals }
}

# Action availability, shared by the GUI buttons and internal_faction_ai_monthly

# Parameters:
# - internal_faction_id: faction id 1-23
internal_faction_can_afford_privilege = {
	OR = {
		AND = {
			check_variable = { global.internal_faction_category^internal_faction_id = 1 }
			set_temp_variable = { internal_faction_cost = gdp_total }
			multiply_temp_variable = { internal_faction_cost = global.internal_faction_cost_grant_gdp_share }
			check_variable = { treasury > internal_faction_cost }
		}
		AND = {
			NOT = { check_variable = { global.internal_faction_category^internal_faction_id = 1 } }
			set_temp_variable = { internal_faction_cost = global.internal_faction_cost_grant_pp }
			has_political_power > internal_faction_cost
		}
	}
}

# Parameters:
# - internal_faction_p: privilege id (faction id - 1) * 4 + k, k 1-3
internal_faction_can_grant_privilege = {
	set_temp_variable = { internal_faction_id = global.internal_faction_policy_faction^internal_faction_p }
	set_temp_variable = { internal_faction_cp = internal_faction_id }
	multiply_temp_variable = { internal_faction_cp = 4 }
	check_variable = { global.internal_faction_policy_kind^internal_faction_p = 1 }
	is_in_array = { array = internal_faction_active value = internal_faction_id }
	NOT = { is_in_array = { array = internal_faction_policies value = internal_faction_p } }
	NOT = { is_in_array = { array = internal_faction_policies value = internal_faction_cp } }
	check_variable = { internal_faction_policy_cooldown^internal_faction_id = 0 }
	internal_faction_can_afford_privilege = yes
}

# Parameters:
# - internal_faction_p: privilege id (faction id - 1) * 4 + k, k 1-3
internal_faction_can_revoke_privilege = {
	set_temp_variable = { internal_faction_id = global.internal_faction_policy_faction^internal_faction_p }
	set_temp_variable = { internal_faction_cost = global.internal_faction_cost_revoke }
	is_in_array = { array = internal_faction_active value = internal_faction_id }
	is_in_array = { array = internal_faction_policies value = internal_faction_p }
	check_variable = { internal_faction_policy_cooldown^internal_faction_id = 0 }
	has_political_power > internal_faction_cost
}

# Parameters:
# - internal_faction_id: faction id 1-23; the military or the intelligence community must be present and not hostile, cracking down on the military needs a positive intelligence community
internal_faction_crackdown_enforcer_ready = {
	OR = {
		AND = {
			check_variable = { internal_faction_id = 7 }
			is_in_array = { internal_faction_active = 9 }
			check_variable = { var = internal_faction_opinion^9 value = 60 compare = greater_than_or_equals }
		}
		AND = {
			NOT = { check_variable = { internal_faction_id = 7 } }
			OR = {
				AND = {
					is_in_array = { internal_faction_active = 7 }
					check_variable = { var = internal_faction_opinion^7 value = 20 compare = greater_than_or_equals }
				}
				AND = {
					is_in_array = { internal_faction_active = 9 }
					check_variable = { var = internal_faction_opinion^9 value = 20 compare = greater_than_or_equals }
				}
			}
		}
	}
}
