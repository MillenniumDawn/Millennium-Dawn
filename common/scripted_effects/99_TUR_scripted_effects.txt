# Runs monthly while Ecevit leads: a decline that speeds up after 2002 and 2004 so resting
# cannot keep him in office forever, one or two health episodes until he is bedridden or
# the party splits, and at most one escalation event per month.
TUR_ecevit_monthly = {
	if = {
		limit = { date < 2002.1.1 }
		subtract_from_variable = { TUR_ecevit_health = 0.75 }
	}
	else_if = {
		limit = { date < 2004.1.1 }
		subtract_from_variable = { TUR_ecevit_health = 1 }
	}
	else = {
		subtract_from_variable = { TUR_ecevit_health = 1.5 }
	}
	TUR_ecevit_health_update = yes
	if = {
		limit = { check_variable = { TUR_ecevit_health < 50 } }
		add_to_variable = { TUR_dsp_plot = 0.5 }
	}
	if = {
		limit = { has_country_flag = TUR_ecevit_cover_up }
		add_to_variable = { TUR_dsp_plot = 0.5 }
	}
	clamp_variable = { var = TUR_dsp_plot min = 0 max = 100 }

	if = {
		limit = {
			check_variable = { TUR_ecevit_health > 14 }
			NOT = { has_country_flag = TUR_dsp_walkout }
		}
		TUR_ecevit_random_episode = yes
		random = {
			chance = 40
			TUR_ecevit_random_episode = yes
		}
	}

	if = {
		limit = {
			check_variable = { TUR_ecevit_health < 1 }
			NOT = { has_country_flag = TUR_ecevit_incapacitated }
		}
		set_country_flag = TUR_ecevit_incapacitated
		country_event = turkish_events.50
	}
	else_if = {
		limit = {
			check_variable = { TUR_dsp_plot > 84 }
			NOT = { has_country_flag = TUR_dsp_walkout }
		}
		set_country_flag = TUR_dsp_walkout
		country_event = turkish_events.49
	}
	else_if = {
		limit = {
			check_variable = { TUR_dsp_plot > 69 }
			NOT = { has_country_flag = Turkey_Ozkan_purged }
		}
		hidden_effect = { set_country_flag = Turkey_Ozkan_purged }
		country_event = turkish_events.52
	}
	else_if = {
		limit = {
			check_variable = { TUR_dsp_plot > 39 }
			NOT = { has_country_flag = TUR_dsp_congress_letter }
		}
		set_country_flag = TUR_dsp_congress_letter
		country_event = turkish_events.51
	}
	else_if = {
		limit = {
			check_variable = { TUR_ecevit_health < 35 }
			NOT = { has_country_flag = TUR_ecevit_hospitalized }
		}
		set_country_flag = TUR_ecevit_hospitalized
		country_event = turkish_events.48
	}
}

# Penalties start below 60 health and reach -36% Political Power and -18% Stability at 0.
TUR_ecevit_health_update = {
	clamp_variable = { var = TUR_ecevit_health min = 0 max = 70 }
	set_variable = { TUR_ecevit_political_power_factor = { value = TUR_ecevit_health subtract = 60 min = 0 multiply = 0.006 } }
	set_variable = { TUR_ecevit_stability_factor = { value = TUR_ecevit_health subtract = 60 min = 0 multiply = 0.003 } }
}

TUR_update_military_standing = {
	set_variable = { TUR_military_standing_value = 0 }

	#Social conservatism - government counts as double negative
	set_variable = { TUR_military_standing_conservatism = ROOT.social_conservatism_government }
	multiply_variable = { TUR_military_standing_conservatism = -2 }
	add_to_variable = { TUR_military_standing_value = TUR_military_standing_conservatism }

	#Military internal faction opinion counts as positive
	set_variable = { TUR_military_standing_opinion = ROOT.the_military_opinion }
	add_to_variable = { TUR_military_standing_value = TUR_military_standing_opinion }

	#Corruption above level 5 is -20 points per level
	set_variable = { TUR_military_standing_corruption = { value = corruption_level subtract = 5 max = 0 multiply = -20 } }
	add_to_variable = { TUR_military_standing_value = TUR_military_standing_corruption }

	#Bad economic cycle is -25 points per level below stable growth
	set_variable = { TUR_military_standing_economy = 0 }
	if = {
		limit = { has_idea = stagnation }
		add_to_variable = { TUR_military_standing_economy = -25 }
	}
	else_if = {
		limit = { has_idea = recession }
		add_to_variable = { TUR_military_standing_economy = -50 }
	}
	else_if = {
		limit = { has_idea = depression }
		add_to_variable = { TUR_military_standing_economy = -75 }
	}
	add_to_variable = { TUR_military_standing_value = TUR_military_standing_economy }

	#Interest rate multiplied by -5
	set_variable = { TUR_military_standing_interest_rate = ROOT.interest_rate }
	round_variable = TUR_military_standing_interest_rate
	multiply_variable = { TUR_military_standing_interest_rate = -5 }
	add_to_variable = { TUR_military_standing_value = TUR_military_standing_interest_rate }

	#Different parties have different base opinions
	set_variable = { TUR_military_standing_party = 0 }
	if = {
		limit = {
			OR = {
				check_variable = { ruling_party = 10 }
				check_variable = { ruling_party = 11 }
				check_variable = { ruling_party = 19 }
			}
		}
		set_variable = { TUR_military_standing_party = -250 }
	}
	else_if = {
		limit = {
			OR = {
				AND = {
					NOT = { has_completed_focus = TUR_beacon_of_islam }
					western_autocrats_are_in_power = yes
				}
				check_variable = { ruling_party = 5 }
				check_variable = { ruling_party = 6 }
				check_variable = { ruling_party = 7 }
				check_variable = { ruling_party = 8 }
				check_variable = { ruling_party = 9 }
				check_variable = { ruling_party = 12 }
				AND = {
					has_completed_focus = TUR_beacon_of_islam
					check_variable = { ruling_party = 13 }
				}
				check_variable = { ruling_party = 14 }
				check_variable = { ruling_party = 17 }
				check_variable = { ruling_party = 20 }

			}
		}
		set_variable = { TUR_military_standing_party = -50 }

	else_if = {

# Effect: TUR_pkk_military_damage
# Purpose: Lowers PKK Strength by force, stopping at 6 per surviving PKK base, plus 10 while Rojava shelters it,
#          unless Strength is already lower. Pressure that would push Strength below that floor destroys the next base, Qandil last.
# Parameters:
# - TUR_pkk_damage: Strength points to remove
TUR_pkk_military_damage = {
	set_temp_variable = { TUR_pkk_floor = { value = TUR_PKK_base_states^num multiply = 6 } }
	if = {
		limit = {
			country_exists = ROJ
			NOT = { ROJ = { is_subject_of = ROOT } }
		}
		add_to_temp_variable = { TUR_pkk_floor = 10 }
	}
	set_temp_variable = { TUR_pkk_floor = { value = TUR_pkk_floor min = TUR_PKK_strength } }

# Effect: TUR_pkk_peace_step
# Purpose: Takes the next rung of the Solution Process and restores the ceasefire. The sixth rung dissolves the PKK.
# Parameters:
# - TUR_pkk_peace_stage (variable): rungs already taken, 0-5
TUR_pkk_peace_step = {
	custom_effect_tooltip = TUR_pkk_peace_step_tt
	TUR_pkk_start_ceasefire = yes
	if = {
		limit = { check_variable = { TUR_pkk_peace_stage = 0 } }
		add_political_power = -25
	}
	else_if = {
		limit = { check_variable = { TUR_pkk_peace_stage = 1 } }
		add_stability = -0.02
		set_temp_variable = { TUR_grievance_change = -10 }
		TUR_pkk_change_grievance = yes
	}
	else_if = {
		limit = { check_variable = { TUR_pkk_peace_stage = 2 } }
		add_political_power = -50
		set_temp_variable = { TUR_grievance_change = -5 }
		TUR_pkk_change_grievance = yes
	}
	else_if = {
		limit = { check_variable = { TUR_pkk_peace_stage = 3 } }
		set_temp_variable = { TUR_pkk_strength_change = -10 }
		TUR_pkk_change_strength = yes
	}
	else_if = {
		limit = { check_variable = { TUR_pkk_peace_stage = 4 } }
		if = {
			limit = { has_idea = TUR_village_guards }
			custom_effect_tooltip = TUR_pkk_disband_guards_tt
			remove_ideas = TUR_village_guards
			medium_expenditure = yes
			add_stability = -0.05
		}
		add_stability = -0.03
		set_temp_variable = { TUR_grievance_change = -10 }
		TUR_pkk_change_grievance = yes
	}
	else = {
		custom_effect_tooltip = TUR_pkk_dissolved_tt
		set_country_flag = TUR_PKK_reconciliation
		set_country_flag = TUR_PKK_gone
		set_variable = { TUR_PKK_strength = 0 }
		add_stability = 0.10
		add_ideas = TUR_peace_in_the_southeast
		hidden_effect = { TUR_pkk_clear_southeast = yes }
	}
	add_to_variable = { TUR_pkk_peace_stage = 1 }
}
