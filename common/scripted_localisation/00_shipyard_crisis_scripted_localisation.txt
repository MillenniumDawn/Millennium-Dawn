# Panel title per country. A new country needs one line here and its own TAG_SHIPYARD_TITLE key.
defined_text = {
	name = shipyard_panel_title
	text = { trigger = { original_tag = USA }
		localization_key = USA_SHIPYARD_TITLE
	}
	text = { trigger = { original_tag = UKR }
		localization_key = UKR_SHIPYARD_TITLE
	}
	text = { trigger = { original_tag = SOV }
		localization_key = SOV_SHIPYARD_TITLE
	}
	text = {
		localization_key = SHIPYARD_CRISIS_TITLE
	}
}

# Live readout for the production tab. shipyard_recalc only refreshes its variables on
# the monthly tick, so the dockyards in use are recomputed here at render time.
defined_text = {
	name = shipyard_readout
	text = {
		trigger = {
			set_temp_variable = {
				shipyard_readout_in_use = {
					value = num_of_naval_factories
					subtract = num_of_available_naval_factories
				}
			}
			check_variable = { shipyard_readout_in_use > shipyard_effective_dockyards }
		}
		localization_key = SHIPYARD_READOUT_OVER
	}
	text = {
		trigger = {
			set_temp_variable = {
				shipyard_readout_in_use = {
					value = num_of_naval_factories
					subtract = num_of_available_naval_factories
				}
			}
		}
		localization_key = SHIPYARD_READOUT_OK
	}
}
