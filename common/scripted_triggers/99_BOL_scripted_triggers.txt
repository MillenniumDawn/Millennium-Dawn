BOL_ai_alt_path = {
	OR = {
		has_global_flag = BOL_MEDIA_LUNA_FOCUS_PATH
		has_global_flag = BOL_REFORMIST_FOCUS_PATH
		has_global_flag = BOL_MILITARY_JUNTA_FOCUS_PATH
		has_global_flag = BOL_WIPHALA_FOCUS_PATH
	}
}

BOL_ai_any_path = {
	OR = {
		has_global_flag = BOL_HISTORICAL_FOCUS_PATH
		BOL_ai_alt_path = yes
	}
}

BOL_ai_historical_path = {
	OR = {
		has_global_flag = BOL_HISTORICAL_FOCUS_PATH
		AND = {
			is_historical_focus_on = yes
			NOT = { BOL_ai_alt_path = yes }
		}
	}
}

BOL_ai_not_historical_path = {
	BOL_ai_alt_path = yes
}
