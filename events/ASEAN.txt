country_event = {
	id = asean.1
	title = asean.1.t
	desc = asean.1.d
	picture = GFX_treaty

	is_triggered_only = yes

	option = {
		name = asean.1.a
		ai_chance = { base = 70 }
		log = "[GetDateText]: [Root.GetName]: Event asean.1 option a (accept ASEAN invitation)"
		add_ideas = ASEAN_Member
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = ASEAN_reforms_modifier } } }
			add_dynamic_modifier = { modifier = ASEAN_reforms_modifier }
		}
		news_event = { id = asean.2 days = 3 }
	}
	option = {
		name = asean.1.b
		ai_chance = { base = 30 }
		log = "[GetDateText]: [Root.GetName]: Event asean.1 option b (decline ASEAN invitation)"
		if = {
			limit = { FROM = { exists = yes } }
			FROM = { country_event = { id = asean.3 days = 3 } }
		}
	}
}
