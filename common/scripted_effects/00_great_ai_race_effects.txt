ai_race_initialize_country = {
	if = {
		limit = { ai_race_eligible_participant = yes }
		if = {
			limit = { NOT = { is_in_array = { global.ai_race_all_initialized = THIS } } }
			add_to_array = { global.ai_race_all_initialized = THIS }
			unlock_decision_category_tooltip = AI_RACE_category
		}
		set_country_flag = AI_RACE_state_initialized
		if = {
			limit = { NOT = { is_in_array = { global.ai_race_participants = THIS } } }
			add_to_array = { global.ai_race_participants = THIS }
		}
		ai_race_repair_country_state = yes
	}
}

					limit = { check_variable = { USA_ai_core_infrastructure_burden > 0 } }

				}
				if = { limit = { check_variable = { energy_difference_variable < 0 } } subtract_from_variable = { ai_race_compute_external = 10 } }

				# Split alignment influence between control and deployment; do not copy it 1:1 into both.
				set_temp_variable = { _usa_align_ctl = { value = USA_ai_core_state_alignment multiply = 7 } } # 70% weight
				set_variable = { ai_race_control_capacity_external = _usa_align_ctl }

					_usa_deploy = {
						value = USA_ai_core_compute_depth multiply = 5
						add = { value = USA_ai_core_public_legitimacy multiply = 4 }
						add = { value = USA_ai_core_state_alignment multiply = 1 } # 10% alignment influence to deployment

				}

ai_race_refresh_participants = {
	clear_array = global.ai_race_participants
	for_each_loop = {
		array = global.ai_race_all_initialized
		value = ai_race_participant
		var:ai_race_participant = {
			clear_variable = ai_race_rank
			if = {
				limit = { ai_race_active = yes }
				add_to_array = { global.ai_race_participants = THIS }
			}
		}
	}
}

ai_race_rebuild_frontier_and_ranking = {
	clear_array = global.ai_race_ranked_participants
	clear_variable = global.ai_race_frontier_capability
	clear_variable = global.ai_race_leader_id
	set_temp_variable = { ai_race_sort_for_purchase = 0 }
	ai_race_sort_participants = yes
	set_temp_variable = { ai_race_next_rank = 1 }
	for_each_loop = {
		array = global.ai_race_sorted_work
		value = ai_race_ranked_country
		var:ai_race_ranked_country = {
			add_to_array = { global.ai_race_ranked_participants = THIS }
			set_variable = { ai_race_rank = ai_race_next_rank }
			if = {
				limit = { check_variable = { ai_race_next_rank = 1 } }
				set_variable = { global.ai_race_leader_id = THIS.id }
			}
			if = {
				limit = { check_variable = { ai_race_effective_capability > global.ai_race_frontier_capability } }
				set_variable = { global.ai_race_frontier_capability = ai_race_effective_capability }
			}
			add_to_temp_variable = { ai_race_next_rank = 1 }
		}
	}
	clear_array = global.ai_race_sorted_work
	for_each_loop = {
		array = global.ai_race_participants
		value = ai_race_participant
		var:ai_race_participant = {
			set_variable = { ai_race_frontier_gap = ai_race_effective_capability }
			subtract_from_variable = { ai_race_frontier_gap = global.ai_race_frontier_capability }
			ai_race_repair_country_state = yes
			ai_race_refresh_gui = yes
		}
	}
}

				ai_race_rebuild_country_metrics = yes

ai_race_monthly_dispatch = {
	if = {
		limit = { ai_race_enabled = yes }
		ai_race_initialize_global = yes
		ai_race_discover_participants = yes
		ai_race_ai_discover_candidates = yes
		ai_race_get_month_token = yes
		if = {
			limit = { NOT = { check_variable = { global.ai_race_last_processed_month = ai_race_current_month } } }
			ai_race_refresh_mode_state = yes
			ai_race_refresh_participants = yes
			ai_race_run_monthly_work = yes
			set_variable = { global.ai_race_last_processed_month = ai_race_current_month }
		}
		if = {
			limit = {
				OR = {
					check_variable = { global.month = 1 }
					check_variable = { global.month = 4 }
					check_variable = { global.month = 7 }
					check_variable = { global.month = 10 }
				}
			}
			set_temp_variable = { ai_race_current_quarter = global.year }
			multiply_temp_variable = { ai_race_current_quarter = 4 }
			if = {
				limit = { check_variable = { global.month = 1 } }
				add_to_temp_variable = { ai_race_current_quarter = 1 }
			}
			else_if = {
				limit = { check_variable = { global.month = 4 } }
				add_to_temp_variable = { ai_race_current_quarter = 2 }
			}
			else_if = {
				limit = { check_variable = { global.month = 7 } }
				add_to_temp_variable = { ai_race_current_quarter = 3 }
			}
			else = {
				add_to_temp_variable = { ai_race_current_quarter = 4 }
			}
			if = {
				limit = {
					NOT = { check_variable = { global.ai_race_last_processed_quarter = ai_race_current_quarter } }
				}
				ai_race_quarterly_reconcile = yes
				set_variable = { global.ai_race_last_processed_quarter = ai_race_current_quarter }
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				has_global_flag = GLOBAL_ai_race_initialized
				check_variable = { global.ai_race_all_initialized^num > 0 }
				check_variable = { global.ai_race_ai_countries^num > 0 }
			}
		}
		ai_race_teardown_global = yes
	}
}

ai_race_sort_participants = {
	clear_array = global.ai_race_sorted_work
	clear_array = global.ai_race_sort_input
	for_each_loop = {
		array = global.ai_race_participants
		value = ai_race_sort_country
		var:ai_race_sort_country = {
			if = {
				limit = {
					OR = {
						check_variable = { ai_race_sort_for_purchase = 0 }
						AND = {
							is_ai = yes
							ai_race_can_review = yes
						}
					}
				}
				add_to_array = { global.ai_race_sort_input = THIS }
			}
		}
	}
	set_temp_variable = { ai_race_sort_remaining = global.ai_race_sort_input^num }
	while_loop_effect = {
		limit = { check_variable = { ai_race_sort_remaining > 0 } }
		set_temp_variable = { ai_race_candidate_found = 0 }
		for_each_loop = {
			array = global.ai_race_sort_input
			value = ai_race_sort_country
			var:ai_race_sort_country = {
				if = {
					limit = { NOT = { is_in_array = { global.ai_race_sorted_work = THIS } } }
					set_temp_variable = { ai_race_this_id = THIS.id }
					if = {
						limit = { ai_race_candidate_is_better = yes }
						set_temp_variable = { ai_race_candidate_found = 1 }
						set_temp_variable = { ai_race_candidate_id = THIS.id }
						set_temp_variable = { ai_race_candidate_stage = ai_race_stage }
						set_temp_variable = { ai_race_candidate_months = ai_race_stage_months }
						set_temp_variable = { ai_race_candidate_capability = ai_race_effective_capability }
					}
				}
			}
		}
		if = {
			limit = { check_variable = { ai_race_candidate_found = 1 } }
			var:ai_race_candidate_id = { add_to_array = { global.ai_race_sorted_work = THIS } }
		}
		subtract_from_temp_variable = { ai_race_sort_remaining = 1 }
	}
	clear_array = global.ai_race_sort_input
}
