add_namespace = STALKER

country_event = {
	id = STALKER.120
	title = STALKER.120.t
	desc = STALKER.120.d
	picture = GFX_libya_militia_training
	is_triggered_only = yes

	trigger = {
		STALKER_scenario_enabled = yes
		has_event_target = STALKER_zone_event_target
		event_target:STALKER_zone_event_target = {
			STALKER_is_active_zone_anchor = yes
			is_controlled_by = ROOT
		}
	}

	option = {
		name = STALKER.120.a
		log = "[GetDateText]: [This.GetName]: STALKER.120.a executed"
		custom_effect_tooltip = STALKER.120.a_tt
		hidden_effect = {
			if = {
				limit = {
					event_target:STALKER_zone_event_target = {
						STALKER_is_active_zone_anchor = yes
						is_controlled_by = ROOT
						check_variable = { STALKER_zone_war_type > 0 }
					}
				}
				ROOT = { add_political_power = -25 }
				event_target:STALKER_zone_event_target = {
					if = {
						limit = { check_variable = { STALKER_zone_war_type = 1 } }
						add_to_variable = { STALKER_zone_duty = 10 }
					}
					else_if = {
						limit = { check_variable = { STALKER_zone_war_type = 2 } }
						add_to_variable = { STALKER_zone_military = 10 }
					}
					else_if = {
						limit = { check_variable = { STALKER_zone_war_type = 3 } }
						add_to_variable = { STALKER_zone_ward = 10 }
						clamp_variable = { var = STALKER_zone_ward min = 0 max = 100 }
					}
					else = {
						add_to_variable = { STALKER_zone_noon = 10 }
						clamp_variable = { var = STALKER_zone_noon min = 0 max = 100 }
					}
					STALKER_clamp_zone_society = yes
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = {
				add = 30
				event_target:STALKER_zone_event_target = { check_variable = { STALKER_zone_containment > 49 } }
			}
			modifier = {
				factor = 0
				has_political_power < 25
			}
		}
	}

	option = {
		name = STALKER.120.b
		log = "[GetDateText]: [This.GetName]: STALKER.120.b executed"
		custom_effect_tooltip = STALKER.120.b_tt
		hidden_effect = {
			if = {
				limit = {
					event_target:STALKER_zone_event_target = {
						STALKER_is_active_zone_anchor = yes
						is_controlled_by = ROOT
						check_variable = { STALKER_zone_war_type > 0 }
					}
				}
				ROOT = { add_political_power = -25 }
				event_target:STALKER_zone_event_target = {
					if = {
						limit = { check_variable = { STALKER_zone_war_type = 1 } }
						add_to_variable = { STALKER_zone_freedom = 10 }
					}
					else_if = {
						limit = { check_variable = { STALKER_zone_war_type = 2 } }
						add_to_variable = { STALKER_zone_bandits = 10 }
					}
					else_if = {
						limit = { check_variable = { STALKER_zone_war_type = 3 } }
						add_to_variable = { STALKER_zone_spark = 10 }
						clamp_variable = { var = STALKER_zone_spark min = 0 max = 100 }
					}
					else = {
						add_to_variable = { STALKER_zone_monolith = 10 }
						clamp_variable = { var = STALKER_zone_monolith min = 0 max = 100 }
					}
					STALKER_clamp_zone_society = yes
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = {
				add = 30
				event_target:STALKER_zone_event_target = { check_variable = { STALKER_zone_exploitation > 29 } }
			}
			modifier = {
				factor = 0
				has_political_power < 25
			}
		}
	}

	option = {
		name = STALKER.120.c
		log = "[GetDateText]: [This.GetName]: STALKER.120.c executed"
		custom_effect_tooltip = STALKER.120.c_tt
		ai_chance = { base = 30 }
	}
}
