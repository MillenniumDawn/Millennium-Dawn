		parent_window_name = "MapmodesInterface_Ingame"

		effects = {
			md_flax_open_button_click = {
				if = {
					limit = { 
						NOT = { has_country_flag = md_show_flax_graph }

					# Defaults to opening the main GDP tab
					set_country_flag = md_show_flax_graph
					update_md_flaxbeard_gdp_graph = yes 
				}

					# If ANY tab is open, close them all
					clr_country_flag = md_show_flax_graph
					clr_country_flag = md_show_flax_graph_gdpc
					clr_country_flag = md_show_flax_graph_prod

					clr_country_flag = md_show_flax_graph_inflation
