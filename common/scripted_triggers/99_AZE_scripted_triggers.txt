AZE_ai_historical_path = {
	OR = {
		is_historical_focus_on = yes
		has_global_flag = AZE_HISTORICAL_FOCUS_PATH
	}
}

AZE_ai_not_historical_path = {
	OR = {
		has_global_flag = AZE_DEMOCRATIC_FOCUS_PATH
		has_global_flag = AZE_JUNTA_FOCUS_PATH
	}
}

AZE_ai_not_democratic_path = {
	NOT = { has_global_flag = AZE_DEMOCRATIC_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = AZE_HISTORICAL_FOCUS_PATH
		has_global_flag = AZE_JUNTA_FOCUS_PATH
	}
}

AZE_ai_not_junta_path = {
	NOT = { has_global_flag = AZE_JUNTA_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = AZE_HISTORICAL_FOCUS_PATH
		has_global_flag = AZE_DEMOCRATIC_FOCUS_PATH
	}
}
