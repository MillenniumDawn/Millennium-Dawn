### No-suicide guards ###

TAJ_no_suicide_afg = {
	allowed = { original_tag = TAJ }
	enable = {
		country_exists = AFG
		OR = {
			has_wargoal_against = AFG
			AFG = { has_wargoal_against = TAJ }
		}
		strength_ratio = {
			tag = AFG
			ratio < 1
		}
