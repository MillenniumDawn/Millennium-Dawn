on_actions = {
	on_monthly_SIA = {
		effect = {
			# UNESCO designates Preah Vihear as World Heritage Site (July 2008)
			# This triggers the border dispute
			if = {
				limit = {
					date > 2008.6.1
					date < 2008.9.1
					NOT = { has_global_flag = preah_vihear_unesco_triggered }
					NOT = { has_global_flag = preah_vihear_resolved }
					country_exists = CBD
					NOT = { has_war_with = CBD }
				}
				set_global_flag = preah_vihear_unesco_triggered
				country_event = preah_vihear.100
				CBD = { country_event = preah_vihear.101 }
			}
		}
	}
}
