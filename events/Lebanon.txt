# 4 August 2020: ammonium nitrate stored for years at the port of Beirut explodes
country_event = {
	id = lebanon.11
	title = lebanon.11.t
	desc = lebanon.11.d
	picture = GFX_bam_earthquake
	is_triggered_only = yes

	trigger = { owns_state = 202 }

	immediate = {
		202 = {
			if = {
				limit = { infrastructure > 0 }
				damage_building = { type = infrastructure damage = 1 }
			}
		}
	}

	# Accept international aid
	option = {
		name = lebanon.11.a
		log = "[GetDateText]: [This.GetName]: lebanon.11.a executed"
		trigger = { country_exists = FRA }
		add_stability = -0.03
		set_temp_variable = { treasury_change = 1 }
		modify_treasury_effect = yes
		set_temp_variable = { percent_change = 5 }
		set_temp_variable = { tag_index = FRA.id }
		change_influence_percentage = yes
		ai_chance = { base = 10 }
	}

	# Rebuild on our own
	option = {
		name = lebanon.11.b
		log = "[GetDateText]: [This.GetName]: lebanon.11.b executed"
		add_stability = -0.05
		set_temp_variable = { treasury_change = -1 }
		modify_treasury_effect = yes
		ai_chance = {
			base = 5
			modifier = { factor = 0.5 ai_has_high_deficit = yes }
			modifier = { factor = 0 has_active_mission = bankruptcy_incoming_collapse }
		}
	}
}
