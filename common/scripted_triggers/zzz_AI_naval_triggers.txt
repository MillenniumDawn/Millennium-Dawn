MD_can_reach_for_invasion = {
	OR = {
		is_neighbor_of = PREV
		AND = {
			controls_a_coastal_state = yes
			PREV = { controls_a_coastal_state = yes }
			OR = {
				AND = {
					is_on_same_continent_as = PREV
					PREV = { has_navy_size = { size > 12 } }
				}
				PREV = {
					has_navy_size = { size > 25 }
					OR = {
						has_idea = intervention_global_interventionism
						has_idea = intervention_neo_imperialism
					}
				}
			}
		}
	}
}
