# How the rest of the world reacts to the Zones: diplomacy before Kaimanov, casualties among foreign teams.
STALKER_monthly_world_pulse = {
	STALKER_refresh_unsc_expected = yes
	STALKER_finish_security_council = yes
	if = {
		limit = {
			NOT = { has_global_flag = STALKER_kaimanov_outcome }
			NOT = { has_global_flag = STALKER_ipsf_formed }
			NOT = { has_global_flag = STALKER_great_power_talks_cooldown }
			any_of_scopes = {
				array = global.STALKER_active_zone_anchors
				check_variable = { STALKER_zone_id = 1 }
				check_variable = { STALKER_zone_containment < 40 }
			}
		}
		random = {
			chance = 10
			set_global_flag = { flag = STALKER_great_power_talks_cooldown value = 1 days = 730 }
			random_scope_in_array = {
				array = global.STALKER_active_zone_anchors
				limit = {
					check_variable = { STALKER_zone_id = 1 }
					controller = { exists = yes }
				}
				controller = { save_global_event_target_as = STALKER_zone_host_target set_variable = { global.STALKER_zone_host_target = THIS } }
			}
			if = {
				limit = { country_exists = USA }
				USA = { country_event = STALKER.132 }
			}
			if = {
				limit = { country_exists = SOV }
				SOV = { country_event = STALKER.133 }
			}
		}
	}
	if = {
		limit = { NOT = { has_global_flag = STALKER_expedition_casualty_cooldown } }
		random = {
			chance = 3
			random_country = {
				limit = {
					has_country_flag = STALKER_zone_research_access
					NOT = { STALKER_controls_active_zone = yes }
				}
				set_global_flag = { flag = STALKER_expedition_casualty_cooldown value = 1 days = 365 }
				save_global_event_target_as = STALKER_casualty_country
				set_variable = { global.STALKER_casualty_country = THIS }
				random_scope_in_array = {
					array = global.STALKER_active_zone_anchors
					limit = { controller = { exists = yes } }
					controller = { save_event_target_as = STALKER_zone_controller_target }
				}
				country_event = STALKER.134
				news_event = STALKER.135
			}
		}
	}
}

# A catastrophic emission (containment already below 20) makes world news, at most once a year.
STALKER_report_major_emission = {
	if = {
		limit = {
			check_variable = { STALKER_zone_containment < 20 }
			NOT = { has_global_flag = STALKER_major_emission_news_cooldown }
			controller = { exists = yes }
		}
		set_global_flag = { flag = STALKER_major_emission_news_cooldown value = 1 days = 365 }
		STALKER_target_zone = yes
		controller = { news_event = STALKER.130 }
	}
}

# One scripted Council session after the worldwide Kaymanov news. Not a UN resolution type.
# Ballots are sent to each permanent member, and to the original Zone's controller when that country is not already seated.
STALKER_schedule_security_council = {
	if = {
		limit = {
			STALKER_scenario_enabled = yes
			NOT = { has_global_flag = STALKER_unsc_scheduled }
		}
		set_global_flag = STALKER_unsc_scheduled
		set_variable = { global.STALKER_unsc_ballots = 0 }
		set_variable = { global.STALKER_unsc_containment = 0 }
		set_variable = { global.STALKER_unsc_access = 0 }
		set_variable = { global.STALKER_unsc_adjourn = 0 }
		set_variable = { global.STALKER_unsc_expected = 0 }
		set_variable = { global.STALKER_unsc_result = 0 }
		for_each_scope_loop = {
			array = global.p5_sc_members
			if = {
				limit = {
					exists = yes
					NOT = { has_country_flag = STALKER_unsc_seat }
				}
				set_country_flag = STALKER_unsc_seat
				country_event = { id = STALKER.137 days = 50 }
				add_to_variable = { global.STALKER_unsc_expected = 1 }
			}
		}
		698 = {
			if = {
				limit = {
					controller = {
						exists = yes
						NOT = { has_country_flag = STALKER_unsc_seat }
					}
				}
				controller = {
					set_country_flag = STALKER_unsc_seat
					country_event = { id = STALKER.137 days = 50 }
				}
				add_to_variable = { global.STALKER_unsc_expected = 1 }
			}
		}
	}
}

STALKER_refresh_unsc_expected = {
	if = {
		limit = {
			has_global_flag = STALKER_unsc_scheduled
			NOT = { has_global_flag = STALKER_unsc_resolved }
		}
		set_variable = { global.STALKER_unsc_expected = global.STALKER_unsc_ballots }
		every_country = {
			limit = {
				has_country_flag = STALKER_unsc_seat
				NOT = { has_country_flag = STALKER_unsc_voted }
			}
			add_to_variable = { global.STALKER_unsc_expected = 1 }
		}
	}
}

STALKER_finish_security_council = {
	STALKER_refresh_unsc_expected = yes
	if = {
		limit = {
			has_global_flag = STALKER_unsc_scheduled
			NOT = { has_global_flag = STALKER_unsc_resolved }
			check_variable = { global.STALKER_unsc_ballots = global.STALKER_unsc_expected }
		}
		set_global_flag = STALKER_unsc_resolved
		set_variable = { global.STALKER_unsc_result = 3 }
		if = {
			limit = {
				check_variable = { global.STALKER_unsc_containment > global.STALKER_unsc_access }
				check_variable = { global.STALKER_unsc_containment > global.STALKER_unsc_adjourn }
			}
			set_variable = { global.STALKER_unsc_result = 1 }
		}
		else_if = {
			limit = {
				check_variable = { global.STALKER_unsc_access > global.STALKER_unsc_containment }
				check_variable = { global.STALKER_unsc_access > global.STALKER_unsc_adjourn }
			}
			set_variable = { global.STALKER_unsc_result = 2 }
			every_country = {
				limit = { has_country_flag = STALKER_unsc_voted_access }
				STALKER_grant_zone_research_access = yes
			}
		}
		news_event = STALKER.138
	}
}
