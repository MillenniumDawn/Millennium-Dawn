update_opec_dirty_variable = {
	if = { limit = { check_variable = { global.opec_dirty_update_var < 10000 } }
		add_to_variable = { global.opec_dirty_update_var = 1 }
	}
	else = {
		set_variable = { global.opec_dirty_update_var = 1 }
	}
}

# Stance ids: 1 Deep Cut, 2 Cut, 3 Hold, 4 Raise, 5 Price War
opec_monthly_update = {
	opec_remove_dead_countries = yes

	set_temp_variable = { opec_price_ratio = { value = global.oil_price divide = global.starting_oil_price } }
	for_each_scope_loop = {
		array = global.opec_members
		if = {
			limit = {
				is_ai = yes
				NOT = { check_variable = { opec_vote > 0 } }
			}
			if = { limit = { check_variable = { opec_price_ratio < 0.75 } }
				set_variable = { opec_vote = 1 }
			}
			else_if = { limit = { check_variable = { opec_price_ratio < 0.95 } }
				set_variable = { opec_vote = 2 }
			}
			else_if = { limit = { OR = { check_variable = { opec_price_ratio > 1.6 } has_war = yes } }
				set_variable = { opec_vote = 4 }
			}
			else = {
				set_variable = { opec_vote = 3 }
			}
		}
	}
	opec_count_votes = yes

	set_temp_variable = { opec_cycle_month = global.month }
	modulo_temp_variable = { opec_cycle_month = 6 }
	set_variable = { global.opec_months_to_conference = { value = 6 subtract = opec_cycle_month } }
	if = { limit = { check_variable = { opec_cycle_month = 0 } }
		opec_open_conference = yes
	}

	update_opec_dirty_variable = yes
}

opec_remove_dead_countries = {
	for_each_loop = {
		array = global.opec_members
		if = { limit = { var:v = { exists = no } }
			add_to_temp_array = { opec_dead_members = v }
		}
	}
	for_each_loop = {
		array = opec_dead_members
		remove_from_array = { global.opec_members = v }
	}
	for_each_loop = {
		array = global.opec_applicants
		if = { limit = { var:v = { exists = no } }
			add_to_temp_array = { opec_dead_applicants = v }
		}
	}
	for_each_loop = {
		array = opec_dead_applicants
		remove_from_array = { global.opec_applicants = v }
	}
}

# Tallies each stance's backing as a share of total member oil production, then ranks the top three
opec_count_votes = {
	clear_array = global.opec_vote_count
	resize_array = { array = global.opec_vote_count value = 0 size = 6 }
	clear_array = global.opec_vote_share
	resize_array = { array = global.opec_vote_share value = 0 size = 6 }

	set_temp_variable = { opec_total_production = 0 }
	for_each_scope_loop = {
		array = global.opec_members
		add_to_temp_variable = { opec_total_production = resource_produced@oil }
		if = { limit = { check_variable = { opec_vote > 0 } }
			set_temp_variable = { opec_member_vote = opec_vote }
			add_to_variable = { global.opec_vote_count^opec_member_vote = 1 }
			add_to_variable = { global.opec_vote_share^opec_member_vote = resource_produced@oil }
		}
	}
	if = { limit = { check_variable = { opec_total_production > 0 } }
		for_loop_effect = {
			start = 1
			end = 6
			value = stance
			set_variable = { global.opec_vote_share^stance = { value = global.opec_vote_share^stance multiply = 100 divide = opec_total_production } }
		}
	}

	clear_array = global.opec_top_proposals
	for_loop_effect = {
		start = 0
		end = 3
		set_temp_variable = { opec_best_stance = 0 }
		set_temp_variable = { opec_best_share = 0 }
		for_loop_effect = {
			start = 1
			end = 6
			value = stance
			if = {
				limit = {
					check_variable = { global.opec_vote_share^stance > opec_best_share }
					NOT = { is_in_array = { global.opec_top_proposals = stance } }
				}
				set_temp_variable = { opec_best_stance = stance }
				set_temp_variable = { opec_best_share = global.opec_vote_share^stance }
			}
		}
		if = { limit = { check_variable = { opec_best_stance > 0 } }
			add_to_array = { global.opec_top_proposals = opec_best_stance }
		}
	}
}

# Freezes the top three proposals into the ballot and asks every member to vote
opec_open_conference = {
	clear_array = global.opec_ballot
	clear_array = global.opec_ballot_shares
	for_each_loop = {
		array = global.opec_top_proposals
		add_to_array = { global.opec_ballot = v }
		add_to_array = { global.opec_ballot_shares = global.opec_vote_share^v }
	}
	if = { limit = { check_variable = { global.opec_ballot^num = 0 } }
		add_to_array = { global.opec_ballot = global.opec_stance }
		add_to_array = { global.opec_ballot_shares = 0 }
	}

	set_global_flag = opec_conference_in_session
	for_each_scope_loop = {
		array = global.opec_members
		clear_variable = opec_conference_vote
		country_event = { id = opec.1 }
	}
	country_event = { id = opec.2 days = 7 }
}

# Adopts the ballot entry with the most member oil production behind it
opec_resolve_conference = {
	set_temp_variable = { opec_total_production = 0 }
	clear_temp_array = opec_ballot_weight
	resize_temp_array = { array = opec_ballot_weight size = 6 }
	for_each_scope_loop = {
		array = global.opec_members
		add_to_temp_variable = { opec_total_production = resource_produced@oil }
		if = { limit = { check_variable = { opec_conference_vote > 0 } }
			set_temp_variable = { opec_member_vote = opec_conference_vote }
			add_to_temp_variable = { opec_ballot_weight^opec_member_vote = resource_produced@oil }
		}
	}

	set_temp_variable = { opec_winning_stance = global.opec_ballot^0 }
	set_temp_variable = { opec_winning_weight = opec_ballot_weight^opec_winning_stance }
	for_each_loop = {
		array = global.opec_ballot
		if = { limit = { check_variable = { opec_ballot_weight^v > opec_winning_weight } }
			set_temp_variable = { opec_winning_stance = v }
			set_temp_variable = { opec_winning_weight = opec_ballot_weight^v }
		}
	}
	set_variable = { global.opec_stance = opec_winning_stance }
	set_variable = { global.opec_result_share = 0 }
	if = { limit = { check_variable = { opec_total_production > 0 } }
		set_variable = { global.opec_result_share = { value = opec_winning_weight multiply = 100 divide = opec_total_production } }
	}

	opec_review_applications = yes
	opec_apply_stance = yes
	recalculate_resources_price = yes
	if = { limit = { check_variable = { global.month = 12 } check_variable = { global.opec_members^num > 0 } }
		opec_rotate_presidency = yes
	}

	clr_global_flag = opec_conference_in_session
	for_each_scope_loop = {
		array = global.opec_members
		clear_variable = opec_vote
		clear_variable = opec_pending_vote
		clear_variable = opec_conference_vote
		country_event = { id = opec.3 }
	}
	opec_count_votes = yes
	update_opec_dirty_variable = yes
}

opec_apply_stance = {
	if = { limit = { check_variable = { global.opec_stance = 1 } }
		set_variable = { global.opec_price_shift = 0.30 }
		set_variable = { global.opec_member_oil_export_modifier = -0.20 }
	}
	else_if = { limit = { check_variable = { global.opec_stance = 2 } }
		set_variable = { global.opec_price_shift = 0.15 }
		set_variable = { global.opec_member_oil_export_modifier = -0.10 }
	}
	else_if = { limit = { check_variable = { global.opec_stance = 4 } }
		set_variable = { global.opec_price_shift = -0.10 }
		set_variable = { global.opec_member_oil_export_modifier = 0.05 }
	}
	else_if = { limit = { check_variable = { global.opec_stance = 5 } }
		set_variable = { global.opec_price_shift = -0.30 }
		set_variable = { global.opec_member_oil_export_modifier = 0.15 }
	}
	else = {
		set_variable = { global.opec_price_shift = 0 }
		set_variable = { global.opec_member_oil_export_modifier = 0 }
	}

	for_each_scope_loop = {
		array = global.opec_members
		if = { limit = { NOT = { has_dynamic_modifier = { modifier = opec_member_modifier } } }
			add_dynamic_modifier = { modifier = opec_member_modifier }
		}
		force_update_dynamic_modifier = yes
	}
}

# Members approve unless at war with the applicant or holding a poor opinion of it. Admission needs 75% of member oil production.
opec_review_applications = {
	for_each_scope_loop = {
		array = global.opec_applicants
		set_temp_variable = { opec_total_production = 0 }
		set_temp_variable = { opec_approving_production = 0 }
		for_each_scope_loop = {
			array = global.opec_members
			add_to_temp_variable = { opec_total_production = resource_produced@oil }
			if = {
				limit = {
					NOT = { has_war_with = PREV }
					has_opinion = { target = PREV value > -25 }
				}
				add_to_temp_variable = { opec_approving_production = resource_produced@oil }
			}
		}
		set_temp_variable = { opec_required_production = { value = opec_total_production multiply = 0.75 } }
		if = {
			limit = {
				check_variable = { var = opec_approving_production value = opec_required_production compare = greater_than_or_equals }
			}
			add_to_temp_array = { opec_admitted = THIS }
		}
	}
	clear_array = global.opec_applicants
	for_each_loop = {
		array = opec_admitted
		add_to_array = { global.opec_members = v }
	}
}

opec_rotate_presidency = {
	set_temp_variable = { opec_next_president_index = 0 }
	for_each_loop = {
		array = global.opec_members
		if = { limit = { check_variable = { v = global.opec_conference_president } }
			set_temp_variable = { opec_next_president_index = { value = i add = 1 } }
			set_temp_variable = { break = 1 }
		}
	}
	set_temp_variable = { opec_member_count = global.opec_members^num }
	modulo_temp_variable = { opec_next_president_index = opec_member_count }
	set_variable = { global.opec_conference_president = global.opec_members^opec_next_president_index }
}

# Country scope: the leaving member
opec_leave = {
	set_temp_variable = { opec_leaver = THIS.id }
	if = {
		limit = {
			check_variable = { global.opec_conference_president = opec_leaver }
			check_variable = { global.opec_members^num > 1 }
		}
		opec_rotate_presidency = yes
	}
	remove_from_array = { global.opec_members = THIS }
	clear_variable = opec_vote
	clear_variable = opec_pending_vote
	clear_variable = opec_conference_vote
	if = { limit = { has_dynamic_modifier = { modifier = opec_member_modifier } }
		remove_dynamic_modifier = { modifier = opec_member_modifier }
	}
	set_country_flag = { flag = opec_left_recently days = 365 value = 1 }
	opec_count_votes = yes
}
