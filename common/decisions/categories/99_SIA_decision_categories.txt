preah_vihear_decisions_category = {
	allowed = {
		OR = {
			original_tag = SIA
			original_tag = CBD
		}
	}

	icon = GFX_decision_category_generic_foreign_policy
	picture = GFX_decision_cat_generic_asia
	priority = 125

	visible = {
		country_exists = SIA
		country_exists = CBD
		date > 2008.7.1
		NOT = { SIA = { is_subject_of = CBD } }
		NOT = { CBD = { is_subject_of = SIA } }
		NOT = { SIA = { is_in_faction_with = CBD } }
		NOT = {
			AND = {
				SIA = { is_subject = yes }
				CBD = { is_subject = yes }
				SIA = { OVERLORD = { has_subject = CBD } }
			}
		}
		OR = {
			NOT = { has_global_flag = GLOBAL_preah_vihear_resolved }
			SIA = { has_country_flag = SIA_border_war_active }
		}
	}
}
