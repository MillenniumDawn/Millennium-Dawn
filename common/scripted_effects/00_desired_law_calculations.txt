# calculate_expected_spending zeroes and rebuilds the expected_* vars, so it is their only
# writer and folds the offsets below in itself. Nothing here may write them.
recalculate_law_desires = {
	# clamp_variable errors on a variable that does not exist yet
	if = { limit = { NOT = { has_variable = bureau_desire_offset } }
		set_variable = { bureau_desire_offset = 0 }
	}
	if = { limit = { NOT = { has_variable = education_desire_offset } }
		set_variable = { education_desire_offset = 0 }
	}
	if = { limit = { NOT = { has_variable = health_desire_offset } }
		set_variable = { health_desire_offset = 0 }
	}
	if = { limit = { NOT = { has_variable = police_desire_offset } }
		set_variable = { police_desire_offset = 0 }
	}
	if = { limit = { NOT = { has_variable = social_desire_offset } }
		set_variable = { social_desire_offset = 0 }
	}
	focus_on_laws_due_to_economic_despair = yes
	# will increase next month, just in case
	increase_spending_if_possible = yes
	ai_reduce_spending_on_deficit = yes
}

focus_on_laws_due_to_economic_despair = {
	if = {
		limit = {
			NOT = { has_country_flag = AI_economic_downspiral }
			OR = {
				has_idea = depression
				has_idea = recession
				ai_has_major_economic_problems = yes
				ai_has_moderate_economic_problems = yes
			}
		}
		set_country_flag = AI_economic_downspiral
		clear_law_desire_offsets = yes
		if = { limit = { is_debug = yes }
			log = "[GetDateText]: [ROOT.GetName]: DEBUG: AI Scripted Trigger: focus_on_laws_due_to_economic_despair fired"
		}
	}
	else_if = {
		limit = {
			has_country_flag = AI_economic_downspiral
			OR = {
				ai_has_acceptable_deficit = yes
				NOT = {
					OR = {
						has_idea = depression
						has_idea = recession
					}
				}
			}
		}
		if = { limit = { is_debug = yes }
			log = "[GetDateText]: [ROOT.GetName]: DEBUG: AI Scripted Trigger: focus_on_laws_due_to_economic_despair removed"
		}
		clr_country_flag = AI_economic_downspiral
	}

	if = {
		limit = {
			check_variable = { corporate_tax_rate > 30 }
		}

		clear_law_desire_offsets = yes
	}
}

# Defence is deliberately left alone: it is driven by security pressure, not by what the
# treasury can spare.
clear_law_desire_offsets = {
	clamp_variable = { var = bureau_desire_offset max = 0 } # deficit cuts survive a crisis
	clamp_variable = { var = education_desire_offset max = 0 }
	clamp_variable = { var = health_desire_offset max = 0 }
	clamp_variable = { var = police_desire_offset max = 0 }
	clamp_variable = { var = social_desire_offset max = 0 }
}

increase_spending_if_possible = {
	if = {
		# if we are good on taxes and not increased spending already
		limit = {
			check_variable = { budget_balance_ratio > 0.03 }
			check_variable = { treasury_rate_with_interest > 0 } # deficit room is no reason to raise expectations
			check_variable = { corporate_tax_rate < 20 }
			check_variable = { population_tax_rate < 25 }
			has_bureau_idea_law_desired = yes
			has_education_law_desired = yes
			has_health_law_desired = yes
			has_police_idea_law_desired = yes
			has_social_law_desired = yes
		}

		set_temp_variable = { desire_change = 1 }
		# A law trimmed for a deficit in the last 180 days is not bumped straight back
		random_list = {
			10 = {
				modifier = { factor = 0 has_country_flag = ai_deficit_trim_bureau }
				change_expected_bureaucracy_spending = yes
			}
			10 = {
				change_expected_education_spending = yes
			}
			10 = {
				modifier = { factor = 0 has_country_flag = ai_deficit_trim_health }
				change_expected_health_spending = yes
			}
			10 = {
				modifier = { factor = 0 has_country_flag = ai_deficit_trim_police }
				# Democracies do not drift toward a police state
				modifier = { factor = 0 has_government = democratic western_autocrats_are_in_power = no }
				change_expected_police_spending = yes
			}
			10 = {
				modifier = { factor = 0 has_country_flag = ai_deficit_trim_social }
				change_expected_social_spending = yes
			}
		}
	}
}

change_expected_bureaucracy_spending = {
	add_to_variable = { bureau_desire_offset = desire_change }
	clamp_variable = {
		var = bureau_desire_offset
		min = -1
		max = 4
	}
}

# Moves each AI spending law one level a month toward its expected level, with no political power cost.
# The law ideas carry no AI weight, so this is the only way the AI changes them.
ai_move_spending_laws_to_expected = {
	# Civil laws move only when the unrounded expected level is 0.75 away, so a level near x.5 does not flip them monthly
	set_temp_variable = { social_gap = { value = expected_welfare_raw subtract = social_law } }
	set_temp_variable = { health_gap = { value = expected_health_raw subtract = health_law } }
	set_temp_variable = { edu_gap = { value = expected_edu_raw subtract = education_law } }
	set_temp_variable = { police_gap = { value = expected_police_raw subtract = police_law } }
	set_temp_variable = { bureau_gap = { value = expected_adm_raw subtract = bureau_law } }

	# A law that a focus, event or protest moved keeps that level for 90 days before a cut
	# Cuts: every law well above its expected level. Education needs a 1.25-level overshoot, so drift never reverses it.
	if = {
		limit = { budget_law_parliament_change_allowed = yes }
		if = {
			limit = {
				check_variable = { social_gap < -0.75 }
				social_decrease_blocked = no
				NOT = { has_country_flag = social_law_changed_outside_ai_pass }
				NOT = { has_country_flag = ai_deficit_trim_social } # on trim cooldown; an event may have raised it
			}
			decrease_social_spending = yes
		}
		if = {
			limit = {
				check_variable = { health_gap < -0.75 }
				health_decrease_blocked = no
				NOT = { has_country_flag = health_law_changed_outside_ai_pass }
				NOT = { has_country_flag = ai_deficit_trim_health }
				# Nationalized healthcare cannot run on the lowest budgets
				NOT = {
					AND = {
						check_variable = { health_law < 4 }
						OR = {
							has_idea = fully_nationalized_healthcare
							has_idea = partially_nationalized_healthcare
						}
					}
				}
				NOT = {
					AND = {
						check_variable = { health_law = 4 }
						has_idea = fully_nationalized_healthcare
					}
				}
			}
			decrease_healthcare_budget = yes
		}
		if = {
			limit = {
				check_variable = { edu_gap < -1.25 }
				edu_decrease_blocked = no
				NOT = { has_country_flag = edu_law_changed_outside_ai_pass }
				NOT = { has_country_flag = ai_deficit_trim_education }
			}
			decrease_education_budget = yes
		}
		if = {
			limit = {
				check_variable = { police_gap < -0.75 }
				police_decrease_blocked = no
				NOT = { has_country_flag = police_law_changed_outside_ai_pass }
				GER_Grundgesetz_allows_budget_law_change = yes
				NOT = { has_country_flag = ai_deficit_trim_police }
			}
			decrease_policing_budget = yes
		}
		if = {
			limit = {
				check_variable = { bureau_gap < -0.75 }
				bureau_decrease_blocked = no
				NOT = { has_country_flag = bureau_law_changed_outside_ai_pass }
				GER_Grundgesetz_allows_budget_law_change = yes
				NOT = { has_country_flag = ai_deficit_trim_bureau }
			}
			decrease_centralization = yes
		}

		# Raises: one civil law a month, only from a balanced budget
		if = {
			limit = {
				check_variable = { budget_balance_ratio > 0 }
				NOT = { has_active_mission = cheap_loan_from_the_imf_mission }
				NOT = { has_active_mission = bankruptcy_incoming_collapse }
			}
			if = {
				limit = {
					check_variable = { social_gap > 0.75 }
					social_increase_blocked = no
					NOT = { has_country_flag = ai_deficit_trim_social }
					NOT = { check_variable = { policy_transition_social_cost > 0.05 } }
				}
				increase_social_spending = yes
			}
			else_if = {
				limit = {
					check_variable = { health_gap > 0.75 }
					health_increase_blocked = no
					NOT = { has_country_flag = ai_deficit_trim_health }
					NOT = { check_variable = { policy_transition_health_cost > 0.05 } }
					# Privatized healthcare cannot run on the highest budgets
					NOT = {
						AND = {
							check_variable = { health_law > 3 }
							OR = {
								has_idea = fully_privatized_healthcare
								has_idea = partially_privatized_healthcare
							}
						}
					}
					NOT = {
						AND = {
							check_variable = { health_law = 3 }
							has_idea = fully_privatized_healthcare
						}
					}
				}
				increase_healthcare_budget = yes
			}
			else_if = {
				limit = {
					check_variable = { edu_gap > 0.75 }
					edu_increase_blocked = no
					NOT = { has_country_flag = ai_deficit_trim_education }
					NOT = { check_variable = { policy_transition_education_cost > 0.05 } }
				}
				increase_education_budget = yes
			}
			else_if = {
				limit = {
					check_variable = { police_gap > 0.75 }
					police_increase_blocked = no
					NOT = { has_country_flag = ai_deficit_trim_police }
					GER_Grundgesetz_allows_budget_law_change = yes
					NOT = { check_variable = { policy_transition_police_cost > 0.05 } }
					# Democracies stop at the standard police budget
					NOT = {
						AND = {
							check_variable = { police_law > 2 }
							has_government = democratic
							western_autocrats_are_in_power = no
						}
					}
				}
				increase_policing_budget = yes
			}
			else_if = {
				limit = {
					check_variable = { bureau_gap > 0.75 }
					bureau_increase_blocked = no
					NOT = { has_country_flag = ai_deficit_trim_bureau }
					GER_Grundgesetz_allows_budget_law_change = yes
					NOT = { check_variable = { policy_transition_bureau_cost > 0.05 } }
				}
				increase_centralization = yes
			}
		}
	}

	# Defence follows security pressure, in war and peace. Threat checks flip its expected level in whole steps,
	# so after a move it holds 180 days; a war can still raise it.
	if = {
		limit = { military_law_parliament_change_allowed = yes }
		if = {
			limit = {
				check_variable = { military_law > 2 } # the step below 2 needs the no military spirit
				check_variable = { military_law > expected_military_law }
				defence_decrease_blocked = no
				NOT = { has_country_flag = defence_law_changed_outside_ai_pass }
				NOT = { has_country_flag = ai_defence_law_moved }
			}
			decrease_military_spending = yes
			set_country_flag = { flag = ai_defence_law_moved days = 180 value = 1 }
		}
		else_if = {
			limit = {
				check_variable = { military_law < expected_military_law }
				defence_increase_blocked = no
				ai_has_major_economic_problems = no # the emergency cuts below would undo it
				NOT = {
					AND = {
						original_tag = GER
						has_idea = german_legacy
						check_variable = { military_law > 3 }
					}
				}
				OR = {
					has_war = yes
					AND = {
						check_variable = { budget_balance_ratio > 0 }
						NOT = { check_variable = { policy_transition_military_cost > 0.05 } }
						NOT = { has_country_flag = ai_defence_law_moved }
					}
				}
			}
			increase_military_spending = yes
			set_country_flag = { flag = ai_defence_law_moved days = 180 value = 1 }
		}
	}
}

ai_reduce_spending_on_deficit = {
	set_country_flag = ai_spending_law_pass
	if = { limit = { is_ai = yes }
		ai_move_spending_laws_to_expected = yes

		# STEP 2: Emergency austerity during MAJOR crisis (interest >12% + rate < -9)
		# Applies even during wartime — if the economy is collapsing, cuts must happen
		# Cut one civilian category per month (the highest one), military only if civilians are all at minimum
		if = {
			limit = {
				ai_has_major_economic_problems = yes
			}

			# Try civilian categories first — pick the highest non-minimum one
			if = {
				limit = {
					check_variable = { social_law > 1 }
					check_variable = { social_law > police_law }
					check_variable = { social_law > education_law }
				}
				decrease_social_spending = yes
				set_country_flag = { flag = ai_deficit_trim_social days = 180 value = 1 } # the pass would raise it straight back
			}
			else_if = {
				limit = {
					check_variable = { health_law > 1 }
					check_variable = { health_law > police_law }
					check_variable = { health_law > education_law }
				}
				decrease_healthcare_budget = yes
				set_country_flag = { flag = ai_deficit_trim_health days = 180 value = 1 }
			}
			else_if = {
				limit = { check_variable = { police_law > 1 } }
				decrease_policing_budget = yes
				set_country_flag = { flag = ai_deficit_trim_police days = 180 value = 1 }
			}
			else_if = {
				limit = { check_variable = { bureau_law > 1 } }
				decrease_centralization = yes
				set_country_flag = { flag = ai_deficit_trim_bureau days = 180 value = 1 }
			}
			# Education goes last: it is the one law the AI does not reverse
			else_if = {
				limit = { check_variable = { education_law > 1 } }
				decrease_education_budget = yes
				set_country_flag = { flag = ai_deficit_trim_education days = 180 value = 1 }
			}
			# Military LAST: only cut if all civilian categories are at minimum (level 1)
			else_if = {
				limit = {
					check_variable = { military_law > 1 }
					check_variable = { social_law < 2 }
					check_variable = { health_law < 2 }
					check_variable = { education_law < 2 }
					check_variable = { police_law < 2 }
					check_variable = { bureau_law < 2 }
				}
				decrease_military_spending = yes
				set_country_flag = { flag = ai_defence_law_moved days = 180 value = 1 }
			}
		}
	}
	clr_country_flag = ai_spending_law_pass
}

# Deficit twin of increase_spending_if_possible: trims one civil law a month with its expected level.
ai_lower_expected_spending_on_deficit = {
	set_country_flag = ai_spending_law_pass # its law changes do not start an outside-change hold
	if = {
		limit = {
			is_ai = yes
			has_war = no
		}
		if = {
			limit = { check_variable = { budget_balance_ratio < -0.03 } }
			set_temp_variable = { desire_change = -1 }
			# Largest budget lines first. Each trim blocks its law for 180 days, so the chain moves down.
			# Expected was rebuilt before this ran; each branch refreshes it so the cut law is not left below it.
			if = {
				limit = {
					check_variable = { social_law > 1 }
					NOT = { has_country_flag = ai_deficit_trim_social }
					NOT = { has_country_flag = ai_deficit_cut_social }
					NOT = { check_variable = { social_law < expected_welfare_spending } }
					NOT = { AND = { check_variable = { social_desire_offset < -0.5 } has_social_law_desired = yes } }
					NOT = { AND = { check_variable = { social_desire_offset > 0 } check_variable = { expected_welfare_spending > 5 } } }
				}
				if = { limit = { has_social_law_desired = yes }
					change_expected_social_spending = yes
					set_country_flag = ai_deficit_cut_social
					set_country_flag = { flag = ai_deficit_trim_social days = 180 value = 1 } # events or AI picks can revert a trim
				}
				decrease_social_spending = yes
				calculate_expected_spending = yes
			}
			else_if = {
				limit = {
					check_variable = { health_law > 1 }
					NOT = { has_country_flag = ai_deficit_trim_health }
					NOT = { has_country_flag = ai_deficit_cut_health }
					NOT = { check_variable = { health_law < expected_health_spending } }
					NOT = { AND = { check_variable = { health_desire_offset < -0.5 } has_health_law_desired = yes } }
					NOT = { AND = { check_variable = { health_desire_offset > 0 } check_variable = { expected_health_spending > 5 } } }
				}
				if = { limit = { has_health_law_desired = yes }
					change_expected_health_spending = yes
					set_country_flag = ai_deficit_cut_health
					set_country_flag = { flag = ai_deficit_trim_health days = 180 value = 1 }
				}
				decrease_healthcare_budget = yes
				calculate_expected_spending = yes
			}
			else_if = {
				limit = {
					check_variable = { police_law > 1 }
					NOT = { has_country_flag = ai_deficit_trim_police }
					NOT = { has_country_flag = ai_deficit_cut_police }
					NOT = { check_variable = { police_law < expected_police_spending } }
					NOT = { AND = { check_variable = { police_desire_offset < -0.5 } has_police_idea_law_desired = yes } }
					NOT = { AND = { check_variable = { police_desire_offset > 0 } check_variable = { expected_police_spending > 4 } } }
				}
				if = { limit = { has_police_idea_law_desired = yes }
					change_expected_police_spending = yes
					set_country_flag = ai_deficit_cut_police
					set_country_flag = { flag = ai_deficit_trim_police days = 180 value = 1 }
				}
				decrease_policing_budget = yes
				calculate_expected_spending = yes
			}
			else_if = {
				limit = {
					check_variable = { bureau_law > 1 }
					NOT = { has_country_flag = ai_deficit_trim_bureau }
					NOT = { has_country_flag = ai_deficit_cut_bureau }
					NOT = { check_variable = { bureau_law < expected_adm_spending } }
					NOT = { AND = { check_variable = { bureau_desire_offset < -0.5 } has_bureau_idea_law_desired = yes } }
					NOT = { AND = { check_variable = { bureau_desire_offset > 0 } check_variable = { expected_adm_spending > 4 } } }
				}
				if = { limit = { has_bureau_idea_law_desired = yes }
					change_expected_bureaucracy_spending = yes
					set_country_flag = ai_deficit_cut_bureau
					set_country_flag = { flag = ai_deficit_trim_bureau days = 180 value = 1 }
				}
				decrease_centralization = yes
				calculate_expected_spending = yes
			}
		}
		# Undo one marked trim; restarts the cooldown in case the step reopens a deficit.
		else_if = {
			limit = {
				check_variable = { budget_balance_ratio > 0.08 }
				OR = {
					has_country_flag = ai_deficit_cut_social
					has_country_flag = ai_deficit_cut_health
					has_country_flag = ai_deficit_cut_police
					has_country_flag = ai_deficit_cut_bureau
				}
			}
			set_temp_variable = { desire_change = 1 }
			random_list = {
				10 = {
					modifier = { factor = 0 NOT = { has_country_flag = ai_deficit_cut_social } }
					clr_country_flag = ai_deficit_cut_social
					set_country_flag = { flag = ai_deficit_trim_social days = 180 value = 1 }
					change_expected_social_spending = yes
					calculate_expected_spending = yes
					if = { limit = { check_variable = { social_law < expected_welfare_spending } }
						increase_social_spending = yes
					}
				}
				10 = {
					modifier = { factor = 0 NOT = { has_country_flag = ai_deficit_cut_health } }
					clr_country_flag = ai_deficit_cut_health
					set_country_flag = { flag = ai_deficit_trim_health days = 180 value = 1 }
					change_expected_health_spending = yes
					calculate_expected_spending = yes
					if = { limit = { check_variable = { health_law < expected_health_spending } }
						increase_healthcare_budget = yes
					}
				}
				10 = {
					modifier = { factor = 0 NOT = { has_country_flag = ai_deficit_cut_police } }
					clr_country_flag = ai_deficit_cut_police
					set_country_flag = { flag = ai_deficit_trim_police days = 180 value = 1 }
					change_expected_police_spending = yes
					calculate_expected_spending = yes
					if = { limit = { check_variable = { police_law < expected_police_spending } }
						increase_policing_budget = yes
					}
				}
				10 = {
					modifier = { factor = 0 NOT = { has_country_flag = ai_deficit_cut_bureau } }
					clr_country_flag = ai_deficit_cut_bureau
					set_country_flag = { flag = ai_deficit_trim_bureau days = 180 value = 1 }
					change_expected_bureaucracy_spending = yes
					calculate_expected_spending = yes
					if = { limit = { check_variable = { bureau_law < expected_adm_spending } }
						increase_centralization = yes
					}
				}
			}
		}
	}
	clr_country_flag = ai_spending_law_pass
}
