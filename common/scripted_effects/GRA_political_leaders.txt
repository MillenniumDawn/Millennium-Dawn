set_leader_GRA = {
	if = { limit = { check_variable = { ruling_party = 4 } }
		if = { limit = { check_variable = { communist_state_leader = 0 } NOT = { check_variable = { b = 1 } } }
			add_to_variable = { communist_state_leader = 1 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Gloria Payne Banfield"
				picture = "gloria_payne_banfield.dds"
				ideology = communist_state
				traits = {
					emerging_communist_state
				}
			}

			if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { communist_state_leader = 1 } }
			set_temp_variable = { b = 1 }
		}
	}
}
