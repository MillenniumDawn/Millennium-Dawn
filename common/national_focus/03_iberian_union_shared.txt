shared_focus = {
	id = IBR_dream_of_iberia
	icon = iberian_summit
	x = 127
	y = 10
	offset = {
		x = -64
		y = 6
		trigger = { original_tag = SPR }
	}
	cost = 5
	search_filters = { FOCUS_FILTER_POLITICAL FOCUS_FILTER_SECTARIANISM }

	available = {
		IBR_controls_iberia = yes
		OR = {
			original_tag = SPR
			has_completed_focus = POR_iberian_union
			AND = {
				original_tag = POR
				NOT = { country_exists = SPR }
			}
		}
		custom_trigger_tooltip = {
			tooltip = IBR_not_usoe_tt
			NOT = { has_country_flag = USoE }
		}
		custom_trigger_tooltip = {
			tooltip = IBR_formables_enabled_tt
			NOT = { has_global_flag = GAME_RULE_disable_formable_nations }
		}
	}

	completion_reward = {
		log = "[GetDateText]: [Root.GetName]: Focus IBR_dream_of_iberia"
		custom_effect_tooltip = IBR_union_cores_tt
		hidden_effect = {
			every_owned_state = {
				limit = {
					OR = {
						is_spain_state = yes
						is_portugal_state = yes
					}
				}
				add_core_of = ROOT
			}
		}
		# A European Federation member keeps its EFS branding; EFS_update_flag never runs twice
		if = {
			limit = {
				has_country_flag = EFS_flag_change
				has_idea = EU_member
			}
			set_cosmetic_tag = EFS_IBR
		}
		else = {
			set_cosmetic_tag = IBR
		}
		custom_effect_tooltip = IBR_dream_of_iberia_tt
		unlock_decision_category_tooltip = IBR_dream_category
		custom_effect_tooltip = {
			localization_key = adds_dynamic_modifier_tt
			MODIFIER = IBR_divided_peninsula
		}
		set_variable = { IBR_union_growth = 0.25 tooltip = IBR_sectarian_growth_change_tt }
		custom_effect_tooltip = IBR_division_stability_tt
		hidden_effect = {
			set_country_flag = is_IBR
			mark_formed_country_formable = yes
			clr_country_flag = dynamic_flag
			clr_country_flag = dynamic_rebel_flag
			IBR_setup_states = yes
			IBR_seed_sectarianism = yes
			IBR_update_average = yes
			add_dynamic_modifier = { modifier = IBR_divided_peninsula }
			IBR_update_map_dirty = yes
			# A Portugal that conquered the peninsula skipped POR_iberian_union, which the Gates of Iberia need
			if = {
				limit = {
					original_tag = POR
					NOT = { has_completed_focus = POR_iberian_union }
				}
				unlock_national_focus = POR_iberian_union
			}
		}
	}

	ai_will_do = { base = 10 }
}
