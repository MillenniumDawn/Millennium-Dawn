add_namespace = sov_war

#The West answers Crimea
country_event = {
	id = sov_war.1
	title = sov_war.1.t
	desc = sov_war.1.d
	picture = GFX_stock_market
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		hidden_effect = {
			if = {
				limit = { NOT = { has_country_flag = SOV_western_sanctions } }
				set_country_flag = SOV_western_sanctions
				add_dynamic_modifier = { modifier = SOV_western_sanctions_modifier }
				SOV_economy_cars_western_lose_script = yes
			}
			SOV_apply_western_sanctions = yes
		}
	}

	option = {
		name = sov_war.1.a
		log = "[GetDateText]: [This.GetName]: sov_war.1.a executed"
		add_ideas = SOV_idea_counter_sanctions
		ai_chance = {
			base = 70
			modifier = { factor = 0.5 SOV_ai_favors_private_business = yes }
		}
	}

	option = {
		name = sov_war.1.b
		log = "[GetDateText]: [This.GetName]: sov_war.1.b executed"
		add_political_power = -25
		custom_effect_tooltip = { localization_key = modifies_dynamic_modifier_tt MODIFIER = SOV_western_sanctions_modifier }
		hidden_effect = {
			set_variable = { SOV_sanctions_invest_cost = { value = SOV_sanctions_invest_cost subtract = 0.04 clamp = { min = 0 max = 0.4 } } }
			set_variable = { SOV_sanctions_roi = { value = SOV_sanctions_roi add = 0.04 clamp = { min = -0.16 max = 0 } } }
			set_variable = { SOV_sanctions_office_tax_gain = { value = SOV_sanctions_office_tax_gain add = 0.04 clamp = { min = -0.16 max = 0 } } }
		}
		custom_effect_tooltip = SOV_sanctions_eras_TT
		set_temp_variable = { temp_opinion = 4 }
		change_oligarchs_opinion = yes
		if = {
			limit = { country_exists = GER }
			GER = { add_opinion_modifier = { target = ROOT modifier = medium_increase } }
		}
		if = {
			limit = { country_exists = ITA }
			ITA = { add_opinion_modifier = { target = ROOT modifier = medium_increase } }
		}
		ai_chance = {
			base = 30
			modifier = { factor = 2 SOV_ai_favors_private_business = yes }
		}
	}
}
