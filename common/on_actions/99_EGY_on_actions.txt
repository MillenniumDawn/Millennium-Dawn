	on_monthly_EGY = {
		effect = {
			coptic_events = yes

				limit = {

						has_idea = defence_00

				limit = {
					EGY_mubarak_in_power = yes

					NOT = { has_country_flag = EGY_sinai_bombing_recent }
				}
				set_temp_variable_to_random = { var = r min = 0 max = 100 }

					limit = { check_variable = { r < EGY_sinai_bombing_risk } }
					set_country_flag = { flag = EGY_sinai_bombing_recent days = 365 value = 1 }

							NOT = { has_country_flag = EGY_sharm_bombed }

					}

						limit = {
							has_country_flag = EGY_sharm_bombed
							NOT = { has_country_flag = EGY_dahab_bombed }
							date > 2006.1.1
						}

					}

						country_event = { id = egypt_mubarak.7 days = 1 }
					}
				}
			}
			if = {
				limit = {
					EGY_mubarak_in_power = yes
					date > 2007.6.1
					NOT = { has_completed_focus = EGY_rafah_on_egyptian_terms }
					NOT = { has_country_flag = EGY_rafah_breach_recent }
				}
				if = {
					limit = { has_completed_focus = EGY_seal_the_rafah_crossing }
					random = {
						chance = 1
						set_country_flag = { flag = EGY_rafah_breach_recent days = 365 value = 1 }

					}
				}

					}
				}
			}
			if = {
				limit = {
					has_completed_focus = EGY_partner_the_tycoons

						has_idea = corruption_level_05

				limit = { country_exists = LBA }

			}
