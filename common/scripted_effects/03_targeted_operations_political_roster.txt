TOP_political_country_opportunities = {
	if = {
		limit = { TOP_country_eligible = yes check_variable = { TOP_political_opportunity_after < global.TOP_clock } }
		set_variable = { TOP_political_opportunity_after = { value = global.TOP_clock add = 90 } }
		set_temp_variable = { TOP_political_candidate = 0 }
		set_temp_variable = { TOP_political_lowest = 80 }
		for_loop_effect = {
			start = 1
			end = global.TOP_registry_capacity
			value = TOP_political_person
			set_temp_variable = { TOP_target = TOP_political_person }
			if = {
				limit = {
					check_variable = { global.TOP_status^TOP_political_person = 1 }
					check_variable = { global.TOP_political^TOP_political_person = 1 }
					check_variable = { global.TOP_civilian^TOP_political_person = 0 }
					set_temp_variable = { TOP_role_target = TOP_political_person }
					TOP_authored_role_eligible = yes
					check_variable = { global.TOP_host^TOP_political_person > 0 }
					TOP_exceptional_authority = yes
					check_variable = { TOP_confidence^TOP_political_person < TOP_political_lowest }
				}
				set_temp_variable = { TOP_political_candidate = TOP_political_person }
				set_temp_variable = { TOP_political_lowest = TOP_confidence^TOP_political_person }
			}
		}
		if = { limit = { check_variable = { TOP_political_candidate > 0 } } TOP_add_target_lead = { TARGET = TOP_political_candidate AMOUNT = 25 } }
	}
}
