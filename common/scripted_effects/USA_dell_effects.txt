USA_oem_storage_import_legacy_linux_state = {
	if = {
		limit = {
			linux_system_enabled = yes
			corporate_history_enabled = yes
			original_tag = USA
			NOT = { has_country_flag = linux_system_usa_storage_legacy_imported }
		}
		clr_country_flag = linux_system_usa_storage_legacy_route_upstream
		clr_country_flag = linux_system_usa_storage_legacy_route_enterprise
		set_variable = { linux_system_usa_storage_legacy_import_level = 0 }
		if = {
			limit = { check_variable = { USA_oem_storage_policy > 0 } }
			set_variable = { linux_system_usa_storage_legacy_import_level = USA_oem_storage_policy }
			clamp_variable = { var = linux_system_usa_storage_legacy_import_level min = 1 max = 2 }
			set_country_flag = linux_system_usa_storage_legacy_route_upstream
		}
		else_if = {
			limit = { check_variable = { USA_oem_storage_policy < 0 } }
			subtract_from_variable = { linux_system_usa_storage_legacy_import_level = USA_oem_storage_policy }
			clamp_variable = { var = linux_system_usa_storage_legacy_import_level min = 1 max = 2 }
			set_country_flag = linux_system_usa_storage_legacy_route_enterprise
		}
		set_country_flag = linux_system_usa_storage_legacy_imported
		linux_system_mark_dirty = yes
	}
}

		limit = { NOT = { has_country_flag = USA_dell_capstone_resolved } }

		if = {
			limit = {
				check_variable = { var = USA_dell_enterprise_pivot value = 8 compare = greater_than_or_equals }
				OR = {
					has_country_flag = USA_dell_ai_server_buildout
					has_country_flag = USA_dell_ai_full_stack
				}
			}
			USA_dell_apply_ai_backbone_capstone = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_dell_enterprise_pivot value = 7 compare = greater_than_or_equals }
				check_variable = { var = USA_dell_governance_legitimacy value = 6 compare = greater_than_or_equals }
			}
			USA_dell_apply_integrated_capstone = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_dell_founder_control value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_dell_balance_sheet_leverage value = 6 compare = greater_than_or_equals }

			USA_dell_apply_founder_empire_capstone = yes
