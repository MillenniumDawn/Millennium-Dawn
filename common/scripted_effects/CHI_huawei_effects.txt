CHI_huawei_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = CHI_huawei_state_initialized } }

		set_variable = { CHI_huawei_carrier_reach = 4 }
		set_variable = { CHI_huawei_standards_leverage = 2 }
		set_variable = { CHI_huawei_consumer_position = 1 }
		set_variable = { CHI_huawei_silicon_autonomy = 2 }
		set_variable = { CHI_huawei_software_ecosystem = 2 }
		set_variable = { CHI_huawei_supply_resilience = 3 }
		set_variable = { CHI_huawei_trusted_market_access = 5 }
		set_country_flag = CHI_huawei_state_initialized
	}

	CHI_huawei_clamp_state = yes
}

CHI_huawei_resolve_capstone = {
	CHI_huawei_initialize_state = yes

	if = {
		limit = { NOT = { CHI_huawei_has_capstone_outcome = yes } }
		if = {
			limit = { CHI_huawei_qualifies_global_connectivity_federation = yes }
			CHI_huawei_apply_global_connectivity_federation = yes
		}
		else_if = {
			limit = { CHI_huawei_qualifies_consumer_ecosystem_power = yes }
			CHI_huawei_apply_consumer_ecosystem_power = yes
		}
		else_if = {
			limit = { CHI_huawei_qualifies_carrier_cloud_utility = yes }
			CHI_huawei_apply_carrier_cloud_utility = yes
		}
		else_if = {
			limit = { CHI_huawei_qualifies_patent_standards_house = yes }
			CHI_huawei_apply_patent_standards_house = yes
		}
		else_if = {
			limit = { CHI_huawei_qualifies_sovereign_full_stack = yes }
			CHI_huawei_apply_sovereign_full_stack = yes
		}
		else = { CHI_huawei_apply_resilient_technology_fortress = yes }
	}
}
