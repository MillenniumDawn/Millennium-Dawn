HKG_naval_contracts_category = {

	# HKG petitions Beijing for a fresh procurement order.
	HKG_contract_solicit_contract = {
		icon = GFX_HKG_contract_beijing

		visible = {
			has_country_flag = HKG_contract_active
			NOT = { has_country_flag = HKG_contract_offer_pending }
		}

		available = {
			country_exists = CHI
			NOT = { has_war_with = CHI }
			custom_trigger_tooltip = {
				tooltip = HKG_contract_slots_free_tt
				check_variable = { HKG_contract_count < HKG_contract_max_slots }
			}
			custom_trigger_tooltip = {
				tooltip = HKG_contract_modules_tt
				HKG_contract_can_build_ships = yes
			}
		}

		cost = 25
		days_re_enable = 90
		fire_only_once = no

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision HKG_contract_solicit_contract"
			custom_effect_tooltip = HKG_contract_tt_solicit
			country_event = { id = HKG_contract.1 days = 2 }
		}

		ai_will_do = {
			base = 50
			modifier = {
				factor = 0
				has_country_flag = HKG_contract_offer_pending
			}
		}
	}

	# --- Per-class deliver decisions ----------------------------------------
	# Delivery is strictly by ship class. transfer_ship moves ONE ship of the
	# given archetype per call; HOI4 exposes no ship iterator and decisions take
	# no text input, so name-based ship selection is impossible. Do not "fix"
	# this into name filtering — class-based transfer is the only robust path.
	# Each click credits the most urgent active contract of that class.

	HKG_contract_deliver_destroyer = {
		icon = GFX_HKG_contract_deliver

		visible = {
			has_country_flag = HKG_contract_active
			HKG_contract_destroyer_active = yes
		}

		available = {
			custom_trigger_tooltip = {
				tooltip = HKG_contract_need_destroyer_tt
				has_navy_size = { archetype = destroyer size > 0 }
			}
		}

		fire_only_once = no

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision HKG_contract_deliver_destroyer"
			custom_effect_tooltip = HKG_contract_tt_deliver
			transfer_ship = { type = destroyer target = CHI }
			custom_effect_tooltip = HKG_contract_tt_deliver_reward
			set_temp_variable = { deliver_class = 1 }
			HKG_contract_credit_delivery = yes
		}

		ai_will_do = { base = 100 }

HKG_contract_chi_category = {

	# Player-controlled PRC commissions a warship contract from Hong Kong.
	HKG_contract_commission_warships = {
		icon = GFX_HKG_contract_beijing

		visible = {
			country_exists = HKG
			HKG = { has_country_flag = HKG_contract_active }
		}

		available = {
			NOT = { has_war_with = HKG }
			custom_trigger_tooltip = {
				tooltip = HKG_contract_commission_ready_tt
				HKG = { HKG_contract_can_solicit = yes }
			}
			custom_trigger_tooltip = {
				tooltip = HKG_contract_commission_yards_tt
				HKG = { HKG_contract_can_build_ships = yes }
			}
		}

		cost = 30
		days_re_enable = 90
		fire_only_once = no

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision HKG_contract_commission_warships"
			HKG = { country_event = { id = HKG_contract.1 days = 2 } }
			custom_effect_tooltip = TT_IF_THEY_ACCEPT
			effect_tooltip = {
				custom_effect_tooltip = HKG_contract_commission_accept_tt
			}
		}

		# base 0: AI-controlled CHI issues contracts through on_weekly instead,
		# so the AI never needs this decision — it exists for the human player.
		ai_will_do = { base = 0 }
	}
}
