MNT_ai_alt_path = {
	OR = {
		has_global_flag = MNT_EUROPEAN_REFORM_FOCUS_PATH
		has_global_flag = MNT_SERBIAN_UNION_FOCUS_PATH
		has_global_flag = MNT_MONARCHIST_RESTORATION_FOCUS_PATH
		has_global_flag = MNT_NEUTRAL_HARBOUR_FOCUS_PATH
		has_global_flag = MNT_SOCIALIST_REVIVAL_FOCUS_PATH
	}
}

MNT_ai_historical_path = {
	OR = {
		has_global_flag = MNT_HISTORICAL_FOCUS_PATH
		AND = {
			is_historical_focus_on = yes
			NOT = { MNT_ai_alt_path = yes }
		}
	}
}

MNT_ai_any_path = {
	OR = {
		MNT_ai_historical_path = yes
		MNT_ai_alt_path = yes
	}
}

MNT_ai_not_historical_path = {
	MNT_ai_alt_path = yes
}

MNT_ai_dukanovic_path = {
	OR = {
		MNT_ai_historical_path = yes
		has_global_flag = MNT_NEUTRAL_HARBOUR_FOCUS_PATH
		has_completed_focus = MNT_dukanovic_rule
	}
}
