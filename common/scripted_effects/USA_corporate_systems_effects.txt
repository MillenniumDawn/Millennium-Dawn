# Linux contributes from generic base state only. Native IBM and storage outcomes
# feed the adapter layer, so reading effective state here would create feedback.
USA_corporate_systems_linux_contribution = {
	if = {
		limit = {
			linux_system_enabled = yes
			has_country_flag = linux_system_initialized
			check_variable = { linux_system_base_deployment > 5 }
		}
		add_to_temp_variable = { USA_oem_contribution_national_compute_stack = 1 }
	}
	if = {
		limit = {
			linux_system_enabled = yes
			has_country_flag = linux_system_initialized
			check_variable = { linux_system_base_stewardship > 6 }
		}
		add_to_temp_variable = { USA_oem_contribution_open_standards = 1 }
	}
	else_if = {
		limit = {
			linux_system_enabled = yes
			has_country_flag = linux_system_initialized
			check_variable = { linux_system_base_stewardship < 3 }
		}
		subtract_from_temp_variable = { USA_oem_contribution_open_standards = 1 }
	}
	if = {
		limit = {
			linux_system_enabled = yes
			has_country_flag = linux_system_initialized
			check_variable = { linux_system_base_assurance > 6 }
		}
		add_to_temp_variable = { USA_oem_contribution_supply_resilience = 1 }
	}
	else_if = {
		limit = {
			linux_system_enabled = yes
			has_country_flag = linux_system_initialized
			check_variable = { linux_system_base_assurance < 3 }
		}
		subtract_from_temp_variable = { USA_oem_contribution_supply_resilience = 1 }
	}
	if = {
		limit = {
			linux_system_enabled = yes
			has_country_flag = linux_system_initialized
			check_variable = { linux_system_base_support_model = 2 }
		}
		add_to_temp_variable = { USA_oem_contribution_vertical_integration = 1 }
	}
	else_if = {
		limit = {
			linux_system_enabled = yes
			has_country_flag = linux_system_initialized
			check_variable = { linux_system_base_support_model = 3 }
		}
		add_to_temp_variable = { USA_oem_contribution_security_control = 1 }
	}
}

	if = {
		limit = {
			original_tag = USA
			NOT = { has_country_flag = collapsed_nation }
			OR = {
				AND = {
					corporate_history_enabled = yes
					has_country_flag = USA_ibm_state_initialized
					USA_corporate_systems_has_meaningful_state = yes
				}
				AND = {
					linux_system_enabled = yes
					has_country_flag = linux_system_initialized
				}
			}
		}
		if = {
			limit = {
				corporate_history_enabled = yes
				has_country_flag = USA_ibm_state_initialized
				USA_corporate_systems_has_meaningful_state = yes
			}
			USA_corporate_systems_rebuild_company_contributions = yes
			USA_corporate_systems_rebuild_effective_axes = yes
		}
		else = {
			USA_corporate_systems_rebuild_linux_only_axes = yes
		}

		set_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_effective_open_standards }
		add_to_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_effective_vertical_integration }
		add_to_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_effective_supply_resilience }
		add_to_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_effective_security_control }
		add_to_temp_variable = { USA_corporate_systems_economic_integration_score = USA_oem_effective_national_compute_stack }

		if = {
			limit = { check_variable = { USA_corporate_systems_economic_integration_score < 15 } }
			if = {
				limit = {
					OR = {
						NOT = { has_idea = USA_corporate_systems_economic_integration_1 }
						has_idea = USA_corporate_systems_economic_integration_2
						has_idea = USA_corporate_systems_economic_integration_3
						has_idea = USA_corporate_systems_economic_integration_4
						has_idea = USA_corporate_systems_economic_integration_5
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_1
			}
		}
		else_if = {
			limit = { check_variable = { USA_corporate_systems_economic_integration_score < 22 } }
			if = {
				limit = {
					OR = {
						has_idea = USA_corporate_systems_economic_integration_1
						NOT = { has_idea = USA_corporate_systems_economic_integration_2 }
						has_idea = USA_corporate_systems_economic_integration_3
						has_idea = USA_corporate_systems_economic_integration_4
						has_idea = USA_corporate_systems_economic_integration_5
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_2
			}
		}
		else_if = {
			limit = { check_variable = { USA_corporate_systems_economic_integration_score < 29 } }
			if = {
				limit = {
					OR = {
						has_idea = USA_corporate_systems_economic_integration_1
						has_idea = USA_corporate_systems_economic_integration_2
						NOT = { has_idea = USA_corporate_systems_economic_integration_3 }
						has_idea = USA_corporate_systems_economic_integration_4
						has_idea = USA_corporate_systems_economic_integration_5
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_3
			}
		}
		else_if = {
			limit = { check_variable = { USA_corporate_systems_economic_integration_score < 38 } }
			if = {
				limit = {
					OR = {
						has_idea = USA_corporate_systems_economic_integration_1
						has_idea = USA_corporate_systems_economic_integration_2
						has_idea = USA_corporate_systems_economic_integration_3
						NOT = { has_idea = USA_corporate_systems_economic_integration_4 }
						has_idea = USA_corporate_systems_economic_integration_5
					}
				}
				USA_corporate_systems_clear_economic_bridge_ideas = yes
				add_ideas = USA_corporate_systems_economic_integration_4
			}
		}
		else = {
			if = {
				limit = {
					OR = {
						has_idea = USA_corporate_systems_economic_integration_1
						has_idea = USA_corporate_systems_economic_integration_2
						has_idea = USA_corporate_systems_economic_integration_3
						has_idea = USA_corporate_systems_economic_integration_4
						NOT = { has_idea = USA_corporate_systems_economic_integration_5 }
