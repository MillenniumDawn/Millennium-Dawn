ENG_referendum_campaign_running = {
	OR = {
		has_active_mission = ENG_scottish_referendum
		has_active_mission = ENG_irish_referendum
		has_active_mission = ENG_welsh_referendum
		has_active_mission = ENG_union_settlement_referendum
	}
}

ENG_referendum_in_progress = {
	OR = {
		ENG_referendum_campaign_running = yes
		has_active_mission = ENG_scottish_referendum_pressure
		has_active_mission = ENG_north_irish_referendum_pressure
		has_active_mission = ENG_welsh_referendum_pressure
	}
}

ENG_settlement_region_lost = {
	OR = {
		AND = {
			check_variable = { ENG_referendum_target = 1 }
			OR = {
				country_exists = SCO
				has_country_flag = ENG_scottish_integrated
				NOT = { ENG_holds_scotland = yes }
			}
		}
		AND = {
			check_variable = { ENG_referendum_target = 2 }
			OR = {
				country_exists = NIR
				has_country_flag = ENG_north_irish_integrated
				NOT = { ENG_holds_north_ireland = yes }
			}
		}
		AND = {
			check_variable = { ENG_referendum_target = 3 }
			OR = {
				country_exists = WAS
				has_country_flag = ENG_welsh_integrated
				NOT = { ENG_holds_wales = yes }
			}
		}
	}
}
