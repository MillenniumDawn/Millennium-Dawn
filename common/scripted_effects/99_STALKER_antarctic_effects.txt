# Treaty observers who reach a station during a polar artifact study seize the samples and report them.
# The cut-short flag outlives the 180-day result event by a day.
STALKER_polar_study_detected = {
	if = {
		limit = { has_dynamic_modifier = { modifier = STALKER_polar_artifact_samples } }
		remove_dynamic_modifier = { modifier = STALKER_polar_artifact_samples }
	}
	set_country_flag = { flag = STALKER_polar_study_cut_short days = 181 value = 1 }
	news_event = STALKER.191
}

# The study's Radiation Laboratory stopped: damage, lost fuel or power, a replaced module or a dismantled station.
# The samples are lost and the result never comes.
STALKER_polar_study_lab_lost = {
	log = "[GetDateText]: [THIS.GetName]: STALKER polar artifact study ended, laboratory down"
	if = {
		limit = { has_dynamic_modifier = { modifier = STALKER_polar_artifact_samples } }
		remove_dynamic_modifier = { modifier = STALKER_polar_artifact_samples }
	}
	set_country_flag = { flag = STALKER_polar_study_cut_short days = 181 value = 1 }
}
