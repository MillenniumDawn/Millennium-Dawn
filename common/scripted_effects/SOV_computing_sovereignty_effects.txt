# Russian computing sovereignty. Five milestones set the route; the closing
# milestone resolves it into one of four computing futures.

SOV_computing_sovereignty_apply_sovereign_stack = {
	if = {
		limit = { SOV_computing_sovereignty_outcome_settled = no }
		add_ideas = SOV_computing_sovereignty_sovereign_computing_stack
	}
}

SOV_computing_sovereignty_apply_chinese_substitution = {
	if = {
		limit = { SOV_computing_sovereignty_outcome_settled = no }
		add_ideas = SOV_computing_sovereignty_chinese_substitution
	}
}

SOV_computing_sovereignty_apply_parallel_imports = {
	if = {
		limit = { SOV_computing_sovereignty_outcome_settled = no }
		add_ideas = SOV_computing_sovereignty_parallel_import_economy
	}
}

SOV_computing_sovereignty_apply_mature_nodes = {
	if = {
		limit = { SOV_computing_sovereignty_outcome_settled = no }
		add_ideas = SOV_computing_sovereignty_mature_node_specialization
	}
}

# One entry point per milestone, called from the yearly pulse. Full presents the
# choice; Outcomes Only writes the historical route silently and grants nothing.
SOV_computing_sovereignty_milestone_2001 = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		country_event = { id = sov_economic.300 days = 151 }
	}
	else_if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
		set_country_flag = SOV_cs_elbrus_domestic_bet
	}
}
