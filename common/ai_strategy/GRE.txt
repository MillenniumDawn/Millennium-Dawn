GRE_antagonize_turkey = {
	allowed = { original_tag = GRE }
	enable = {
		country_exists = TUR
		TUR = { neutrality_neutral_muslim_brotherhood_are_in_power = yes }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = antagonize id = "TUR" value = 50 }
}

GRE_counter_turkish_alignment = {
	allowed = { original_tag = GRE }
	enable = {
		country_exists = TUR
		TUR = {
			OR = {
				has_completed_focus = TUR_right_wing_victory
				has_completed_focus = TUR_great_east
			}
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = antagonize id = "TUR" value = 50 }
	ai_strategy = { type = contain id = "TUR" value = 50 }
	ai_strategy = { type = front_unit_request tag = TUR value = 10 }
}

GRE_support_albania = {
	allowed = { original_tag = GRE }
	enable = {
		country_exists = ALB
		NOT = { has_wargoal_against = ALB }
		ALB = {
			NOT = { has_government = fascism }
			NOT = { has_government = nationalist }
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = support id = "ALB" value = 50 }
	ai_strategy = { type = befriend id = "ALB" value = 50 }
}

GRE_protect_cyprus = {
	allowed = { original_tag = GRE }
	enable = {
		country_exists = CYP
		NOT = { has_war_with = CYP }
		NOT = { has_wargoal_against = CYP }
		CYP = { NOT = { has_opinion = { target = GRE value < 0 } } }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = befriend id = "CYP" value = 100 }
	ai_strategy = { type = protect id = "CYP" value = 50 }
}
