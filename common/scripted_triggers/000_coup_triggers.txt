can_military_attempt_coup = {
	custom_trigger_tooltip = {
		tooltip = can_military_attempt_coup_TT
		NOT = {
			OR = {
				# could be used for country content as well as countries that have no military at game start
				is_in_array = { global.military_cant_coup = THIS.id }
				has_idea = no_military
			}
			# can't coup themselves militarily, use counter-coup for this instead
			nationalist_military_junta_are_in_power = yes
		}
	}
}
