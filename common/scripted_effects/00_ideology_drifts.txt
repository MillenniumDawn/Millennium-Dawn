reset_democratic_coalition_ideas = {

}

# OUTLOOK is the 0..4 numeric outlook ID; table slot 0 means no coalition idea.
update_coalition_outlook_idea = {
	set_temp_variable = {
		coalition_idea_level = {
			value = { value = $SHARE$ greater_than = 0.01 }
			add = { value = $SHARE$ greater_than = 0.07 }
			add = { value = $SHARE$ greater_than = 0.11 }
			add = { value = $SHARE$ greater_than = 0.15 }
			add = { value = $SHARE$ greater_than = 0.20 }
			add = { value = $SHARE$ greater_than = 0.25 }
			add = { value = $SHARE$ greater_than = 0.30 }
			add = { value = $SHARE$ greater_than = 0.35 }
			add = { value = $SHARE$ greater_than = 0.40 }
		}
	}
	set_temp_variable = { coalition_idea_target_token = global.$IDEAS$^coalition_idea_level }
	if = {
		limit = {
			NOT = { check_variable = { coalition_applied_tokens^$OUTLOOK$ = 0 } }
			NOT = { check_variable = { coalition_applied_tokens^$OUTLOOK$ = coalition_idea_target_token } }
		}
		remove_ideas = var:coalition_applied_tokens^$OUTLOOK$
	}
	if = {
		limit = {
			check_variable = { coalition_idea_level > 0 }
			NOT = { has_idea = var:coalition_idea_target_token }
		}
		add_ideas = var:coalition_idea_target_token
	}
	set_variable = { coalition_applied_tokens^$OUTLOOK$ = coalition_idea_target_token }
}

clear_coalition_count_ideas = {
	for_each_loop = {
		array = coalition_applied_tokens
		value = coalition_clear_token
		if = {
			limit = { NOT = { check_variable = { coalition_clear_token = 0 } } }
			remove_ideas = var:coalition_clear_token
		}
	}
	clear_array = coalition_applied_tokens
}

add_own_ideology_drift = {
	if = {
		limit = {
			has_government = fascism
			is_major = no
		}
		add_ideas = ruling_party_salafist
		if = {
			limit = {
				NOT = { original_tag = NIG }
				NOT = { has_idea = sunni }
			}
			add_ideas = sunni
		}
	}
	else_if = {
		limit = {
			has_government = fascism
			is_major = yes
		}
		add_ideas = major_ruling_party_salafist
	}

	if = {
		limit = {
			has_government = democratic
			is_major = no

economic_cycle_drift_popularity = {
	hidden_effect = {
		set_temp_variable = { economic_cycle_drift_cycle_id = -1 }
		set_temp_variable = { economic_cycle_drift_outlook_id = -1 }

		if = {
			limit = { has_idea = economic_boom }
			set_temp_variable = { economic_cycle_drift_cycle_id = 4 }
		}
		else_if = {
			limit = { has_idea = fast_growth }
			set_temp_variable = { economic_cycle_drift_cycle_id = 3 }
		}
		else_if = {
			limit = { has_idea = stagnation }
			set_temp_variable = { economic_cycle_drift_cycle_id = 2 }
		}
		else_if = {
			limit = { has_idea = recession }
			set_temp_variable = { economic_cycle_drift_cycle_id = 1 }
		}
		else_if = {
			limit = { has_idea = depression }
			set_temp_variable = { economic_cycle_drift_cycle_id = 0 }
		}

		if = {
			limit = { has_government = democratic }
			set_temp_variable = { economic_cycle_drift_outlook_id = 0 }
		}
		else_if = {
			limit = { has_government = nationalist }
			set_temp_variable = { economic_cycle_drift_outlook_id = 1 }
		}
		else_if = {
			limit = { has_government = communism }
			set_temp_variable = { economic_cycle_drift_outlook_id = 2 }
		}
		else_if = {
			limit = { has_government = neutrality }
			set_temp_variable = { economic_cycle_drift_outlook_id = 3 }
		}
		else_if = {
			limit = { has_government = fascism }
			set_temp_variable = { economic_cycle_drift_outlook_id = 4 }
		}

		if = {
			limit = {
				check_variable = { economic_cycle_drift_cycle_id > -1 }
				check_variable = { economic_cycle_drift_outlook_id > -1 }
			}
			set_temp_variable = { economic_cycle_drift_index = economic_cycle_drift_cycle_id }
			multiply_temp_variable = { economic_cycle_drift_index = 5 }
			add_to_temp_variable = { economic_cycle_drift_index = economic_cycle_drift_outlook_id }
			set_temp_variable = {
				economic_cycle_drift_target_token = global.economic_cycle_drift_tokens^economic_cycle_drift_index
			}
		}

		if = {
			limit = { has_variable = economic_cycle_drift_applied_token }
			if = {
				limit = {
					OR = {
						check_variable = { economic_cycle_drift_cycle_id = -1 }
						check_variable = { economic_cycle_drift_outlook_id = -1 }
					}
				}
				remove_ideas = var:economic_cycle_drift_applied_token
				clear_variable = economic_cycle_drift_applied_token
			}
			else_if = {
				limit = {
					NOT = {
						check_variable = {
							economic_cycle_drift_applied_token = economic_cycle_drift_target_token
						}
					}
				}
				remove_ideas = var:economic_cycle_drift_applied_token
				clear_variable = economic_cycle_drift_applied_token
			}
		}

		if = {
			limit = {
				check_variable = { economic_cycle_drift_cycle_id > -1 }
				check_variable = { economic_cycle_drift_outlook_id > -1 }
				NOT = { has_variable = economic_cycle_drift_applied_token }
			}
			add_ideas = var:economic_cycle_drift_target_token
			set_variable = { economic_cycle_drift_applied_token = economic_cycle_drift_target_token }
		}
	}
}
