set_leader_SIA = {
	if = { limit = { has_country_flag = set_Neutral_conservatism }
		# Bhumjaithai holds this slot only during the caretaker storyline; otherwise it is Chart Thai.
		if = { limit = { has_country_flag = SIA_anutin_caretaker }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Anutin Charnvirakul"
				desc = "SIA_anutin_charnvirakul_desc"
				picture = "anuthin_charnvirakul.dds"
				ideology = Neutral_conservatism
				traits = {
					neutrality_Neutral_conservatism
					skilled_lobbyist
				}
			}

			set_temp_variable = { b = 1 }
		}
		else = {
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Banharn Silpa-archa"
				desc = "SIA_banharn_silpaarcha_desc"
				picture = "generic.dds"
				ideology = Neutral_conservatism
				traits = {
					neutrality_Neutral_conservatism
					businessman
				}
			}

			set_temp_variable = { b = 1 }
		}
	}
	# Date-gated rather than counter-gated so a re-election re-seats the premier of the day
	# instead of advancing past them.
	else_if = { limit = { has_country_flag = set_neutral_Social }
		if = { limit = { date < 2011.8.5 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Thaksin Shinawatra"
				desc = "SIA_thaksin_shinawatra_desc"
				picture = "Thaksin_Shinawatra.dds"
				ideology = neutral_Social
				traits = {
					neutrality_neutral_Social
					businessman
				}
			}

			set_temp_variable = { b = 1 }
		}
		else_if = { limit = { date < 2023.8.22 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Yingluck Shinawatra"
				desc = "SIA_yingluck_shinawatra_desc"
				picture = "Yingluck_Shinawatra.dds"
				ideology = neutral_Social
				traits = {
					neutrality_neutral_Social
					businessman
					likeable
				}
			}

			set_temp_variable = { b = 1 }
		}
		else_if = { limit = { date < 2024.8.16 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Srettha Thavisin"
				desc = "SIA_srettha_thavisin_desc"
				picture = "generic.dds"
				ideology = neutral_Social
				traits = {
					neutrality_neutral_Social
					businessman
				}
			}

			set_temp_variable = { b = 1 }
		}
		else_if = { limit = { date < 2025.9.1 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Paetongtarn Shinawatra"
				desc = "SIA_paetongtarn_shinawatra_desc"
				picture = "generic.dds"
				ideology = neutral_Social
				traits = {
					neutrality_neutral_Social
					likeable
				}
			}

			set_temp_variable = { b = 1 }
		}
		else = {
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Police LT.Gen. Virot Paoin"
				desc = "SIA_virot_paoin_desc"
				picture = "virot_paoin.dds"
				ideology = neutral_Social
				traits = {
					neutrality_neutral_Social
					military_career
				}
			}

			set_temp_variable = { b = 1 }
		}
	}
	else_if = { limit = { has_country_flag = set_Monarchist }
		if = { limit = { check_variable = { Monarchist_leader = 0 } }
			add_to_variable = { Monarchist_leader = 1 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Rama IX Bhumibol Adulyadej"
				desc = "SIA_bhumibol_adulyadej_desc"
				picture = "King_Bhumibol.dds"
				ideology = Monarchist
				traits = {
					nationalist_Monarchist
					hands_that_crafted_the_nation
				}
			}

			if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { Monarchist_leader = 1 } }
			if = { limit = { date < 2016.1.2 } set_temp_variable = { b = 1 } } #skip if 2017
		}
		if = { limit = { check_variable = { Monarchist_leader = 1 } NOT = { check_variable = { b = 1 } } }
			add_to_variable = { Monarchist_leader = 1 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Rama X Maha Vajiralongkorn"
				desc = "SIA_vajiralongkorn_desc"
				picture = "Maha_Vajiralongkorn.dds"
				ideology = Monarchist
				traits = {
					nationalist_Monarchist
					capable
				}
			}

			if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { Monarchist_leader = 1 } }
			set_temp_variable = { b = 1 }
		}
	}
	else_if = { limit = { has_country_flag = set_conservatism }
		if = { limit = { check_variable = { conservatism_leader = 0 } }
			add_to_variable = { conservatism_leader = 1 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Chuan Leekpai"
				desc = "SIA_chuan_leekpai_desc"
				picture = "Chuan_Leekpai.dds"
				ideology = conservatism
				traits = {
					western_conservatism
					staunch_constitutionalist
				}
			}

			if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { conservatism_leader = 1 } }
			set_temp_variable = { b = 1 }
		}
		if = { limit = { check_variable = { conservatism_leader = 1 } NOT = { check_variable = { b = 1 } } }
			add_to_variable = { conservatism_leader = 1 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Abhisit Vejjajiva"
				desc = "SIA_abhisit_vejjajiva_desc"
				picture = "abhisit_vejjajiva.dds"
				ideology = conservatism
				traits = {
					western_conservatism
					educator
				}
			}

			if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { conservatism_leader = 1 } }
			set_temp_variable = { b = 1 }
		}
	}
	# Date-gated: Future Forward, then Move Forward, then the People's Party.
	else_if = { limit = { has_country_flag = set_socialism }
		if = { limit = { date < 2020.2.21 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Thanathorn Juangroongruangkit"
				desc = "SIA_thanathorn_juangroongruangkit_desc"
				picture = "generic.dds"
				ideology = socialism
				traits = {
					western_socialism
					businessman
				}
			}

			set_temp_variable = { b = 1 }
		}
		else_if = { limit = { date < 2024.9.1 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Pita Limjaroenrat"
				desc = "SIA_pita_limjaroenrat_desc"
				picture = "generic.dds"
				ideology = socialism
				traits = {
					western_socialism
					geopolitical_thinker
				}
			}

			set_temp_variable = { b = 1 }
		}
		else = {
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Natthaphong Ruengpanyawut"
				desc = "SIA_natthaphong_ruengpanyawut_desc"
				picture = "generic.dds"
				ideology = socialism
				traits = {
					western_socialism
					capable
				}
			}

			set_temp_variable = { b = 1 }
		}
	}
	else_if = { limit = { has_country_flag = set_Nat_Autocracy date < 2013.1.1 NOT = { has_country_flag = SIA_leader_sonthi } }
		hidden_effect = { kill_country_leader = yes }

		create_country_leader = {
			name = "Sonthi Boonyaratglin"
			desc = "SIA_sonthi_boonyaratglin_desc"
			picture = "Sonthi_Boonyaratglin.dds"
			ideology = Nat_Autocracy
			traits = {
				military_career
				nationalist_Nat_Autocracy
			}
		}

		set_country_flag = SIA_leader_sonthi
		set_temp_variable = { b = 1 }
	}
	else_if = { limit = { has_country_flag = set_Nat_Autocracy }
		if = { limit = { check_variable = { Nat_Autocracy_leader = 0 } }
			add_to_variable = { Nat_Autocracy_leader = 1 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Prayut Chan-o-cha"
				desc = "SIA_prayut_chanocha_desc"
				picture = "Prayut_Chan-o-cha.dds"
				ideology = Nat_Autocracy
				traits = {
					military_career
					nationalist_Nat_Autocracy
					pro_china
					capable
					ruthless
				}
			}

			if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { Nat_Autocracy_leader = 1 } }
			set_temp_variable = { b = 1 }
		}
	}
	else_if = { limit = { has_country_flag = set_Nat_Populism }
		if = { limit = { check_variable = { Nat_Populism_leader = 0 } }
			add_to_variable = { Nat_Populism_leader = 1 }
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Chavalit Yongchaiyudh"
				desc = "SIA_chavalit_yongchaiyudh_desc"
				picture = "generic.dds"
				ideology = Nat_Populism
				traits = {
					nationalist_Nat_Populism
					military_career
				}
			}

			if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { Nat_Populism_leader = 1 } }
			set_temp_variable = { b = 1 }
		}
	}
	# "Songkhram Phaendin" is a FICTIONAL name for the ahistorical Pan-Thai path, not a real person.
	else_if = { limit = { has_country_flag = set_Nat_Fascism }
		if = { limit = {
				NOT = { has_country_flag = SIA_leader_songkhram }
				has_game_rule = { rule = SIA_alt_history_thailand option = SIA_alt_history_on }
			}
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Songkhram Phaendin"
				desc = "SIA_songkhram_phaendin_desc"
				picture = "generic.dds"
				ideology = Nat_Fascism
				traits = {
					military_career
					nationalist_Nat_Fascism
				}
			}

			set_country_flag = SIA_leader_songkhram
			set_temp_variable = { b = 1 }
		}
	}
	# "Somchai Ratchaphruek" is a FICTIONAL CPT-cadre archetype for the revival path, not a real person.
	else_if = { limit = { has_country_flag = set_Communist-State }
		if = { limit = {
				NOT = { has_country_flag = SIA_leader_somchai }
				has_game_rule = { rule = SIA_alt_history_thailand option = SIA_alt_history_on }
			}
			hidden_effect = { kill_country_leader = yes }

			create_country_leader = {
				name = "Somchai Ratchaphruek"
				desc = "SIA_somchai_ratchaphruek_desc"
				picture = "generic.dds"
				ideology = Communist-State
				traits = {
					emerging_Communist-State
					legendary_guerrilla_leader
				}
			}

			set_country_flag = SIA_leader_somchai
			set_temp_variable = { b = 1 }
		}
	}
}
