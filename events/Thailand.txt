# The 2001 General Election
country_event = {
	id = thailand.1
	title = thailand.1.t
	desc = thailand.1.d
	is_triggered_only = yes

	trigger = {
		tag = SIA
	}

	# The three choices only tilt the ballot: each boosts one party's popularity and
	# its ideology-group outlook. The in-game election then decides who governs, and
	# only the matching trunk-entry focus (thai_rak_thai / old_order / democrats_endure)
	# unlocks once that party actually rules. No option forces a government.

	# A - Back Thai Rak Thai (Pheu Thai, neutrality, idx 18)
	option = {
		name = thailand.1.a
		log = "[GetDateText]: [Root.GetName]: event thailand.1.a"
		set_temp_variable = { party_index = 18 }
		set_temp_variable = { party_popularity_increase = 0.15 }
		set_temp_variable = { temp_outlook_increase = 0.10 }
		add_relative_party_popularity = yes
		recalculate_party = yes

		add_to_variable = { SIA_coup_risk = 5 }

# The War on Drugs (Story 2.4 - path-forked; CAP-6)
# Dual entry preserved: fired by SIA_war_on_drugs (Thaksin Branch-A node OR the
# Golden Triangle interdiction tail). The two original options are widened and a
# reform decriminalization option (C) is added. Coup-risk dynamic modifier and the
# SIA_war_on_drugs focus block are untouched; the hard line now couples to story
# 2.3's Deep South intensity floor (SIA_ds_set_intensity + SIA_ds_raise_floor).
country_event = {
	id = thailand.3
	title = thailand.3.t
	desc = thailand.3.d
	is_triggered_only = yes

	trigger = {
		tag = SIA
	}

	# A - Hard line (extrajudicial campaign). Historical default, always available.
	option = {
		name = thailand.3.a
		log = "[GetDateText]: [Root.GetName]: event thailand.3.a"
		add_ideas = SIA_drug_war_stain
		add_timed_idea = { idea = SIA_war_on_drugs_campaign days = 365 }
		set_party_index_to_ruling_party = yes
		set_temp_variable = { party_popularity_increase = 0.10 }
		add_relative_party_popularity = yes
		# Iron-Fist drug-war coupling: only when the Deep South crisis is
		# live. The campaign suppresses intensity (-10, floor-aware) but its
		# brutality permanently raises the grievance floor (+6) - reusing story
		# 2.3's helpers, not a reimplementation.
		if = {
			limit = { has_country_flag = SIA_ds_active }
			set_temp_variable = { ds_delta = -10 }
			SIA_ds_set_intensity = yes
			set_temp_variable = { ds_floor_delta = 6 }
			SIA_ds_raise_floor = yes
		}
		# Populist hard line: coup risk + the 2,800-killings reckoning.
		if = {
			limit = { has_country_flag = SIA_path_populist }
			add_to_variable = { SIA_coup_risk = 5 }
			random_list = {
				70 = { }
				30 = { country_event = { id = thailand.361 days = 7 } }
			}
		}
		hidden_effect = {
			country_event = { id = thailand.68 days = 150 random_days = 60 }
		}
		ai_chance = {
			base = 1
			modifier = {
				factor = 5
				is_historical_focus_on = yes
			}
		}
	}

	# B - Rule of law / treatment (restrained; royalist & default exit).
	option = {
		name = thailand.3.b
		log = "[GetDateText]: [Root.GetName]: event thailand.3.b"
		add_stability = 0.02
		set_country_flag = SIA_drug_war_restrained
		if = {
			limit = { has_completed_focus = SIA_golden_triangle_interdiction }
			set_temp_variable = { treasury_change = 2 }
			modify_treasury_effect = yes
			USA = { add_opinion_modifier = { target = SIA modifier = small_increase } }
			ENG = { add_opinion_modifier = { target = SIA modifier = small_increase } }
		}
		ai_chance = {
			base = 1
		}
	}

	# C - Crops not poppies (reform alternative-development). Reform path only.
	option = {
		name = thailand.3.c
		log = "[GetDateText]: [Root.GetName]: event thailand.3.c"
		# B-1 guard: hide if the Golden Triangle path already fired .362 (A/B remain).
		trigger = {
			has_country_flag = SIA_path_reform
			NOT = { has_country_flag = SIA_362_fired }
		}
		set_country_flag = SIA_drug_war_restrained
		country_event = { id = thailand.362 days = 1 }
		ai_chance = {
			base = 1
			modifier = {
				factor = 3
				is_historical_focus_on = yes
			}
		}
	}
}
