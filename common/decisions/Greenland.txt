GRN_national_decisions = {
	GRN_decision_independence_war = {
		fire_only_once = yes
		activation = {
			DEN = {
				has_war_with = GRN
				has_war_with = FAO
			}
		}
		visible = { always = yes }
		available = {
			1161 = { is_controlled_by = GRN }
			1162 = { is_controlled_by = GRN }
			2 = { is_controlled_by = FAO }
		}
		days_mission_timeout = 180
		is_good = no
		cancel_trigger = {
			GRN = { is_subject = yes }
			FAO = { is_subject = yes }
		}
		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision timeout GRN_decision_independence_war"
			if = {
				limit = {
					DEN = { controls_province = 14381 }
					DEN = { controls_province = 12512 }
					DEN = { controls_province = 14940 }
					DEN = { controls_province = 14386 }
					DEN = { controls_province = 14387 }
					DEN = { controls_province = 14379 }
					DEN = { controls_province = 14384 }
					DEN = { controls_province = 13003 }
				}
				DEN = { 
					annex_country = { target = GRN } 
					annex_country = { target = FAO } 
				}
			}
			else_if = {
				limit = {
					OR = {
						GRN = {
							controls_province = 6287
						}
						FAO = {
							controls_province = 6287
						}
					}
				}
				white_peace = DEN
				white_peace = GRN
				white_peace = FAO
			}
		}
		timeout_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision timeout GRN_decision_independence_war"
			
			white_peace = DEN
			white_peace = GRN
			white_peace = FAO
		}
	}
}
