@government_ideology_drift_pool = 0.02
@government_ideology_drift_cap = 0.10
@dominant_outlook_support = 0.65
@dominant_outlook_acceptance = 30

update_coalition_diplomatic_acceptance = {
	set_current_outlook_id = yes
	set_outlook_popularity_array = yes
	clear_temp_array = coalition_seats_by_outlook
	resize_temp_array = { array = coalition_seats_by_outlook size = 5 }
	for_each_loop = {
		array = gov_coalition_array
		value = coalition_partner_slot
		set_temp_variable = { outlook_party_index = coalition_partner_slot }
		set_outlook_from_party_index = yes
		if = { limit = { NOT = { check_variable = { outlook_id = current_outlook_id } } }
			add_to_temp_variable = { coalition_seats_by_outlook^outlook_id = party_pop_elect_array^coalition_partner_slot }
		}
	}

	clear_temp_array = coalition_acceptance_by_outlook
	resize_temp_array = { array = coalition_acceptance_by_outlook size = 5 }
	for_loop_effect = {
		end = 5
		value = coalition_outlook_slot
		set_temp_variable = { coalition_seat_share = coalition_seats_by_outlook^coalition_outlook_slot }
		if = { limit = { check_variable = { outlook_popularity_by_id^coalition_outlook_slot > @dominant_outlook_support } }
			set_temp_variable = { coalition_acceptance_by_outlook^coalition_outlook_slot = @dominant_outlook_acceptance }
		}
		else_if = { limit = { check_variable = { coalition_seat_share > 0.01 } }
			set_temp_variable = {
				coalition_acceptance_by_outlook^coalition_outlook_slot = {
					value = 1
					add = { value = coalition_seat_share greater_than = 0.07 }
					add = { value = coalition_seat_share greater_than = 0.11 }
					add = { value = coalition_seat_share greater_than = 0.15 }
					add = { value = coalition_seat_share greater_than = 0.20 }
					add = { value = coalition_seat_share greater_than = 0.25 }
					multiply = 5
				}
			}
		}
	}
	set_variable = { gov_pop_democratic_acceptance = coalition_acceptance_by_outlook^0 }
	set_variable = { gov_pop_communism_acceptance = coalition_acceptance_by_outlook^1 }
	set_variable = { gov_pop_fascism_acceptance = coalition_acceptance_by_outlook^2 }
	set_variable = { gov_pop_neutrality_acceptance = coalition_acceptance_by_outlook^3 }
	set_variable = { gov_pop_nationalist_acceptance = coalition_acceptance_by_outlook^4 }
}

update_government_ideology_drift = {
	clear_temp_array = gov_drift_support
	resize_temp_array = { array = gov_drift_support size = 5 }
	set_temp_variable = { gov_drift_total_support = 0 }
	set_temp_variable = { gov_drift_ruler = -1 }
	set_current_outlook_id = yes
	if = {
		limit = { check_variable = { current_outlook_id > -1 } }
		if = {
			limit = { has_variable = ruling_party }
			set_temp_variable = { gov_drift_ruler = ruling_party }
			set_temp_variable = { gov_drift_total_support = { value = party_pop_array^gov_drift_ruler max = 0 } }
			set_temp_variable = { gov_drift_support^current_outlook_id = gov_drift_total_support }
		}
		for_each_loop = {
			array = gov_coalition_array
			value = gov_drift_partner
			if = {
				limit = {
					NOT = { check_variable = { gov_drift_partner = gov_drift_ruler } }
					check_variable = { party_pop_array^gov_drift_partner > 0 }
				}
				set_temp_variable = { outlook_party_index = gov_drift_partner }
				set_outlook_from_party_index = yes
				add_to_temp_variable = { gov_drift_support^outlook_id = party_pop_array^gov_drift_partner }
				add_to_temp_variable = { gov_drift_total_support = party_pop_array^gov_drift_partner }
			}
		}
		# Without polling data, keep all drift with the sitting government.
		if = {
			limit = { check_variable = { gov_drift_total_support = 0 } }
			set_temp_variable = { gov_drift_support^current_outlook_id = 1 }
			set_temp_variable = { gov_drift_total_support = 1 }
		}
	}
	clamp_temp_variable = { var = gov_drift_total_support min = 0.00001 }
	set_temp_variable = { gov_drift_pool = { value = @government_ideology_drift_pool min = @government_ideology_drift_cap } }
	set_variable = { gov_pop_democratic_drift = { value = gov_drift_support^0 divide = gov_drift_total_support multiply = gov_drift_pool } }
	set_variable = { gov_pop_communism_drift = { value = gov_drift_support^1 divide = gov_drift_total_support multiply = gov_drift_pool } }
	set_variable = { gov_pop_fascism_drift = { value = gov_drift_support^2 divide = gov_drift_total_support multiply = gov_drift_pool } }
	set_variable = { gov_pop_neutrality_drift = { value = gov_drift_support^3 divide = gov_drift_total_support multiply = gov_drift_pool } }
	set_variable = { gov_pop_nationalist_drift = { value = gov_drift_support^4 divide = gov_drift_total_support multiply = gov_drift_pool } }
	if = { limit = { NOT = { has_dynamic_modifier = { modifier = government_popularity_modifier } } }
		add_dynamic_modifier = { modifier = government_popularity_modifier }
	}
}

economic_cycle_drift_popularity = {
	hidden_effect = {
		set_temp_variable = { economic_cycle_drift_cycle_id = -1 }
		set_temp_variable = { economic_cycle_drift_outlook_id = -1 }

		if = {
			limit = { has_idea = economic_boom }
			set_temp_variable = { economic_cycle_drift_cycle_id = 4 }
		}
		else_if = {
			limit = { has_idea = fast_growth }
			set_temp_variable = { economic_cycle_drift_cycle_id = 3 }
		}
		else_if = {
			limit = { has_idea = stagnation }
			set_temp_variable = { economic_cycle_drift_cycle_id = 2 }
		}
		else_if = {
			limit = { has_idea = recession }
			set_temp_variable = { economic_cycle_drift_cycle_id = 1 }
		}
		else_if = {
			limit = { has_idea = depression }
			set_temp_variable = { economic_cycle_drift_cycle_id = 0 }
		}

		if = {
			limit = { has_government = democratic }
			set_temp_variable = { economic_cycle_drift_outlook_id = 0 }
		}
		else_if = {
			limit = { has_government = nationalist }
			set_temp_variable = { economic_cycle_drift_outlook_id = 1 }
		}
		else_if = {
			limit = { has_government = communism }
			set_temp_variable = { economic_cycle_drift_outlook_id = 2 }
		}
		else_if = {
			limit = { has_government = neutrality }
			set_temp_variable = { economic_cycle_drift_outlook_id = 3 }
		}
		else_if = {
			limit = { has_government = fascism }
			set_temp_variable = { economic_cycle_drift_outlook_id = 4 }
		}

		if = {
			limit = {
				check_variable = { economic_cycle_drift_cycle_id > -1 }
				check_variable = { economic_cycle_drift_outlook_id > -1 }
			}
			set_temp_variable = { economic_cycle_drift_index = economic_cycle_drift_cycle_id }
			multiply_temp_variable = { economic_cycle_drift_index = 5 }
			add_to_temp_variable = { economic_cycle_drift_index = economic_cycle_drift_outlook_id }
			set_temp_variable = {
				economic_cycle_drift_target_token = global.economic_cycle_drift_tokens^economic_cycle_drift_index
			}
		}

		if = {
			limit = { has_variable = economic_cycle_drift_applied_token }
			if = {
				limit = {
					OR = {
						check_variable = { economic_cycle_drift_cycle_id = -1 }
						check_variable = { economic_cycle_drift_outlook_id = -1 }
					}
				}
				remove_ideas = var:economic_cycle_drift_applied_token
				clear_variable = economic_cycle_drift_applied_token
			}
			else_if = {
				limit = {
					NOT = {
						check_variable = {
							economic_cycle_drift_applied_token = economic_cycle_drift_target_token
						}
					}
				}
				remove_ideas = var:economic_cycle_drift_applied_token
				clear_variable = economic_cycle_drift_applied_token
			}
		}

		if = {
			limit = {
				check_variable = { economic_cycle_drift_cycle_id > -1 }
				check_variable = { economic_cycle_drift_outlook_id > -1 }
				NOT = { has_variable = economic_cycle_drift_applied_token }
			}
			add_ideas = var:economic_cycle_drift_target_token
			set_variable = { economic_cycle_drift_applied_token = economic_cycle_drift_target_token }
		}
	}
}
