# Border War Mechanic
# Shared scripted effects for the RAJ-PAK and RAJ-CHI border conflict systems.
# Per-country: border_readiness, border_war_progress
# Global: border_tension

border_war_add_readiness = {
	set_temp_variable = { readiness_gain = border_war_add_readiness_amount }
	randomize_temp_variable = {
		var = readiness_gain
		min = -3
		max = 5
		integer = yes
	}
	add_to_variable = { border_readiness = readiness_gain }
	clamp_variable = { var = border_readiness min = 0 max = 100 }
}

border_war_add_tension = {
	set_temp_variable = { tension_gain = border_war_add_tension_amount }
	randomize_temp_variable = {
		var = tension_gain
		min = -5
		max = 8
		integer = yes
	}
	add_to_variable = { global.border_tension = tension_gain }
	clamp_variable = { var = global.border_tension min = 0 max = 100 }
}

border_war_add_progress = {
	set_temp_variable = { progress_gain = border_war_add_progress_amount }
	randomize_temp_variable = {
		var = progress_gain
		min = -2
		max = 4
		integer = yes
	}
	add_to_variable = { border_war_progress = progress_gain }
	clamp_variable = { var = border_war_progress min = 0 max = 100 }
}

border_war_subtract_progress = {
	set_temp_variable = { progress_loss = border_war_subtract_progress_amount }
	randomize_temp_variable = {
		var = progress_loss
		min = -2
		max = 4
		integer = yes
	}
	subtract_from_variable = { border_war_progress = progress_loss }
	clamp_variable = { var = border_war_progress min = 0 max = 100 }
}

# One RNG roll per month per country; replaces seven separate decay blocks.
border_war_decay = {
	if = {
		limit = {
			is_border_war_eligible = yes
		}
		randomize_temp_variable = {
			var = border_war_decay_roll
			min = 0
			max = 99
			integer = yes
		}
		if = {
			limit = { check_variable = { border_war_decay_roll < 30 } }
			subtract_from_variable = { border_readiness = 5 }
		}
		else_if = {
			limit = { check_variable = { border_war_decay_roll < 60 } }
			subtract_from_variable = { border_war_progress = 5 }
		}
		else = {
			subtract_from_variable = { global.border_tension = 5 }
		}
		clamp_variable = { var = border_readiness min = 0 max = 100 }
		clamp_variable = { var = border_war_progress min = 0 max = 100 }
		clamp_variable = { var = global.border_tension min = 0 max = 100 }
	}
}

# Set the recently-active flag so the exp-grind decisions become available.
