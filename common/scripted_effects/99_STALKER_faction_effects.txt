# Scientists follow exploitation and flee bandits; the military follows containment and resents Freedom.
STALKER_update_zone_factions = {
	set_temp_variable = {
		STALKER_scientist_target = {
			value = STALKER_zone_exploitation
			multiply = 0.4
			subtract = { value = STALKER_zone_bandits multiply = 0.3 }
			add = 10
			clamp = { min = 0 max = 100 }
		}
	}
	add_to_variable = { STALKER_zone_scientists = { value = STALKER_scientist_target subtract = STALKER_zone_scientists multiply = 0.1 } }
	set_temp_variable = {
		STALKER_military_target = {
			value = STALKER_zone_containment
			multiply = 0.5
			subtract = { value = STALKER_zone_freedom multiply = 0.2 }
			add = 10
			clamp = { min = 0 max = 100 }
		}
	}
	add_to_variable = { STALKER_zone_military = { value = STALKER_military_target subtract = STALKER_zone_military multiply = 0.1 } }
	STALKER_clamp_zone_society = yes
	# The Ward and the Spark exist once their own variables are set. A Ward charter must not create the Spark.
	if = {
		limit = { has_variable = STALKER_zone_ward }
		set_temp_variable = {
			STALKER_ward_target = {
				value = STALKER_zone_containment
				multiply = 0.4
				add = { value = STALKER_zone_scientists multiply = 0.2 }
				add = 10
				clamp = { min = 0 max = 100 }
			}
		}
		add_to_variable = { STALKER_zone_ward = { value = STALKER_ward_target subtract = STALKER_zone_ward multiply = 0.1 } }
		clamp_variable = { var = STALKER_zone_ward min = 0 max = 100 }
		add_to_variable = { STALKER_zone_bandits = { value = STALKER_zone_ward multiply = -0.02 } }
		STALKER_clamp_zone_society = yes
	}
	if = {
		limit = { has_variable = STALKER_zone_spark }
		set_temp_variable = {
			STALKER_spark_target = {
				value = STALKER_zone_activity
				multiply = 0.4
				subtract = { value = STALKER_zone_containment multiply = 0.3 }
				add = 10
				clamp = { min = 0 max = 100 }
			}
		}
		add_to_variable = { STALKER_zone_spark = { value = STALKER_spark_target subtract = STALKER_zone_spark multiply = 0.1 } }
		clamp_variable = { var = STALKER_zone_spark min = 0 max = 100 }
		add_to_variable = { STALKER_zone_activity = { value = STALKER_zone_spark multiply = 0.02 } }
		clamp_variable = { var = STALKER_zone_activity min = 0 max = 100 }
		STALKER_clamp_zone_society = yes
	}
	# Clear Sky studies the emissions and damps activity. From 2011 the dated assault owns its fall.
	if = {
		limit = {
			has_variable = STALKER_zone_clear_sky
			check_variable = { STALKER_zone_clear_sky > 0 }
		}
		set_temp_variable = {
			STALKER_clear_sky_target = {
				value = STALKER_zone_activity
				multiply = 0.3
				subtract = { value = STALKER_zone_bandits multiply = 0.2 }
				add = 10
				clamp = { min = 1 max = 100 }
			}
		}
		add_to_variable = { STALKER_zone_clear_sky = { value = STALKER_clear_sky_target subtract = STALKER_zone_clear_sky multiply = 0.1 } }
		clamp_variable = { var = STALKER_zone_clear_sky min = 0 max = 100 }
		add_to_variable = { STALKER_zone_activity = { value = STALKER_zone_clear_sky multiply = -0.02 } }
		clamp_variable = { var = STALKER_zone_activity min = 0 max = 100 }
		if = {
			limit = { NOT = { has_global_flag = STALKER_strelok_turned_back } }
			add_to_variable = { STALKER_zone_clear_sky_months = 1 }
			if = {
				limit = {
					check_variable = { STALKER_zone_clear_sky_months > 35 }
					NOT = { has_global_flag = STALKER_2011_dispatched }
					date < 2010.12.1
					controller = { exists = yes }
				}
				random = {
					chance = 10
					set_variable = { STALKER_zone_clear_sky = 0 }
					set_global_flag = STALKER_scar_survived
					STALKER_target_zone = yes
					controller = { country_event = STALKER.122 }
				}
			}
		}
	}
	# Noon, former Monolith soldiers, wears down its old brothers and draws on the stalkers around it.
	if = {
		limit = { has_variable = STALKER_zone_noon }
		set_temp_variable = {
			STALKER_noon_target = {
				value = STALKER_zone_stalkers
				multiply = 0.2
				add = 5
				clamp = { min = 0 max = 100 }
			}
		}
		add_to_variable = { STALKER_zone_noon = { value = STALKER_noon_target subtract = STALKER_zone_noon multiply = 0.1 } }
		clamp_variable = { var = STALKER_zone_noon min = 0 max = 100 }
		add_to_variable = { STALKER_zone_monolith = { value = STALKER_zone_noon multiply = -0.03 } }
		STALKER_clamp_zone_society = yes
	}
	if = {
		limit = { check_variable = { STALKER_zone_war_type > 0 } }
		STALKER_progress_zone_war = yes
	}
	else_if = {
		limit = {
			NOT = { has_state_flag = STALKER_zone_war_cooldown }
			controller = { exists = yes }
		}
		STALKER_check_zone_war_start = yes
	}
}

# Rivals go to war when both are strong and neither clearly leads: Duty and Freedom (1), the military and bandits (2),
# the Ward and the Spark (3), or Noon and the Monolith (4).
STALKER_check_zone_war_start = {
	set_temp_variable = { STALKER_war_gap = { value = STALKER_zone_duty subtract = STALKER_zone_freedom } }
	if = {
		limit = {
			check_variable = { STALKER_zone_duty > 34 }
			check_variable = { STALKER_zone_freedom > 34 }
			check_variable = { STALKER_war_gap < 10 }
			check_variable = { STALKER_war_gap > -10 }
		}
		random = {
			chance = 10
			set_variable = { STALKER_zone_war_type = 1 }
			STALKER_begin_zone_war = yes
		}
	}
	else = {
		set_temp_variable = { STALKER_war_gap = { value = STALKER_zone_military subtract = STALKER_zone_bandits } }
		if = {
			limit = {
				check_variable = { STALKER_zone_military > 34 }
				check_variable = { STALKER_zone_bandits > 34 }
				check_variable = { STALKER_war_gap < 10 }
				check_variable = { STALKER_war_gap > -10 }
			}
			random = {
				chance = 10
				set_variable = { STALKER_zone_war_type = 2 }
				STALKER_begin_zone_war = yes
			}
		}
		else_if = {
			limit = { has_variable = STALKER_zone_ward }
			set_temp_variable = { STALKER_war_gap = { value = STALKER_zone_ward subtract = STALKER_zone_spark } }
			if = {
				limit = {
					check_variable = { STALKER_zone_ward > 34 }
					check_variable = { STALKER_zone_spark > 34 }
					check_variable = { STALKER_war_gap < 10 }
					check_variable = { STALKER_war_gap > -10 }
				}
				random = {
					chance = 10
					set_variable = { STALKER_zone_war_type = 3 }
					STALKER_begin_zone_war = yes
				}
			}
		}
	}
	# Noon runs smaller than the other factions, so its war with the Monolith needs less strength to start.
	if = {
		limit = {
			check_variable = { STALKER_zone_war_type = 0 }
			has_variable = STALKER_zone_noon
			check_variable = { STALKER_zone_noon > 24 }
			check_variable = { STALKER_zone_monolith > 24 }
		}
		set_temp_variable = { STALKER_war_gap = { value = STALKER_zone_noon subtract = STALKER_zone_monolith } }
		if = {
			limit = {
				check_variable = { STALKER_war_gap < 15 }
				check_variable = { STALKER_war_gap > -15 }
			}
			random = {
				chance = 10
				set_variable = { STALKER_zone_war_type = 4 }
				STALKER_begin_zone_war = yes
			}
		}
	}
}

STALKER_begin_zone_war = {
	set_variable = { STALKER_zone_war_stage = 1 }
	log = "[GetDateText]: STALKER war type [?STALKER_zone_war_type] begins in [THIS.GetName]"
	STALKER_target_zone = yes
	controller = { country_event = STALKER.120 }
}

# Each war month bleeds the Zone; every other month on average the war moves on, and after stage 3 it is decided.
STALKER_progress_zone_war = {
	add_to_variable = { STALKER_zone_stalkers = -3 }
	add_to_variable = { STALKER_zone_containment = -3 }
	STALKER_clamp_zone_society = yes
	random = {
		chance = 50
		add_to_variable = { STALKER_zone_war_stage = 1 }
	}
	if = {
		limit = { check_variable = { STALKER_zone_war_stage > 3 } }
		STALKER_resolve_zone_war = yes
	}
}

STALKER_resolve_zone_war = {
	set_temp_variable = { STALKER_war_winner = 2 }
	if = {
		limit = {
			check_variable = { STALKER_zone_war_type = 1 }
			check_variable = { STALKER_zone_duty > STALKER_zone_freedom }
		}
		set_temp_variable = { STALKER_war_winner = 1 }
	}
	else_if = {
		limit = {
			check_variable = { STALKER_zone_war_type = 2 }
			check_variable = { STALKER_zone_military > STALKER_zone_bandits }
		}
		set_temp_variable = { STALKER_war_winner = 1 }
	}
	else_if = {
		limit = {
			check_variable = { STALKER_zone_war_type = 3 }
			check_variable = { STALKER_zone_ward > STALKER_zone_spark }
		}
		set_temp_variable = { STALKER_war_winner = 1 }
	}
	else_if = {
		limit = {
			check_variable = { STALKER_zone_war_type = 4 }
			check_variable = { STALKER_zone_noon > STALKER_zone_monolith }
		}
		set_temp_variable = { STALKER_war_winner = 1 }
	}
	if = {
		limit = { check_variable = { STALKER_zone_war_type = 1 } }
		if = {
			limit = { check_variable = { STALKER_war_winner = 1 } }
			if = {
				limit = { check_variable = { STALKER_zone_id = 1 } }
				add_to_variable = { STALKER_zone_duty = 15 }
				add_to_variable = { STALKER_zone_freedom = -25 }
			}
		}
		else = {
			if = {
				limit = { check_variable = { STALKER_zone_id = 1 } }
				add_to_variable = { STALKER_zone_freedom = 15 }
				add_to_variable = { STALKER_zone_duty = -25 }
			}
		}
	}
	else_if = {
		limit = { check_variable = { STALKER_zone_war_type = 4 } }
		if = {
			limit = { check_variable = { STALKER_war_winner = 1 } }
			add_to_variable = { STALKER_zone_noon = 15 }
			add_to_variable = { STALKER_zone_monolith = -25 }
		}
		else = {
			add_to_variable = { STALKER_zone_monolith = 15 }
			add_to_variable = { STALKER_zone_noon = -25 }
		}
		clamp_variable = { var = STALKER_zone_noon min = 0 max = 100 }
	}
	else_if = {
		limit = { check_variable = { STALKER_zone_war_type = 3 } }
		if = {
			limit = { check_variable = { STALKER_war_winner = 1 } }
			add_to_variable = { STALKER_zone_ward = 15 }
			add_to_variable = { STALKER_zone_spark = -25 }
		}
		else = {
			add_to_variable = { STALKER_zone_spark = 15 }
			add_to_variable = { STALKER_zone_ward = -25 }
		}
		clamp_variable = { var = STALKER_zone_ward min = 0 max = 100 }
		clamp_variable = { var = STALKER_zone_spark min = 0 max = 100 }
	}
	else = {
		if = {
			limit = { check_variable = { STALKER_war_winner = 1 } }
			add_to_variable = { STALKER_zone_military = 15 }
			add_to_variable = { STALKER_zone_bandits = -25 }
		}
		else = {
			add_to_variable = { STALKER_zone_bandits = 15 }
			add_to_variable = { STALKER_zone_military = -25 }
		}
	}
	STALKER_clamp_zone_society = yes
	set_variable = { STALKER_zone_war_winner = STALKER_war_winner }
	set_variable = { STALKER_zone_last_war_type = STALKER_zone_war_type }
	set_state_flag = { flag = STALKER_zone_war_cooldown value = 1 days = 730 }
	STALKER_target_zone = yes
	if = {
		limit = { controller = { exists = yes } }
		controller = { country_event = STALKER.121 }
	}
	set_variable = { STALKER_zone_war_type = 0 }
	set_variable = { STALKER_zone_war_stage = 0 }
}

# Beating the Monolith back can break some fighters' conditioning; the first defectors found Noon.
STALKER_monolith_defections = {
	random = {
		chance = 50
		if = {
			limit = {
				NOT = { has_variable = STALKER_zone_noon }
				controller = { exists = yes }
			}
			STALKER_target_zone = yes
			controller = { country_event = STALKER.123 }
		}
		add_to_variable = { STALKER_zone_noon = 10 }
		clamp_variable = { var = STALKER_zone_noon min = 0 max = 100 }
	}
}

# ---- 99_STALKER_faction_wars_effects.txt (consolidated) ----
# GSC's pre-launch Faction Wars Discord event (November 2024): SIRCAA asked every faction to bring in Tearlings,
# thanked the faction that gathered the most, and paid every faction a share.
STALKER_monthly_faction_wars_pulse = {
	if = {
		limit = {
			STALKER_scenario_enabled = yes
			check_variable = { global.STALKER_active_zone_anchors^num > 0 }
		}
		for_each_scope_loop = {
			array = global.STALKER_active_zone_anchors
			if = {
				limit = {
					STALKER_is_active_zone_anchor = yes
					has_variable = STALKER_zone_ward
				}
				if = {
					limit = { has_variable = STALKER_zone_tearling_months }
					STALKER_gather_tearlings = yes
				}
				else_if = {
					limit = {
						has_global_flag = STALKER_2022_outcome_resolved
						NOT = { has_state_flag = STALKER_tearling_cooldown }
						controller = { exists = yes }
					}
					random = {
						chance = 5
						STALKER_start_tearling_drive = yes
					}
				}
			}
		}
	}
}

STALKER_start_tearling_drive = {
	set_variable = { STALKER_zone_tearling_months = 3 }
	set_variable = { STALKER_zone_tearling_backed = 0 }
	clear_variable = STALKER_zone_tearling_backer
	set_variable = { STALKER_zone_tearlings_stalkers = 0 }
	set_variable = { STALKER_zone_tearlings_duty = 0 }
	set_variable = { STALKER_zone_tearlings_freedom = 0 }
	set_variable = { STALKER_zone_tearlings_scientists = 0 }
	set_variable = { STALKER_zone_tearlings_bandits = 0 }
	STALKER_target_zone = yes
	controller = { country_event = STALKER.150 }
}

# Each faction gathers in proportion to its strength, with luck; the faction the controller backs gathers 5 more.
STALKER_gather_tearlings = {
	set_temp_variable_to_random = { var = STALKER_tearling_roll min = 0 max = 6 integer = yes }
	add_to_variable = { STALKER_zone_tearlings_stalkers = { value = STALKER_zone_stalkers multiply = 0.2 add = STALKER_tearling_roll } }
	if = {
		limit = { check_variable = { STALKER_zone_id = 1 } }
		set_temp_variable_to_random = { var = STALKER_tearling_roll min = 0 max = 6 integer = yes }
		add_to_variable = { STALKER_zone_tearlings_duty = { value = STALKER_zone_duty multiply = 0.2 add = STALKER_tearling_roll } }
		set_temp_variable_to_random = { var = STALKER_tearling_roll min = 0 max = 6 integer = yes }
		add_to_variable = { STALKER_zone_tearlings_freedom = { value = STALKER_zone_freedom multiply = 0.2 add = STALKER_tearling_roll } }
	}
	set_temp_variable_to_random = { var = STALKER_tearling_roll min = 0 max = 6 integer = yes }
	add_to_variable = { STALKER_zone_tearlings_scientists = { value = STALKER_zone_scientists multiply = 0.2 add = STALKER_tearling_roll } }
	set_temp_variable_to_random = { var = STALKER_tearling_roll min = 0 max = 6 integer = yes }
	add_to_variable = { STALKER_zone_tearlings_bandits = { value = STALKER_zone_bandits multiply = 0.2 add = STALKER_tearling_roll } }
	if = {
		limit = { check_variable = { STALKER_zone_tearling_backed = 1 } }
		add_to_variable = { STALKER_zone_tearlings_stalkers = 5 }
	}
	else_if = {
		limit = {
			check_variable = { STALKER_zone_id = 1 }
			check_variable = { STALKER_zone_tearling_backed = 2 }
		}
		add_to_variable = { STALKER_zone_tearlings_duty = 5 }
	}
	else_if = {
		limit = {
			check_variable = { STALKER_zone_id = 1 }
			check_variable = { STALKER_zone_tearling_backed = 3 }
		}
		add_to_variable = { STALKER_zone_tearlings_freedom = 5 }
	}
	else_if = {
		limit = { check_variable = { STALKER_zone_tearling_backed = 4 } }
		add_to_variable = { STALKER_zone_tearlings_scientists = 5 }
	}
	add_to_variable = { STALKER_zone_tearling_months = -1 }
	if = {
		limit = { check_variable = { STALKER_zone_tearling_months < 1 } }
		STALKER_resolve_tearling_drive = yes
	}
}

# Leader codes: 1 loners, 2 Duty, 3 Freedom, 4 Ecologists, 5 bandits. Ties go to the earlier code.
STALKER_resolve_tearling_drive = {
	round_variable = STALKER_zone_tearlings_stalkers
	round_variable = STALKER_zone_tearlings_duty
	round_variable = STALKER_zone_tearlings_freedom
	round_variable = STALKER_zone_tearlings_scientists
	round_variable = STALKER_zone_tearlings_bandits
	set_variable = { STALKER_zone_tearling_leader = 1 }
	set_variable = { STALKER_zone_tearling_best = STALKER_zone_tearlings_stalkers }
	if = {
		limit = { check_variable = { STALKER_zone_tearlings_duty > STALKER_zone_tearling_best } }
		set_variable = { STALKER_zone_tearling_leader = 2 }
		set_variable = { STALKER_zone_tearling_best = STALKER_zone_tearlings_duty }
	}
	if = {
		limit = { check_variable = { STALKER_zone_tearlings_freedom > STALKER_zone_tearling_best } }
		set_variable = { STALKER_zone_tearling_leader = 3 }
		set_variable = { STALKER_zone_tearling_best = STALKER_zone_tearlings_freedom }
	}
	if = {
		limit = { check_variable = { STALKER_zone_tearlings_scientists > STALKER_zone_tearling_best } }
		set_variable = { STALKER_zone_tearling_leader = 4 }
		set_variable = { STALKER_zone_tearling_best = STALKER_zone_tearlings_scientists }
	}
	if = {
		limit = { check_variable = { STALKER_zone_tearlings_bandits > STALKER_zone_tearling_best } }
		set_variable = { STALKER_zone_tearling_leader = 5 }
		set_variable = { STALKER_zone_tearling_best = STALKER_zone_tearlings_bandits }
	}
	add_to_variable = { STALKER_zone_stalkers = 3 }
	if = {
		limit = { check_variable = { STALKER_zone_id = 1 } }
		add_to_variable = { STALKER_zone_duty = 3 }
		add_to_variable = { STALKER_zone_freedom = 3 }
	}
	add_to_variable = { STALKER_zone_scientists = 3 }
	add_to_variable = { STALKER_zone_bandits = 3 }
	if = {
		limit = { check_variable = { STALKER_zone_tearling_leader = 1 } }
		add_to_variable = { STALKER_zone_stalkers = 10 }
	}
	else_if = {
		limit = { check_variable = { STALKER_zone_tearling_leader = 2 } }
		if = {
			limit = { check_variable = { STALKER_zone_id = 1 } }
			add_to_variable = { STALKER_zone_duty = 10 }
		}
	}
	else_if = {
		limit = { check_variable = { STALKER_zone_tearling_leader = 3 } }
		if = {
			limit = { check_variable = { STALKER_zone_id = 1 } }
			add_to_variable = { STALKER_zone_freedom = 10 }
		}
	}
	else_if = {
		limit = { check_variable = { STALKER_zone_tearling_leader = 4 } }
		add_to_variable = { STALKER_zone_scientists = 10 }
	}
	else = {
		add_to_variable = { STALKER_zone_bandits = 10 }
	}
	add_to_variable = { STALKER_zone_ward = 5 }
	clamp_variable = { var = STALKER_zone_ward min = 0 max = 100 }
	STALKER_clamp_zone_society = yes
	set_state_flag = { flag = STALKER_tearling_cooldown value = 1 days = 730 }
	set_variable = {
		global.STALKER_tearlings_total = {
			value = STALKER_zone_tearlings_stalkers
			add = STALKER_zone_tearlings_duty
			add = STALKER_zone_tearlings_freedom
			add = STALKER_zone_tearlings_scientists
			add = STALKER_zone_tearlings_bandits
		}
	}
	set_variable = { global.STALKER_tearlings_best = STALKER_zone_tearling_best }
	STALKER_target_zone = yes
	log = "[GetDateText]: Tearling exchange resolved in [THIS.GetName]: leader [?STALKER_zone_tearling_leader], total [?global.STALKER_tearlings_total]"
	if = {
		limit = { controller = { exists = yes } }
		controller = { country_event = STALKER.151 }
	}
	if = {
		limit = {
			has_variable = STALKER_zone_tearling_backer
			var:STALKER_zone_tearling_backer = { exists = yes }
		}
		if = {
			limit = { check_variable = { STALKER_zone_tearling_backed = STALKER_zone_tearling_leader } }
			var:STALKER_zone_tearling_backer = { add_political_power = 50 }
		}
		var:STALKER_zone_tearling_backer = {
			if = {
				limit = { NOT = { controls_state = PREV } }
				country_event = STALKER.151
			}
		}
	}
	clear_variable = STALKER_zone_tearling_months
	clear_variable = STALKER_zone_tearlings_stalkers
	clear_variable = STALKER_zone_tearlings_duty
	clear_variable = STALKER_zone_tearlings_freedom
	clear_variable = STALKER_zone_tearlings_scientists
	clear_variable = STALKER_zone_tearlings_bandits
	clear_variable = STALKER_zone_tearling_best
}

# ---- 99_STALKER_mutant_effects.txt (consolidated) ----
# Mutants drift toward a level set by activity against containment, and prey on the stalkers already inside.
STALKER_update_zone_mutants = {
	set_temp_variable = {
		STALKER_mutant_target = {
			value = STALKER_zone_activity
			subtract = STALKER_zone_containment
			multiply = 0.4
			add = 15
			clamp = { min = 0 max = 100 }
		}
	}
	add_to_variable = { STALKER_zone_mutants = { value = STALKER_mutant_target subtract = STALKER_zone_mutants multiply = 0.1 } }
	clamp_variable = { var = STALKER_zone_mutants min = 0 max = 100 }
	add_to_variable = { STALKER_zone_stalkers = { value = STALKER_zone_mutants multiply = -0.05 } }
	STALKER_clamp_zone_society = yes
	if = {
		limit = {
			check_variable = { STALKER_zone_mutants > 49 }
			NOT = { has_state_flag = STALKER_mutant_breach_cooldown }
			controller = { exists = yes }
		}
		random = {
			chance = 5
			if = {
				limit = { has_state_flag = STALKER_fortified_perimeter }
				random = {
					chance = 50
					STALKER_mutant_breach = yes
				}
			}
			else = {
				STALKER_mutant_breach = yes
			}
		}
	}
	if = {
		limit = { controller = { exists = yes } }
		if = {
			limit = {
				check_variable = { STALKER_zone_mutants > 59 }
				NOT = { has_state_flag = STALKER_pseudogiant_cooldown }
			}
			random = {
				chance = 5
				set_state_flag = { flag = STALKER_pseudogiant_cooldown value = 1 days = 365 }
				STALKER_target_zone = yes
				controller = { country_event = STALKER.93 }
			}
		}
		if = {
			limit = {
				check_variable = { STALKER_zone_mutants > 39 }
				check_variable = { STALKER_zone_activity > 29 }
				NOT = { has_state_flag = STALKER_controller_surge_cooldown }
			}
			random = {
				chance = 5
				set_state_flag = { flag = STALKER_controller_surge_cooldown value = 1 days = 365 }
				STALKER_target_zone = yes
				controller = { country_event = STALKER.92 }
			}
		}
		if = {
			limit = {
				check_variable = { STALKER_zone_mutants > 29 }
				NOT = { has_state_flag = STALKER_bloodsucker_lair_cooldown }
			}
			random = {
				chance = 5
				set_state_flag = { flag = STALKER_bloodsucker_lair_cooldown value = 1 days = 365 }
				STALKER_target_zone = yes
				controller = { country_event = STALKER.91 }
			}
		}
		if = {
			limit = {
				check_variable = { STALKER_zone_exploitation > 25 }
				check_variable = { STALKER_zone_mutants > 20 }
				NOT = { has_state_flag = STALKER_bio_harvest_cooldown }
			}
			random = {
				chance = 5
				set_state_flag = { flag = STALKER_bio_harvest_cooldown value = 1 days = 365 }
				STALKER_target_zone = yes
				controller = { country_event = STALKER.94 }
			}
		}
	}
}

# A breach needs a neighbouring state the Zone's controller also holds; without one, nothing happens and no cooldown starts.
STALKER_mutant_breach = {
	controller = { save_event_target_as = STALKER_mutant_controller }
	STALKER_target_zone = yes
	random_neighbor_state = {
		limit = {
			is_controlled_by = event_target:STALKER_mutant_controller
			NOT = { has_dynamic_modifier = { modifier = STALKER_mutant_incursion } }
		}
		add_dynamic_modifier = { modifier = STALKER_mutant_incursion days = 60 }
		PREV = { set_state_flag = { flag = STALKER_mutant_breach_cooldown value = 1 days = 365 } }
		event_target:STALKER_mutant_controller = { country_event = STALKER.90 }
	}
}

# ---- 99_STALKER_mdst_effects.txt (consolidated) ----
# The wider-network reveal shows how deep MDST already reaches; nobody sees it before then.
STALKER_expose_mdst_penetration = {
	if = {
		limit = { NOT = { has_variable = STALKER_mdst_penetration } }
		set_temp_variable = { STALKER_mdst_roll = 0 }
		randomize_temp_variable = { var = STALKER_mdst_roll min = 20 max = 50 distribution = uniform }
		set_variable = { STALKER_mdst_penetration = { value = STALKER_mdst_roll round = yes } }
		add_to_array = { global.STALKER_mdst_aware_countries = THIS }
	}
}

# An unchecked network recruits a little each month and steals research in proportion to its reach.
STALKER_monthly_mdst_pulse = {
	for_each_scope_loop = {
		array = global.STALKER_mdst_aware_countries
		if = {
			limit = {
				exists = yes
				has_variable = STALKER_mdst_penetration
			}
			add_to_variable = { STALKER_mdst_penetration = 1 }
			clamp_variable = { var = STALKER_mdst_penetration min = 0 max = 80 }
			if = {
				limit = {
					check_variable = { STALKER_mdst_penetration > 9 }
					NOT = { has_country_flag = STALKER_mdst_theft_cooldown }
				}
				set_temp_variable = { STALKER_mdst_theft_chance = { value = STALKER_mdst_penetration divide = 5 } }
				random = {
					chance = STALKER_mdst_theft_chance
					set_country_flag = { flag = STALKER_mdst_theft_cooldown value = 1 days = 180 }
					country_event = STALKER.80
				}
			}
		}
	}
}

# ---- 99_STALKER_xlab_effects.txt (consolidated) ----
# Investigation knowledge belongs to the government that funds it, so progress lives on ROOT, not the Zone.
STALKER_run_xlab_expedition = {
	ROOT = {
		add_to_variable = {
			STALKER_xlab_progress = {
				value = PREV.STALKER_zone_stalkers
				add = PREV.STALKER_zone_exploitation
				multiply = 0.1
				add = 20
				subtract = { value = PREV.STALKER_zone_monolith multiply = 0.2 }
				add = { value = PREV.STALKER_zone_scientists multiply = 0.1 }
				clamp = { min = 5 max = 100 }
			}
		}
	}
	random = {
		chance = 25
		add_to_variable = { STALKER_zone_stalkers = -5 }
		STALKER_clamp_zone_society = yes
	}
	if = {
		limit = {
			ROOT = {
				check_variable = { STALKER_xlab_progress > 99.99 }
				check_variable = { STALKER_xlab_stage < 4 }
			}
		}
		STALKER_target_zone = yes
		ROOT = { STALKER_advance_xlab_stage = yes }
	}
}

STALKER_advance_xlab_stage = {
	set_variable = { STALKER_xlab_progress = 0 }
	add_to_variable = { STALKER_xlab_stage = 1 }
	add_to_variable = { STALKER_xlab_research_factor = 0.01 }
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = STALKER_xlab_findings } } }
		add_dynamic_modifier = { modifier = STALKER_xlab_findings }
	}
	if = {
		limit = { check_variable = { STALKER_xlab_stage < 4 } }
		add_to_variable = { STALKER_artifact_stockpile = 5 }
	}
	if = {
		limit = { check_variable = { STALKER_xlab_stage = 1 } }
		country_event = STALKER.10
	}
	else_if = {
		limit = { check_variable = { STALKER_xlab_stage = 2 } }
		country_event = STALKER.11
	}
	else_if = {
		limit = { check_variable = { STALKER_xlab_stage = 3 } }
		country_event = STALKER.12
	}
	else = {
		STALKER_expose_mdst_penetration = yes
		country_event = STALKER.13
	}
}
