SYR_decrease_divided_syria = {
	if = { limit = { has_idea = divided_syria_3 }
		remove_ideas = divided_syria_3
	}
	else_if = { limit = { has_idea = divided_syria_2 }
		swap_ideas = {
			add_idea = divided_syria_3
			remove_idea = divided_syria_2
		}
	}
	else_if = { limit = { has_idea = divided_syria_1 }
		swap_ideas = {
			add_idea = divided_syria_2
			remove_idea = divided_syria_1
		}
	}
	else_if = { limit = { has_idea = divided_syria }
		swap_ideas = {
			add_idea = divided_syria_1
			remove_idea = divided_syria
		}
	}
}

# State scope: adds the state's infrastructure and internet to minority_status
SYR_add_minority_state_development = {
	if = {
		limit = { is_owned_by = SYR }
		add_to_temp_variable = {
			var = minority_status
			value = {
				value = infrastructure_level
				multiply = 2.5
				add = {
					value = building_level@internet_station
					multiply = 1.5
				}
			}
		}
	}
}

SYR_alawite_political_activity_modifier = {
	set_temp_variable = {
		minority_status = {
			value = 10
			subtract = corruption_level
			multiply = 2.5
		}
	}
	SYR_add_religious_freedom_terms = yes
	189 = { SYR_add_minority_state_development = yes }
	158 = { SYR_add_minority_state_development = yes }
	SYR_add_federalization_bonus = yes
	add_to_temp_variable = { minority_status = SYR_alawite_trust }
	if = { limit = { has_decision = SYR_reassure_the_alawite_coast }
		add_to_temp_variable = { minority_status = 10 }
	}
	if = { limit = { has_country_flag = SYR_alawite_reserved_seats }
		add_to_temp_variable = { minority_status = 30 }
	}

	#assads in power effect
	if = {
		limit = { NOT = { has_completed_focus = SYR_kick_out_al_assads } }
		add_to_temp_variable = { minority_status = 40 }
	}

	set_variable = {
		alawite_status = {
			value = minority_status
			clamp = { min = 0 max = 100 }
		}
	}
	set_variable = {
		alawite_status_stability_modifier = {
			value = alawite_status
			subtract = 50
			multiply = 0.001
		}
	}
	set_variable = {
		alawite_status_drift_defense = {
			value = alawite_status
			subtract = 50
			multiply = -0.002
		}
	}

	# The has_idea pair mirrors the modifier's remove_trigger, so a federalised Syria does not get it re-attached
	if = {
		limit = {
			any_owned_state = { is_core_of = ALA }
			NOT = { has_idea = syria_federal_government }
			NOT = { has_idea = syria_bureaucratic_government }
			NOT = { has_dynamic_modifier = { modifier = alawite_political_inclusiveness_dynamic_modifier } }
		}
		add_dynamic_modifier = { modifier = alawite_political_inclusiveness_dynamic_modifier }
	}
	else_if = {
		limit = {
			has_dynamic_modifier = { modifier = alawite_political_inclusiveness_dynamic_modifier }
			NOT = { any_owned_state = { is_core_of = ALA } }
		}
		remove_dynamic_modifier = { modifier = alawite_political_inclusiveness_dynamic_modifier }
	}
}

# A fallen Islamic State returns six months to a year later, at most twice, against whoever holds the desert by then
SYR_isis_resurgence_tick = {
	if = {
		limit = {
			has_global_flag = ISIL_declared
			NOT = { country_exists = ISI }
			NOT = { check_variable = { global.isis_resurgences > 1 } }
		}
		if = {
			limit = { NOT = { has_global_flag = ISIS_resurgence_queued } }
			set_global_flag = ISIS_resurgence_queued
		}
		else_if = {
			limit = { has_global_flag = { flag = ISIS_resurgence_queued days > 179 } }
			random = {
				chance = 25
				192 = { OWNER = { country_event = { id = SyriaCivilWar.40 days = 1 random_days = 10 } } }
			}
		}
	}
	# The six months count from the fall, not from a flag set while the Islamic State still stood
	else_if = {
		limit = {
			country_exists = ISI
			has_global_flag = ISIS_resurgence_queued
		}
		clr_global_flag = ISIS_resurgence_queued
	}
}

# Effect: SYR_defectors_join_fsa
# Purpose: Turns a wave of regime defectors into new FSA brigades at the FSA capital, carrying their rifles and anti-tank weapons
# Parameters:
# - SYR_defector_count: brigades that reach the FSA
SYR_defectors_join_fsa = {
	set_temp_variable = { defector_manpower = { value = SYR_defector_count multiply = -1500 } }
	set_temp_variable = { defector_rifles = { value = SYR_defector_count multiply = 600 } }
	set_temp_variable = { defector_at = { value = SYR_defector_count multiply = 60 } }
	set_temp_variable = { defector_rifles_lost = { value = defector_rifles multiply = -1 } }
	set_temp_variable = { defector_at_lost = { value = defector_at multiply = -1 } }
	add_manpower = defector_manpower
	add_equipment_to_stockpile = { type = infantry_weapons_type amount = defector_rifles_lost }
	add_equipment_to_stockpile = { type = L_AT_Equipment amount = defector_at_lost }
	FSA = {
		if = {
			limit = { NOT = { has_template = "Defected Army Brigade" } }
			division_template = {
				name = "Defected Army Brigade"
				is_locked = yes
				division_names_group = SYR_FSA_UNITS
				regiments = {
					Mot_Inf_Bat = { x = 0 y = 0 }
					Mot_Inf_Bat = { x = 0 y = 1 }
					Mot_Inf_Bat = { x = 0 y = 2 }
				}
				support = {
					L_Engi_Comp = { x = 0 y = 0 }
				}
			}
		}
		add_equipment_to_stockpile = { type = infantry_weapons_type amount = defector_rifles producer = SYR }
		add_equipment_to_stockpile = { type = L_AT_Equipment amount = defector_at producer = SYR }
		set_temp_variable = { defectors_left = SYR_defector_count }
		while_loop_effect = {
			limit = { check_variable = { defectors_left > 0 } }
			capital_scope = {
				create_unit = {
					division = "name = \"Defected Army Brigade\" division_template = \"Defected Army Brigade\" start_experience_factor = 0.5 start_equipment_factor = 0.7"
					owner = FSA
				}
			}
			subtract_from_temp_variable = { defectors_left = 1 }
		}
	}
}
