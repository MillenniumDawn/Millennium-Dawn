###########SHERIFF UP - CHANGE EFFECT###########
sheriff_holding_change_positive = {
	if = {
		limit = {
			check_variable = { PMR_sheriff_level > 6 }
			NOT = { has_country_flag = PMR_sheriff_finish }
		}
		add_popularity = {
			ideology = neutrality
			popularity = 0.07
		}
		set_temp_variable = { party_index = 15 }
		set_temp_variable = { party_popularity_increase = 0.07 }
		change_relative_party_popularity = yes
		add_political_power = 150
	}
	else_if = {
		limit = {
			NOT = { has_country_flag = PMR_sheriff_finish }
		}
		add_to_variable = { PMR_sheriff_level = 1 }
		custom_effect_tooltip = PMR_sheriff_grows_tt
		PMR_sheriff_apply_tier = yes
	}
}

PMR_partner_withdrawal_check = {
	if = {
		limit = {
			has_completed_focus = PMR_ge_invest
			NOT = { has_country_flag = PMR_ger_invest }
			OR = {
				has_war_with = GER
				NOT = { country_exists = GER }
			}
		}
		set_country_flag = PMR_ger_invest
		random_owned_state = {
			limit = { industrial_complex > 0 }
			remove_building = {
				type = industrial_complex
				level = 1
			}
		}
	}
	if = {
		limit = {
			has_completed_focus = PMR_aus_invest
