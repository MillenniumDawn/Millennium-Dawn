		set_variable = { comp_check = 1 }
	}
	else_if = { limit = { has_country_flag = md_sel }
		set_variable = { comp_check = 4 }
	}
}

un_recog_setup_check = {
	if = {
		limit = {
			OR = {
				has_idea = Lacks_International_Recognition
				has_idea = Non_State_Actor
			}
		}
		if = {
			limit = {
				NOT = {
					is_in_array = { global.Unrecognised_States = THIS }
					THIS = { has_idea = Lacks_International_Recognition }
				}
			}
			add_to_array = { global.Unrecognised_States = THIS }
		} else = {
			if = {
				limit = { is_in_array = { global.Unrecognised_States = THIS } }
				remove_from_array = { global.Unrecognised_States = THIS }
			}
			add_to_array = {
				array = global.Unrecognised_States
				value = THIS
				index = 0
			}
		}
	}else = {
		add_to_array = { global.UN_general_assembly = THIS }
	}
}

				NOT = { has_idea = sc_economic_sanctions }
			}
			add_to_array = { global.can_be_sc_sanctioned = THIS }
		}
	}
}

security_council_vote_finished = {
	if = {
		limit = {
			has_passed_sc_vote = yes
		}
		if = {
			limit = {
				check_variable = { global.current_sc_vote_type = 5 }
			}
			push_next_vote_to_ga = yes
		}
		if = {
			limit = {
				check_variable = { global.current_sc_vote_type = 1 }
			}
		}
		if = {
			limit = {
				check_variable = { global.current_sc_vote_type = 2 }
			}
		}
		if = {
			limit = {
				check_variable = { global.current_sc_vote_type = 3 }
			}
		}
		if = {
			limit = {
				check_variable = { global.current_sc_vote_type = 4 }
			}
		}
	} else = {
		news_event = recognition.122
	}

	#clear all arrays and variables
	clear_array = global.security_council_vote_yes
	clear_array = global.security_council_vote_no
	clear_array = global.security_council_vote_abstain
	clear_variable = global.current_sc_vote_type
	clear_variable = global.current_sc_vote_nation
	clr_global_flag = current_ongoing_sc_vote

	if = {
		limit = {
			has_global_flag = security_council_veto
		}
		clr_global_flag = security_council_veto
	}

	set_global_flag = {
		flag = un_sc_vote_cooldown
		days = 8
		value = 1
	}

	missile_ui_update = yes
}

push_next_vote_to_ga = {
	if = {
		limit = {
			OR = {
				has_global_flag = current_ongoing_ga_vote
				has_global_flag = un_ga_vote_cooldown
				check_variable = { global.ga_vote_type_tracker^num > 0 }
			}
		}
		add_to_array = { global.ga_vote_type_tracker = global.current_sc_vote_type }
		add_to_array = { global.ga_vote_nation_tracker = global.current_sc_vote_nation }
	} else = {
		set_variable = { global.current_ga_vote_type = global.current_sc_vote_type }
		set_variable = { global.current_ga_vote_nation = global.current_sc_vote_nation }

		if = {
			limit = {
				check_variable = { global.current_ga_vote_type = 5 }
			}
			var:global.current_ga_vote_nation = {
				country_event = recognition.13
			}
		}
	}
}

general_assembly_vote_finished = {
	#Do the actions based on the type of vote
	#Add nation to the UN
	if = {
		limit = {
			check_variable = { global.general_assembly_vote_yes^num > global.general_assembly_vote_no^num }
		}
		if = {
			limit = {
				check_variable = { global.current_ga_vote_type = 5 }
				un_ga_2_3_required = yes
			}
			var:global.current_ga_vote_nation = {
				country_event = recognition.71
			}
		} else = {
			var:global.current_ga_vote_nation = {
				country_event = recognition.72
			}
		}
	} else = {
		news_event = recognition.73
	}
	#Propose non-binding resolution
	#This will be an idea that players, after voting, can choose to accept or not
	#NO votes who accept will get an opinion boost, those who refuse get debuffs


	#clear all arrays and variables
	clear_array = global.general_assembly_vote_yes
	clear_array = global.general_assembly_vote_no
	clear_array = global.general_assembly_vote_abstain
	clear_variable = global.current_ga_vote_type
	clear_variable = global.current_ga_vote_nation
	clr_global_flag = current_ongoing_ga_vote

	set_global_flag = {
		flag = un_ga_vote_cooldown
		days = 50
		value = 1
	}
	missile_ui_update = yes
}
