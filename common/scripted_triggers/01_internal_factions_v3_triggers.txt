# Opinion tiers on if_opinion^if_id
if_tier_hostile = {
	check_variable = { if_opinion^if_id < 20 }
}
if_tier_negative = {
	check_variable = { var = if_opinion^if_id value = 20 compare = greater_than_or_equals }
	check_variable = { if_opinion^if_id < 40 }
}
if_tier_indifferent = {
	check_variable = { var = if_opinion^if_id value = 40 compare = greater_than_or_equals }
	check_variable = { if_opinion^if_id < 60 }
}
if_tier_positive = {
	check_variable = { var = if_opinion^if_id value = 60 compare = greater_than_or_equals }
	check_variable = { if_opinion^if_id < 80 }
}
if_tier_enthusiastic = {
	check_variable = { var = if_opinion^if_id value = 80 compare = greater_than_or_equals }
}

# Influence tiers on if_influence^if_id
