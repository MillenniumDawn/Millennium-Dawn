	USA_palantir_corporate_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = USA_palantir_corporate_capstone_resolved } }
		if = {
			limit = { check_variable = { USA_palantir_corporate_capstone_choice = 1 } }
			USA_palantir_corporate_apply_defense_operating_system = yes
		}
		else_if = {
			limit = { check_variable = { USA_palantir_corporate_capstone_choice = 2 } }
			USA_palantir_corporate_apply_dual_use_operating_platform = yes
		}
		else_if = {
			limit = { check_variable = { USA_palantir_corporate_capstone_choice = 3 } }
			USA_palantir_corporate_apply_accountable_public_data_infrastructure = yes
		}
		else_if = {
			limit = { check_variable = { USA_palantir_corporate_capstone_choice = 4 } }
			USA_palantir_corporate_apply_commercial_ai_platform = yes
		}
		else = { USA_palantir_corporate_apply_dual_use_operating_platform = yes }
	}
}
