TAI_pc_industry_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = TAI_pc_industry_state_initialized } }
		set_variable = { TAI_pc_component_depth = 7 }
		set_variable = { TAI_pc_global_brand_reach = 4 }
		set_variable = { TAI_pc_platform_independence = 3 }
		set_variable = { TAI_pc_systems_breadth = 4 }
		set_country_flag = TAI_pc_industry_state_initialized
	}
	TAI_pc_industry_clamp_state = yes
}

TAI_pc_industry_clamp_state = {
	set_temp_variable = { corp_value = TAI_pc_component_depth }
	corporate_history_clamp_value = yes
	set_variable = { TAI_pc_component_depth = corp_value }
	set_temp_variable = { corp_value = TAI_pc_global_brand_reach }

	set_variable = { TAI_pc_global_brand_reach = corp_value }
	set_temp_variable = { corp_value = TAI_pc_platform_independence }

TAI_pc_industry_resolve_capstone = {
	TAI_pc_industry_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = TAI_pc_industry_capstone_resolved } }
		if = {
			limit = {
				check_variable = { var = TAI_pc_component_depth value = 8 compare = greater_than_or_equals }
				check_variable = { var = TAI_pc_global_brand_reach value = 8 compare = greater_than_or_equals }
				check_variable = { var = TAI_pc_platform_independence value = 8 compare = greater_than_or_equals }
				check_variable = { var = TAI_pc_systems_breadth value = 8 compare = greater_than_or_equals }
				has_country_flag = TAI_via_third_x86_ecosystem
				has_country_flag = TAI_gigabyte_independent_systems
				OR = {
					has_country_flag = TAI_asus_brand_oem_separation
					has_country_flag = TAI_asus_vertical_integration
				}
				OR = {
					has_country_flag = TAI_msi_consumer_notebooks
					has_country_flag = TAI_msi_professional_systems
				}
				OR = {
					has_country_flag = TAI_acer_premium_systems_reset
					has_country_flag = TAI_acer_margin_reset
				}
			}
			TAI_pc_industry_apply_full_spectrum_computing_ecosystem = yes
		}
		else_if = {
			limit = {
				check_variable = { var = TAI_pc_global_brand_reach value = 9 compare = greater_than_or_equals }
				check_variable = { var = TAI_pc_systems_breadth value = 8 compare = greater_than_or_equals }
				has_country_flag = TAI_acer_profitability_discipline
				has_country_flag = TAI_asus_brand_oem_separation
				OR = {
					has_country_flag = TAI_msi_consumer_notebooks
					has_country_flag = TAI_msi_professional_systems
				}
				OR = {
					has_country_flag = TAI_acer_margin_reset
					has_country_flag = TAI_acer_premium_systems_reset
				}
			}
			TAI_pc_industry_apply_branded_systems_powerhouse = yes
		}
		else_if = {
			limit = {
				check_variable = { var = TAI_pc_component_depth value = 9 compare = greater_than_or_equals }
				has_country_flag = TAI_gigabyte_independent_components
				has_country_flag = TAI_asus_component_leadership
				has_country_flag = TAI_msi_component_specialization
				OR = {
					has_country_flag = TAI_gigabyte_joint_venture_suspended
					has_country_flag = TAI_gigabyte_independent_systems
				}
				OR = {
					has_country_flag = TAI_asus_component_oem_retreat
					has_country_flag = TAI_asus_brand_oem_separation
				}
				OR = {
					has_country_flag = TAI_msi_gaming_notebooks
					has_country_flag = TAI_msi_consumer_notebooks
					has_country_flag = TAI_msi_professional_systems
				}
				OR = {
					has_country_flag = TAI_via_nano_focused_platform
					has_country_flag = TAI_via_ip_licensing_platform
				}
			}
			TAI_pc_industry_apply_global_component_commonwealth = yes
		}
		else_if = {
			limit = {
				OR = {
					has_country_flag = TAI_gigabyte_joint_venture_revived
					has_country_flag = TAI_asus_component_oem_retreat
					has_country_flag = TAI_msi_gaming_notebooks
					has_country_flag = TAI_via_nano_focused_platform
					has_country_flag = TAI_via_ip_licensing_platform
					has_country_flag = TAI_acer_margin_reset
					has_country_flag = TAI_acer_market_share_drive
				}
			}
			TAI_pc_industry_apply_fragmented_margin_squeeze = yes
		}
		else = { TAI_pc_industry_apply_fragmented_margin_squeeze = yes }
	}
}
