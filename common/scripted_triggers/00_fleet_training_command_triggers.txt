@MD_ftc_exercise_xp_min = 449

@MD_ftc_rnd_liaison_xp_min = 599
@MD_ftc_fleet_min_ships = 4
@MD_ftc_command_power_min = 74.99

MD_ftc_can_start_exercise_capital = {
	has_navy_experience > @MD_ftc_exercise_xp_min
	command_power > @MD_ftc_command_power_min
	NOT = { has_idea = MD_ftc_exercise_capital }
	OR = {
		NOT = { has_no_capital_ships = yes }
		has_navy_size = { unit = carrier size > 0 }
		has_navy_size = { unit = helicopter_operator size > 0 }
	}
}

MD_ftc_can_start_exercise_escort = {
	has_navy_experience > @MD_ftc_exercise_xp_min
	command_power > @MD_ftc_command_power_min
	NOT = { has_idea = MD_ftc_exercise_escort }
	OR = {
		has_navy_size = { unit = frigate size > 0 }
		has_navy_size = { unit = corvette size > 0 }
		has_navy_size = { unit = patrol_boat size > 0 }
		has_navy_size = { unit = screen_destroyer size > 0 }
		has_navy_size = { unit = stealth_frigate size > 0 }
		has_navy_size = { unit = stealth_corvette size > 0 }
	}
}

MD_ftc_can_upgrade_academy = {
	OR = {
		AND = {
			check_variable = { MD_ftc_academy_level < 1 }
			has_navy_experience > @MD_ftc_upgrade_xp_min_1
		}
		AND = {
			check_variable = { MD_ftc_academy_level = 1 }
			has_navy_experience > @MD_ftc_upgrade_xp_min_2
		}
		AND = {
			check_variable = { MD_ftc_academy_level = 2 }
			has_navy_experience > @MD_ftc_upgrade_xp_min_3
		}
	}
}

	OR = {
		AND = {
			has_country_flag = MD_ftc_doctrine_blue_water
			NOT = { has_doctrine = blue_water }
		}
		AND = {
			has_country_flag = MD_ftc_doctrine_green_water
			NOT = { has_doctrine = green_water }
		}
		AND = {
			has_country_flag = MD_ftc_doctrine_jeune_ecole
			NOT = { has_doctrine = jeune_ecole }
		}
		AND = {
			has_country_flag = MD_ftc_doctrine_guerre_de_course
			NOT = { has_doctrine = guerre_de_course }
		}
	}
}
