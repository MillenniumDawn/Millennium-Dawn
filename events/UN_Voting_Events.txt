#event to scope SC votes
country_event = {
	id = UNSC.100

	is_triggered_only = yes
	hidden = yes

	immediate = {
		random_scope_in_array = {
			array = global.security_council_members
			set_global_flag = current_ongoing_sc_vote
			set_country_flag = has_sc_mission
			activate_mission = UN_SC_voting_mission
		}
		for_each_loop = {
			array = global.security_council_members
			value = v

			var:v = {
				meta_effect = {
					text = {
						country_event = UNSC.[ID]
					}
					ID = "[?global.current_sc_vote_type]"
				}
				if = {
					limit ={
						NOT = { has_country_flag = has_sc_mission }
					}
					activate_mission = UN_SC_voting_mission_result_timer
				}
			}
		}
	}
}

#Pressure to end war
country_event = {
	id = UNSC.1
	title = UNSC.1.t
	desc = UNSC.1.d
	picture = GFX_UN_gen_assembly

	is_triggered_only = yes

	#Yes
	option = {
		name = UNSC.1.a
		log = "[GetDateText]: [THIS.GetName] has voted yes"
		add_to_array = { global.security_council_vote_yes = THIS }

		missile_ui_update = yes

		ai_chance = {
			base = 0
			modifier = {
				add = 20
				has_opinion = {
					target = FROM
					value < -20
				}
			}
			modifier = {
				add = 20
				is_embargoed_by = FROM
			}
			modifier = {
				add = 30
				is_neighbor_of = FROM
			}
			modifier = {
				add = 50
				is_in_array = { potential_and_current_enemies = FROM }
			}
		}
	}
	#No
	option = {
		name = UNSC.1.b
		add_to_array = { global.security_council_vote_no = THIS }
		log = "[GetDateText]: [THIS.GetName] has voted no"
		if = {
			limit = {
				is_in_array = { global.p5_sc_members = THIS }
			}
			set_global_flag = security_council_veto
		}

		missile_ui_update = yes

		ai_chance = {
			base = 0

			modifier = {
				add = 10
				is_ally_with = FROM
			}

			modifier = {
				add = 10
				is_in_faction_with = FROM
			}

			modifier = {
				add = 10
				FROM = {
					has_same_ideology = yes
				}
			}

			modifier = {
				add = 20
				has_opinion = {
					target = FROM
					value > 20
				}
			}

			modifier = {
				add = 20
				has_war = yes
			}

			modifier = {
				add = 200
				is_puppet_of = FROM
			}

			modifier = {
				add = 20
				is_guaranteed_by = FROM
			}

			modifier = {
				add = 10
				has_non_aggression_pact_with = FROM
			}

			modifier = {
				add = 10
				has_guaranteed = FROM
			}
		}
	}
	#Abstain
	option = {
		name = UNSC.1.c
		add_to_array = { global.security_council_vote_abstain = THIS }
		log = "[GetDateText]: [THIS.GetName] has abstained from voting"

		missile_ui_update = yes
		ai_chance = {
			base = 0
			modifier = {
				add = 5
				has_opinion = {
					target = FROM
					value < 20
				}
				has_opinion = {
					target = FROM
					value > -20
				}
			}
			modifier = {
				add = 1000
				has_war_with = FROM
			}
		}
	}
}
