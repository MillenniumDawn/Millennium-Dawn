# Operation Blue Tiger: Silicon Island
# Workforce (Millennium Dawn's staffing check), energy and blockade conditions for a new microchip plant.
TAI_can_build_microchip_plant = {
	custom_trigger_tooltip = {
		tooltip = TAI_can_build_microchip_plant_tt
		can_staff_an_microchip_plant = yes
		NOT = { has_an_energy_deficit = yes }
		NOT = {
			any_country = {
				tag = CHI
				has_country_flag = CHI_TAI_full_blockade_active
			}
		}
	}
}

# Two fossil-fuel power stations need about 15,000 workers (Millennium Dawn: 7,500 per plant).
TAI_can_staff_power_stations = {
	custom_trigger_tooltip = {
		tooltip = TAI_can_staff_power_stations_tt
		check_variable = { total_unemployed_k > 15 }
	}
}
