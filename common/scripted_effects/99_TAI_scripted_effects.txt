# Operation Blue Tiger milestone 3: monthly pressure update
# Blockade Burden reads Beijing's real blockade state; the economic loss itself is charged once, by China's blockade spirits.
TAI_update_pressure = {
	set_variable = { TAI_blockade_burden = 0 }
	if = {
		limit = { country_exists = CHI }
		if = {
			limit = { has_war_with = CHI }
			set_variable = { TAI_blockade_burden = 85 }
		}
		else_if = {
			limit = { has_idea = TAI_blockade_economic_crisis }
			set_variable = { TAI_blockade_burden = 70 }
		}
		else_if = {
			limit = { has_idea = TAI_blockade_partial }
			set_variable = { TAI_blockade_burden = 40 }
		}
	}
	if = {
		limit = { check_variable = { TAI_blockade_burden > 0 } }
		if = { limit = { has_idea = TAI_strategic_reserves } subtract_from_variable = { TAI_blockade_burden = 10 } }
		if = { limit = { has_idea = TAI_strategic_fuel_reserve } subtract_from_variable = { TAI_blockade_burden = 5 } }
		if = { limit = { has_idea = TAI_emergency_doctrine } subtract_from_variable = { TAI_blockade_burden = 10 } }
		if = { limit = { has_idea = TAI_pacific_lifeline } subtract_from_variable = { TAI_blockade_burden = 10 } }
		if = { limit = { has_country_flag = TAI_shipping_guaranteed } subtract_from_variable = { TAI_blockade_burden = 10 } }
		if = { limit = { has_country_flag = TAI_energy_rationed } subtract_from_variable = { TAI_blockade_burden = 5 } }
		clamp_variable = { var = TAI_blockade_burden min = 0 max = 100 }

		set_temp_variable = { TAI_ptn_change = TAI_blockade_burden }
		divide_temp_variable = { TAI_ptn_change = 20 }
		if = { limit = { has_stability > 0.6 } subtract_from_temp_variable = { TAI_ptn_change = 1 } }
		if = { limit = { has_idea = TAI_impossible_to_coerce } subtract_from_temp_variable = { TAI_ptn_change = 1 } }
		if = { limit = { CHI = { check_variable = { CHI_taiwan_us_support > 60 } } } subtract_from_temp_variable = { TAI_ptn_change = 1 } }
		add_to_variable = { TAI_pressure_to_negotiate = TAI_ptn_change }

		# Coercion paradox: pressure on a legitimate government hardens identity and draws support
		if = {
			limit = {
				check_variable = { TAI_blockade_burden > 39 }
				has_stability > 0.5
			}
			if = {
				limit = { check_variable = { TAI_taiwanese_identity < 75 } }
				add_to_variable = { TAI_taiwanese_identity = 0.5 }
			}
			if = {
				limit = { CHI = { check_variable = { CHI_taiwan_us_support < 90 } } }
				CHI = { add_to_variable = { CHI_taiwan_us_support = 0.5 } }
			}
		}
	}
	else = {
		subtract_from_variable = { TAI_pressure_to_negotiate = 2 }
	}
	clamp_variable = { var = TAI_pressure_to_negotiate min = 0 max = 100 }

	if = {
		limit = {
			country_exists = CHI
			NOT = { has_war_with = CHI }
			check_variable = { TAI_negotiation_stage = 0 }
			check_variable = { TAI_pressure_to_negotiate > 69 }
			check_variable = { TAI_blockade_burden > 39 }
			NOT = { has_country_flag = TAI_refused_terms }
			NOT = { has_country_flag = TAI_terms_offered }
		}
		set_country_flag = { flag = TAI_terms_offered days = 60 value = 1 }
		country_event = { id = blue_tiger.40 days = 1 }
	}

	if = {
		limit = {
			has_country_flag = TAI_strait_war
			NOT = { has_war_with = CHI }
		}
		clr_country_flag = TAI_strait_war
	}
}

TAI_count_mainland_control = {
	set_variable = { TAI_mainland_states_total = 0 }
	set_variable = { TAI_mainland_states_held = 0 }
	every_state = {
		limit = {
			is_core_of = CHI
			NOT = { ROC_taiwan_island_state = yes }
		}
		ROOT = { add_to_variable = { TAI_mainland_states_total = 1 } }
		if = {
			limit = { is_controlled_by = ROOT }
			ROOT = { add_to_variable = { TAI_mainland_states_held = 1 } }
		}
	}
	set_variable = { TAI_mainland_control = 0 }
	if = {
		limit = { check_variable = { TAI_mainland_states_total > 0 } }
		set_variable = { TAI_mainland_control = TAI_mainland_states_held }
		multiply_variable = { TAI_mainland_control = 100 }
		divide_variable = { TAI_mainland_control = TAI_mainland_states_total }
	}
}

				AND = {
					has_war_with = CHI
					CHI = { has_capitulated = yes }
				}
				AND = {
					NOT = { country_exists = CHI }
					ROC_holds_mainland = yes
				}
			}
		}
		set_country_flag = TAI_mainland_falling
		country_event = { id = reclamation.5 days = 1 }
	}
	else_if = {
		limit = {
			has_country_flag = TAI_reclamation_war
			NOT = { has_country_flag = TAI_mainland_falling }
			OR = {
				NOT = { country_exists = CHI }
				NOT = { has_war_with = CHI }
			}
		}
		clr_country_flag = TAI_reclamation_war
		country_event = { id = reclamation.18 days = 1 }
	}
}

		limit = { NOT = { has_variable = ROC_regions_restored } }
		set_variable = { ROC_regions_restored = 0 }
	}
	if = {
		limit = { NOT = { has_variable = ROC_admin_reforms } }
		set_variable = { ROC_admin_reforms = 0 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = ROC_provisional_administration } } }

	}
	ROC_update_administration = yes
}

ROC_update_administration = {
	set_temp_variable = { ROC_admin_points = ROC_regions_restored }
	add_to_temp_variable = { ROC_admin_points = ROC_admin_reforms }
	set_variable = { ROC_pa_factory = ROC_admin_points }
	multiply_variable = { ROC_pa_factory = 0.04 }
	subtract_from_variable = { ROC_pa_factory = 0.32 }
	clamp_variable = { var = ROC_pa_factory max = 0 }
	set_variable = { ROC_pa_stability = ROC_admin_points }
	multiply_variable = { ROC_pa_stability = 0.02 }
	subtract_from_variable = { ROC_pa_stability = 0.16 }
	clamp_variable = { var = ROC_pa_stability max = 0 }
	set_variable = { ROC_pa_pp = ROC_admin_points }
	multiply_variable = { ROC_pa_pp = 0.025 }
	subtract_from_variable = { ROC_pa_pp = 0.20 }
	clamp_variable = { var = ROC_pa_pp max = 0 }
	set_variable = { ROC_pa_conscription = ROC_admin_points }
	multiply_variable = { ROC_pa_conscription = 0.06 }
	subtract_from_variable = { ROC_pa_conscription = 0.48 }
	clamp_variable = { var = ROC_pa_conscription max = 0 }

OBT_strait_update = {
	if = {
		limit = {
			country_exists = CHI
			country_exists = TAI
		}
		set_variable = { global.OBT_strait_fill = CHI.CHI_taiwan_strait_tension }
		clamp_variable = { var = global.OBT_strait_fill min = 0 max = 100 }
		if = {
			limit = { check_variable = { global.OBT_strait_fill > 79 } }
			set_temp_variable = { OBT_new_stage = 4 }
		}
		else_if = {
			limit = { check_variable = { global.OBT_strait_fill > 59 } }
			set_temp_variable = { OBT_new_stage = 3 }
		}
		else_if = {
			limit = { check_variable = { global.OBT_strait_fill > 39 } }
			set_temp_variable = { OBT_new_stage = 2 }
		}
		else_if = {
			limit = { check_variable = { global.OBT_strait_fill > 19 } }
			set_temp_variable = { OBT_new_stage = 1 }
		}
		else = {
			set_temp_variable = { OBT_new_stage = 0 }
		}
		if = {
			limit = {
				NOT = { has_global_flag = OBT_strait_initialized }
			}
			set_global_flag = OBT_strait_initialized
			set_variable = { global.OBT_strait_stage = OBT_new_stage }
			CHI = { OBT_strait_apply_stage = yes }
			TAI = { OBT_strait_apply_stage = yes }
			OBT_strait_init_snapshot = yes
		}
		else_if = {
			limit = {
				NOT = { check_variable = { global.OBT_strait_stage = OBT_new_stage } }
			}
			set_temp_variable = { OBT_old_stage = global.OBT_strait_stage }
			set_variable = { global.OBT_strait_stage = OBT_new_stage }
			CHI = { OBT_strait_apply_stage = yes }
			TAI = { OBT_strait_apply_stage = yes }
			if = {
				limit = {
					OR = {
						NOT = { has_global_flag = OBT_strait_event_cooldown }
						has_global_flag = { flag = OBT_strait_event_cooldown days > 179 }
					}
					NOT = { CHI = { has_war_with = TAI } }
				}
				clr_global_flag = OBT_strait_event_cooldown
				set_global_flag = OBT_strait_event_cooldown
				if = {
					limit = { check_variable = { OBT_new_stage > OBT_old_stage } }
					CHI = { country_event = { id = obt_strait.1 days = 1 } }
					TAI = { country_event = { id = obt_strait.3 days = 1 } }
				}
				else = {
					CHI = { country_event = { id = obt_strait.2 days = 1 } }
					TAI = { country_event = { id = obt_strait.4 days = 1 } }
				}
			}
		}
	}
}
