ISI_setup_hvt_pool = {
	for_loop_effect = {
		start = 1
		end = 13
		compare = less_than_or_equals

		add_to_array = { global.ISI_hvt_pool = v }
	}

	set_global_flag = ISI_hvt_pool_seeded
}

ISI_burn_hvt = {
	if = {
		limit = { TOP_enabled = yes }
		random_scope_in_array = {
			array = global.countries
			limit = { exists = yes has_war_with = ISI is_subject = no }
			random_list = {
				1 = { trigger = { TOP_target_active_16 = yes } TOP_add_target_lead = { TARGET = 16 AMOUNT = 40 } }
				1 = { trigger = { TOP_target_active_17 = yes } TOP_add_target_lead = { TARGET = 17 AMOUNT = 40 } }
				1 = { trigger = { TOP_target_active_18 = yes } TOP_add_target_lead = { TARGET = 18 AMOUNT = 40 } }
				1 = { trigger = { TOP_target_active_19 = yes } TOP_add_target_lead = { TARGET = 19 AMOUNT = 40 } }
				1 = { trigger = { TOP_target_active_20 = yes } TOP_add_target_lead = { TARGET = 20 AMOUNT = 40 } }
				1 = { trigger = { TOP_target_active_21 = yes } TOP_add_target_lead = { TARGET = 21 AMOUNT = 40 } }
				1 = { trigger = { TOP_target_active_22 = yes } TOP_add_target_lead = { TARGET = 22 AMOUNT = 40 } }
				1 = { trigger = { TOP_target_active_23 = yes } TOP_add_target_lead = { TARGET = 23 AMOUNT = 40 } }
				1 = { trigger = { TOP_target_active_24 = yes } TOP_add_target_lead = { TARGET = 24 AMOUNT = 40 } }
				1 = { trigger = { TOP_target_active_25 = yes } TOP_add_target_lead = { TARGET = 25 AMOUNT = 40 } }
				1 = { trigger = { TOP_target_active_26 = yes } TOP_add_target_lead = { TARGET = 26 AMOUNT = 40 } }
				1 = { trigger = { TOP_target_active_27 = yes } TOP_add_target_lead = { TARGET = 27 AMOUNT = 40 } }
				1 = { trigger = { TOP_target_active_28 = yes } TOP_add_target_lead = { TARGET = 28 AMOUNT = 40 } }
			}
		}
		set_country_flag = { flag = ISI_hvt_cooldown value = 1 days = 60 }
	}
	else = {
		random_country = {
			limit = {
				has_war_with = ISI
				is_subject = no
				OR = {
					original_tag = SOV
					original_tag = USA
					is_in_faction_with = USA
				}
			}

			set_temp_variable = { ISI_hvt_max = { value = global.ISI_hvt_pool^num subtract = 1 } }

			set_temp_variable_to_random = {
				var = ISI_hvt_slot
				min = 0
				max = ISI_hvt_max
				integer = yes
			}

			set_temp_variable = { ISI_hvt_id = global.ISI_hvt_pool^ISI_hvt_slot }
			remove_from_array = { array = global.ISI_hvt_pool index = ISI_hvt_slot }

			ISI = {
				add_timed_idea = { idea = ISI_decapitated_leadership days = 120 }
				add_war_support = -0.02
				set_country_flag = { flag = ISI_hvt_cooldown value = 1 days = 60 }

				ISI_retire_hvt_character = yes
			}

			if = {
				limit = { original_tag = SOV }
				ISI_hvt_news_russian = yes
			}
			else = {
				ISI_hvt_news_coalition = yes
			}
		}
	}
}
