# Do not open a war with Venezuela while materially outmatched
GUY_avoid_losing_war_VEN = {
	allowed = { original_tag = GUY }
	enable = {
		country_exists = VEN
		has_wargoal_against = VEN
		NOT = { has_war_with = VEN }
		strength_ratio = { tag = VEN ratio < 1 }
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = declare_war id = "VEN" value = -4000 }
}
