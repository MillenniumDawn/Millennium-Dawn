SOV_computing_sovereignty_update_band = {
	if = {
		limit = {
			check_variable = {
				var = SOV_computing_sovereignty_sovereignty
				value = 2.5
				compare = less_than
			}
		}
		if = {
			limit = { NOT = { has_idea = SOV_computing_sovereignty_import_dependent_core } }
			SOV_computing_sovereignty_clear_band = yes
			add_ideas = SOV_computing_sovereignty_import_dependent_core
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = SOV_computing_sovereignty_sovereignty
				value = 5
				compare = less_than
			}
		}
		if = {
			limit = { NOT = { has_idea = SOV_computing_sovereignty_hybrid_industrial_base } }
			SOV_computing_sovereignty_clear_band = yes
			add_ideas = SOV_computing_sovereignty_hybrid_industrial_base
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = SOV_computing_sovereignty_sovereignty
				value = 7
				compare = less_than
			}
		}
		if = {
			limit = { NOT = { has_idea = SOV_computing_sovereignty_localized_stack } }
			SOV_computing_sovereignty_clear_band = yes
			add_ideas = SOV_computing_sovereignty_localized_stack
		}
	}
	else_if = {
		limit = {
			check_variable = {
				var = SOV_computing_sovereignty_sovereignty
				value = 8.5
				compare = less_than
			}

			SOV_computing_sovereignty_clear_band = yes
			add_ideas = SOV_computing_sovereignty_sovereign_computing_base
		}

	else = {
		if = {
			limit = { NOT = { has_idea = SOV_computing_sovereignty_near_autarkic_stack } }
			SOV_computing_sovereignty_clear_band = yes
			add_ideas = SOV_computing_sovereignty_near_autarkic_stack
		}
	}
}

SOV_computing_sovereignty_recalculate_state = {
	if = {
		limit = { corporate_history_enabled = yes }
		SOV_computing_sovereignty_initialize_state = yes

		set_temp_variable = { SOV_computing_sovereignty_temp_architecture = SOV_computing_sovereignty_architecture }
		multiply_temp_variable = { SOV_computing_sovereignty_temp_architecture = 0.25 }
		set_temp_variable = { SOV_computing_sovereignty_temp_fabrication = SOV_computing_sovereignty_fabrication }
		multiply_temp_variable = { SOV_computing_sovereignty_temp_fabrication = 0.30 }
		set_temp_variable = { SOV_computing_sovereignty_temp_software = SOV_computing_sovereignty_software }
		multiply_temp_variable = { SOV_computing_sovereignty_temp_software = 0.15 }
		set_temp_variable = { SOV_computing_sovereignty_temp_systems = SOV_computing_sovereignty_systems_integration }
		multiply_temp_variable = { SOV_computing_sovereignty_temp_systems = 0.15 }
		set_temp_variable = { SOV_computing_sovereignty_temp_supply = SOV_computing_sovereignty_supply_tooling }
		multiply_temp_variable = { SOV_computing_sovereignty_temp_supply = 0.15 }

		set_variable = { SOV_computing_sovereignty_sovereignty = SOV_computing_sovereignty_temp_architecture }
		add_to_variable = { SOV_computing_sovereignty_sovereignty = SOV_computing_sovereignty_temp_fabrication }
		add_to_variable = { SOV_computing_sovereignty_sovereignty = SOV_computing_sovereignty_temp_software }
		add_to_variable = { SOV_computing_sovereignty_sovereignty = SOV_computing_sovereignty_temp_systems }
		add_to_variable = { SOV_computing_sovereignty_sovereignty = SOV_computing_sovereignty_temp_supply }
		clamp_variable = { var = SOV_computing_sovereignty_sovereignty min = 0 max = 10 }

		set_temp_variable = { SOV_computing_sovereignty_temp_dependence = 10 }
		subtract_from_temp_variable = { SOV_computing_sovereignty_temp_dependence = SOV_computing_sovereignty_sovereignty }
		set_temp_variable = { SOV_computing_sovereignty_temp_access_gap = 10 }
		subtract_from_temp_variable = { SOV_computing_sovereignty_temp_access_gap = SOV_computing_sovereignty_foreign_access }
		multiply_temp_variable = { SOV_computing_sovereignty_temp_dependence = SOV_computing_sovereignty_temp_access_gap }
		multiply_temp_variable = { SOV_computing_sovereignty_temp_dependence = 0.1 }
		set_variable = { SOV_computing_sovereignty_sanction_exposure = SOV_computing_sovereignty_temp_dependence }
		clamp_variable = { var = SOV_computing_sovereignty_sanction_exposure min = 0 max = 10 }

		SOV_computing_sovereignty_update_band = yes
		if = {
			limit = { has_country_flag = SOV_computing_sovereignty_event_18_resolved }
			SOV_computing_sovereignty_select_terminal_outcome = yes
		}
	}
}

		limit = { SOV_computing_sovereignty_sovereign_stack_qualified = yes }
		if = {
			limit = { NOT = { has_idea = SOV_computing_sovereignty_sovereign_computing_stack } }
			SOV_computing_sovereignty_apply_sovereign_stack_outcome = yes
		}
		if = {
			limit = { has_idea = SOV_computing_sovereignty_technology_embargo }
			remove_ideas = SOV_computing_sovereignty_technology_embargo
		}
	}
	else_if = {

		}
	}

		limit = { SOV_computing_sovereignty_parallel_import_qualified = yes }
		if = {
			limit = { NOT = { has_idea = SOV_computing_sovereignty_parallel_import_economy } }

		}

			SOV_computing_sovereignty_apply_industrial_retrenchment_outcome = yes
		}

			}
		}
		add_to_variable = { SOV_computing_sovereignty_project_success_chance = 10 }
	}
	if = {
		limit = {
			check_variable = {

				value = 8
				compare = greater_than_or_equals
			}
		}
		add_to_variable = { SOV_computing_sovereignty_project_success_chance = 5 }
	}
	if = {
		limit = {
			OR = {
				AND = { has_country_flag = SOV_computing_sovereignty_project_90nm_active has_idea = SOV_computing_sovereignty_domestic_180nm_capacity }

				AND = { has_country_flag = SOV_computing_sovereignty_project_28nm_hvm_active has_idea = SOV_computing_sovereignty_65nm_hvm_capacity }
				AND = { has_country_flag = SOV_computing_sovereignty_project_16nm_hvm_active has_idea = SOV_computing_sovereignty_28nm_hvm_capacity }
			}
		}
		add_to_variable = { SOV_computing_sovereignty_project_success_chance = 15 }
	}
	if = {
		limit = {
			check_variable = {
				var = SOV_computing_sovereignty_foreign_access
				value = 7
				compare = greater_than_or_equals
			}
		}
		add_to_variable = { SOV_computing_sovereignty_project_success_chance = 10 }
	}
	else_if = {
		limit = {
			check_variable = {
				var = SOV_computing_sovereignty_foreign_access
				value = 3
				compare = less_than
			}
		}
		add_to_variable = { SOV_computing_sovereignty_project_success_chance = -25 }
	}
	if = {
		limit = { has_idea = SOV_computing_sovereignty_foreign_process_license }
		add_to_variable = { SOV_computing_sovereignty_project_success_chance = 15 }
	}
	if = {

			check_variable = {
				var = SOV_computing_sovereignty_angstrem_t_debt
				value = 4
				compare = greater_than_or_equals
			}
		}
		add_to_variable = { SOV_computing_sovereignty_project_success_chance = -10 }
	}
	if = {
		limit = {
			has_country_flag = SOV_computing_sovereignty_project_28nm_hvm_active
			NOT = { has_idea = SOV_computing_sovereignty_65nm_hvm_capacity }
		}
		add_to_variable = { SOV_computing_sovereignty_project_success_chance = -20 }
	}

SOV_computing_sovereignty_resolve_industrial_project = {
	if = {
		limit = {
			OR = {
				AND = {
					has_country_flag = SOV_computing_sovereignty_project_foreign_8c_tapeout_active
					NOT = {
						OR = {
							SOV_computing_sovereignty_has_28nm_or_better_domestic_capacity = yes
							SOV_computing_sovereignty_has_external_foundry_route = yes
						}
					}
				}
				AND = {
					has_country_flag = SOV_computing_sovereignty_project_foreign_16c_tapeout_active
					NOT = {
						OR = {
							has_idea = SOV_computing_sovereignty_16nm_hvm_capacity
							SOV_computing_sovereignty_has_external_foundry_route = yes
						}
					}
				}
			}
		}
		SOV_computing_sovereignty_apply_industrial_project_failure = yes
	}
	else = {
		set_temp_variable_to_random = { var = SOV_computing_sovereignty_project_roll min = 1 max = 100 integer = yes }
		if = {
			limit = {
				check_variable = {
					var = SOV_computing_sovereignty_project_roll
					value = SOV_computing_sovereignty_project_success_chance
					compare = less_than_or_equals
				}
			}
			SOV_computing_sovereignty_apply_industrial_project_success = yes
		}
		else = { SOV_computing_sovereignty_apply_industrial_project_failure = yes }
	}
	SOV_computing_sovereignty_clear_industrial_project = yes
}
