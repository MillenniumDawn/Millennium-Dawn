add_namespace = opec

# Conference vote, sent to every member
country_event = {
	id = opec.1
	title = opec.1.t
	desc = opec.1.d
	picture = GFX_generic_oil_field
	is_triggered_only = yes
	timeout_days = 6

	option = {
		name = opec.1.a
		log = "[GetDateText]: [This.GetName]: opec.1.a executed"
		set_variable = { opec_conference_vote = global.opec_ballot^0 }
		custom_effect_tooltip = OPEC_BALLOT_1_EFFECTS_TT
		ai_chance = {
			base = 1
			modifier = { add = 10 check_variable = { opec_vote = global.opec_ballot^0 } }
		}
	}

	option = {
		name = opec.1.b
		trigger = { check_variable = { global.opec_ballot^num > 1 } }
		log = "[GetDateText]: [This.GetName]: opec.1.b executed"
		set_variable = { opec_conference_vote = global.opec_ballot^1 }
		custom_effect_tooltip = OPEC_BALLOT_2_EFFECTS_TT
		ai_chance = {
			base = 1
			modifier = { add = 10 check_variable = { opec_vote = global.opec_ballot^1 } }
		}
	}

	option = {
		name = opec.1.c
		trigger = { check_variable = { global.opec_ballot^num > 2 } }
		log = "[GetDateText]: [This.GetName]: opec.1.c executed"
		set_variable = { opec_conference_vote = global.opec_ballot^2 }
		custom_effect_tooltip = OPEC_BALLOT_3_EFFECTS_TT
		ai_chance = {
			base = 1
			modifier = { add = 10 check_variable = { opec_vote = global.opec_ballot^2 } }
		}
	}
}
