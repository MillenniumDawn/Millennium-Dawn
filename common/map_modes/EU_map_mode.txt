scripted_map_modes = {
	EU_map_mode = {
		top = {
			type = country
			color = {
				set_temp_variable = { red = 0.100 }
				set_temp_variable = { blue = 0.850 }
				set_temp_variable = { green = 0.400 }

				FROM = {
					is_in_array = { array = global.EU_member value = THIS }
					if = {
						limit = {
							OR = {
								has_idea = EU_commission_president
								has_idea = EU_parliament_president
								has_idea = EU_council_president
							}
						}
						set_temp_variable = { red = 0.900 }
						set_temp_variable = { blue = 0.100 }
						set_temp_variable = { green = 0.150 }
					}
					else_if = { limit = { has_idea = EU_president }
						set_temp_variable = { red = 0.950 }
						set_temp_variable = { blue = 0.950 }
						set_temp_variable = { green = 0.950 }
					}
					else_if = {
						limit = {
							OR = {
								has_idea = EUU_euroarmy_sc
								has_idea = EUU_euronavy_sc
								has_idea = frontex_ExDir
							}
						}
						set_temp_variable = { red = 0.150 }
						set_temp_variable = { blue = 0.900 }
						set_temp_variable = { green = 0.000 }
					}
					else_if = {
						limit = {
							OR = {
								has_idea = EU_foreign_minister
								has_idea = EUU_ambassador_unsc
							}
						}
						set_temp_variable = { red = 0.550 }
						set_temp_variable = { blue = 1.000 }
						set_temp_variable = { green = 0.800 }
					}
					else_if = {
						limit = {
							OR = {
								has_idea = ECB_president
								has_idea = EU_finance_minister
							}
						}
						set_temp_variable = { red = 0.200 }
						set_temp_variable = { blue = 0.450 }
						set_temp_variable = { green = 0.700 }
					}
				}
				set_temp_variable = { alpha = 1.0 }
				set_temp_variable = { thickness = 10.0 }
			}
			targets = {
				target_array = global.EU_potential
			}
		}
		bottom = {
			type = country
			color = {
				set_temp_variable = { red = 0.950 }
				set_temp_variable = { blue = 0.300 }
				set_temp_variable = { green = 0.850 }

				set_temp_variable = { alpha = 1.0 }
				FROM = {
					is_in_array = {
						array = global.EU_potential
						value = THIS
					}
				}
			}
			targets = {
				target_array = global.EU_potential
			}
		}

		far_text = country
		near_text = none

		update_daily = no
	}
}
