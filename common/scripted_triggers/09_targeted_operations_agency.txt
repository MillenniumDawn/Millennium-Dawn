top_agency_person_ready = {
	TOP_country_eligible = yes
	check_variable = { TOP_operation_subject_kind = 1 }
	check_variable = { TOP_arg_target > 0 }
	check_variable = { TOP_arg_target < global.TOP_registry_capacity }
	set_temp_variable = { TOP_arg_method = top_agency_method }
	set_temp_variable = { TOP_arg_state = TOP_case_state^TOP_arg_target }
	TOP_case_binding_valid = yes
	check_variable = { TOP_case_host^TOP_arg_target = FROM }
	set_temp_variable = { TOP_access_method = top_agency_method }
	set_temp_variable = { TOP_access_state = TOP_arg_state }
	set_temp_variable = { TOP_access_host = FROM }
	TOP_person_access_available = yes
	if = {
		limit = { check_variable = { top_agency_method = 5 } }
		check_variable = { TOP_case_host_posture^TOP_arg_target = 4 }
		check_variable = { TOP_case_consent^TOP_arg_target > 0 }
	}
}

	check_variable = { top_agency_state^top_agency_key = TOP_case_state^TOP_arg_target }
	top_agency_person_ready = yes
}
