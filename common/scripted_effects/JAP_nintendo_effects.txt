	if = {
		limit = { NOT = { has_country_flag = JAP_nintendo_state_initialized } }

		set_variable = { JAP_nintendo_platform_strength = 4 }
		set_variable = { JAP_nintendo_developer_ecosystem = 4 }
		set_variable = { JAP_nintendo_digital_services = 0 }

JAP_nintendo_resolve_capstone = {
	JAP_nintendo_initialize_state = yes

	if = {
		limit = { NOT = { has_country_flag = JAP_nintendo_capstone_resolved } }
		if = {
			limit = {
				check_variable = { var = JAP_nintendo_platform_strength value = 8 compare = greater_than_or_equals }
				check_variable = { var = JAP_nintendo_digital_services value = 7 compare = greater_than_or_equals }
				check_variable = { var = JAP_nintendo_ip_reach value = 7 compare = greater_than_or_equals }
			}
			JAP_nintendo_apply_integrated_entertainment_ecosystem = yes
		}
		else_if = {
			limit = {
				check_variable = { var = JAP_nintendo_hardware_resilience value = 8 compare = greater_than_or_equals }
				check_variable = { var = JAP_nintendo_platform_strength value = 7 compare = greater_than_or_equals }
			}
			JAP_nintendo_apply_hardware_platform_sovereignty = yes
		}
		else_if = {
			limit = {
				check_variable = { var = JAP_nintendo_developer_ecosystem value = 8 compare = greater_than_or_equals }
				check_variable = { var = JAP_nintendo_digital_services value = 6 compare = greater_than_or_equals }
			}
			JAP_nintendo_apply_open_developer_commonwealth = yes
		}
		else_if = {
			limit = { check_variable = { var = JAP_nintendo_ip_reach value = 9 compare = greater_than_or_equals } }
			JAP_nintendo_apply_global_ip_licensor = yes
		}
		else = { JAP_nintendo_apply_disciplined_hybrid_platform = yes }
	}
}
