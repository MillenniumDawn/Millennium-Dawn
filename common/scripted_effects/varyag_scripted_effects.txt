varyag_send_hull_to_china = {
	clr_global_flag = varyag_dispute_open
	CHI = { country_event = { id = varyag.5 days = 122 } }
	TUR = { country_event = varyag.7 }
	if = {
		limit = { country_exists = UKR }
		UKR = { country_event = varyag.7 }
	}
}

varyag_allow_standard_passage = {
	if = {
		limit = { check_variable = { var = treasury value = 1.0 compare = greater_than_or_equals } }
		set_temp_variable = { treasury_change = -1.0 }
		modify_treasury_effect = yes
		TUR = {
			set_temp_variable = { treasury_change = 1.0 }
			modify_treasury_effect = yes
		}
	}
	varyag_send_hull_to_china = yes
}

varyag_close_without_delivery = {
	clr_global_flag = varyag_dispute_open
	CHI = {
		set_country_flag = CHI_varyag_hull_lost
		if = {
			limit = { NOT = { has_tech = carrier_hull_2 } }
			add_tech_bonus = { name = varyag_blueprints bonus = 0.25 uses = 1 technology = carrier_hull_2 }
		}
		country_event = varyag.10
	}
}

varyag_refuse_transit = {
	varyag_close_without_delivery = yes
	if = {
		limit = { country_exists = UKR }
		UKR = {
			set_country_flag = UKR_varyag_return_offer
			country_event = varyag.13
		}
	}
}
