SOV_shipyard_setup = {
	set_variable = { shipyard_crisis_level = 20 }
	set_variable = { shipyard_drift_early = 0.2 }
	set_variable = { shipyard_drift_late = 0.45 }
	set_variable = { shipyard_late_year = 2014 }
	set_variable = { shipyard_cap_output = 0.3 }
	set_variable = { shipyard_cap_repair = 0.5 }
	set_variable = { shipyard_cap_refit = 0.35 }
	set_variable = { shipyard_cost_base = 12 }
	set_variable = { shipyard_drydock_stage = 0 }
	set_variable = { shipyard_workforce_stage = 0 }
	set_variable = { shipyard_eq_steps = 0 }
	add_dynamic_modifier = { modifier = shipyard_crisis_modifier }
	shipyard_recalc = yes
}
