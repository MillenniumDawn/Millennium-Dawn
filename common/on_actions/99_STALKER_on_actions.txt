on_actions = {
	on_startup = {
		effect = {
			STALKER_initialize_zone_registry = yes
		}
	}

	on_monthly = {
		effect = {
			STALKER_check_2022_aftermath = yes
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					STALKER_scenario_enabled = yes
					FROM.FROM = { STALKER_is_active_zone_anchor = yes }
				}
				FROM = {
					if = {
						limit = { exists = yes }
						STALKER_refresh_zone_administration = yes
					}
				}
				ROOT = {
					if = {
						limit = { exists = yes }
						STALKER_refresh_zone_administration = yes
					}
				}
				FROM.FROM = {
					if = {
						limit = {
							has_state_flag = STALKER_zone_policy_pending
							controller = { exists = yes }
						}
						save_event_target_as = STALKER_zone_event_target
						controller = { country_event = STALKER.1 }
					}
					if = {
						limit = {
							STALKER_is_active_zone_anchor = yes
							check_variable = { STALKER_zone_id = 1 }
							has_state_flag = STALKER_2022_aftermath_due
							controller = { exists = yes }
						}
						controller = { country_event = STALKER.37 }
					}
				}
			}
		}
	}
}
