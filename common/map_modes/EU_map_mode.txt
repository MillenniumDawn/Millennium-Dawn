scripted_map_modes = {
	EU_map_mode = {
		top = {
			type = country
			color = {
				FROM = {
					is_in_array = { array = global.EU_member value = THIS }
					if = {
						limit = {
							OR = {
								has_idea = EU_commission_office_idea
								has_idea = EU_parliament_office_idea
								has_idea = EU_council_office_idea
							}
						}
						set_temp_variable = { red = 0.900 }
						set_temp_variable = { green = 0.150 }
						set_temp_variable = { blue = 0.100 }
					}
					else_if = { limit = { has_idea = EU_EF_president }
						set_temp_variable = { red = 0.950 }
						set_temp_variable = { green = 0.950 }
						set_temp_variable = { blue = 0.950 }
					}
					else_if = {
						limit = {
							OR = {
								has_idea = EU_army_office_idea
								has_idea = EU_navy_office_idea
								has_idea = EU_frontex_office_idea
							}
						}
						set_temp_variable = { red = 0.670 }
						set_temp_variable = { green = 0.320 }
						set_temp_variable = { blue = 0.930 }
					}
					else_if = {
						limit = {
							OR = {
								has_idea = EU_high_repr_office_idea
								has_idea = EU_UN_ambassador_office_idea
							}
						}
						set_temp_variable = { red = 0.630 }
						set_temp_variable = { green = 0.770 }
						set_temp_variable = { blue = 1.000 }
					}
					else_if = {
						limit = {
							OR = {
								has_idea = EU_ECB_office_idea
								has_idea = EU_ECOFIN_office_idea
							}
						}
						set_temp_variable = { red = 0.150 }
						set_temp_variable = { green = 0.570 }
						set_temp_variable = { blue = 0.450 }
					}
					else = {
						set_temp_variable = { red = 0.000 }
						set_temp_variable = { green = 0.200 }
						set_temp_variable = { blue = 0.600 }
					}
				}
				set_temp_variable = { alpha = 1.0 }
			}
			targets = {
				target_array = global.EU_member
			}
		}
		bottom = {
			type = country
			color = {
				set_temp_variable = { red = 1.000 }
				set_temp_variable = { green = 0.800 }
				set_temp_variable = { blue = 0.000 }
				set_temp_variable = { alpha = 1.0 }
				# Members are drawn by the top layer, so exclude them from this layer.
				FROM = { NOT = { is_in_array = { array = global.EU_member value = THIS } } }
			}
			targets = {
				target_array = global.EU_potential
			}
		}

		far_text = country
		near_text = none

		update_daily = no
	}
}
