USA_ibm_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = USA_ibm_state_initialized } }

		set_variable = { USA_oem_open_standards = 5 }
		set_variable = { USA_oem_vertical_integration = 5 }
		set_variable = { USA_oem_supply_resilience = 5 }
		set_variable = { USA_oem_security_control = 5 }
		set_variable = { USA_oem_national_compute_stack = 5 }
		set_variable = { USA_ibm_faction_systems = 5 }
		set_variable = { USA_ibm_faction_services = 5 }
		set_variable = { USA_ibm_faction_open_platform = 5 }
		set_variable = { USA_ibm_faction_financial = 5 }
		set_variable = { USA_ibm_faction_research = 5 }
		set_country_flag = USA_ibm_state_initialized
	}

	USA_ibm_clamp_state = yes
}

USA_ibm_clamp_state = {
	clamp_variable = { var = USA_oem_open_standards min = 0 max = 10 }
	clamp_variable = { var = USA_oem_vertical_integration min = 0 max = 10 }
	clamp_variable = { var = USA_oem_supply_resilience min = 0 max = 10 }
	clamp_variable = { var = USA_oem_security_control min = 0 max = 10 }
	clamp_variable = { var = USA_oem_national_compute_stack min = 0 max = 10 }
	clamp_variable = { var = USA_ibm_faction_systems min = 0 max = 10 }
	clamp_variable = { var = USA_ibm_faction_services min = 0 max = 10 }
	clamp_variable = { var = USA_ibm_faction_open_platform min = 0 max = 10 }
	clamp_variable = { var = USA_ibm_faction_financial min = 0 max = 10 }
	clamp_variable = { var = USA_ibm_faction_research min = 0 max = 10 }
}

USA_ibm_clear_capstone_outcome = {
	remove_ideas = {
		USA_ibm_open_hybrid_flywheel
		USA_ibm_big_blue_intact
		USA_ibm_fortress_z
		USA_ibm_cloud_without_iron
		USA_ibm_american_systems_ministry
		USA_ibm_stack_in_name_only
	}

	if = {
		limit = { has_country_flag = USA_ibm_outcome_open_hybrid }
		clr_country_flag = USA_ibm_outcome_open_hybrid
	}
	if = {
		limit = { has_country_flag = USA_ibm_outcome_big_blue_intact }
		clr_country_flag = USA_ibm_outcome_big_blue_intact
	}
	if = {
		limit = { has_country_flag = USA_ibm_outcome_fortress_z }
		clr_country_flag = USA_ibm_outcome_fortress_z
	}
	if = {
		limit = { has_country_flag = USA_ibm_outcome_cloud_without_iron }
		clr_country_flag = USA_ibm_outcome_cloud_without_iron
	}
	if = {
		limit = { has_country_flag = USA_ibm_outcome_systems_ministry }
		clr_country_flag = USA_ibm_outcome_systems_ministry
	}
	if = {
		limit = { has_country_flag = USA_ibm_outcome_failed }
		clr_country_flag = USA_ibm_outcome_failed
	}
}

USA_ibm_resolve_capstone = {
	USA_ibm_initialize_state = yes

	if = {
		limit = { NOT = { has_country_flag = USA_ibm_capstone_resolved } }

		USA_ibm_clear_capstone_outcome = yes

		if = {
			limit = {
				OR = {
					AND = {
						has_country_flag = USA_ibm_pc_retained
						has_country_flag = USA_ibm_x86_retained
					}
					AND = {
						has_country_flag = USA_ibm_federal_systems_arm
						has_country_flag = USA_ibm_watsonx_regulated
					}
				}
				NOT = {
					OR = {
						AND = {
							has_country_flag = USA_ibm_pc_retained
							has_country_flag = USA_ibm_x86_retained
							OR = {
								has_country_flag = USA_ibm_redhat_neutral
								has_country_flag = USA_ibm_redhat_absorbed
							}
							OR = {
								has_country_flag = USA_ibm_services_retained
								has_country_flag = USA_ibm_federal_systems_arm
							}
							check_variable = { var = USA_oem_vertical_integration value = 5 compare = greater_than_or_equals }
						}
						AND = {
							has_country_flag = USA_ibm_fortress_z_path
							check_variable = { var = USA_oem_security_control value = 6 compare = greater_than_or_equals }
							OR = {
								has_country_flag = USA_ibm_silicon_retained
								has_country_flag = USA_ibm_federal_systems_arm
								has_country_flag = USA_ibm_hashicorp_strategic
								has_country_flag = USA_ibm_quantum_classified
							}
						}
						AND = {
							has_country_flag = USA_ibm_pc_divested
							has_country_flag = USA_ibm_x86_divested
							has_country_flag = USA_ibm_redhat_neutral
							has_country_flag = USA_ibm_kyndryl_spun
							has_country_flag = USA_ibm_watsonx_services
							check_variable = { var = USA_oem_open_standards value = 5 compare = greater_than_or_equals }
						}
						AND = {
							has_country_flag = USA_ibm_federal_systems_arm
							has_country_flag = USA_ibm_watsonx_regulated
							OR = {
								has_country_flag = USA_ibm_watson_state
								has_country_flag = USA_ibm_watson_public_sector
							}
							OR = {
								has_country_flag = USA_ibm_hashicorp_strategic
								has_country_flag = USA_ibm_quantum_classified
							}
							check_variable = { var = USA_oem_security_control value = 7 compare = greater_than_or_equals }
							check_variable = { var = USA_oem_vertical_integration value = 4 compare = greater_than_or_equals }
						}
						AND = {
							has_country_flag = USA_ibm_redhat_neutral
							has_country_flag = USA_ibm_kyndryl_spun
							OR = {
								has_country_flag = USA_ibm_watsonx_open
								has_country_flag = USA_ibm_watsonx_services
								has_country_flag = USA_ibm_watsonx_regulated
							}
							check_variable = { var = USA_oem_open_standards value = 7 compare = greater_than_or_equals }
						}
					}
				}
			}
			set_country_flag = USA_ibm_outcome_failed
			country_event = { id = USA_ibm_events.11 days = 1 }
		}
		else_if = {
			limit = {
				has_country_flag = USA_ibm_pc_retained
				has_country_flag = USA_ibm_x86_retained
				OR = {
					has_country_flag = USA_ibm_redhat_neutral
					has_country_flag = USA_ibm_redhat_absorbed
				}
				OR = {
					has_country_flag = USA_ibm_services_retained
					has_country_flag = USA_ibm_federal_systems_arm
				}
				check_variable = { var = USA_oem_vertical_integration value = 5 compare = greater_than_or_equals }
			}
			set_country_flag = USA_ibm_outcome_big_blue_intact
			add_ideas = USA_ibm_big_blue_intact
		}
		else_if = {
			limit = {
				has_country_flag = USA_ibm_fortress_z_path
				check_variable = { var = USA_oem_security_control value = 6 compare = greater_than_or_equals }
				OR = {
					has_country_flag = USA_ibm_silicon_retained
					has_country_flag = USA_ibm_federal_systems_arm
					has_country_flag = USA_ibm_hashicorp_strategic
					has_country_flag = USA_ibm_quantum_classified
				}
			}
			set_country_flag = USA_ibm_outcome_fortress_z
			add_ideas = USA_ibm_fortress_z
		}
		else_if = {
			limit = {
				has_country_flag = USA_ibm_pc_divested
				has_country_flag = USA_ibm_x86_divested
				has_country_flag = USA_ibm_redhat_neutral
				has_country_flag = USA_ibm_kyndryl_spun
				has_country_flag = USA_ibm_watsonx_services
				check_variable = { var = USA_oem_open_standards value = 5 compare = greater_than_or_equals }
			}
			set_country_flag = USA_ibm_outcome_cloud_without_iron
			add_ideas = USA_ibm_cloud_without_iron
		}
		else_if = {
			limit = {
				has_country_flag = USA_ibm_federal_systems_arm
				has_country_flag = USA_ibm_watsonx_regulated
				OR = {
					has_country_flag = USA_ibm_watson_state
					has_country_flag = USA_ibm_watson_public_sector
				}
				OR = {
					has_country_flag = USA_ibm_hashicorp_strategic
					has_country_flag = USA_ibm_quantum_classified
				}
				check_variable = { var = USA_oem_security_control value = 7 compare = greater_than_or_equals }
				check_variable = { var = USA_oem_vertical_integration value = 4 compare = greater_than_or_equals }
			}
			set_country_flag = USA_ibm_outcome_systems_ministry
			add_ideas = USA_ibm_american_systems_ministry
		}
		else_if = {
			limit = {
				has_country_flag = USA_ibm_redhat_neutral
				has_country_flag = USA_ibm_kyndryl_spun
				OR = {
					has_country_flag = USA_ibm_watsonx_open
					has_country_flag = USA_ibm_watsonx_services
					has_country_flag = USA_ibm_watsonx_regulated
				}
				check_variable = { var = USA_oem_open_standards value = 7 compare = greater_than_or_equals }
			}
			set_country_flag = USA_ibm_outcome_open_hybrid
			add_ideas = USA_ibm_open_hybrid_flywheel
		}
		else = {
			set_country_flag = USA_ibm_outcome_failed
			country_event = { id = USA_ibm_events.11 days = 1 }
		}

		set_country_flag = USA_ibm_capstone_resolved
	}
}
