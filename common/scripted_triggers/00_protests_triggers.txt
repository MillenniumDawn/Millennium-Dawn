MD_protests_can_give_promises = {
	has_political_power > 99
	custom_trigger_tooltip = {
		tooltip = MD_protests_strength_above_50_tt
		check_variable = { protest_strength > 50 }
	}
	custom_trigger_tooltip = {
		tooltip = MD_protests_promises_cooldown_tt
		NOT = { has_country_flag = MD_protests_promises_cooldown }
	}
}

MD_protests_can_delve_demands = {
	has_political_power > 99
	custom_trigger_tooltip = {
		tooltip = MD_protests_strength_above_75_tt
		check_variable = { protest_strength > 75 }
	}
	custom_trigger_tooltip = {
		tooltip = MD_protests_has_unmet_demand_tt
		MD_protests_has_unmet_demand = yes
	}
	custom_trigger_tooltip = {
		tooltip = MD_protests_demands_cooldown_tt
		NOT = { has_country_flag = MD_protests_demands_cooldown }
	}
}

MD_protests_can_bribe = {
	has_political_power > 49
	custom_trigger_tooltip = {
		tooltip = MD_protests_treasury_above_0_1_tt
		check_variable = { treasury > 0.1 }
	}
	custom_trigger_tooltip = {
		tooltip = MD_protests_strength_below_50_tt
		check_variable = { protest_strength < 50 }
	}
	custom_trigger_tooltip = {
		tooltip = MD_protests_bribe_cooldown_tt
		NOT = { has_country_flag = MD_protests_bribe_cooldown }
	}
}

# A red square on the spending grid: a law funded below what citizens expect.
MD_protests_has_unmet_demand = {
	OR = {
		check_variable = { social_law < expected_welfare_spending }
		check_variable = { health_law < expected_health_spending }
		check_variable = { education_law < expected_edu_spending }
		check_variable = { police_law < expected_police_spending }
		check_variable = { bureau_law < expected_adm_spending }
		# military_law is the idea number + 1, so equal still means underfunded.
		AND = {
			NOT = { has_idea = no_military }
			NOT = { check_variable = { military_law > expected_military_sp } }
		}
	}
}
