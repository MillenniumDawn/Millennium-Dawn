# Polish industrial sovereignty. Five milestones set the route; the closing
# milestone resolves it into one of four industrial models.

POL_apply_sovereign_industrial_republic = {
	if = {
		limit = { POL_industrial_sovereignty_outcome_settled = no }
		add_ideas = POL_sovereign_industrial_republic
	}
}

# One entry point per milestone, called from the yearly pulse. Full presents the
# choice; Outcomes Only writes the historical route silently and grants nothing.
POL_industrial_milestone_2001 = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		country_event = { id = poland.300 days = 119 }
	}
	else_if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
		set_country_flag = POL_is_combines_privatised
	}
}
