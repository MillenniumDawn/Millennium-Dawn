# The header click fills ledger_sort_values in country_array order; a repeat click sorts descending
create_sorted_array = {
	if = {
		limit = { check_variable = { last_sort_var = sort_var } }
		clear_variable = last_sort_var
		set_temp_variable = { ledger_sort_descending = 1 }
	}
	else = {
		set_variable = { last_sort_var = sort_var }
	}
	for_each_loop = {
		array = country_array
		add_to_temp_array = { ledger_sort_nations = v }
	}
	clear_array = country_array
	set_temp_variable = { ledger_sort_end = ledger_sort_nations^num }
	for_loop_effect = {
		end = ledger_sort_end
		if = {
			limit = { check_variable = { ledger_sort_descending = 1 } }
			find_highest_in_array = { array = ledger_sort_values index = ledger_sort_idx }
		}
		else = {
			find_lowest_in_array = { array = ledger_sort_values index = ledger_sort_idx }
		}
		set_temp_variable = { ledger_sort_nation = ledger_sort_nations^ledger_sort_idx }
		add_to_array = { country_array = ledger_sort_nation }
		remove_from_temp_array = { array = ledger_sort_nations index = ledger_sort_idx }
		remove_from_temp_array = { array = ledger_sort_values index = ledger_sort_idx }
	}
	update_ledger_dirty_var = yes
}

open_ledger = {
	if = {
		limit = { check_variable = { ledger_open > 0 } }
		clear_variable = ledger_open
	}
	else = {
		update_ledger_country_array = yes
		# One entry so the header list draws the current page's header
		clear_array = ledger_header_array
		add_to_array = { ledger_header_array = THIS }
		set_variable = { ledger_open = 1 }
	}
	update_ledger_dirty_var = yes
}

update_ledger_country_array = {
	clear_variable = last_sort_var
	clear_array = country_array
	every_country = {
		set_variable = {
			consumer_goods_factories = {
				value = num_of_civilian_factories
				multiply = actual_consumer_goods_percent_display
			}
		}

		set_variable = {
			used_consumer_goods_factories = {
				value = num_of_civilian_factories
				subtract = consumer_goods_factories
			}
		}

		set_variable = {
			effective_civilian_factories = {
				value = used_consumer_goods_factories
				multiply = {
					value = 1
					add = modifier@production_speed_buildings_factor
				}
			}
		}

		set_variable = {
			effective_military_factories = {
				value = num_of_military_factories
				multiply = {
					value = 1
					add = modifier@industrial_capacity_factory
				}
			}
		}

		set_variable = {
			effective_naval_factories = {
				value = num_of_naval_factories
				multiply = {
					value = 1
					add = modifier@industrial_capacity_dockyard
				}
			}
		}

		set_variable = {
			num_controlled_states_including_subjects = {
				value = 0
				every_collection = {
					named_collection = country_and_all_subjects_collection
					add = num_of_controlled_states
				}
			}
		}

		add_to_array = {
			PREV.country_array = THIS
		}
	}

	update_ledger_dirty_var = yes
}

update_ledger_faction_array = {
	clear_variable = last_sort_var
	clear_array = country_array
	every_country = {
		limit = { is_faction_leader = yes }
		set_variable = {
			faction_battalions = {
				value = 0
				every_collection = {
					named_collection = faction_members_collection
					add = num_battalions
				}
			}
		}

		set_variable = {
			faction_ships = {
				value = 0
				every_collection = {
					named_collection = faction_members_collection
					add = num_ships
				}
			}
		}

		set_variable = {
			faction_air = {
				value = 0
				every_collection = {
					named_collection = faction_members_collection
					add = deployed_airforce_manpower_k
				}
			}
		}

		set_variable = {
			faction_industry = {
				value = 0
				every_collection = {
					named_collection = faction_members_collection
					add = num_of_factories
				}
			}
		}

		set_variable = {
			faction_controlled_states = {
				value = 0
				every_collection = {
					named_collection = faction_members_collection
					add = num_of_controlled_states
				}
			}
		}

		add_to_array = {
			PREV.country_array = THIS
		}
	}
	update_ledger_dirty_var = yes
}

update_ledger_dirty_var = {
	set_variable = {
		ledger_ui_dirty_var = {
			value = ledger_ui_dirty_var
			add = 1
			mod = 1000001
		}
	}
}
