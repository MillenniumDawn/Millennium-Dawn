	title = STALKER.140.t
	desc = STALKER.140.d
	picture = GFX_crime_fighting
	is_triggered_only = yes

	trigger = {
		STALKER_scenario_enabled = yes
		has_event_target = STALKER_zone_event_target
		event_target:STALKER_zone_event_target = {
			STALKER_is_active_zone_anchor = yes
			is_controlled_by = ROOT
		}
	}

	option = {
		name = STALKER.140.a
		log = "[GetDateText]: [This.GetName]: STALKER.140.a executed"
		custom_effect_tooltip = STALKER.140.a_tt
		hidden_effect = {
			if = {
				limit = {
					event_target:STALKER_zone_event_target = {
						STALKER_is_active_zone_anchor = yes
						is_controlled_by = ROOT
					}
				}
				ROOT = { add_political_power = -25 }
				event_target:STALKER_zone_event_target = {
					add_to_variable = { STALKER_zone_foreign_intel = -25 }
					clamp_variable = { var = STALKER_zone_foreign_intel min = 0 max = 100 }
				}
			}
		}
		ai_chance = {
			base = 60
			modifier = {
				factor = 0
				has_political_power < 25
			}
		}
	}

	option = {
		name = STALKER.140.b
		log = "[GetDateText]: [This.GetName]: STALKER.140.b executed"
		custom_effect_tooltip = STALKER.140.b_tt
		ai_chance = { base = 40 }
	}
}

	desc = STALKER.141.d
	picture = GFX_crime_fighting
	is_triggered_only = yes

	trigger = {
		STALKER_scenario_enabled = yes
		has_event_target = STALKER_zone_event_target
		event_target:STALKER_zone_event_target = {
			STALKER_is_active_zone_anchor = yes
			check_variable = { STALKER_zone_id = 1 }
			is_controlled_by = ROOT
		}
	}

	option = {
		name = STALKER.141.a
		log = "[GetDateText]: [This.GetName]: STALKER.141.a executed"

						is_controlled_by = ROOT
					}
				}
				add_political_power = -25
				event_target:STALKER_zone_event_target = {
					add_to_variable = { STALKER_zone_mercenaries = 8 }
					add_to_variable = { STALKER_zone_military = -3 }

			base = 40
			modifier = {
				add = 20
				event_target:STALKER_zone_event_target = { check_variable = { STALKER_zone_containment < 40 } }

				factor = 0

						check_variable = { STALKER_zone_id = 1 }
						is_controlled_by = ROOT
					}
				}
				event_target:STALKER_zone_event_target = {
					add_to_variable = { STALKER_zone_mercenaries = -15 }

			}
