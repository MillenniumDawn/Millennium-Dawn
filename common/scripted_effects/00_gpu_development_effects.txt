gpu_refresh_accelerator_demand = {
	if = {
		limit = { has_idea = gpu_accelerator_demand }
		remove_ideas = gpu_accelerator_demand
	}
	add_timed_idea = { idea = gpu_accelerator_demand days = 730 }
}

# One entry point per milestone, called once from the yearly pulse rather than
# per country: the events gate on original_tag themselves. Full presents the
# choice; Outcomes Only records the beat silently and grants nothing.
gpu_development_milestone_2000 = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		USA = { country_event = { id = gpu_development.1 days = 110 } }
		CAN = { country_event = { id = gpu_development.1 days = 110 } }
		TAI = { country_event = { id = gpu_development.1 days = 110 } }
	}
	else_if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
		USA = {
			if = {
				limit = { NOT = { has_country_flag = collapsed_nation } }
				set_country_flag = gpu_development_1_resolved
			}
		}
		CAN = { set_country_flag = gpu_development_1_resolved }
		TAI = { set_country_flag = gpu_development_1_resolved }
	}
}
