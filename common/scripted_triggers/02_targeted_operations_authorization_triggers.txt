TOP_review_pending = {
	OR = {
		check_variable = { TOP_proposal_stage > 0 }
		check_variable = { TOP_review_event_open > 0 }
	}
}

TOP_enhanced_review_government = {
	OR = {
		western_conservatism_are_in_power = yes
		western_liberals_are_in_power = yes
		western_social_democrats_are_in_power = yes
	}
}

TOP_review_lethal = {
	OR = {
		check_variable = { TOP_proposal_method = 1 }
		check_variable = { TOP_proposal_method = 3 }
		check_variable = { TOP_proposal_method = 7 }
	}
}

TOP_review_valid = {
	TOP_country_eligible = yes
	check_variable = { TOP_proposal_target > 0 }
	check_variable = { TOP_proposal_target < global.TOP_status^num }
	check_variable = { TOP_proposal_method > 0 }
	check_variable = { TOP_proposal_method < 8 }
	check_variable = { TOP_proposal_expires > global.TOP_clock }
	check_variable = { global.TOP_status^TOP_proposal_target = 1 }
	check_variable = { TOP_known^TOP_proposal_target > 0 }
	check_variable = { TOP_confidence^TOP_proposal_target > 59 }
	check_variable = { TOP_lead_age^TOP_proposal_target < 57 }
	check_variable = { TOP_lead_state^TOP_proposal_target = TOP_proposal_state }
	check_variable = { TOP_proposal_state > 0 }
	check_variable = { TOP_proposal_host > 0 }

	TOP_case_can_review = { TARGET = TOP_proposal_target HOST = TOP_proposal_host }
	check_variable = { TOP_case_sequence^TOP_proposal_target = TOP_proposal_case_sequence }
	if = {
		limit = { check_variable = { TOP_case_phase^TOP_proposal_target = 2 } }
		check_variable = { TOP_case_method^TOP_proposal_target = TOP_proposal_method }
		check_variable = { TOP_case_state^TOP_proposal_target = TOP_proposal_state }
	}
	if = {
		limit = { check_variable = { TOP_proposal_method < 3 } }
		TOP_native_binding_unused = { TARGET = TOP_proposal_target METHOD = TOP_proposal_method STATE = TOP_proposal_state }
	}
	has_political_power > 49
	set_temp_variable = { TOP_review_expected_host = TOP_proposal_host }
	var:TOP_proposal_state = { is_controlled_by = var:TOP_review_expected_host }
	var:TOP_proposal_host = { exists = yes }
	set_temp_variable = { TOP_target = TOP_proposal_target }
	TOP_exceptional_authority = yes
	if = {
		limit = {
			OR = {
				check_variable = { TOP_proposal_method = 2 }
				check_variable = { TOP_proposal_method = 4 }
			}
		}
		has_tech = special_forces_tech_1
	}
	if = {
		limit = { check_variable = { TOP_proposal_method = 3 } }
		has_tech = decryption1
	}
	if = {
		limit = { check_variable = { TOP_proposal_method = 7 } }
		original_tag = SOV
		has_tech = decryption2
		check_variable = { TOP_confidence^TOP_proposal_target > 89 }
	}
	if = {
		limit = { check_variable = { TOP_proposal_method > 2 } }
		OR = {
			NOT = { check_variable = { TOP_proposal_host = THIS } }
			AND = { check_variable = { TOP_proposal_method = 3 } TOP_authored_capture_override = { TARGET = TOP_proposal_target } }
		}
	}
	if = {
		limit = { check_variable = { TOP_proposal_method = 6 } }
		set_temp_variable = { TOP_facility_state = TOP_proposal_state }
		set_temp_variable = { TOP_facility_kind = TOP_proposal_facility }
		TOP_facility_available = yes
	}
}

TOP_review_staff_ready = {
	TOP_review_valid = yes
	check_variable = { TOP_proposal_civilian_assurance > 59 }
	if = {
		limit = {
			OR = {
				TOP_enhanced_review_government = yes
				check_variable = { TOP_proposal_rigor = 2 }
				check_variable = { global.TOP_political^TOP_proposal_target = 1 }
			}
		}
		check_variable = { TOP_confidence^TOP_proposal_target > 79 }
		check_variable = { TOP_proposal_civilian_assurance > 79 }
		if = {
			limit = { TOP_review_lethal = yes }
			OR = {
				check_variable = { TOP_proposal_capture_feasible = 0 }
				TOP_authored_capture_override = { TARGET = TOP_proposal_target }
			}
		}
	}
}

TOP_review_can_approve = {

	if = {
		limit = { check_variable = { TOP_proposal_method = 5 } }
		check_variable = { TOP_proposal_consent = 1 }
	}
}
