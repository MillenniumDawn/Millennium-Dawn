on_actions = {
	#ROOT is new controller #FROM is old controller #FROM.FROM is state ID
	on_state_control_changed = {
		effect = {
			if = {
				limit = { FROM.FROM = { state = 27 } }
				27 = {
					if = {
						limit = {
							ROOT = { original_tag = IRE }
							NOT = { has_dynamic_modifier = { modifier = IRE_dublin_transport } }
						}
						add_dynamic_modifier = { modifier = IRE_dublin_transport }
					}
					else_if = {
						limit = {
							NOT = { ROOT = { original_tag = IRE } }
							has_dynamic_modifier = { modifier = IRE_dublin_transport }
						}
						remove_dynamic_modifier = { modifier = IRE_dublin_transport }
					}
				}
			}
		}
	}
}
