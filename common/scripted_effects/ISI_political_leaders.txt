set_leader_ISI = {
	if = {
		limit = { TOP_enabled = yes }
		set_temp_variable = { TOP_target = global.TOP_group_leader^3 }
		TOP_apply_office_successor = yes
	}
	else = {
		if = { limit = { check_variable = { ruling_party = 11 } }

			if = { limit = { check_variable = { Caliphate_leader = 0 } }
				set_variable = { Caliphate_leader = 1 }
				if = {
					limit = { OR = { TOP_enabled = no check_variable = { global.TOP_status^32 < 2 } } }
					kill_country_leader = yes

					create_country_leader = {
						name = "Abu Bakr al-Baghdadi"
						picture = "Abu_Bakr_al-Baghdadi.dds"
						ideology = Caliphate
						traits = {
							guerrilla_leader
							salafist_Caliphate
							qureshi
							zealous
							ruthless
							war_criminal
						}
					}
				}

				if = { limit = { has_country_flag = do_not_retire } subtract_from_variable = { Caliphate_leader = 1 } }
				set_temp_variable = { b = 1 }
			}
		}
	}
}
