TOP_log_milestone = {
	set_temp_variable = { TOP_history_row = TOP_history_cursor }
	if = { limit = { is_in_array = { TOP_history_rows = TOP_history_row } } remove_from_array = { TOP_history_rows = TOP_history_row } }
	if = { limit = { is_in_array = { TOP_visible_history = TOP_history_row } } remove_from_array = { TOP_visible_history = TOP_history_row } }
	set_variable = { TOP_history_subject_kind^TOP_history_row = TOP_history_kind }
	set_variable = { TOP_history_subject_id^TOP_history_row = TOP_history_id }
	set_variable = { TOP_history_package_cycle^TOP_history_row = TOP_history_current_cycle }
	set_variable = { TOP_history_case_sequence^TOP_history_row = TOP_history_current_case }
	set_variable = { TOP_history_day^TOP_history_row = global.TOP_clock }
	set_variable = { TOP_history_event^TOP_history_row = TOP_history_current_event }
	set_variable = { TOP_history_reason^TOP_history_row = TOP_history_current_reason }
	add_to_array = { TOP_history_rows = TOP_history_row }
	if = {
		limit = {
			check_variable = { TOP_tab = 6 }
			OR = {
				AND = { check_variable = { TOP_history_kind = 1 } check_variable = { TOP_selected_kind = 1 } check_variable = { TOP_selected = TOP_history_id } }
				AND = { check_variable = { TOP_history_kind = 2 } check_variable = { TOP_selected_kind = 2 } check_variable = { TOP_selected_organization = TOP_history_id } }
			}
		}
		add_to_array = { TOP_visible_history = TOP_history_row }
	}
	if = { limit = { check_variable = { TOP_tab = 6 } } TOP_refresh_view = yes }
	add_to_variable = { TOP_history_cursor = 1 }
	if = { limit = { check_variable = { TOP_history_cursor > 127 } } set_variable = { TOP_history_cursor = 0 } }
	set_temp_variable = { TOP_history_current_reason = 0 }
}

TOP_seed_public_subjects = {
	if = {
		limit = { is_ai = no TOP_enabled = yes }
		for_each_loop = {
			array = global.TOP_active_targets
			value = top_public_target
			set_temp_variable = { TOP_role_target = top_public_target }
			if = {
				limit = {
					check_variable = { global.TOP_public_identity^top_public_target = 1 }
					check_variable = { global.TOP_status^top_public_target = 1 }
					TOP_authored_role_eligible = yes
				}
				set_variable = { TOP_known^top_public_target = 1 }
				set_variable = { TOP_assessment^top_public_target = 1 }
				if = {
					limit = { check_variable = { TOP_identity_confidence^top_public_target < 100 } }
					set_variable = { TOP_identity_confidence^top_public_target = 100 }
					set_variable = { TOP_identity_source^top_public_target = 1 }
				}
				if = {
					limit = { check_variable = { TOP_lead_host^top_public_target = 0 } }
					set_variable = { TOP_lead_host^top_public_target = global.TOP_host^top_public_target }
				}
				if = { limit = { NOT = { is_in_array = { TOP_dossiers = top_public_target } } } add_to_array = { TOP_dossiers = top_public_target } }
				set_temp_variable = { TOP_target = top_public_target }
				TOP_update_person_compat_confidence = yes
			}
		}
		for_loop_effect = {
			start = 1
			end = global.TOP_group_class^num
			compare = less_than
			value = top_public_group
			set_temp_variable = { TOP_group_target = top_public_group }
			if = {
				limit = {
					check_variable = { global.TOP_group_public_identity^top_public_group = 1 }
					TOP_group_operational = yes
				}
				if = {
					limit = { check_variable = { global.TOP_group_state^top_public_group = 0 } check_variable = { global.TOP_group_host^top_public_group > 0 } }
					var:global.TOP_group_host^top_public_group = {
						set_temp_variable = { TOP_public_group_id = PREV.top_public_group }
						if = { limit = { num_of_controlled_states > 0 } random_controlled_state = { set_variable = { global.TOP_group_state^PREV.TOP_public_group_id = THIS } } }
					}
				}
				set_variable = { TOP_org_known^top_public_group = 1 }
				if = {
					limit = { check_variable = { TOP_org_verification^top_public_group < 100 } }
					set_variable = { TOP_org_verification^top_public_group = 100 }
					set_variable = { TOP_org_verification_source^top_public_group = 1 }
				}
				if = {
					limit = { check_variable = { TOP_org_lead_host^top_public_group = 0 } }
					set_variable = { TOP_org_lead_host^top_public_group = global.TOP_group_host^top_public_group }
				}
				if = { limit = { NOT = { is_in_array = { TOP_organization_dossiers = top_public_group } } } add_to_array = { TOP_organization_dossiers = top_public_group } }
			}
		}
	}
}

		limit = { check_variable = { TOP_new_pressure_tier > global.TOP_pressure_tier^TOP_target } }

			set_temp_variable = { TOP_pressure_event_defender = global.TOP_host^TOP_target }

					set_temp_variable = { TOP_pressure_enqueue_kind = 1 }
