				value = v
				index = i

						limit = {
							exists = yes
							SUB_is_rf_subject = yes
						}
						if = {
							limit = { NOT = { has_idea = CHE_republic_rf } }
							add_ideas = CHE_republic_rf

							limit = {
								SOV = { has_country_flag = SOV_united_russia }

							set_country_flag = SUB_united_russia

					}
				}
