AST_ai_alt_path = {
	OR = {
		has_global_flag = AST_COSTELLO_FOCUS_PATH
		has_global_flag = AST_INDEPENDENT_FOCUS_PATH
	}
}

AST_ai_historical_path = {
	OR = {
		has_global_flag = AST_HISTORICAL_FOCUS_PATH
		AND = {
			is_historical_focus_on = yes
			NOT = { AST_ai_alt_path = yes }
		}
	}
}

AST_ai_western_path = {
	OR = {
		AST_ai_historical_path = yes
		has_global_flag = AST_COSTELLO_FOCUS_PATH
	}
}

AST_ai_not_western_path = {
	has_global_flag = AST_INDEPENDENT_FOCUS_PATH
}

AST_ai_costello_path = {
	has_global_flag = AST_COSTELLO_FOCUS_PATH
}

AST_ai_not_costello_path = {
	NOT = { has_global_flag = AST_COSTELLO_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = AST_HISTORICAL_FOCUS_PATH
		has_global_flag = AST_INDEPENDENT_FOCUS_PATH
	}
}

AST_ai_not_independent_path = {
	NOT = { has_global_flag = AST_INDEPENDENT_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = AST_HISTORICAL_FOCUS_PATH
		has_global_flag = AST_COSTELLO_FOCUS_PATH
	}
}
