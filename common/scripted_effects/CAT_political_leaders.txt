set_leader_CAT = {
	if = { limit = { check_variable = { ruling_party = 1 } }

			hidden_effect = { kill_country_leader = yes }

				name = "Xavier García Albiol"
				picture = "xavier_garcia_albiol.dds"

				traits = {
					western_conservatism
				}
			}

		}
