# -----------------------------------------------------------------------------
# Initialise — adds the umbrella idea + 8 standing debuffs and seeds the
# picks counter. Idempotent (each idea add is gated). Called from on_startup.
# -----------------------------------------------------------------------------
NKO_spa_initialise = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_idea = NKO_spa_umbrella } }
			add_ideas = NKO_spa_umbrella
		}

		if = {
			limit = {
				NOT = { has_idea = NKO_spa_debuff_industry }
				NOT = { has_idea = NKO_spa_bonus_industry }
			}
			add_ideas = NKO_spa_debuff_industry
		}
		if = {
			limit = {
				NOT = { has_idea = NKO_spa_debuff_production }
				NOT = { has_idea = NKO_spa_bonus_production }
			}
			add_ideas = NKO_spa_debuff_production
		}
		if = {
			limit = {
				NOT = { has_idea = NKO_spa_debuff_military }
				NOT = { has_idea = NKO_spa_bonus_military }
			}
			add_ideas = NKO_spa_debuff_military
		}
		if = {
			limit = {
				NOT = { has_idea = NKO_spa_debuff_economy }
				NOT = { has_idea = NKO_spa_bonus_economy }
			}
			add_ideas = NKO_spa_debuff_economy
		}
		if = {
			limit = {
				NOT = { has_idea = NKO_spa_debuff_research }
			}
			add_ideas = NKO_spa_debuff_research
		}
		if = {
			limit = {
				NOT = { has_idea = NKO_spa_debuff_agriculture }
				NOT = { has_idea = NKO_spa_bonus_agriculture }
			}
			add_ideas = NKO_spa_debuff_agriculture
		}
		if = {
			limit = {
				NOT = { has_idea = NKO_spa_debuff_foreign }
				NOT = { has_idea = NKO_spa_bonus_foreign }
			}
			add_ideas = NKO_spa_debuff_foreign
		}
		if = {
			limit = {
				NOT = { has_idea = NKO_spa_debuff_construction }
				NOT = { has_idea = NKO_spa_bonus_construction }
			}
			add_ideas = NKO_spa_debuff_construction
		}

		# Seed the picks counter at 0 so the .gui's [?picks_made|0] token has
		# a real value to render on the very first frame.
		set_variable = { NKO_spa_picks_made = 0 }
	}
}

# =============================================================================
# Second Economic Committee — arms & manpower export mechanic.
# Unlocked by the NKO_create_komid focus (calls NKO_arms_export_initialize).
# The player supplies any nation at war via the targeted panel decisions, or
# fulfils Russia's munitions request (NKO_export.1). Each DISTINCT nation
# supplied grants a permanent, uncapped +1% factory-output bonus
# (NKO_arms_export_modifiers scaled by NKO_arms_export_output_var) and raises
# sanctions pressure.
#
# Calling convention for a deal: the caller sets temp var NKO_arms_client_id to
# the client's id, saves the client as event_target:NKO_arms_client, then calls
# one NKO_export_do_* effect. The do_* effect handles the recipient buff, the
# NKO payment, and the deal registration.
# =============================================================================

NKO_arms_export_initialize = {
	set_country_flag = NKO_arms_export_active
	if = { limit = { NOT = { has_variable = NKO_arms_export_clients } }  set_variable = { NKO_arms_export_clients = 0 } }
	if = { limit = { NOT = { has_variable = NKO_arms_export_deals } }    set_variable = { NKO_arms_export_deals = 0 } }
	if = { limit = { NOT = { has_variable = NKO_arms_export_heat } }     set_variable = { NKO_arms_export_heat = 0 } }
	if = { limit = { NOT = { has_variable = NKO_arms_export_revenue } }  set_variable = { NKO_arms_export_revenue = 0 } }
	if = { limit = { NOT = { has_variable = NKO_arms_export_output_var } } set_variable = { NKO_arms_export_output_var = 0 } }
	if = { limit = { NOT = { has_variable = NKO_arms_export_gui_dirty } } set_variable = { NKO_arms_export_gui_dirty = 0 } }
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = NKO_arms_export_modifiers } } }
		add_dynamic_modifier = { modifier = NKO_arms_export_modifiers }
	}
	NKO_arms_export_rebuild_targets = yes
	NKO_arms_contract_initialize = yes
}

# Rebuild the targetable-clients array — every existing country at war that is
# neither NKO nor an enemy of NKO. Driven weekly from 99_NKO_on_actions.txt and
# once at unlock. Global array so the targeted decisions can read it.
NKO_arms_export_rebuild_targets = {
	clear_array = global.NKO_arms_export_targets
	every_country = {
		limit = {
			NOT = { original_tag = NKO }
			exists = yes
			has_war = yes
			is_subject = no
			NOT = { has_war_with = NKO }
		}
		add_to_array = { global.NKO_arms_export_targets = THIS }
	}
}

# Register one concluded deal: count it, add sanctions heat (temp var
# NKO_arms_export_deal_heat), and — only for a NEW distinct client — credit the
# permanent factory-output bonus. Then run the sanctions and capstone watchdogs.
NKO_arms_export_register_deal = {
	add_to_variable = { var = NKO_arms_export_deals  value = 1 }
	add_to_variable = { var = NKO_arms_export_heat   value = NKO_arms_export_deal_heat }
	if = {
		limit = { NOT = { is_in_array = { array = NKO_arms_export_supplied_array  value = NKO_arms_client_id } } }
		add_to_array = { NKO_arms_export_supplied_array = NKO_arms_client_id }
		add_to_variable = { var = NKO_arms_export_clients     value = 1 }
		add_to_variable = { var = NKO_arms_export_output_var  value = 0.01 }
	}
	NKO_arms_export_check_sanctions = yes
	NKO_arms_export_check_capstone = yes
	if = {
		limit = { is_ai = no }
		add_to_variable = { var = NKO_arms_export_gui_dirty  value = 1 }
	}
}

# Every 5 points of accumulated heat tightens sanctions one level.
NKO_arms_export_check_sanctions = {
	if = {
		limit = { check_variable = { NKO_arms_export_heat > 4 } }
		subtract_from_variable = { var = NKO_arms_export_heat  value = 5 }
		NKO_sanctions_negative = yes
	}
}

# Capstone unlocks once ten distinct nations have been supplied.
NKO_arms_export_check_capstone = {
	if = {
		limit = {
			NOT = { has_country_flag = NKO_arms_export_capstone_ready }
			check_variable = { NKO_arms_export_clients > 9 }
		}
		set_country_flag = NKO_arms_export_capstone_ready
	}
}

# Pick the contract class (gated by standing bands) and stage its terms.
NKO_arms_contract_generate_offer = {
	random_list = {
		30 = { set_variable = { NKO_arms_contract_offer_class = 1 } }
		25 = { set_variable = { NKO_arms_contract_offer_class = 2 } }
		25 = {
			set_variable = { NKO_arms_contract_offer_class = 3 }
			modifier = { factor = 0  NOT = { check_variable = { NKO_arms_contract_standing > NKO_arms_contract_band_low } } }
		}
		20 = {
			set_variable = { NKO_arms_contract_offer_class = 4 }
			modifier = { factor = 0  NOT = { check_variable = { NKO_arms_contract_standing > NKO_arms_contract_band_mid } } }
		}
	}
	NKO_arms_contract_set_offer_terms = yes
	add_to_variable = { NKO_arms_contract_gui_dirty = 1 }
}

NKO_arms_contract_set_offer_terms = {
	if = {
		limit = { check_variable = { NKO_arms_contract_offer_class = 1 } }
		set_variable = { NKO_arms_contract_offer_quota = 3 }
		set_variable = { NKO_arms_contract_offer_deadline = 1095 }
		set_variable = { NKO_arms_contract_offer_funds = 4 }
		set_variable = { NKO_arms_contract_offer_pp = 20 }
	}
	else_if = {
		limit = { check_variable = { NKO_arms_contract_offer_class = 2 } }
		set_variable = { NKO_arms_contract_offer_quota = 2 }
		set_variable = { NKO_arms_contract_offer_deadline = 1095 }
		set_variable = { NKO_arms_contract_offer_funds = 6 }
		set_variable = { NKO_arms_contract_offer_pp = 25 }
	}
	else_if = {
		limit = { check_variable = { NKO_arms_contract_offer_class = 3 } }
		set_variable = { NKO_arms_contract_offer_quota = 2 }
		set_variable = { NKO_arms_contract_offer_deadline = 1095 }
		set_variable = { NKO_arms_contract_offer_funds = 10 }
		set_variable = { NKO_arms_contract_offer_pp = 35 }
	}
	else = {
		set_variable = { NKO_arms_contract_offer_quota = 1 }
		set_variable = { NKO_arms_contract_offer_deadline = 1460 }
		set_variable = { NKO_arms_contract_offer_funds = 16 }
		set_variable = { NKO_arms_contract_offer_pp = 50 }
	}
	if = {
		limit = { check_variable = { NKO_arms_contract_standing > NKO_arms_contract_band_high } }
		multiply_variable = { NKO_arms_contract_offer_funds = 1.25 }
		multiply_variable = { NKO_arms_contract_offer_pp = 1.25 }
		multiply_variable = { NKO_arms_contract_offer_deadline = 1.25 }
		round_variable = NKO_arms_contract_offer_funds
		round_variable = NKO_arms_contract_offer_pp
		round_variable = NKO_arms_contract_offer_deadline
	}
	set_variable = { NKO_arms_contract_offer_advance = NKO_arms_contract_offer_funds }
	multiply_variable = { NKO_arms_contract_offer_advance = 0.4 }
	round_variable = NKO_arms_contract_offer_advance
}

NKO_arms_contract_accept_offer = {
	set_temp_variable = { free_slot = -1 }
	for_loop_effect = {
		start = 0
		end = 3
		if = {
			limit = {
				check_variable = { free_slot < 0 }
				check_variable = { NKO_arms_contract_occupied_array^v = 0 }
			}
			set_temp_variable = { free_slot = v }
		}
	}
	if = {
		limit = { check_variable = { free_slot > -1 } }
		set_variable = { NKO_arms_contract_occupied_array^free_slot = 1 }
		set_variable = { NKO_arms_contract_class_array^free_slot = NKO_arms_contract_offer_class }
		set_variable = { NKO_arms_contract_quota_array^free_slot = NKO_arms_contract_offer_quota }
		set_variable = { NKO_arms_contract_total_array^free_slot = NKO_arms_contract_offer_quota }
		set_variable = { NKO_arms_contract_deadline_array^free_slot = NKO_arms_contract_offer_deadline }
		set_variable = { NKO_arms_contract_pp_array^free_slot = NKO_arms_contract_offer_pp }
		set_variable = { NKO_arms_contract_partner_array^free_slot = NKO_arms_contract_offer_partner }
		set_temp_variable = { final_pay = NKO_arms_contract_offer_funds }
		subtract_from_temp_variable = { final_pay = NKO_arms_contract_offer_advance }
		set_variable = { NKO_arms_contract_funds_array^free_slot = final_pay }
		add_to_variable = { NKO_arms_contract_count = 1 }
		set_temp_variable = { treasury_change = NKO_arms_contract_offer_advance }
		modify_treasury_effect = yes
		add_to_variable = { NKO_arms_export_revenue = NKO_arms_contract_offer_advance }
		if = {
			limit = { check_variable = { NKO_arms_contract_count > 1 } }
			for_loop_effect = {
				start = 0
				end = 3
				if = {
					limit = { check_variable = { NKO_arms_contract_occupied_array^v = 1 } }
					add_to_variable = { NKO_arms_contract_deadline_array^v = NKO_arms_contract_grace }
				}
			}
		}
		NKO_arms_contract_refresh_gui = yes
	}
}

# Credit one delivered batch against the most-urgent slot of the delivered class
# (temp var deliver_class). Fulfils the slot when its quota reaches zero.
NKO_arms_contract_credit_delivery = {
	set_temp_variable = { target_slot = -1 }
	set_temp_variable = { best_deadline = 999999 }
	for_loop_effect = {
		start = 0
		end = 3
		if = {
			limit = {
				check_variable = { NKO_arms_contract_occupied_array^v = 1 }
				check_variable = { NKO_arms_contract_class_array^v = deliver_class }
				check_variable = { NKO_arms_contract_quota_array^v > 0 }
				check_variable = { NKO_arms_contract_deadline_array^v < best_deadline }
			}
			set_temp_variable = { target_slot = v }
			set_temp_variable = { best_deadline = NKO_arms_contract_deadline_array^v }
		}
	}
	if = {
		limit = { check_variable = { target_slot > -1 } }
		subtract_from_variable = { NKO_arms_contract_quota_array^target_slot = 1 }
		if = {
			limit = { check_variable = { NKO_arms_contract_quota_array^target_slot < 1 } }
			set_temp_variable = { done_slot = target_slot }
			NKO_arms_contract_fulfill_slot = yes
		}
	}
	NKO_arms_contract_refresh_gui = yes
}

NKO_arms_contract_fulfill_slot = {
	log = "[GetDateText]: [Root.GetName]: NKO_arms_contract_fulfill_slot - contract paid out"
	set_temp_variable = { treasury_change = NKO_arms_contract_funds_array^done_slot }
	modify_treasury_effect = yes
	add_to_variable = { NKO_arms_export_revenue = NKO_arms_contract_funds_array^done_slot }
	set_temp_variable = { pp_reward = NKO_arms_contract_pp_array^done_slot }
	add_political_power = pp_reward
	add_to_variable = { NKO_arms_contract_standing = NKO_arms_contract_standing_gain }
	clamp_variable = { var = NKO_arms_contract_standing  min = 0  max = 100 }
	add_to_variable = { NKO_arms_contract_completed = 1 }
	add_timed_idea = { idea = NKO_arms_contract_revenue days = 365 }
	# Count the contracting partner as a distinct client (capstone / +1% output / sanctions).
	set_temp_variable = { NKO_arms_client_id = NKO_arms_contract_partner_array^done_slot }
	set_temp_variable = { NKO_arms_export_deal_heat = 1 }
	NKO_arms_export_register_deal = yes
	NKO_arms_contract_clear_slot = yes
	subtract_from_variable = { NKO_arms_contract_count = 1 }
	country_event = { id = NKO_export.11 days = 1 }
}

NKO_arms_contract_weekly_tick = {
	set_temp_variable = { NKO_contract_tick_changed = 0 }
	for_loop_effect = {
		start = 0
		end = 3
		if = {
			limit = { check_variable = { NKO_arms_contract_occupied_array^v = 1 } }

		limit = { check_variable = { NKO_contract_tick_changed = 1 } }
