# One political map and per-country filters reused by the four decision categories.
# ROOT is the viewing country. Current share and historical data live exclusively on UKR.
scripted_gui = {
	UKR_crimea_triangle_gui = {
		context_type = decision_category
		window_name = "UKR_crimea_triangle_window"
		visible = {
			OR = { tag = UKR tag = SOV tag = TUR tag = CRM }
			UKR = { has_country_flag = UKR_crimean_bop_initialized_flag }
		}
		# Marker coordinate mirrors are published by UKR_crimean_bop_update_display.
		# Deliberately no dirty = token: the lightweight GUI must reflect
		# other countries' missions and country flags without waiting on ROOT's dirty var.
		properties = {
			UKR_crimea_triangle_marker = { x = UKR_crimean_bop_influence_marker_x y = UKR_crimean_bop_influence_marker_y }
		}
		effects = {
			UKR_crimean_bop_ukrainian_filter_button_click = {
				if = {
					limit = { has_country_flag = UKR_crimean_bop_ukrainian_filter_hidden_flag }
					clr_country_flag = UKR_crimean_bop_ukrainian_filter_hidden_flag
				}
				else = { set_country_flag = UKR_crimean_bop_ukrainian_filter_hidden_flag }
				UKR_crimean_bop_refresh_filter_counts = yes
			}
			UKR_crimean_bop_tatar_filter_button_click = {
				if = {
					limit = { has_country_flag = UKR_crimean_bop_tatar_filter_hidden_flag }
					clr_country_flag = UKR_crimean_bop_tatar_filter_hidden_flag
				}
				else = { set_country_flag = UKR_crimean_bop_tatar_filter_hidden_flag }
				UKR_crimean_bop_refresh_filter_counts = yes
			}
			UKR_crimean_bop_russian_filter_button_click = {
				if = {
					limit = { has_country_flag = UKR_crimean_bop_russian_filter_hidden_flag }
					clr_country_flag = UKR_crimean_bop_russian_filter_hidden_flag
				}
				else = { set_country_flag = UKR_crimean_bop_russian_filter_hidden_flag }
				UKR_crimean_bop_refresh_filter_counts = yes
			}
		}

		ai_enabled = { always = no } # AI takes decisions through regular decision ai_will_do.
	}
}
