CAN_ati_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = CAN_ati_state_initialized } }
		set_variable = { CAN_ati_architecture_lead = 4 }
		set_variable = { CAN_ati_canadian_research_base = 5 }
		set_variable = { CAN_ati_amd_synergy = 0 }
		set_variable = { CAN_ati_semicustom_reach = 1 }
		set_variable = { CAN_ati_compute_ecosystem = 1 }
		set_country_flag = CAN_ati_state_initialized
	}
	CAN_ati_clamp_state = yes
}

CAN_ati_clamp_state = {
	set_temp_variable = { corp_value = CAN_ati_architecture_lead }
	corporate_history_clamp_value = yes
	set_variable = { CAN_ati_architecture_lead = corp_value }
	set_temp_variable = { corp_value = CAN_ati_canadian_research_base }
	corporate_history_clamp_value = yes
	set_variable = { CAN_ati_canadian_research_base = corp_value }
	set_temp_variable = { corp_value = CAN_ati_amd_synergy }
	corporate_history_clamp_value = yes
	set_variable = { CAN_ati_amd_synergy = corp_value }
	set_temp_variable = { corp_value = CAN_ati_semicustom_reach }
	corporate_history_clamp_value = yes
	set_variable = { CAN_ati_semicustom_reach = corp_value }
	set_temp_variable = { corp_value = CAN_ati_compute_ecosystem }
	corporate_history_clamp_value = yes
	set_variable = { CAN_ati_compute_ecosystem = corp_value }
}

CAN_ati_resolve_capstone = {
	CAN_ati_initialize_state = yes
	if = {
		limit = { NOT = { has_country_flag = CAN_ati_capstone_resolved } }
		if = {
			limit = { CAN_ati_integrated_compute_engine_qualified = yes }
			CAN_ati_apply_integrated_compute_engine = yes
		}
		else_if = {
			limit = { CAN_ati_research_compact_qualified = yes }
			CAN_ati_apply_research_compact = yes
		}
		else_if = {
			limit = { CAN_ati_semicustom_network_qualified = yes }
			CAN_ati_apply_semicustom_network = yes
		}
		else_if = {
			limit = { CAN_ati_open_accelerator_ecosystem_qualified = yes }
			CAN_ati_apply_open_accelerator_ecosystem = yes
		}
		else = { CAN_ati_apply_absorbed_legacy = yes }
	}
}

CAN_ati_reconstruct_history = {

		limit = {
			date > 2002.7.18

		}
		CAN_ati_record_r300_architecture = yes
	}
	if = {
		limit = {
			date > 2003.8.14
			NOT = { OR = { has_country_flag = CAN_ati_xbox_semicustom has_country_flag = CAN_ati_xbox_pc_priority } }
		}
		CAN_ati_record_xbox_semicustom = yes
	}
	if = {
		limit = {

		limit = {
			date > 2010.8.30
			NOT = { OR = { has_country_flag = CAN_ati_radeon_brand_retained has_country_flag = CAN_ati_amd_brand_unified } }
		}
		CAN_ati_record_radeon_brand = yes
	}
	if = {
		limit = {
			date > 2011.1.4
			NOT = { OR = { has_country_flag = CAN_ati_fusion_integrated has_country_flag = CAN_ati_discrete_graphics_priority } }
		}
		CAN_ati_record_fusion = yes
	}
	if = {
		limit = {
			date > 2013.11.15
			NOT = { OR = { has_country_flag = CAN_ati_console_semicustom has_country_flag = CAN_ati_pc_architecture_priority } }
		}
		CAN_ati_record_console_generation = yes
	}
	if = {
		limit = {
			date > 2015.9.9
			NOT = { OR = { has_country_flag = CAN_ati_rtg_autonomous has_country_flag = CAN_ati_rtg_integrated } }
		}
		CAN_ati_record_rtg_autonomy = yes
	}
	if = {
		limit = {
			date > 2016.6.13
			NOT = { OR = { has_country_flag = CAN_ati_polaris_efficiency has_country_flag = CAN_ati_polaris_performance } }
		}
		CAN_ati_record_polaris_efficiency = yes
	}
	if = {
		limit = {
			date > 2019.7.7
			NOT = { OR = { has_country_flag = CAN_ati_rdna_architecture has_country_flag = CAN_ati_rdna_open_ecosystem } }

		limit = {
			date > 2023.12.6
			NOT = { OR = { has_country_flag = CAN_ati_mi300_compute_stack has_country_flag = CAN_ati_rocm_open_ecosystem } }
		}
		CAN_ati_record_mi300_compute = yes
	}

	CAN_ati_clamp_state = yes
	if = {
		limit = {
			date > 2026.9.1
			NOT = { has_country_flag = CAN_ati_capstone_resolved }
		}
		CAN_ati_resolve_capstone = yes
	}
	if = {
		limit = { date > 2026.9.1 }
		set_country_flag = CAN_ati_reconstruct_complete
	}
}
