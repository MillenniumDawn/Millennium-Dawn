		}
	}
}

		remove_ideas = HLS_idea_holy_war
	}
}

	desc = vatican.11.d
	picture = GFX_politics_demands
	is_triggered_only = yes

	option = { ##### Agree to join in the crusades
		name = vatican.11.o1
		log = "[GetDateText]: [This.GetName]: vatican_event.11.o1 executed"
		ai_chance = {
			base =  30
		}
		create_wargoal = {
			type = crusade_wargoal
			target = FROM
		}
		send_equipment = { target = HLS type = Inf_equipment amount = 300 }
		add_political_power = 50
	}

	option = { ##### Disagree to join the crusades, but still send help

			base =  45
		}

	}

	}
}

country_event = {
	id = vatican.16
	title = vatican.16.t
	desc = vatican.16.d
	picture = GFX_united_states_dollar
	is_triggered_only = yes
	trigger = {
		tag = HLS
		OR = {
			has_idea = HLS_idea_holy_see
			has_idea = christian
		}
	}

	option = { #Ah Yes. FUNDS.
		name = vatican.16.o1
		log = "[GetDateText]: [This.GetName]: vatican.16.o1"
		if = {
			limit = {
				num_of_factories < 20
			}
			set_temp_variable = { treasury_change = 0.200 }
		}
		else_if = {
			limit = {
				num_of_factories > 20
				num_of_factories < 40
			}
			set_temp_variable = { treasury_change = 0.500 }
		}
		else_if = {
			limit = {
				num_of_factories > 40
				num_of_factories < 60
			}
			set_temp_variable = { treasury_change = 0.750 }
		}
		else_if = {
			limit = {
				num_of_factories > 60
				num_of_factories < 80
			}
			set_temp_variable = { treasury_change = 2.0 }
		}
		else_if = {
			limit = {
				num_of_factories > 80
				num_of_factories < 100
			}
			set_temp_variable = { treasury_change = 3.0 }
		}
		else_if = {
			limit = {
				num_of_factories > 100
				num_of_factories < 125
			}
			set_temp_variable = { treasury_change = 5.0 }
		}
		else_if = {
			limit = {
				num_of_factories > 125
			}
			set_temp_variable = { treasury_change = 10.0 }
		}
		modify_treasury_effect = yes
		hidden_effect = {
			country_event = { id = vatican.16 days = 365 }
		}
		ai_chance = {
			base =  1
		}
	}
}
