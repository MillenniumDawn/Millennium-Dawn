# Operation Blue Tiger: focus tree navigation
# Re-checks which focus tree branches are shown (allow_branch) whenever the ruling party changes,
# using Millennium Dawn's update_focus_tree_obsolete_branches (respects the Obsolete focus branches rule).
TAI_refresh_branch_visibility = {
	if = {
		limit = {
			has_completed_focus = TAI_df_special_division
			NOT = { has_country_flag = TAI_nav_chose_df_special_division }
		}
		set_country_flag = TAI_nav_chose_df_special_division
		update_focus_tree_obsolete_branches = yes
	}
	if = {
		limit = {
			has_completed_focus = TAI_df_ordinary_prosecutors
			NOT = { has_country_flag = TAI_nav_chose_df_ordinary_prosecutors }
		}
		set_country_flag = TAI_nav_chose_df_ordinary_prosecutors
		update_focus_tree_obsolete_branches = yes
	}
	if = {
		limit = {
			has_completed_focus = TAI_second_repbulic_constitution
			NOT = { has_country_flag = TAI_nav_chose_second_repbulic_constitution }
		}
		set_country_flag = TAI_nav_chose_second_repbulic_constitution
		update_focus_tree_obsolete_branches = yes
	}
	if = {
		limit = {
			has_completed_focus = TAI_su_kmt_roc_on_taiwan
			NOT = { has_country_flag = TAI_nav_chose_su_kmt_roc_on_taiwan }
		}
		set_country_flag = TAI_nav_chose_su_kmt_roc_on_taiwan
		update_focus_tree_obsolete_branches = yes
	}
	# Games where Form the Cabinet was completed before this was added
	if = {
		limit = {
			has_completed_focus = TAI_su_form_cabinet
			NOT = { has_country_flag = TAI_first_cabinet_formed }
		}
		set_country_flag = TAI_first_cabinet_formed
		update_focus_tree_obsolete_branches = yes
	}
	if = {
		limit = {
			NOT = { check_variable = { TAI_visible_ruling_party = ruling_party } }
		}
		set_variable = { TAI_visible_ruling_party = ruling_party }
		update_focus_tree_obsolete_branches = yes
	}
}
