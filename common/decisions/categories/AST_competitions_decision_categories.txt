## Pacific influence competitions - tracker only.
## The category holds no decisions of its own; the individual contests keep
## their decisions in their own categories. What it carries is the panel that
## lists every registered competition, which is attached here because a
## decision_category scripted GUI has to hang off a category to be drawn at
## all. Hence visible_when_empty - there is nothing in it but the panel.
AST_pacific_competitions = {
	icon = power_struggle
	priority = 100

	scripted_gui = AST_competition_panel_gui

	allowed = {
		original_tag = AST
	}

	visible = {
		original_tag = AST
		OR = {
			check_variable = { global.AST_active_competitions^num > 0 }
			check_variable = { global.AST_resolved_competitions^num > 0 }
		}
	}

	visible_when_empty = yes
}
