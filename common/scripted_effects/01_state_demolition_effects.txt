state_demolition_rebuild_list = {
	clear_array = ROOT.state_demolition_buildings
	if = { limit = { naval_headquarters > 0 }
		add_to_array = { ROOT.state_demolition_buildings = token:naval_headquarters }
	}
	if = { limit = { fossil_powerplant > 0 }
		add_to_array = { ROOT.state_demolition_buildings = token:fossil_powerplant }
	}
	if = { limit = { nuclear_reactor > 0 }
		add_to_array = { ROOT.state_demolition_buildings = token:nuclear_reactor }
	}
	if = { limit = { renewable_energy_infra > 0 }
		add_to_array = { ROOT.state_demolition_buildings = token:renewable_energy_infra }
	}
	if = { limit = { air_base > 0 }
		add_to_array = { ROOT.state_demolition_buildings = token:air_base }
	}
	if = { limit = { radar_station > 0 }
		add_to_array = { ROOT.state_demolition_buildings = token:radar_station }
	}
	if = { limit = { anti_air_building > 0 }
		add_to_array = { ROOT.state_demolition_buildings = token:anti_air_building }

	if = { limit = { fuel_silo > 0 }
		add_to_array = { ROOT.state_demolition_buildings = token:fuel_silo }
	}
}

state_demolition_confirm = {
	if = { limit = { state_demolition_can_confirm = yes }
		log = "[GetDateText]: [ROOT.GetName]: Demolition requested for up to [?ROOT.state_demolition_amount] levels of [?ROOT.state_demolition_building.GetTokenKey] in [THIS.GetName]"
		for_loop_effect = {
			start = 1
			end = ROOT.state_demolition_amount
			value = state_demolition_iteration
			state_demolition_remove_one = yes
		}
		ROOT = {
			clear_variable = state_demolition_building
			ingame_update_setup = yes
		}
		state_demolition_rebuild_list = yes
	}
}

state_demolition_remove_one = {
