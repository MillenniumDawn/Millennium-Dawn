	SIA_pv_attack_preah_vihear = {
		icon = GFX_decision_generic_military

		allowed = {
			original_tag = SIA
		}

		highlight_states = {
			highlight_states_trigger = {
				state = 512
			}
		}

		visible = {

		}

		cost = 0
		days_remove = 10
		days_re_enable = 45

		available = {
			NOT = { has_war_with = CBD }
			NOT = { has_country_flag = preah_vihear_clash_active }
			NOT = { has_country_flag = preah_vihear_settled_cooldown }
			custom_trigger_tooltip = {
				tooltip = preah_vihear_tension_TT
				check_variable = { GLOBAL.preah_vihear_tension > 40 }
			}
			custom_trigger_tooltip = {
				tooltip = preah_vihear_readiness_TT
				check_variable = { preah_vihear_readiness > 50 }
			}
			OR = {
				has_government = nationalist
				has_idea = intervention_regional_interventionism
				has_idea = intervention_global_interventionism
			}
		}

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision SIA_pv_attack_preah_vihear"
			add_threat = 1
			set_country_flag = preah_vihear_clash_active
			CBD = {
				country_event = preah_vihear.5
			}
			news_event = preah_vihear_news.3
		}

		remove_effect = {
			log = "[GetDateText]: [Root.GetName]: Remove decision SIA_pv_attack_preah_vihear"
			if = {
				limit = {
					OR = {
						has_border_war_between = { attacker = 506 defender = 512 }
						has_border_war_between = { attacker = 512 defender = 506 }
					}
				}
				clr_country_flag = preah_vihear_clash_active
			}
			else = {
				if = {
					limit = {
						controls_state = 506
					}
					start_border_war = {
						change_state_after_war = no
						attacker = {
							state = 506
							num_provinces = 1
							on_win = preah_vihear.10
							on_lose = preah_vihear.11
							on_cancel = preah_vihear.12
							dig_in_factor = 1
							terrain_factor = 1
						}
						defender = {
							state = 512
							num_provinces = 1
							on_win = preah_vihear.11
							on_lose = preah_vihear.10
							on_cancel = preah_vihear.12
						}
					}
				}
				else = {
					clr_country_flag = preah_vihear_clash_active
				}
			}
		}

		ai_will_do = {
			base = 1
			modifier = {
				factor = 3
				has_government = nationalist
			}
			modifier = {
				factor = 2
				date > 2008.1.1
				date < 2012.1.1
			}
			modifier = {
				factor = 0.1
				has_war = yes
			}
