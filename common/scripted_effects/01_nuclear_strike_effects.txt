# Effect: nuclear_strike_devastate_state
# Purpose: Destroys a share of the economic buildings in the state in scope, cuts its productivity and damages its military sites
# Parameters:
# - nuclear_destroy_share: share of each building's levels to remove, 0-1
# - nuclear_site_damage_max: highest share of air base and rocket site levels damaged, 0-1; below 1 one level always survives

			set_temp_variable = { nuclear_levels_lost = { value = building_level@industrial_complex multiply = nuclear_destroy_share round = yes } }
			if = { limit = { check_variable = { nuclear_levels_lost > 0 } }

			}

				meta_effect = {

					LEVELS = "[?nuclear_levels_lost|0]"

		# Air bases and rocket sites: a roll between half and all of nuclear_site_damage_max
		set_temp_variable = { nuclear_site_damage_min = { value = nuclear_site_damage_max multiply = 0.5 } }
		if = { limit = { air_base > 0 }
			set_temp_variable_to_random = { var = nuclear_site_roll min = nuclear_site_damage_min max = nuclear_site_damage_max }
			set_temp_variable = { nuclear_levels_lost = { value = building_level@air_base multiply = nuclear_site_roll round = yes } }
			if = { limit = { check_variable = { nuclear_site_damage_max < 1 } }
				set_temp_variable = { nuclear_levels_kept = { value = building_level@air_base subtract = 1 } }
				clamp_temp_variable = { var = nuclear_levels_lost max = nuclear_levels_kept }
			}
			if = { limit = { check_variable = { nuclear_levels_lost > 0 } }
				meta_effect = {
					text = { damage_building = { type = air_base damage = [LEVELS] } }
					LEVELS = "[?nuclear_levels_lost|0]"
				}
			}
		}
		if = { limit = { rocket_site > 0 }
			set_temp_variable_to_random = { var = nuclear_site_roll min = nuclear_site_damage_min max = nuclear_site_damage_max }
			set_temp_variable = { nuclear_levels_lost = { value = building_level@rocket_site multiply = nuclear_site_roll round = yes } }
			if = { limit = { check_variable = { nuclear_site_damage_max < 1 } }
				set_temp_variable = { nuclear_levels_kept = { value = building_level@rocket_site subtract = 1 } }
				clamp_temp_variable = { var = nuclear_levels_lost max = nuclear_levels_kept }
			}
			if = { limit = { check_variable = { nuclear_levels_lost > 0 } }
				meta_effect = {
					text = { damage_building = { type = rocket_site damage = [LEVELS] } }
					LEVELS = "[?nuclear_levels_lost|0]"
				}
			}
		}
		if = { limit = { radar_station > 0 }
			damage_building = { type = radar_station damage = 10 }
		}
		if = { limit = { anti_air_building > 0 }
			damage_building = { type = anti_air_building damage = 10 }
		}

		# The caller refreshes the owner's economy once, after every state is done
		set_temp_variable = { nuclear_productivity_lost = { value = productivity_state_var multiply = nuclear_destroy_share } }
		subtract_from_variable = { productivity_state_var = nuclear_productivity_lost }
		clamp_variable = { var = productivity_state_var min = 100 }

		add_dynamic_modifier = {
			modifier = nuclear_fallout_severe
			days = 365
		}
	}
}

# Effect: nuclear_strike_political_fallout
# Purpose: Charges the country in scope the domestic and international cost of a nuclear strike it launched
# Parameters:
# - nuclear_strike_severity: 1 tactical, 3 state, 6 strategic
nuclear_strike_political_fallout = {
	# Every strike within 180 days of the last one costs half as much again, up to double
	set_temp_variable = { nuclear_days_since_strike = global.num_days }
	subtract_from_temp_variable = { nuclear_days_since_strike = nuclear_last_strike_day }
	set_temp_variable = { nuclear_strikes_before = nuclear_recent_strikes }
	if = { limit = { check_variable = { nuclear_days_since_strike > 180 } }
		set_temp_variable = { nuclear_strikes_before = 0 }
	}
	set_temp_variable = { nuclear_escalation = { value = nuclear_strikes_before multiply = 0.5 add = 1 min = 2 } }

	# How much of the backlash the government has to answer for at home
	set_temp_variable = { nuclear_domestic_share = 1 }
	if = { limit = { has_totalitarian_government = yes }
		set_temp_variable = { nuclear_domestic_share = 0.15 }
	}
	else_if = { limit = { has_autocratic_government = yes }
		set_temp_variable = { nuclear_domestic_share = 0.35 }
	}
	else_if = { limit = { has_elections = no }
		set_temp_variable = { nuclear_domestic_share = 0.65 }
	}
	if = { limit = { has_war_support > 0.7 }
		multiply_temp_variable = { nuclear_domestic_share = 0.75 }
	}
	else_if = { limit = { has_war_support < 0.3 }
		multiply_temp_variable = { nuclear_domestic_share = 1.25 }
	}
	if = { limit = { check_variable = { government_coalition_strength > 0.6 } }
		multiply_temp_variable = { nuclear_domestic_share = 0.75 }
	}
	else_if = { limit = { check_variable = { government_coalition_strength < 0.4 } }
		multiply_temp_variable = { nuclear_domestic_share = 1.25 }
	}

	# Answering a nuclear strike costs far less than starting one
	if = { limit = { has_country_flag = has_been_nuked }
		multiply_temp_variable = { nuclear_domestic_share = 0.5 }
		if = {
			limit = {
				has_war_support > 0.5
				check_variable = { government_coalition_strength > 0.5 }
			}
			multiply_temp_variable = { nuclear_domestic_share = 0.4 }
		}
	}

	# Half the political power cost is fixed, half follows the domestic share
	set_temp_variable = { nuclear_pp_cost = { value = nuclear_domestic_share multiply = 0.5 add = 0.5 multiply = nuclear_strike_severity multiply = -25 } }
	if = { limit = { has_country_flag = has_been_nuked }
		multiply_temp_variable = { nuclear_pp_cost = 0.5 }
	}

	set_temp_variable = { nuclear_stability_loss = { value = nuclear_strike_severity multiply = -0.03 multiply = nuclear_domestic_share multiply = nuclear_escalation max = -0.3 } }
	set_temp_variable = { nuclear_war_support_loss = { value = nuclear_stability_loss multiply = 0.5 } }
	set_temp_variable = { nuclear_protest_gain = { value = nuclear_strike_severity multiply = 5 multiply = nuclear_domestic_share multiply = nuclear_escalation } }
	if = { limit = { check_variable = { police_law > 3 } }
		multiply_temp_variable = { nuclear_protest_gain = 0.5 }
	}

	custom_effect_tooltip = nuclear_strike_political_fallout_tt
	if = { limit = { has_country_flag = has_been_nuked }
		custom_effect_tooltip = nuclear_strike_retaliation_tt
	}
	else = {
		custom_effect_tooltip = nuclear_strike_first_use_tt
	}
	add_political_power = nuclear_pp_cost

	hidden_effect = {
		set_variable = { nuclear_recent_strikes = nuclear_strikes_before }
		add_to_variable = { nuclear_recent_strikes = 1 }
		set_variable = { nuclear_last_strike_day = global.num_days }

		meta_effect = {
			text = {
				add_stability = [STABILITY]
				add_war_support = [WAR_SUPPORT]
			}
			STABILITY = "[?nuclear_stability_loss|3]"
			WAR_SUPPORT = "[?nuclear_war_support_loss|3]"
		}
		add_to_variable = { protest_strength = nuclear_protest_gain }
		clamp_variable = { var = protest_strength min = 0 max = 100 }

		# The ruling party loses the same share of its own support as the country loses stability, so it never drops below zero
		set_party_index_to_ruling_party = yes
		set_temp_variable = { party_popularity_increase = { value = party_pop_array^party_index multiply = nuclear_stability_loss } }
		change_relative_party_popularity = yes

		if = { limit = { has_country_flag = has_been_nuked }
			every_other_country = {
				add_opinion_modifier = { target = PREV modifier = nuclear_strike_retaliation }
			}
		}
		else = {
			if = { limit = { check_variable = { nuclear_strike_severity > 5 } }
				every_other_country = {
					add_opinion_modifier = { target = PREV modifier = nuclear_strike_strategic_first_use }
				}
			}
			else_if = { limit = { check_variable = { nuclear_strike_severity > 2 } }
				every_other_country = {
					add_opinion_modifier = { target = PREV modifier = nuclear_strike_state_first_use }
				}
			}
			else = {
				every_other_country = {
					add_opinion_modifier = { target = PREV modifier = nuclear_strike_tactical_first_use }
				}
			}

			# State and strategic first use go to a Security Council vote; UNSC.1 weighs the flag
			if = { limit = { check_variable = { nuclear_strike_severity > 2 } }
				set_country_flag = { flag = nuclear_first_use_flag value = 1 days = 730 }
				if = {
					limit = {
						NOT = { has_global_flag = sc_action_against@THIS }
						OR = {
							NOT = { has_dynamic_modifier = { modifier = united_nations_security_council_sanctions } }
							NOT = { has_idea = unsc_arms_embargo }
						}
					}
					if = { limit = { NOT = { has_dynamic_modifier = { modifier = united_nations_security_council_sanctions } } }
						set_temp_variable = { sc_new_vote_type = 2 }
					}
					else = {
						set_temp_variable = { sc_new_vote_type = 4 }
					}
					set_temp_variable = { sc_new_vote_nation = THIS.id }
					set_global_flag = sc_action_against@THIS
					update_sc_vote = yes
				}
			}
		}
	}
}
