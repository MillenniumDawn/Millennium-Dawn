TOP_end_visit = {
	set_temp_variable = { TOP_visit_target = TOP_arg_target }
	set_temp_variable = { TOP_visit_story_id = TOP_arg_story }
	set_temp_variable = { TOP_visit_trigger_target = TOP_visit_target }
	set_temp_variable = { TOP_visit_trigger_story = TOP_visit_story_id }
	if = {
		limit = { TOP_visit_can_end = yes }
		set_temp_variable = { TOP_visit_restore_state = 0 }
		set_temp_variable = { TOP_visit_restore_host = global.TOP_visit_return_host^TOP_visit_target }
		if = {
			limit = {
				check_variable = { global.TOP_status^TOP_visit_target = 1 }
				check_variable = { global.TOP_host^TOP_visit_target = global.TOP_visit_host^TOP_visit_target }
				check_variable = { global.TOP_state^TOP_visit_target = global.TOP_visit_state^TOP_visit_target }
				check_variable = { global.TOP_visit_return_state^TOP_visit_target > 0 }
			}
			var:global.TOP_visit_return_state^TOP_visit_target = {
				if = {
					limit = { is_controlled_by = var:TOP_visit_restore_host }
					set_temp_variable = { PREV.TOP_visit_restore_state = THIS }
				}
			}
		}
		if = {
			limit = {
				check_variable = { global.TOP_status^TOP_visit_target = 1 }
				check_variable = { global.TOP_host^TOP_visit_target = global.TOP_visit_host^TOP_visit_target }
				check_variable = { global.TOP_state^TOP_visit_target = global.TOP_visit_state^TOP_visit_target }
				check_variable = { TOP_visit_restore_state = 0 }
				check_variable = { TOP_visit_restore_host > 0 }
				var:TOP_visit_restore_host = { exists = yes num_of_controlled_states > 0 }
			}
			var:TOP_visit_restore_host = {
				random_controlled_state = { set_temp_variable = { PREV.TOP_visit_restore_state = THIS } }
			}
		}
		if = {
			limit = { check_variable = { TOP_visit_restore_state > 0 } }
			set_variable = { global.TOP_state^TOP_visit_target = TOP_visit_restore_state }
			set_variable = { global.TOP_host^TOP_visit_target = TOP_visit_restore_host }
		}
		remove_from_array = { global.TOP_active_visits = TOP_visit_target }
		set_temp_variable = { TOP_arg_target = TOP_visit_target }
		TOP_clear_visit = yes
	}
}

TOP_process_visits = {
	TOP_setup_extended_runtime = yes
	clear_array = global.TOP_planned_visits_next
	for_each_loop = {
		array = global.TOP_planned_visits
		value = TOP_visit_target
		set_temp_variable = { TOP_arg_target = TOP_visit_target }
		TOP_revalidate_planned_visit = yes
		if = {
			limit = { check_variable = { TOP_visit_revalidated = 1 } }
			add_to_array = { global.TOP_planned_visits_next = TOP_visit_target }
		}
		else_if = {
			limit = {
				check_variable = { global.TOP_visit_status^TOP_visit_target = 1 }
				check_variable = { global.TOP_visit_planned_token^TOP_visit_target = global.TOP_visit_token^TOP_visit_target }
			}
			set_temp_variable = { TOP_arg_target = TOP_visit_target }
			TOP_clear_visit = yes
		}
	}
	clear_array = global.TOP_planned_visits
	for_each_loop = {
		array = global.TOP_planned_visits_next
		value = TOP_visit_target
		add_to_array = { global.TOP_planned_visits = TOP_visit_target }
	}

	clear_array = global.TOP_active_visits_next
	for_each_loop = {
		array = global.TOP_active_visits
		value = TOP_visit_target
		set_temp_variable = { TOP_arg_target = TOP_visit_target }
		TOP_revalidate_active_visit = yes
		if = {
			limit = { check_variable = { TOP_visit_revalidated = 1 } }
			add_to_array = { global.TOP_active_visits_next = TOP_visit_target }
		}
		else_if = {
			limit = {
				check_variable = { global.TOP_visit_status^TOP_visit_target = 2 }
				check_variable = { global.TOP_visit_active_token^TOP_visit_target = global.TOP_visit_token^TOP_visit_target }
			}
			set_temp_variable = { TOP_visit_story_id = global.TOP_visit_story^TOP_visit_target }
			set_temp_variable = { TOP_visit_cleanup_host = global.TOP_visit_host^TOP_visit_target }
			if = {
				limit = {
					check_variable = { TOP_visit_cleanup_host > 0 }
					var:TOP_visit_cleanup_host = { exists = yes }
				}
				var:TOP_visit_cleanup_host = {
					set_temp_variable = { TOP_arg_target = PREV.TOP_visit_target }
					set_temp_variable = { TOP_arg_story = PREV.TOP_visit_story_id }
					TOP_end_visit = yes
				}
			}
			else = {
				set_temp_variable = { TOP_arg_target = TOP_visit_target }
				TOP_clear_visit = yes
			}
		}
	}
	clear_array = global.TOP_active_visits
	for_each_loop = {
		array = global.TOP_active_visits_next
		value = TOP_visit_target
		add_to_array = { global.TOP_active_visits = TOP_visit_target }
	}
}

TOP_snapshot_crisis_coalition_balance = {
	set_variable = { global.TOP_crisis_actor_coalition_anchor = global.TOP_crisis_actor }
	set_variable = { global.TOP_crisis_victim_coalition_anchor = global.TOP_crisis_protection }
	if = {
		limit = {
			check_variable = { global.TOP_crisis_actor_faction_leader > 0 }
			var:global.TOP_crisis_actor_faction_leader = { exists = yes }
		}
		set_variable = { global.TOP_crisis_actor_coalition_anchor = global.TOP_crisis_actor_faction_leader }
	}
	if = {
		limit = {
			check_variable = { global.TOP_crisis_protection_faction_leader > 0 }
			var:global.TOP_crisis_protection_faction_leader = { exists = yes }
		}
		set_variable = { global.TOP_crisis_victim_coalition_anchor = global.TOP_crisis_protection_faction_leader }
	}
	# -1 favors the victim coalition, 0 is parity, and 1 favors the actor coalition.
	set_variable = { global.TOP_crisis_coalition_balance = 0 }
	if = {
		limit = {
			NOT = { check_variable = { global.TOP_crisis_actor_coalition_anchor = global.TOP_crisis_victim_coalition_anchor } }
			var:global.TOP_crisis_victim_coalition_anchor = {
				var:global.TOP_crisis_actor_coalition_anchor = {
					strength_ratio = { tag = PREV ratio > 1.25 }
				}
			}
		}
		set_variable = { global.TOP_crisis_coalition_balance = 1 }
	}
	else_if = {
		limit = {
			NOT = { check_variable = { global.TOP_crisis_actor_coalition_anchor = global.TOP_crisis_victim_coalition_anchor } }
			var:global.TOP_crisis_victim_coalition_anchor = {
				var:global.TOP_crisis_actor_coalition_anchor = {
					strength_ratio = { tag = PREV ratio < 0.80 }
				}
			}
		}
		set_variable = { global.TOP_crisis_coalition_balance = -1 }
	}
}

				set_variable = { global.TOP_crisis_host_complicit = 1 }
				add_to_variable = { global.TOP_crisis_tension = 20 }
				clamp_variable = { var = global.TOP_crisis_tension min = 0 max = 100 }
			}
			set_variable = { global.TOP_crisis_existing_war = 0 }
			set_variable = { global.TOP_crisis_actor_response = 0 }
			set_variable = { global.TOP_crisis_actor_faction_leader = 0 }
			set_variable = { global.TOP_crisis_protection_faction_leader = 0 }
			var:global.TOP_crisis_actor = {
				if = {
					limit = { has_war_with = var:global.TOP_crisis_protection }
					set_variable = { global.TOP_crisis_existing_war = 1 }
				}
				if = {
					limit = { is_in_faction = yes }
					faction_leader = { set_variable = { global.TOP_crisis_actor_faction_leader = THIS } }
				}
				set_variable = { TOP_crisis_token = global.TOP_crisis_token }
				set_variable = { TOP_crisis_counterpart = global.TOP_crisis_protection }
				set_variable = { TOP_crisis_target = global.TOP_crisis_target }
			}
			var:global.TOP_crisis_protection = {
				if = {
					limit = { is_in_faction = yes }
					faction_leader = { set_variable = { global.TOP_crisis_protection_faction_leader = THIS } }
				}
				set_variable = { TOP_crisis_token = global.TOP_crisis_token }
				set_variable = { TOP_crisis_counterpart = global.TOP_crisis_actor }
				set_variable = { TOP_crisis_target = global.TOP_crisis_target }
				set_variable = { TOP_crisis_host_complicit = global.TOP_crisis_host_complicit }
				country_event = { id = TOP_crisis.1 days = 1 }
			}
			TOP_snapshot_crisis_coalition_balance = yes
			if = {
				limit = {
					check_variable = { global.TOP_crisis_host > 0 }
					NOT = { check_variable = { global.TOP_crisis_host = global.TOP_crisis_actor } }
					NOT = { check_variable = { global.TOP_crisis_host = global.TOP_crisis_protection } }
				}
				var:global.TOP_crisis_host = {
					set_variable = { TOP_crisis_token = global.TOP_crisis_token }
					set_variable = { TOP_crisis_counterpart = global.TOP_crisis_protection }
					set_variable = { TOP_crisis_target = global.TOP_crisis_target }
					set_variable = { TOP_crisis_host_complicit = global.TOP_crisis_host_complicit }
					country_event = { id = TOP_crisis.4 days = 1 }
				}
			}
			if = {
				limit = {
					check_variable = { global.TOP_crisis_actor_faction_leader > 0 }
					NOT = { check_variable = { global.TOP_crisis_actor_faction_leader = global.TOP_crisis_actor } }
					NOT = { check_variable = { global.TOP_crisis_actor_faction_leader = global.TOP_crisis_protection } }
					NOT = { check_variable = { global.TOP_crisis_actor_faction_leader = global.TOP_crisis_host } }
					var:global.TOP_crisis_actor_faction_leader = { exists = yes }
				}
				var:global.TOP_crisis_actor_faction_leader = {
					set_variable = { TOP_crisis_token = global.TOP_crisis_token }
					set_variable = { TOP_crisis_counterpart = global.TOP_crisis_protection }
					set_variable = { TOP_crisis_target = global.TOP_crisis_target }
					country_event = { id = TOP_crisis.7 days = 1 }
				}
			}
			if = {
				limit = {
					check_variable = { global.TOP_crisis_protection_faction_leader > 0 }
					NOT = { check_variable = { global.TOP_crisis_protection_faction_leader = global.TOP_crisis_actor } }
					NOT = { check_variable = { global.TOP_crisis_protection_faction_leader = global.TOP_crisis_protection } }
					NOT = { check_variable = { global.TOP_crisis_protection_faction_leader = global.TOP_crisis_host } }
					NOT = { check_variable = { global.TOP_crisis_protection_faction_leader = global.TOP_crisis_actor_faction_leader } }
					var:global.TOP_crisis_protection_faction_leader = { exists = yes }
				}
				var:global.TOP_crisis_protection_faction_leader = {
					set_variable = { TOP_crisis_token = global.TOP_crisis_token }
					set_variable = { TOP_crisis_counterpart = global.TOP_crisis_actor }
					set_variable = { TOP_crisis_target = global.TOP_crisis_target }
					country_event = { id = TOP_crisis.7 days = 1 }
				}
			}
			news_event = { id = TOP_crisis.10 days = 1 }
		}
		else_if = {
			limit = { check_variable = { global.TOP_crisis_active = 1 } }
			add_to_variable = { global.TOP_crisis_tension = global.TOP_crisis_repeat_tension }
			clamp_variable = { var = global.TOP_crisis_tension min = 0 max = 100 }
			if = {
				limit = {
					check_variable = { global.TOP_crisis_stage = 2 }
					TOP_crisis_tension_ultimatum = yes
				}
				set_variable = { global.TOP_crisis_stage = 3 }
				var:global.TOP_crisis_protection = { country_event = { id = TOP_crisis.3 days = 1 } }
			}
			news_event = { id = TOP_crisis.11 days = 1 }
		}
	}
	set_variable = { global.TOP_crisis_candidate_actor = 0 }
	set_variable = { global.TOP_crisis_candidate_protection = 0 }
	set_variable = { global.TOP_crisis_candidate_host = 0 }
}

TOP_settle_crisis = {
	if = {
		limit = { TOP_crisis_event_valid = yes }
		var:global.TOP_crisis_protection = {
			add_opinion_modifier = { target = var:global.TOP_crisis_actor modifier = TOP_crisis_settlement }
		}
		var:global.TOP_crisis_actor = {
			add_opinion_modifier = { target = var:global.TOP_crisis_protection modifier = TOP_crisis_settlement }
		}
		if = {
			limit = { check_variable = { global.TOP_crisis_actor_response = 1 } }
			var:global.TOP_crisis_actor = { add_political_power = global.TOP_crisis_settlement_apology_pp }
		}
		else_if = {
			limit = { check_variable = { global.TOP_crisis_actor_response = 2 } }
			var:global.TOP_crisis_actor = { add_political_power = global.TOP_crisis_settlement_denial_pp }
		}
		else = { var:global.TOP_crisis_actor = { add_political_power = global.TOP_crisis_settlement_defiance_pp } }
		TOP_cleanup_crisis = yes
	}
}

TOP_process_crisis = {
	if = {
		limit = {
			check_variable = { global.TOP_crisis_active = 1 }
			check_variable = { global.TOP_crisis_until < global.TOP_clock }
		}
		if = {
			limit = { TOP_crisis_tension_ultimatum = yes }
			set_variable = { global.TOP_crisis_stage = 3 }
			set_variable = { global.TOP_crisis_until = global.TOP_clock }
			add_to_variable = { global.TOP_crisis_until = 30 }
			var:global.TOP_crisis_protection = { country_event = { id = TOP_crisis.3 days = 1 } }
		}
		else = {
			var:global.TOP_crisis_protection = {
				add_opinion_modifier = { target = var:global.TOP_crisis_actor modifier = harsh_sanctions }
			}
			var:global.TOP_crisis_actor = {
				add_opinion_modifier = { target = var:global.TOP_crisis_protection modifier = harsh_sanctions }
				add_stability = -0.01
			}
			TOP_cleanup_crisis = yes
		}
	}
}
