on_actions = {
	on_startup = {
		effect = {
			POL = { POL_shipyard_setup = yes }
			TUR = { TUR_shipyard_setup = yes }
			CHI = { CHI_shipyard_setup = yes }
			JAP = { JAP_shipyard_setup = yes }
			KOR = { KOR_shipyard_setup = yes }
			every_country = {
				limit = { NOT = { has_dynamic_modifier = { modifier = shipyard_crisis_modifier } } }
				shipyard_refresh_facilities = yes
			}
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = shipyard_crisis_modifier } } }
				shipyard_refresh_facilities = yes
			}
		}
	}

	on_monthly_POL = {
		effect = {
			shipyard_monthly = yes
		}
	}

	on_monthly_TUR = {
		effect = {
			shipyard_monthly = yes
		}
	}

	on_monthly_CHI = {
		effect = {
			shipyard_monthly = yes
		}
	}

	on_monthly_JAP = {
		effect = {
			shipyard_monthly = yes
		}
	}

	on_monthly_KOR = {
		effect = {
			shipyard_monthly = yes
		}
	}
}
