# -------------------------------------------------------------
# Instant 15-Year V2 Fast-Forward Simulator
# Run in HOI4 console via: effect md_test_fast_forward_15_years = yes
# -------------------------------------------------------------
md_test_fast_forward_15_years = {
	# Ensure arrays are initialized
	md_submod_debt_graph_init = yes
	md_submod_inflation_graph_init = yes
	md_submod_pop_graph_init = yes
	md_submod_budget_graph_init = yes
	md_submod_energy_graph_init = yes
	md_submod_defense_graph_init = yes
	md_submod_trade_graph_init = yes

	# Start from baseline values and simulate 10 years (120 months)
	set_temp_variable = { sim_gdp = 1200 }
	set_temp_variable = { sim_gdpc = 1000 }
	set_temp_variable = { sim_prod = 50 }
	set_temp_variable = { sim_debt = 500 }
	set_temp_variable = { sim_inf = 2.5 }
	set_temp_variable = { sim_pop = 1200 }
	set_temp_variable = { sim_rev = 150 }
	set_temp_variable = { sim_exp = 140 }
	set_temp_variable = { sim_bal = 10 }
	set_temp_variable = { sim_egen = 200 }
	set_temp_variable = { sim_econ = 180 }
	set_temp_variable = { sim_ebal = 20 }
	set_temp_variable = { sim_def = 80 }
	set_temp_variable = { sim_defpct = 2.5 }
	set_temp_variable = { sim_exp_trade = 120 }
	set_temp_variable = { sim_imp_trade = 100 }
	set_temp_variable = { sim_bal_trade = 20 }

	for_loop_effect = {
		start = 0
		end = 120

		set_variable = { gdp_array^v = sim_gdp }
		set_variable = { gdpc_array^v = sim_gdpc }
		set_variable = { prod_array^v = sim_prod }
		set_variable = { debt_array^v = sim_debt }
		set_variable = { inflation_array^v = sim_inf }
		set_variable = { pop_array^v = sim_pop }
		set_variable = { workforce_array^v = sim_pop }
		set_variable = { employment_array^v = sim_pop }
		set_variable = { revenue_array^v = sim_rev }
		set_variable = { expense_array^v = sim_exp }
		set_variable = { budget_balance_array^v = sim_bal }
		set_variable = { energy_gen_array^v = sim_egen }
		set_variable = { energy_con_array^v = sim_econ }
		set_variable = { energy_bal_array^v = sim_ebal }
		set_variable = { defense_spend_array^v = sim_def }
		set_variable = { defense_pct_array^v = sim_defpct }
		set_variable = { trade_exports_array^v = sim_exp_trade }
		set_variable = { trade_imports_array^v = sim_imp_trade }
		set_variable = { trade_balance_array^v = sim_bal_trade }

		# Progressive monthly growth
		add_to_temp_variable = { sim_gdp = 15 }
		add_to_temp_variable = { sim_gdpc = 10 }
		add_to_temp_variable = { sim_prod = 0.25 }
		add_to_temp_variable = { sim_debt = 5 }
		add_to_temp_variable = { sim_pop = 1.2 }
		add_to_temp_variable = { sim_rev = 1.5 }
		add_to_temp_variable = { sim_exp = 1.3 }
		add_to_temp_variable = { sim_bal = 0.2 }
		add_to_temp_variable = { sim_egen = 1.8 }
		add_to_temp_variable = { sim_econ = 1.7 }
		add_to_temp_variable = { sim_ebal = 0.1 }
		add_to_temp_variable = { sim_def = 0.5 }
		add_to_temp_variable = { sim_exp_trade = 1.4 }
		add_to_temp_variable = { sim_imp_trade = 1.2 }
		add_to_temp_variable = { sim_bal_trade = 0.2 }
	}

	# Trigger the official 10-year shift logic!
	gdp_graph_shift_array = yes

	# Now simulate an extra 3 years (36 months) after the shift into slots 108..119
	for_loop_effect = {
		start = 108
		end = 120

		set_variable = { gdp_array^v = sim_gdp }
		set_variable = { gdpc_array^v = sim_gdpc }
		set_variable = { prod_array^v = sim_prod }
		set_variable = { debt_array^v = sim_debt }
		set_variable = { inflation_array^v = sim_inf }
		set_variable = { pop_array^v = sim_pop }
		set_variable = { workforce_array^v = sim_pop }
		set_variable = { employment_array^v = sim_pop }
		set_variable = { revenue_array^v = sim_rev }
		set_variable = { expense_array^v = sim_exp }
		set_variable = { budget_balance_array^v = sim_bal }
		set_variable = { energy_gen_array^v = sim_egen }
		set_variable = { energy_con_array^v = sim_econ }
		set_variable = { energy_bal_array^v = sim_ebal }
		set_variable = { defense_spend_array^v = sim_def }
		set_variable = { defense_pct_array^v = sim_defpct }
		set_variable = { trade_exports_array^v = sim_exp_trade }
		set_variable = { trade_imports_array^v = sim_imp_trade }
		set_variable = { trade_balance_array^v = sim_bal_trade }

		add_to_temp_variable = { sim_gdp = 18 }
		add_to_temp_variable = { sim_gdpc = 12 }
		add_to_temp_variable = { sim_prod = 0.3 }
		add_to_temp_variable = { sim_debt = 6 }
	}

	set_variable = { current_gdp_graph_end = 107 }

	# Refresh all 10 graphs
	update_md_flaxbeard_gdp_graph = yes
	update_md_flaxbeard_gdpc_graph = yes
	update_md_flaxbeard_prod_graph = yes
	update_md_flaxbeard_debt_graph = yes
	update_md_flaxbeard_inflation_graph = yes
	update_md_flaxbeard_pop_graph = yes
	update_md_flaxbeard_budget_graph = yes
	update_md_flaxbeard_energy_graph = yes
	update_md_flaxbeard_defense_graph = yes
	update_md_flaxbeard_trade_graph = yes
	gdp_graph_update_dirty_var = yes
}
