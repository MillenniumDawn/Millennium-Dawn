collaboration_government_decisions = {
	instantiate_collaboration = {
		icon = generic_political_discourse

		target_root_trigger = {
			has_rule = can_create_collaboration_government
			is_subject = no
		}
		target_array = occupied_countries
		target_non_existing = yes

		visible = {
			is_available_to_collaboration_government = yes
			has_core_occupation_modifier = {
				occupied_country_tag = FROM
				modifier = compliance_60
			}
		}

		available = {
			has_core_occupation_modifier = {
				occupied_country_tag = FROM
				modifier = compliance_80
			}
		}

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision instantiate_collaboration target [From.GetName]"
			set_temp_variable = { country_to_initiate = FROM }
			instantiate_collaboration_government = yes
		}

		ai_will_do = {
			base = 10
			# Only in a war or once world tension is rising
			modifier = {
				has_war = no
				threat < 0.35
				factor = 0
			}
			# Larger occupations first
			modifier = {
				set_temp_variable = { num_cores_we_control = 0 }
				all_of_scopes = {
					array = FROM.core_states
					if = {
						limit = { is_controlled_by = PREV }
						add_to_temp_variable = { num_cores_we_control = 1 }
					}
				}
				add = num_cores_we_control
			}
			# Each collaboration government already held raises the cores needed for the next
			modifier = {
				set_temp_variable = { num_min_cores = 1.5 }
				all_of_scopes = {
					array = subjects
					if = {
						limit = { has_autonomy_state = autonomy_collaboration_government }
						add_to_temp_variable = { num_min_cores = 0.7 }
					}
				}
				check_variable = { num_cores_we_control < num_min_cores }
				factor = 0
			}
			# Needs at least half of the country, not a border state or an enclave
			modifier = {
				set_temp_variable = { half_of_cores = FROM.core_states^num }
				multiply_temp_variable = { half_of_cores = 0.5 }
				check_variable = { num_cores_we_control < half_of_cores }
				factor = 0
			}
			modifier = {
				any_core_state = {
					OR = {
						is_controlled_by = ROOT
						is_owned_by = ROOT
					}
					is_core_of = FROM
				}
				factor = 0
			}
		}
	}
}
