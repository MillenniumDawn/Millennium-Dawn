	fire_only_once = yes

		original_tag = GUY

			base = 6
			modifier = { factor = 1.5 has_opinion = { target = USA value > 20 } }
			modifier = { factor = 0.4 has_opinion = { target = USA value < -20 } }
		}
	}

	# Rally CARICOM

		name = guyana_essequibo.4.b
		log = "[GetDateText]: [This.GetName]: guyana_essequibo.4.b executed"
		add_stability = 0.02

				has_country_flag = caribbean_nation_flag

			add_opinion_modifier = { target = ROOT modifier = took_stand_for_us }
		}
		ai_chance = { base = 5 }

		name = guyana_essequibo.4.c
		log = "[GetDateText]: [This.GetName]: guyana_essequibo.4.c executed"
		trigger = { country_exists = BRA }
		add_opinion_modifier = { target = BRA modifier = small_increase }

			base = 4
			modifier = { factor = 1.5 BRA = { has_country_flag = BRA_guyana_link_funded } }
			modifier = { factor = 1.5 has_opinion = { target = BRA value > 30 } }
			modifier = { factor = 0.4 has_opinion = { target = BRA value < 0 } }
		}
