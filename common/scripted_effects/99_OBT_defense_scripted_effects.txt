OBT_proc_update = {
	# hold reasons: 1 relationship broken, 2 war with the United States, 3 Strait closed by war
	set_variable = { OBT_proc_hold = 0 }
	if = { limit = { OBT_proc_us_channel_closed = yes } set_variable = { OBT_proc_hold = 1 } }
	else_if = { limit = { country_exists = USA has_war_with = USA } set_variable = { OBT_proc_hold = 2 } }
	else_if = { limit = { OBT_proc_strait_closed = yes } set_variable = { OBT_proc_hold = 3 } }

	# approvals
	if = { limit = { check_variable = { OBT_pk_stage = 0 } OBT_proc_package_approved = yes } set_variable = { OBT_pk_stage = 1 } }
	if = { limit = { check_variable = { OBT_pp_stage = 0 } OBT_proc_package_approved = yes } set_variable = { OBT_pp_stage = 1 } }
	if = { limit = { check_variable = { OBT_ps_stage = 0 } OBT_proc_package_approved = yes } set_variable = { OBT_ps_stage = 1 } }
	if = { limit = { check_variable = { OBT_pa_stage = 0 } has_global_flag = OBT_2001_aegis } set_variable = { OBT_pa_stage = 1 } }

	# Kidd-class destroyers
	if = {
		limit = { check_variable = { OBT_pk_stage = 1 } has_country_flag = OBT_budget_kidd }
		set_variable = { OBT_pk_stage = 2 }
	}
	else_if = {
		limit = { check_variable = { OBT_pk_stage = 2 } has_completed_focus = TAI_na_kidd_destroyers }
		set_variable = { OBT_pk_stage = 3 }
		set_variable = { OBT_pk_days = 930 }
	}
	else_if = {
		limit = { OR = { check_variable = { OBT_pk_stage = 3 } check_variable = { OBT_pk_stage = 4 } } }
		if = { limit = { check_variable = { OBT_proc_hold = 0 } } subtract_from_variable = { OBT_pk_days = 30 } }
		if = {
			limit = { check_variable = { OBT_pk_stage = 3 } check_variable = { OBT_pk_days < 1 } OBT_proc_has_port = yes }
			OBT_deliver_kidd_pair_1 = yes
			set_variable = { OBT_pk_stage = 4 }
			set_variable = { OBT_pk_days = 320 }
			country_event = { id = obt_proc.1 days = 1 }
		}
		else_if = {
			limit = { check_variable = { OBT_pk_stage = 4 } check_variable = { OBT_pk_days < 1 } OBT_proc_has_port = yes }
			OBT_deliver_kidd_pair_2 = yes
			set_variable = { OBT_pk_stage = 5 }
			set_variable = { OBT_pk_days = 180 }
			country_event = { id = obt_proc.2 days = 1 }
		}
	}
	else_if = {
		limit = { check_variable = { OBT_pk_stage = 5 } }
		subtract_from_variable = { OBT_pk_days = 30 }
		if = {
			limit = { check_variable = { OBT_pk_days < 1 } }
			set_variable = { OBT_pk_stage = 6 }
			add_to_variable = { TAI_military_confidence = 5 tooltip = TAI_military_confidence_tt }
			clamp_variable = { var = TAI_military_confidence min = 0 max = 100 }
			navy_experience = 25
		}
	}

	# P-3C Orion maritime patrol aircraft
	if = {
		limit = { check_variable = { OBT_pp_stage = 1 } has_country_flag = OBT_budget_full }
		set_variable = { OBT_pp_stage = 2 }
	}
	else_if = {
		limit = { check_variable = { OBT_pp_stage = 2 } has_completed_focus = TAI_na_maritime_patrol }
		set_variable = { OBT_pp_stage = 3 }
		set_variable = { OBT_pp_days = 600 }
	}
	else_if = {
		limit = { OR = { check_variable = { OBT_pp_stage = 3 } check_variable = { OBT_pp_stage = 4 } check_variable = { OBT_pp_stage = 5 } } }
		if = { limit = { check_variable = { OBT_proc_hold = 0 } } subtract_from_variable = { OBT_pp_days = 30 } }
		if = {
			limit = { check_variable = { OBT_pp_stage = 3 } check_variable = { OBT_pp_days < 1 } }
			set_variable = { OBT_pp_stage = 4 }
			set_variable = { OBT_pp_days = 1500 }
			country_event = { id = obt_proc.10 days = 1 }
		}
		else_if = {
			limit = { OR = { check_variable = { OBT_pp_stage = 4 } check_variable = { OBT_pp_stage = 5 } } check_variable = { OBT_pp_days < 1 } }
			OBT_deliver_p3c_batch = yes
			if = {
				limit = { check_variable = { OBT_pp_n > 11 } }
				set_variable = { OBT_pp_stage = 6 }
				set_variable = { OBT_pp_days = 180 }
				country_event = { id = obt_proc.12 days = 1 }
			}
			else = {
				set_variable = { OBT_pp_stage = 5 }
				set_variable = { OBT_pp_days = 540 }
				if = { limit = { has_country_flag = OBT_pp_second_line } set_variable = { OBT_pp_days = 270 } }
				country_event = { id = obt_proc.11 days = 1 }
			}
		}
	}
	else_if = {
		limit = { check_variable = { OBT_pp_stage = 6 } }
		subtract_from_variable = { OBT_pp_days = 30 }
		if = {
			limit = { check_variable = { OBT_pp_days < 1 } }
			set_variable = { OBT_pp_stage = 7 }
			add_to_variable = { TAI_military_confidence = 3 tooltip = TAI_military_confidence_tt }
			clamp_variable = { var = TAI_military_confidence min = 0 max = 100 }
			air_experience = 20
		}
	}

	# Diesel-electric submarines
	if = {
		limit = { check_variable = { OBT_ps_stage = 1 } has_country_flag = OBT_budget_full }
		set_variable = { OBT_ps_stage = 2 }
	}
	else_if = {
		limit = { check_variable = { OBT_ps_stage = 2 } has_completed_focus = TAI_na_submarine_question }
		set_variable = { OBT_ps_stage = 3 }
		country_event = { id = obt_proc.20 days = 1 }
	}
	else_if = {
		limit = { check_variable = { OBT_ps_stage = 3 } date > 2009.1.1 NOT = { has_country_flag = OBT_ps_negotiating } }
		set_variable = { OBT_ps_stage = 98 }
		country_event = { id = obt_proc.21 days = 1 }
	}
	else_if = {
		limit = { OR = { check_variable = { OBT_ps_stage = 5 } check_variable = { OBT_ps_stage = 6 } } }
		if = { limit = { check_variable = { OBT_proc_hold = 0 } } subtract_from_variable = { OBT_ps_days = 30 } }
		if = {
			limit = { check_variable = { OBT_ps_days < 1 } OBT_proc_has_port = yes }
			OBT_deliver_sub_pair = yes
			if = {
				limit = { check_variable = { OBT_ps_n > 7 } }
				set_variable = { OBT_ps_stage = 7 }
				set_variable = { OBT_ps_days = 180 }
				country_event = { id = obt_proc.25 days = 1 }
			}
			else = {
				set_variable = { OBT_ps_stage = 6 }
				set_variable = { OBT_ps_days = 365 }
				if = { limit = { has_country_flag = OBT_ps_second_line } set_variable = { OBT_ps_days = 180 } }
				country_event = { id = obt_proc.24 days = 1 }
			}
		}
	}
	else_if = {
		limit = { check_variable = { OBT_ps_stage = 7 } }
		subtract_from_variable = { OBT_ps_days = 30 }
		if = {
			limit = { check_variable = { OBT_ps_days < 1 } }
			set_variable = { OBT_ps_stage = 8 }
			add_to_variable = { TAI_military_confidence = 5 tooltip = TAI_military_confidence_tt }
			clamp_variable = { var = TAI_military_confidence min = 0 max = 100 }
			navy_experience = 30
		}
	}

	# Aegis destroyers (alternate history)
	if = {
		limit = { OR = { check_variable = { OBT_pa_stage = 2 } check_variable = { OBT_pa_stage = 3 } } }
		if = { limit = { check_variable = { OBT_proc_hold = 0 } } subtract_from_variable = { OBT_pa_days = 30 } }
		if = {
			limit = { check_variable = { OBT_pa_days < 1 } OBT_proc_has_port = yes }
			OBT_deliver_aegis_pair = yes
			if = {
				limit = { check_variable = { OBT_pa_stage = 3 } }
				set_variable = { OBT_pa_stage = 4 }
				set_variable = { OBT_pa_days = 180 }
			}
			else = {
				set_variable = { OBT_pa_stage = 3 }
				set_variable = { OBT_pa_days = 365 }
			}
			country_event = { id = obt_proc.30 days = 1 }
		}
	}
	else_if = {
		limit = { check_variable = { OBT_pa_stage = 4 } }
		subtract_from_variable = { OBT_pa_days = 30 }
		if = {
			limit = { check_variable = { OBT_pa_days < 1 } }
			set_variable = { OBT_pa_stage = 5 }
			add_to_variable = { TAI_military_confidence = 5 tooltip = TAI_military_confidence_tt }
			clamp_variable = { var = TAI_military_confidence min = 0 max = 100 }
		}
	}
