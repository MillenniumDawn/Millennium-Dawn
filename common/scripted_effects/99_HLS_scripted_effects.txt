# Annex wargoal on $TARGET$ unless it is gone, already at war with us, or a faction ally
HLS_holy_wargoal = {
	if = {
		limit = {
			country_exists = $TARGET$
			NOT = { has_war_with = $TARGET$ }
			NOT = { is_in_faction_with = $TARGET$ }
		}
		create_wargoal = {
			type = annex_everything
			target = $TARGET$
		}
	}
}
