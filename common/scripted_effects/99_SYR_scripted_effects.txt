SYR_decrease_divided_syria = {
	if = { limit = { has_idea = divided_syria_3 }
		remove_ideas = divided_syria_3
	}
	else_if = { limit = { has_idea = divided_syria_2 }
		swap_ideas = {
			add_idea = divided_syria_3
			remove_idea = divided_syria_2
		}
	}
	else_if = { limit = { has_idea = divided_syria_1 }
		swap_ideas = {
			add_idea = divided_syria_2
			remove_idea = divided_syria_1
		}
	}
	else_if = { limit = { has_idea = divided_syria }
		swap_ideas = {
			add_idea = divided_syria_1
			remove_idea = divided_syria
		}
	}
}

# State scope: adds the state's infrastructure and internet to minority_status
syr_add_minority_state_development = {
	if = {
		limit = { is_owned_by = SYR }
		add_to_temp_variable = {
			var = minority_status
			value = {
				value = infrastructure_level
				multiply = 2.5
				add = {
					value = building_level@internet_station
					multiply = 1.5
				}
			}
		}
	}
}

syr_add_religious_freedom_terms = {
	add_to_temp_variable = {
		var = minority_status
		value = {
			value = 100
			subtract = social_conservatism_society
			multiply = 0.25
		}
	}
	if = { limit = { has_completed_focus = SYR_islamist_renaissance }
		subtract_from_temp_variable = { minority_status = 10 }
	}
	if = { limit = { has_completed_focus = SYR_religious_values }
		subtract_from_temp_variable = { minority_status = 20 }
	}
	if = { limit = { has_completed_focus = SYR_empower_sharia_courts }
		subtract_from_temp_variable = { minority_status = 20 }
	}
}

syr_add_federalization_bonus = {
	if = {
		limit = {
			power_balance_value = {
				id = Syria_centralization_balance_of_power_category
				value < -0.5
			}
		}
		add_to_temp_variable = { minority_status = 20 }
	}
	if = {
		limit = {
			power_balance_value = {
				id = Syria_centralization_balance_of_power_category
				value < -0.9
			}
		}
		add_to_temp_variable = { minority_status = 20 }
	}
}

SYR_alawite_political_activity_modifier = {
	set_temp_variable = {
		minority_status = {
			value = 10
			subtract = corruption_level
			multiply = 2.5
		}
	}
	syr_add_religious_freedom_terms = yes
	189 = { syr_add_minority_state_development = yes }
	158 = { syr_add_minority_state_development = yes }
	syr_add_federalization_bonus = yes

	#assads in power effect
	if = {
		limit = { NOT = { has_completed_focus = SYR_kick_out_al_assads } }
		add_to_temp_variable = { minority_status = 40 }
	}

	set_variable = {
		alawite_status = {
			value = minority_status
			clamp = { min = 0 max = 100 }
		}
	}
	set_variable = {
		alawite_status_stability_modifier = {
			value = alawite_status
			subtract = 50
			multiply = 0.001
		}
	}
	set_variable = {
		alawite_status_drift_defense = {
			value = alawite_status
			subtract = 50
			multiply = -0.002
		}
	}

	# The has_idea pair mirrors the modifier's remove_trigger, so a federalised Syria does not get it re-attached
	if = {
		limit = {
			any_owned_state = { is_core_of = ALA }
			NOT = { has_idea = syria_federal_government }
			NOT = { has_idea = syria_bureaucratic_government }
			NOT = { has_dynamic_modifier = { modifier = alawite_political_inclusiveness_dynamic_modifier } }
		}
		add_dynamic_modifier = { modifier = alawite_political_inclusiveness_dynamic_modifier }
	}
	else_if = {
		limit = {
			has_dynamic_modifier = { modifier = alawite_political_inclusiveness_dynamic_modifier }
			NOT = { any_owned_state = { is_core_of = ALA } }
		}
		remove_dynamic_modifier = { modifier = alawite_political_inclusiveness_dynamic_modifier }
	}
}
