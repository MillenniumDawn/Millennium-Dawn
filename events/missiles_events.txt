			limit = { is_ai = no }

		}
	}
}

country_event = {
	id = satellites.10
	title = satellites.10.t
	desc = satellites.10.d
	picture = GFX_space_debris
	is_triggered_only = yes
	option = {
		name = satellites.10.a
		log = "[GetDateText]: [Root.GetName]: event satellites.10.a"
		hidden_effect = {
			set_variable = { MD_satellite_access_type = 1 }
			set_variable = { MD_satellite_renegotiation_rate = FROM.MD_GNSS_mil_standard_rate }
			accept_satellite_access_renegotiation = yes
		}
		ai_chance = { base = 50 modifier = { add = 50 check_variable = { treasury > 10 } } modifier = { add = 25 check_variable = { FROM.var_GNSS_mil_system_idx > ROOT.var_GNSS_mil_system_idx } } modifier = { add = 25 has_opinion = { target = FROM value > 49 } } }
	}
	option = {
		name = satellites.10.b
		log = "[GetDateText]: [Root.GetName]: event satellites.10.b"
		hidden_effect = {
			set_variable = { MD_satellite_access_type = 1 }
			decline_satellite_access_renegotiation = yes
		}
		ai_chance = { base = 10 modifier = { add = 100 check_variable = { treasury < 5 } } }
	}
}
