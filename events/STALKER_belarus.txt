country_event = {
	id = STALKER.63
	title = STALKER.63.t
	desc = STALKER.63.d
	picture = GFX_SLO_peacekeepers
	is_triggered_only = yes

	trigger = {
		STALKER_scenario_enabled = yes
		has_event_target = STALKER_commission_proposer
		event_target:STALKER_commission_proposer = { exists = yes }
		NOT = { has_global_flag = STALKER_joint_zone_commission }
		698 = {
			STALKER_is_active_zone_anchor = yes
			is_controlled_by = ROOT
		}
	}

	option = {
		name = STALKER.63.a
		log = "[GetDateText]: [This.GetName]: STALKER.63.a executed"
		custom_effect_tooltip = STALKER.63.a_tt
		hidden_effect = {
			if = {
				limit = {
					NOT = { has_global_flag = STALKER_joint_zone_commission }
					698 = {
						STALKER_is_active_zone_anchor = yes
						is_controlled_by = ROOT
					}
				}
				set_global_flag = STALKER_joint_zone_commission
				698 = {
					add_to_variable = { STALKER_zone_containment = 10 }
					add_to_variable = { STALKER_zone_bandits = -5 }
					STALKER_clamp_zone_society = yes
				}
				event_target:STALKER_commission_proposer = {
					add_opinion_modifier = { target = ROOT modifier = STALKER_granted_zone_access }
					STALKER_grant_zone_research_access = yes
					add_to_variable = { STALKER_polesie_security = 10 }
					clamp_variable = { var = STALKER_polesie_security min = 0 max = 100 }
					country_event = STALKER.65
				}
			}
		}
		ai_chance = {
			base = 50
			modifier = {
				add = 30
				has_opinion = { target = event_target:STALKER_commission_proposer value > 24 }
			}
			modifier = {
				factor = 0.2
				has_opinion = { target = event_target:STALKER_commission_proposer value < 0 }
			}
		}
	}

	option = {
		name = STALKER.63.b
		log = "[GetDateText]: [This.GetName]: STALKER.63.b executed"
		custom_effect_tooltip = STALKER.63.b_tt
		hidden_effect = {
			event_target:STALKER_commission_proposer = {
				add_opinion_modifier = { target = ROOT modifier = STALKER_refused_zone_access }
				country_event = STALKER.66
			}
		}
		ai_chance = {
			base = 30
			modifier = {
				add = 40
				has_opinion = { target = event_target:STALKER_commission_proposer value < 0 }
			}
		}
	}
}
