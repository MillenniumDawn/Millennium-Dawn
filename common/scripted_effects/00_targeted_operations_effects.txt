TOP_add_target_lead = {
	if = {
		limit = { TOP_enabled = yes }
		TOP_initialize_global = yes
		TOP_country_initialize = yes
		set_temp_variable = { TOP_target = TOP_arg_target }
		if = {
			limit = { check_variable = { global.TOP_status^TOP_target = 1 } }
			if = {
				limit = { check_variable = { TOP_known^TOP_target = 0 } }
				if = {
					limit = { check_variable = { global.TOP_public_identity^TOP_target = 1 } }
					set_variable = { TOP_identity_confidence^TOP_target = 100 }
					set_variable = { TOP_identity_source^TOP_target = 1 }
					set_variable = { TOP_location_confidence^TOP_target = 0 }
					set_variable = { TOP_pattern_confidence^TOP_target = 0 }
				}
				else = {
					set_variable = { TOP_identity_confidence^TOP_target = 15 }
					set_variable = { TOP_location_confidence^TOP_target = 15 }
					set_variable = { TOP_identity_source^TOP_target = 2 }
					set_variable = { TOP_location_source^TOP_target = 2 }
					set_variable = { TOP_pattern_confidence^TOP_target = 0 }
				}
			}
			set_variable = { TOP_known^TOP_target = 1 }
			set_variable = { TOP_assessment^TOP_target = 1 }
			set_temp_variable = { TOP_old_identity = TOP_identity_confidence^TOP_target }
			set_temp_variable = { TOP_old_location = TOP_location_confidence^TOP_target }
			set_temp_variable = { TOP_old_pattern = TOP_pattern_confidence^TOP_target }
			add_to_variable = { TOP_identity_confidence^TOP_target = TOP_arg_amount }
			add_to_variable = { TOP_location_confidence^TOP_target = TOP_arg_amount }
			add_to_variable = { TOP_pattern_confidence^TOP_target = TOP_arg_amount }
			clamp_variable = { var = TOP_identity_confidence^TOP_target min = 0 max = 100 }
			clamp_variable = { var = TOP_location_confidence^TOP_target min = 0 max = 100 }
			clamp_variable = { var = TOP_pattern_confidence^TOP_target min = 0 max = 100 }
			if = { limit = { check_variable = { TOP_identity_confidence^TOP_target > TOP_old_identity } } set_variable = { TOP_identity_source^TOP_target = 2 } }
			if = { limit = { check_variable = { TOP_location_confidence^TOP_target > TOP_old_location } } set_variable = { TOP_location_source^TOP_target = 2 } }
			if = { limit = { check_variable = { TOP_pattern_confidence^TOP_target > TOP_old_pattern } } set_variable = { TOP_pattern_source^TOP_target = 2 } }
			set_temp_variable = { TOP_old_lead_state = TOP_lead_state^TOP_target }
			set_temp_variable = { TOP_old_lead_host = TOP_lead_host^TOP_target }
			set_temp_variable = { TOP_old_lead_clock = TOP_lead_report_clock^TOP_target }
			set_variable = { TOP_lead_age^TOP_target = 0 }
			set_variable = { TOP_lead_report_clock^TOP_target = global.TOP_clock }
			set_variable = { TOP_lead_state^TOP_target = global.TOP_state^TOP_target }
			set_variable = { TOP_lead_host^TOP_target = global.TOP_host^TOP_target }
			if = {
				limit = { OR = { NOT = { check_variable = { TOP_lead_state^TOP_target = TOP_old_lead_state } } NOT = { check_variable = { TOP_lead_host^TOP_target = TOP_old_lead_host } } NOT = { check_variable = { TOP_lead_report_clock^TOP_target = TOP_old_lead_clock } } } }
				set_variable = { TOP_lead_source^TOP_target = 2 }
			}
			set_temp_variable = { TOP_group_target = global.TOP_affiliation^TOP_target }
			if = {
				limit = { check_variable = { TOP_group_target > 0 } }
				set_temp_variable = { TOP_arg_group = TOP_group_target }
				TOP_discover_organization = yes
			}
			if = {
				limit = { NOT = { is_in_array = { TOP_dossiers = TOP_target } } }
				add_to_array = { TOP_dossiers = TOP_target }
			}
			TOP_update_person_compat_confidence = yes
		}
	}
}

TOP_process_collection_assignments = {
	clear_array = TOP_collecting_subjects_remaining
	for_each_loop = {
		array = TOP_collecting_subjects
		value = top_collection_subject
		if = {
			limit = { check_variable = { top_collection_subject < 1000 } }
			set_temp_variable = { TOP_target = top_collection_subject }
			set_temp_variable = { TOP_role_target = TOP_target }
			if = {
				limit = {
					check_variable = { TOP_target > 0 }
					check_variable = { TOP_target < global.TOP_status^num }
					check_variable = { global.TOP_status^TOP_target = 1 }
					check_variable = { TOP_package_state^TOP_target = 2 }
					TOP_authored_role_eligible = yes
				}
				add_to_array = { TOP_collecting_subjects_remaining = top_collection_subject }
				TOP_collect_person_pulse = yes
			}
			else_if = {
				limit = {
					check_variable = { TOP_target > 0 }
					check_variable = { TOP_target < global.TOP_status^num }
					check_variable = { TOP_package_state^TOP_target = 2 }
				}
				set_variable = { TOP_package_state^TOP_target = 1 }
				set_temp_variable = { TOP_arg_target = TOP_target }
				set_temp_variable = { TOP_history_current_event = 3 }
				set_temp_variable = { TOP_history_current_reason = 23 }
				TOP_log_person_milestone = yes
			}
		}
		else = {
			set_temp_variable = { TOP_group_target = { value = top_collection_subject subtract = 1000 } }
			if = {
				limit = {
					check_variable = { TOP_group_target > 0 }
					check_variable = { TOP_group_target < global.TOP_group_class^num }
					check_variable = { TOP_org_package_state^TOP_group_target = 2 }

				}

				limit = {

					check_variable = { TOP_group_target < global.TOP_group_class^num }
					check_variable = { TOP_org_package_state^TOP_group_target = 2 }

				set_temp_variable = { TOP_arg_group = TOP_group_target }

				set_temp_variable = { TOP_history_current_reason = 23 }
				TOP_log_organization_milestone = yes
			}
		}
	}
	clear_array = TOP_collecting_subjects
	for_each_loop = {
