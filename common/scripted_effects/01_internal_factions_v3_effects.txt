# Activates faction if_id; when all 4 slots are full it takes the slot of if_replace_id, or of the lowest-influence active faction when if_replace_id is 0
if_add_faction = {
	if = {
		limit = { NOT = { is_in_array = { array = if_active value = if_id } } }
		if = {
			limit = { check_variable = { if_active^num < 4 } }
			add_to_array = { if_active = if_id }
		}
		else = {
			if = {
				limit = { NOT = { is_in_array = { array = if_active value = if_replace_id } } }
				set_temp_variable = { if_replace_id = if_active^0 }
				for_each_loop = {
					array = if_active
					value = if_v
					if = {
						limit = { check_variable = { if_influence^if_v < if_influence^if_replace_id } }
						set_temp_variable = { if_replace_id = if_v }
					}
				}
			}
			set_temp_variable = { if_dm_id = if_replace_id }
			if_detach_dynmod = yes
			set_temp_variable = { if_clear_id = if_replace_id }
			if_clear_faction_state = yes
			for_each_loop = {
				array = if_active
				index = if_i
				value = if_v
				if = {
					limit = { check_variable = { if_v = if_replace_id } }
					set_variable = { if_active^if_i = if_id }
				}
			}
		}
		set_variable = { if_opinion^if_id = 50 }
		set_variable = { if_influence^if_id = global.if_base_influence^if_id }
		set_temp_variable = { if_dm_id = if_id }
		if_attach_dynmod = yes
		if_apply_faction_modifiers = yes
		update_if_dirty_variable = yes
	}
	set_temp_variable = { if_replace_id = 0 }
}

# Sets if_replace_id to the lowest-influence active faction whose idea is not held, 0 when none
if_pick_rule_filled_slot = {
	set_temp_variable = { if_replace_id = 0 }
	for_each_loop = {
		array = if_active
		value = if_v
		set_temp_variable = { if_tok = global.if_token^if_v }
		if = {
			limit = {
				NOT = { has_idea = var:if_tok }
				OR = {
					check_variable = { if_replace_id = 0 }
					check_variable = { if_influence^if_v < if_influence^if_replace_id }
				}
			}
			set_temp_variable = { if_replace_id = if_v }
		}
	}
}

# Parameters:

	}
}

if_seed_random_fill = {
	clear_temp_array = if_pool
	for_loop_effect = {
		start = 1
		end = 24
		value = if_id
		if = {
			limit = {
				NOT = { is_in_array = { array = if_active value = if_id } }
				if_is_available = yes
			}
			add_to_temp_array = { if_pool = if_id }
		}
	}
	while_loop_effect = {
		limit = {
			check_variable = { if_active^num < 4 }
			check_variable = { if_pool^num > 0 }
		}
		set_temp_variable = { if_last = if_pool^num }
		subtract_from_temp_variable = { if_last = 1 }
		set_temp_variable_to_random = { var = if_r min = 0 max = if_last integer = yes }
		set_temp_variable = { if_id = if_pool^if_r }
		if = {
			limit = { if_is_available = yes }
			if_add_faction = yes
		}
		remove_from_temp_array = { array = if_pool index = if_r }
	}
}

# Parameters:
# - nation_to_copy_from: country whose if_active, if_opinion and if_influence are copied; factions backing the new ruling party mirror opinion around 50
if_copy_factions = {
	if_init_arrays = yes
	clear_array = if_active
	for_each_loop = {
		array = var:nation_to_copy_from.if_active
		value = if_v
		add_to_array = { if_active = if_v }
		set_temp_variable = { if_dm_id = if_v }
		if_attach_dynmod = yes
		set_variable = { if_opinion^if_v = var:nation_to_copy_from.if_opinion^if_v }
		set_variable = { if_influence^if_v = var:nation_to_copy_from.if_influence^if_v }
		set_temp_variable = { if_aff = if_v }
		multiply_temp_variable = { if_aff = 13 }
		add_to_temp_variable = { if_aff = global.ip_party_policy_group^ruling_party }
		if = {
			limit = { check_variable = { global.if_affinity^if_aff = 1 } }
			set_variable = { if_opinion^if_v = 100 }
			subtract_from_variable = { if_opinion^if_v = var:nation_to_copy_from.if_opinion^if_v }
		}
	}
	if_apply_modifiers = yes
	update_if_dirty_variable = yes
}

# Parameters (set by if_monthly_tick):
# - if_v: faction id
# - if_group: policy group of the ruling party
# - if_mil_term, if_pol_term, if_edu_term, if_soc_term, if_corp_term, if_pop_term: hoisted law terms
# - if_priv_count, if_crackdown: temp arrays built by if_monthly_tick from if_policies
if_compute_opinion_target = {
	set_temp_variable = { if_t = 50 }
	if = {
		limit = { check_variable = { global.if_affinity^if_aff = 1 } }
		add_to_temp_variable = { if_t = 15 }
	}
	else_if = {
		limit = { check_variable = { global.if_affinity^if_aff = -1 } }
		subtract_from_temp_variable = { if_t = 10 }
	}
	if = {
		limit = { check_variable = { gov_coalition_array^num > 0 } }
		set_temp_variable = { if_col = 0 }
		for_each_loop = {
			array = gov_coalition_array
			value = if_party
			set_temp_variable = { if_aff_col = if_v }
			multiply_temp_variable = { if_aff_col = 13 }
			add_to_temp_variable = { if_aff_col = global.ip_party_policy_group^if_party }
			if = {
				limit = { check_variable = { global.if_affinity^if_aff_col = 1 } }
				add_to_temp_variable = { if_col = 15 }
			}
			else_if = {
				limit = { check_variable = { global.if_affinity^if_aff_col = -1 } }
				subtract_from_temp_variable = { if_col = 10 }
			}
		}
		divide_temp_variable = { if_col = gov_coalition_array^num }
		multiply_temp_variable = { if_col = 0.5 }
		add_to_temp_variable = { if_t = if_col }
	}
	set_temp_variable = { if_x = if_priv_count^if_v }
	multiply_temp_variable = { if_x = 10 }
	clamp_temp_variable = { var = if_x min = 0 max = 20 }
	add_to_temp_variable = { if_t = if_x }
	if = {
		limit = { check_variable = { if_crackdown^if_v = 1 } }
		subtract_from_temp_variable = { if_t = 15 }
	}
	set_temp_variable = { if_x = 0 }

		array = if_active

				NOT = { check_variable = { if_j = if_v } }

			set_temp_variable = { if_r = if_v }

			add_to_temp_variable = { if_r = if_j }
			if = {
				limit = { check_variable = { global.if_rival^if_r = 1 } }
				set_temp_variable = { if_y = if_priv_count^if_j }
				multiply_temp_variable = { if_y = 5 }
				subtract_from_temp_variable = { if_x = if_y }
			}
		}
	}
	clamp_temp_variable = { var = if_x min = -10 max = 0 }
	add_to_temp_variable = { if_t = if_x }
	set_temp_variable = { if_law = global.if_pref_military_law^if_v }
	multiply_temp_variable = { if_law = if_mil_term }
	set_temp_variable = { if_x = global.if_pref_police_law^if_v }
	multiply_temp_variable = { if_x = if_pol_term }
	add_to_temp_variable = { if_law = if_x }
	set_temp_variable = { if_x = global.if_pref_education_law^if_v }
	multiply_temp_variable = { if_x = if_edu_term }
	add_to_temp_variable = { if_law = if_x }
	set_temp_variable = { if_x = global.if_pref_social_law^if_v }
	multiply_temp_variable = { if_x = if_soc_term }
	add_to_temp_variable = { if_law = if_x }
	set_temp_variable = { if_x = global.if_pref_corporate_tax^if_v }

	add_to_temp_variable = { if_law = if_x }
	set_temp_variable = { if_x = global.if_pref_population_tax^if_v }

# Monthly: recompute targets for every active faction, drift opinion (global.if_drift_rate) and influence toward them, tick cooldowns, run the AI routine
if_monthly_tick = {
	if = {
		limit = { check_variable = { if_active^num > 0 } }
		if_init_arrays = yes
		set_temp_variable = { if_group = global.ip_party_policy_group^ruling_party }
		set_temp_variable = { if_mil_term = military_law }
		subtract_from_temp_variable = { if_mil_term = 5 }
		set_temp_variable = { if_pol_term = police_law }
		subtract_from_temp_variable = { if_pol_term = 3 }
		multiply_temp_variable = { if_pol_term = 2 }
		set_temp_variable = { if_edu_term = education_law }
		subtract_from_temp_variable = { if_edu_term = 3 }
		multiply_temp_variable = { if_edu_term = 2 }
		set_temp_variable = { if_soc_term = social_law }
		subtract_from_temp_variable = { if_soc_term = 3 }
		multiply_temp_variable = { if_soc_term = 2 }
		set_temp_variable = { if_corp_term = corporate_tax_rate }
		subtract_from_temp_variable = { if_corp_term = 25 }
		multiply_temp_variable = { if_corp_term = 0.2 }
		set_temp_variable = { if_pop_term = population_tax_rate }
		subtract_from_temp_variable = { if_pop_term = 25 }
		multiply_temp_variable = { if_pop_term = 0.2 }
		set_temp_variable = { if_oil_share = 0 }
		if = {
			limit = { check_variable = { gdp_total > 0 } }
			set_temp_variable = { if_oil_share = oil_exports }
			divide_temp_variable = { if_oil_share = gdp_total }
		}
		set_temp_variable = { if_war = 0 }
		if = {
			limit = { has_war = yes }
			set_temp_variable = { if_war = 10 }
		}
		resize_temp_array = { array = if_priv_count size = 24 }
		resize_temp_array = { array = if_crackdown size = 24 }
		for_each_loop = {
			array = if_policies
			value = if_p
			set_temp_variable = { if_pf = global.if_policy_faction^if_p }
			if = {
				limit = { check_variable = { global.if_policy_kind^if_p = 1 } }
				add_to_temp_variable = { if_priv_count^if_pf = 1 }
			}
			else = {
				set_temp_variable = { if_crackdown^if_pf = 1 }
			}
		}
		for_each_loop = {
			array = if_active
			value = if_v
			set_temp_variable = { if_aff = if_v }
			multiply_temp_variable = { if_aff = 13 }
			add_to_temp_variable = { if_aff = if_group }
			if_compute_opinion_target = yes
			if_compute_influence_target = yes
			set_temp_variable = { if_d = if_target^if_v }
			subtract_from_temp_variable = { if_d = if_opinion^if_v }
			multiply_temp_variable = { if_d = global.if_drift_rate }
			add_to_variable = { if_opinion^if_v = if_d }
			clamp_variable = { var = if_opinion^if_v min = 0 max = 100 }
			set_temp_variable = { if_d = if_inf_target^if_v }
			subtract_from_temp_variable = { if_d = if_influence^if_v }
			multiply_temp_variable = { if_d = @if_influence_drift_rate }
			add_to_variable = { if_influence^if_v = if_d }
			clamp_variable = { var = if_influence^if_v min = 0 max = 100 }
			if = {
				limit = { check_variable = { if_policy_cooldown^if_v > 0 } }
				subtract_from_variable = { if_policy_cooldown^if_v = 1 }
			}
			if = {
				limit = { check_variable = { if_swap_cooldown^if_v > 0 } }
				subtract_from_variable = { if_swap_cooldown^if_v = 1 }
			}
		}
		if_party_push = yes
		if_apply_modifiers = yes
		if = {
			limit = { is_ai = yes }
			if_ai_monthly = yes
		}
		update_if_dirty_variable = yes
	}
}
