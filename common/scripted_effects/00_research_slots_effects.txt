## SCOPE: Country
research_slot_init_for_country = {
	set_research_slots = 0
	set_variable = { research_slots_from_gdp = 0 }
	set_variable = { next_slot_from_gdp = 1 }
	set_variable = { research_slots_from_gdp_c = 0 }
	set_variable = { next_slot_from_gdp_c = 1 }

	# Thresholds only rise, so stopping at the first unmet slot grants every met one
	while_loop_effect = {
		limit = {
			check_variable = { next_slot_from_gdp < 10 }
			check_variable = { gdp_total > GLOBAL.gdp_slot^next_slot_from_gdp }
		}
		add_research_slot = 1
		add_to_variable = { research_slots_from_gdp = 1 }
		add_to_variable = { next_slot_from_gdp = 1 }
	}
	while_loop_effect = {
		limit = {
			check_variable = { next_slot_from_gdp_c < 7 }
			check_variable = { gdp_per_capita > GLOBAL.gdp_c_slot^next_slot_from_gdp_c }
			check_variable = { var = GLOBAL.gdp_c_slot_civs^next_slot_from_gdp_c value = industrial_complex_total compare = less_than_or_equals }
		}
		add_research_slot = 1
		add_to_variable = { research_slots_from_gdp_c = 1 }
		add_to_variable = { next_slot_from_gdp_c = 1 }
	}
}

## SCOPE: Country
# Runs after the weekly GDP refresh instead of letting the engine poll the missions' activation
activate_research_slot_loss_missions = {
	if = {
		limit = {
			NOT = { has_active_mission = gdp_lose_slot }
			check_variable = { gdp_total < GLOBAL.gdp_slot^research_slots_from_gdp }
		}
		activate_mission = gdp_lose_slot
	}
	if = {
		limit = {
			NOT = { has_active_mission = gdp_c_lose_slot }
			OR = {
				check_variable = { gdp_per_capita < GLOBAL.gdp_c_slot^research_slots_from_gdp_c }
				check_variable = { industrial_complex_total < GLOBAL.gdp_c_slot_civs^research_slots_from_gdp_c }
			}
		}
		activate_mission = gdp_c_lose_slot
	}
}

add_research_buff = {
	if = {
		limit = { has_idea = research_idea_1 }
		swap_ideas = {
			remove_idea = research_idea_1
			add_idea = research_idea_2
		}
	}
	else_if = {
		limit = { has_idea = research_idea_2 }

		swap_ideas = {
			remove_idea = research_idea_2
			add_idea = research_idea_3
		}
	}
	else_if = {
		limit = { has_idea = research_idea_3 }
		add_political_power = 150
	}
	else = { add_ideas = research_idea_1 }
}
