# Foreign services follow the access agreements and the artifact money into every Zone.
STALKER_monthly_intel_pulse = {
	for_each_scope_loop = {
		array = global.STALKER_active_zone_anchors
		if = {
			limit = {
				STALKER_is_active_zone_anchor = yes
				NOT = { has_state_flag = STALKER_zone_policy_pending }
				controller = { exists = yes }
			}
			set_temp_variable = {
				STALKER_intel_target = {
					value = global.STALKER_research_access_count
					multiply = 5
					add = { value = STALKER_zone_exploitation multiply = 0.4 }
					clamp = { min = 0 max = 100 }
				}
			}
			add_to_variable = { STALKER_zone_foreign_intel = { value = STALKER_intel_target subtract = STALKER_zone_foreign_intel multiply = 0.1 } }
			clamp_variable = { var = STALKER_zone_foreign_intel min = 0 max = 100 }
			if = {
				limit = { check_variable = { STALKER_zone_foreign_intel > 49 } }
				controller = {
					if = {
						limit = { has_variable = STALKER_mdst_penetration }
						add_to_variable = { STALKER_mdst_penetration = 1 }
						clamp_variable = { var = STALKER_mdst_penetration min = 0 max = 80 }
					}
				}
				if = {
					limit = { NOT = { has_state_flag = STALKER_foreign_agents_cooldown } }
					random = {
						chance = 10
						set_state_flag = { flag = STALKER_foreign_agents_cooldown value = 1 days = 365 }
						STALKER_target_zone = yes
						controller = { country_event = STALKER.140 }
					}
				}
			}
		}
	}
}
