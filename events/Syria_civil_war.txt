add_namespace = SyriaCivilWar

# Regime officers defect to the Free Syrian Army
country_event = {
	id = SyriaCivilWar.1
	title = SyriaCivilWar.1.t
	desc = SyriaCivilWar.1.d
	picture = GFX_arab_spring_defection
	is_triggered_only = yes

	trigger = {
		has_war_with = FSA
		FSA = { num_of_controlled_states > 0 }
	}

	immediate = {
		# One brigade at 90% support, up to four once support collapses
		set_temp_variable = { regime_support = party_pop_array^ruling_party }
		set_temp_variable = { wave_size = { value = 1 subtract = regime_support multiply = 4 add = 0.5 } }
		set_variable = { SYR_defector_wave_size = wave_size }
		round_variable = SYR_defector_wave_size
		clamp_variable = { var = SYR_defector_wave_size min = 1 max = 4 }
	}

	# Hunt down the deserters
	option = {
		name = SyriaCivilWar.1.a
		log = "[GetDateText]: [This.GetName]: SyriaCivilWar.1.a executed ([?SYR_defector_wave_size] brigades)"
		set_temp_variable = { SYR_defector_count = SYR_defector_wave_size }
		SYR_defectors_join_fsa = yes
		add_stability = -0.02
		set_temp_variable = { party_popularity_increase = -0.01 }
		change_relative_party_popularity = yes
		custom_effect_tooltip = SYR_defections_suppressed_tt
		set_country_flag = { flag = SYR_defections_suppressed days = 90 value = 1 }
		ai_chance = {
			base = 10
			modifier = { factor = 2 SYR_ai_hardline_path = yes }
			modifier = { factor = 0 has_stability < 0.25 }
		}
	}

	# Offer an amnesty to those who return
	option = {
		name = SyriaCivilWar.1.b
		log = "[GetDateText]: [This.GetName]: SyriaCivilWar.1.b executed ([?SYR_defector_wave_size] brigades)"
		add_political_power = -50
		custom_effect_tooltip = SYR_defectors_amnesty_tt
		set_temp_variable = { SYR_defector_count = SYR_defector_wave_size }
		subtract_from_temp_variable = { SYR_defector_count = 1 }
		if = {
			limit = { check_variable = { SYR_defector_count > 0 } }
			SYR_defectors_join_fsa = yes
		}
		ai_chance = {
			base = 10
			modifier = { factor = 2 SYR_ai_moderate_path = yes }
			modifier = { factor = 0.25 has_active_mission = bankruptcy_incoming_collapse }
		}
	}

	# Blame foreign conspirators
	option = {
		name = SyriaCivilWar.1.c
		log = "[GetDateText]: [This.GetName]: SyriaCivilWar.1.c executed ([?SYR_defector_wave_size] brigades)"
		set_temp_variable = { SYR_defector_count = SYR_defector_wave_size }
		SYR_defectors_join_fsa = yes
		add_war_support = 0.01
		set_temp_variable = { party_popularity_increase = -0.01 }
		change_relative_party_popularity = yes
		ai_chance = { base = 5 }
	}
}

# The Islamic State regroups in the desert. ROOT owns the state it rises in
country_event = {
	id = SyriaCivilWar.40
	title = SyriaCivilWar.40.t
	desc = SyriaCivilWar.40.d
	picture = GFX_SYR_tahrir_al_sham
	is_triggered_only = yes

	trigger = {
		NOT = { country_exists = ISI }
		any_owned_state = {
			OR = {
				state = 192
				state = 995
				state = 561
				state = 994
			}
		}
	}

	immediate = {
		hidden_effect = {
			clr_global_flag = ISIS_resurgence_queued
			add_to_variable = { global.isis_resurgences = 1 }
			random_owned_state = {
				limit = {
					OR = {
						state = 192
						state = 995
						state = 561
						state = 994
					}
				}
				save_event_target_as = isis_resurgence_state
			}
			# Leftover cores would hand the new insurgency every state we own
			every_state = {
				limit = { is_core_of = ISI }
				remove_core_of = ISI
			}
			event_target:isis_resurgence_state = { add_core_of = ISI }
			release = ISI
			ISI = {
				ingame_startup_nations_effect = yes
				# Its history capital is North Raqqa, which it may not hold
				if = { limit = { owns_state = 192 } set_capital = { state = 192 } }
				else_if = { limit = { owns_state = 995 } set_capital = { state = 995 } }
				else_if = { limit = { owns_state = 561 } set_capital = { state = 561 } }
				else_if = { limit = { owns_state = 994 } set_capital = { state = 994 } }
				if = {
					limit = { NOT = { has_template = "Liwa Milisha" } }
					division_template = {
						name = "Liwa Milisha"
						regiments = {
							Mot_Militia_Bat = { x = 0 y = 0 }
							Mot_Militia_Bat = { x = 0 y = 1 }
							Mot_Militia_Bat = { x = 0 y = 2 }
						}
					}
				}
				# The resurgence state is ISI's only state, so it is also its capital
				capital_scope = {
					create_unit = {
						division = "name = \"Liwa Milisha\" division_template = \"Liwa Milisha\" start_experience_factor = 0.5"
						owner = ISI
						count = 3
					}
				}
				declare_war_on = { target = ROOT type = annex_everything }
			}
			news_event = { id = SyriaCivilWar.41 hours = 6 }
		}
	}

	option = {
		name = SyriaCivilWar.40.a
		log = "[GetDateText]: [This.GetName]: SyriaCivilWar.40.a executed"
		custom_effect_tooltip = SYR_isis_resurgence_tt
	}
}
