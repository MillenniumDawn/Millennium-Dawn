# Siemens corporate history. Five milestones set the route; the closing
# milestone resolves it into one of four shapes for the group.

GER_siemens_apply_focused_industrials = {
	if = {
		limit = { GER_siemens_outcome_settled = no }
		add_ideas = GER_siemens_focused_industrials
	}
}

# One entry point per milestone, called from the yearly pulse. Full presents the
# choice; Outcomes Only writes the historical route silently and grants nothing.
GER_siemens_milestone_2001 = {
	if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = full } }
		country_event = { id = germany_yearly.300 days = 119 }
	}
	else_if = {
		limit = { has_game_rule = { rule = rule_corporate_history option = outcomes_only } }
		set_country_flag = GER_siemens_integrated_conglomerate
	}
}
