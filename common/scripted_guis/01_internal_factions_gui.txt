scripted_gui = {
	internal_faction_open_topbar_button = {
		context_type = player_context
		window_name = "internal_faction_open_topbar_window"

		visible = { is_ai = no }

		ai_enabled = { always = no }

		effects = {
			internal_faction_open_topbar_btn_click = { internal_faction_toggle_window = yes }
		}

		properties = {
			internal_faction_open_topbar_btn = { x = global.var_missiles_button_x }
		}
	}

	internal_faction_open_politics_button = {
		context_type = player_context
		window_name = "internal_faction_open_politics_window"
		parent_window_token = politics_tab

		ai_enabled = { always = no }

		effects = {
			internal_faction_open_politics_btn_click = { internal_faction_toggle_window = yes }
		}
	}

	internal_faction_main_gui = {
		context_type = player_context
		window_name = "internal_faction_main_window"

		dirty = global.internal_faction_ui_dirty

		visible = { has_country_flag = internal_faction_window_open }

		ai_enabled = { always = no }

		dynamic_lists = {
			internal_faction_faction_grid = {
				array = internal_faction_active
				entry_container = "internal_faction_faction_entry"
			}
			internal_faction_inactive_grid = {
				array = internal_faction_inactive
				entry_container = "internal_faction_inactive_entry"
			}
		}

		effects = {
			internal_faction_close_btn_click = { clr_country_flag = internal_faction_window_open }
			internal_faction_policy_1_btn_click = {
				set_temp_variable = { internal_faction_p = v }
				multiply_temp_variable = { internal_faction_p = 4 }
				subtract_from_temp_variable = { internal_faction_p = 3 }
				internal_faction_toggle_privilege = yes
			}
			internal_faction_policy_2_btn_click = {
				set_temp_variable = { internal_faction_p = v }
				multiply_temp_variable = { internal_faction_p = 4 }
				subtract_from_temp_variable = { internal_faction_p = 2 }
				internal_faction_toggle_privilege = yes
			}
			internal_faction_policy_3_btn_click = {
				set_temp_variable = { internal_faction_p = v }
				multiply_temp_variable = { internal_faction_p = 4 }
				subtract_from_temp_variable = { internal_faction_p = 1 }
				internal_faction_toggle_privilege = yes
			}
			internal_faction_policy_4_btn_click = {
				set_temp_variable = { internal_faction_id = v }
				internal_faction_toggle_crackdown = yes
			}
			internal_faction_appease_btn_click = {
				set_temp_variable = { internal_faction_id = v }
				internal_faction_appease = yes
			}
			internal_faction_support_btn_click = {
				set_temp_variable = { internal_faction_id = v }
				internal_faction_support_faction = yes
			}
		}

		triggers = {
			internal_faction_policy_1_btn_click_enabled = {
				set_temp_variable = { internal_faction_p = v }
				multiply_temp_variable = { internal_faction_p = 4 }
				subtract_from_temp_variable = { internal_faction_p = 3 }
				OR = {
					internal_faction_can_grant_privilege = yes
					internal_faction_can_revoke_privilege = yes
				}
			}
			internal_faction_policy_2_btn_click_enabled = {
				set_temp_variable = { internal_faction_p = v }
				multiply_temp_variable = { internal_faction_p = 4 }
				subtract_from_temp_variable = { internal_faction_p = 2 }
				OR = {
					internal_faction_can_grant_privilege = yes
					internal_faction_can_revoke_privilege = yes
				}
			}
			internal_faction_policy_3_btn_click_enabled = {
				set_temp_variable = { internal_faction_p = v }
				multiply_temp_variable = { internal_faction_p = 4 }
				subtract_from_temp_variable = { internal_faction_p = 1 }
				OR = {
					internal_faction_can_grant_privilege = yes
					internal_faction_can_revoke_privilege = yes
				}
			}
			internal_faction_policy_4_btn_click_enabled = {
				set_temp_variable = { internal_faction_id = v }
				OR = {
					internal_faction_can_enact_crackdown = yes
					internal_faction_can_lift_crackdown = yes
				}
			}
			internal_faction_appease_btn_click_enabled = {
				set_temp_variable = { internal_faction_id = v }
				internal_faction_can_appease = yes
			}
			internal_faction_support_btn_click_enabled = {
				set_temp_variable = { internal_faction_id = v }
				internal_faction_can_support = yes
			}
		}

		properties = {
			internal_faction_entry_icon = { image = "GFX_idea_[?global.internal_faction_token^v.GetTokenKey]" }
			internal_faction_entry_opinion_bar = { frame = internal_faction_opinion^v }
			internal_faction_entry_influence_bar = { frame = internal_faction_influence^v }
			internal_faction_inactive_icon = { image = "GFX_idea_[?global.internal_faction_token^v.GetTokenKey]" }
			internal_faction_inactive_influence_bar = { frame = internal_faction_influence^v }
			internal_faction_policy_1_btn = { image = "[internal_faction_policy_1_icon]" }
			internal_faction_policy_2_btn = { image = "[internal_faction_policy_2_icon]" }
			internal_faction_policy_3_btn = { image = "[internal_faction_policy_3_icon]" }
			internal_faction_policy_4_btn = { image = "[internal_faction_policy_4_icon]" }
		}
	}
}
