		icon = GFX_decision_dockyard_dec

		cost = 60

		available = {
			custom_trigger_tooltip = {
				tooltip = ukr_yard_free_slipway_tt
				check_variable = { shipyard_dockyards < shipyard_effective_dockyards }
			}
		}

		custom_cost_trigger = { check_variable = { treasury > ukr_cost_dockyard } }
		custom_cost_text = ukr_cost_dockyard_tt

		days_remove = 120
		days_re_enable = 90

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision UKR_yard_restore_dockyard"
			set_temp_variable = { treasury_change = ukr_cost_dockyard }
			multiply_temp_variable = { treasury_change = -1 }
			modify_treasury_effect = yes
			custom_effect_tooltip = ukr_yard_restore_dockyard_effect_tt
		}

		remove_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision UKR_yard_restore_dockyard"
			hidden_effect = {
				# An existing yard with room to grow first, any other valid coast after.
				if = {
					limit = {
						any_controlled_state = {
							dockyard_random_build_loc = yes
							check_variable = { building_level@dockyard > 0 }
							free_building_slots = { building = dockyard size > 0 include_locked = no }
						}
					}
					random_controlled_state = {
						limit = {
							dockyard_random_build_loc = yes
							check_variable = { building_level@dockyard > 0 }
							free_building_slots = { building = dockyard size > 0 include_locked = no }
						}
						add_building_construction = {
							type = dockyard
							level = 1
							instant_build = yes
						}
					}
				}
				else = {
					set_temp_variable = { skip_payment = 1 }
					one_random_dockyard = yes
				}
				shipyard_recalc = yes
			}
		}

		ai_will_do = {
			base = 8
			modifier = { factor = 0.5 ai_has_high_deficit = yes }
			modifier = { factor = 0 has_active_mission = bankruptcy_incoming_collapse }
		}
	}
