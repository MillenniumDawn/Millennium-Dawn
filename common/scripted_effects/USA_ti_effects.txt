USA_ti_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = USA_ti_state_initialized } }

		set_variable = { USA_ti_embedded_reach = 5 }
		set_variable = { USA_ti_dsp_leadership = 5 }
		set_variable = { USA_ti_manufacturing_control = 5 }
		set_variable = { USA_ti_defense_integration = 5 }
		set_country_flag = USA_ti_state_initialized
	}

	USA_ti_clamp_state = yes
	USA_ti_normalize_state = yes
}

USA_ti_clamp_state = {
	clamp_variable = { var = USA_ti_embedded_reach min = 0 max = 10 }
	clamp_variable = { var = USA_ti_dsp_leadership min = 0 max = 10 }
	clamp_variable = { var = USA_ti_manufacturing_control min = 0 max = 10 }
	clamp_variable = { var = USA_ti_defense_integration min = 0 max = 10 }
}

USA_ti_normalize_state = {
	if = {
		limit = { has_country_flag = USA_ti_defense_sold }
		clr_country_flag = USA_ti_defense_retained
		clr_country_flag = USA_ti_defense_affiliate
	}
	else_if = {
		limit = { has_country_flag = USA_ti_defense_retained }
		clr_country_flag = USA_ti_defense_affiliate
	}
	else_if = {
		limit = { has_country_flag = USA_ti_event_12_resolved }
		set_country_flag = USA_ti_defense_sold
	}

	if = {
		limit = { has_country_flag = USA_ti_omap_neutral }
		clr_country_flag = USA_ti_omap_stack
		clr_country_flag = USA_ti_omap_open
	}
	else_if = {
		limit = { has_country_flag = USA_ti_omap_stack }
		clr_country_flag = USA_ti_omap_open
	}
	else_if = {
		limit = { has_country_flag = USA_ti_event_13_resolved }
		set_country_flag = USA_ti_omap_neutral
	}

	if = {
		limit = { has_country_flag = USA_ti_kilby_central }
		clr_country_flag = USA_ti_kilby_product_labs
		clr_country_flag = USA_ti_kilby_public_consortium
	}
	else_if = {
		limit = { has_country_flag = USA_ti_kilby_product_labs }
		clr_country_flag = USA_ti_kilby_public_consortium
	}
	else_if = {
		limit = { has_country_flag = USA_ti_kilby_public_consortium }
		clr_country_flag = USA_ti_kilby_central
		clr_country_flag = USA_ti_kilby_product_labs
	}
	else_if = {
		limit = { has_country_flag = USA_ti_event_14_resolved }
		set_country_flag = USA_ti_kilby_central
	}

	if = {
		limit = { has_country_flag = USA_ti_national_integrated }
		clr_country_flag = USA_ti_national_federation
		clr_country_flag = USA_ti_national_constrained
	}
	else_if = {
		limit = { has_country_flag = USA_ti_national_federation }
		clr_country_flag = USA_ti_national_constrained
	}
	else_if = {
		limit = { has_country_flag = USA_ti_national_constrained }
		clr_country_flag = USA_ti_national_integrated
		clr_country_flag = USA_ti_national_federation
	}
	else_if = {
		limit = { has_country_flag = USA_ti_event_15_resolved }
		set_country_flag = USA_ti_national_integrated
	}

	if = {
		limit = { has_country_flag = USA_ti_omap_exited }
		clr_country_flag = USA_ti_omap_mobile_bet
		clr_country_flag = USA_ti_omap_embedded_edge
	}
	else_if = {
		limit = { has_country_flag = USA_ti_omap_mobile_bet }
		clr_country_flag = USA_ti_omap_embedded_edge
	}
	else_if = {
		limit = { has_country_flag = USA_ti_omap_embedded_edge }
		clr_country_flag = USA_ti_omap_exited
		clr_country_flag = USA_ti_omap_mobile_bet
	}
	else_if = {
		limit = { has_country_flag = USA_ti_event_16_resolved }
		set_country_flag = USA_ti_omap_embedded_edge
	}

	if = {
		limit = { has_country_flag = USA_ti_chips_milestones }
		clr_country_flag = USA_ti_chips_public_capacity
		clr_country_flag = USA_ti_chips_corporate_control
	}
	else_if = {
		limit = { has_country_flag = USA_ti_chips_public_capacity }
		clr_country_flag = USA_ti_chips_corporate_control
	}
	else_if = {
		limit = { has_country_flag = USA_ti_chips_corporate_control }
		clr_country_flag = USA_ti_chips_milestones
		clr_country_flag = USA_ti_chips_public_capacity
	}
	else_if = {
		limit = { has_country_flag = USA_ti_event_18_resolved }
		set_country_flag = USA_ti_chips_milestones
	}

	if = {
		limit = { has_country_flag = USA_ti_fabs_accelerated }
		clr_country_flag = USA_ti_fabs_staged
		clr_country_flag = USA_ti_fabs_strategic_offtake
	}
	else_if = {
		limit = { has_country_flag = USA_ti_fabs_staged }
		clr_country_flag = USA_ti_fabs_strategic_offtake
	}
	else_if = {
		limit = { has_country_flag = USA_ti_event_19_resolved }
		set_country_flag = USA_ti_fabs_staged
	}

	if = {

		clr_country_flag = USA_ti_silabs_strategic_review
	}
	else_if = {
		limit = { has_country_flag = USA_ti_event_20_resolved }
		set_country_flag = USA_ti_silabs_integration_plan
	}
}

USA_ti_reconstruct_history = {
	USA_ti_initialize_state = yes

	if = {
		limit = {
			date > 1997.7.2
			NOT = { has_country_flag = USA_ti_event_12_resolved }
		}
		set_country_flag = USA_ti_event_12_resolved
		set_country_flag = USA_ti_defense_sold
		add_to_variable = { USA_ti_embedded_reach = 1 }
		add_to_variable = { USA_ti_defense_integration = -3 }
	}
	if = {
		limit = {
			date > 1999.12.31
			NOT = { has_country_flag = USA_ti_event_13_resolved }
		}
		set_country_flag = USA_ti_event_13_resolved
		set_country_flag = USA_ti_omap_neutral
		add_to_variable = { USA_ti_embedded_reach = 1 }
		add_to_variable = { USA_ti_dsp_leadership = 1 }
	}
	if = {
		limit = {
			date > 2008.1.1
			NOT = { has_country_flag = USA_ti_event_14_resolved }
		}
		set_country_flag = USA_ti_event_14_resolved
		set_country_flag = USA_ti_kilby_central
		add_to_variable = { USA_ti_embedded_reach = 1 }
		add_to_variable = { USA_ti_dsp_leadership = 1 }
	}
	if = {
		limit = {
			date > 2011.9.23
			NOT = { has_country_flag = USA_ti_event_15_resolved }
		}
		set_country_flag = USA_ti_event_15_resolved
		set_country_flag = USA_ti_national_integrated
		add_to_variable = { USA_ti_embedded_reach = 1 }
		add_to_variable = { USA_ti_manufacturing_control = 2 }
	}
	if = {
		limit = {
			date > 2012.11.14
			NOT = { has_country_flag = USA_ti_event_16_resolved }
		}
		set_country_flag = USA_ti_event_16_resolved
		set_country_flag = USA_ti_omap_embedded_edge
		add_to_variable = { USA_ti_embedded_reach = 2 }
		add_to_variable = { USA_ti_dsp_leadership = -1 }
		add_to_variable = { USA_ti_manufacturing_control = 1 }
	}
	if = {
		limit = {
			date > 2021.6.30
			NOT = { has_country_flag = USA_ti_event_17_resolved }
			OR = {
				has_country_flag = USA_stack_txn_lehi_sold_to_ti
				has_country_flag = USA_stack_txn_lehi_retained_by_micron
				has_country_flag = USA_stack_txn_lehi_shared_reserve
			}
		}
		set_country_flag = USA_ti_event_17_resolved
		if = {
			limit = { has_country_flag = USA_stack_txn_lehi_sold_to_ti }
			add_to_variable = { USA_ti_embedded_reach = 1 }
			add_to_variable = { USA_ti_manufacturing_control = 2 }
		}
		else_if = {
			limit = { has_country_flag = USA_stack_txn_lehi_shared_reserve }
			add_to_variable = { USA_ti_dsp_leadership = 1 }
			add_to_variable = { USA_ti_manufacturing_control = 1 }
		}
		else = {
			add_to_variable = { USA_ti_manufacturing_control = -1 }
		}
	}
	if = {
		limit = {
			date > 2024.12.20
			NOT = { has_country_flag = USA_ti_event_18_resolved }
		}
		set_country_flag = USA_ti_event_18_resolved
		set_country_flag = USA_ti_chips_milestones
		set_country_flag = USA_stack_ti_chips_award
		add_to_variable = { USA_ti_manufacturing_control = 1 }
	}
	if = {
		limit = {
			date > 2025.6.18
			NOT = { has_country_flag = USA_ti_event_19_resolved }
		}
		set_country_flag = USA_ti_event_19_resolved
		set_country_flag = USA_ti_fabs_staged
		set_country_flag = USA_ti_seven_fab_plan_announced
		add_to_variable = { USA_ti_manufacturing_control = 1 }
	}
	if = {
		limit = {
			date > 2025.12.1
			has_country_flag = USA_ti_event_19_resolved
		}
		set_country_flag = USA_ti_sm1_initial_production
	}
	if = {
		limit = {
			date > 2026.2.4
			NOT = { has_country_flag = USA_ti_event_20_resolved }
		}
		set_country_flag = USA_ti_event_20_resolved
		set_country_flag = USA_ti_silabs_agreement_pending
		set_country_flag = USA_ti_silabs_integration_plan
		set_country_flag = USA_ti_sm1_initial_production
		add_to_variable = { USA_ti_embedded_reach = 1 }
		add_to_variable = { USA_ti_dsp_leadership = 1 }
	}

	USA_ti_normalize_state = yes
	USA_ti_clamp_state = yes
	USA_ti_resolve_capstone = yes
}

USA_ti_clear_capstone = {
	remove_ideas = {
		USA_ti_analog_hegemony
		USA_ti_silicon_arsenal
		USA_ti_world_as_signal
		USA_ti_foundational_foundry
		USA_ti_calculator_republic
		USA_ti_everything_connected
	}

}

USA_ti_resolve_capstone = {
	USA_ti_initialize_state = yes

	if = {
		limit = { has_country_flag = USA_ti_event_20_resolved }

		USA_ti_normalize_state = yes
		USA_ti_clamp_state = yes
		USA_ti_clear_capstone = yes
		clr_country_flag = USA_stack_ti_foundational_viable
		if = {
			limit = {
				check_variable = { var = USA_ti_embedded_reach value = 7 compare = greater_than_or_equals }
				check_variable = { var = USA_ti_manufacturing_control value = 7 compare = greater_than_or_equals }
				has_country_flag = USA_stack_ti_chips_award
				has_country_flag = USA_ti_sm1_initial_production
			}
			set_country_flag = USA_stack_ti_foundational_viable
		}

		if = {
			limit = {
				OR = {
					has_country_flag = USA_ti_defense_retained
					has_country_flag = USA_ti_defense_affiliate
				}
				check_variable = { var = USA_ti_defense_integration value = 7 compare = greater_than_or_equals }
				check_variable = { var = USA_ti_manufacturing_control value = 6 compare = greater_than_or_equals }
			}
			add_ideas = USA_ti_silicon_arsenal
		}
		else_if = {
			limit = {
				has_country_flag = USA_ti_omap_mobile_bet
				OR = {
					has_country_flag = USA_ti_omap_stack
					has_country_flag = USA_ti_omap_open
				}
				check_variable = { var = USA_ti_dsp_leadership value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_ti_embedded_reach value = 6 compare = greater_than_or_equals }
			}
			add_ideas = USA_ti_world_as_signal
		}
		else_if = {
			limit = {
				OR = {
					has_country_flag = USA_ti_fabs_accelerated
					has_country_flag = USA_ti_fabs_strategic_offtake
				}
				has_country_flag = USA_ti_seven_fab_plan_announced
				has_country_flag = USA_ti_sm1_initial_production
				check_variable = { var = USA_ti_manufacturing_control value = 8 compare = greater_than_or_equals }
			}
			add_ideas = USA_ti_foundational_foundry
		}
		else_if = {
			limit = {
				has_country_flag = USA_ti_kilby_public_consortium
				check_variable = { var = USA_ti_embedded_reach value = 7 compare = greater_than_or_equals }
			}
			add_ideas = USA_ti_calculator_republic
		}
		else_if = {
			limit = {
				has_country_flag = USA_ti_silabs_strategic_review
				has_country_flag = USA_ti_silabs_agreement_pending
				check_variable = { var = USA_ti_embedded_reach value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_ti_dsp_leadership value = 7 compare = greater_than_or_equals }
			}
			add_ideas = USA_ti_everything_connected
		}
		else = {
			add_ideas = USA_ti_analog_hegemony
		}

		set_country_flag = USA_ti_capstone_resolved
	}

	USA_physical_compute_stack_resolve = yes
}
