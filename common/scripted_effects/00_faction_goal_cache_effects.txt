	set_temp_variable = { faction_goal_marine_battalions_cache_temp = 0 }
	set_temp_variable = { faction_goal_airborne_battalions_cache_temp = 0 }
	set_temp_variable = { faction_goal_submarines_cache_temp = 0 }

							has_faction_goal = faction_goal_transform_nato_forces
							NOT = { has_completed_faction_goal = faction_goal_transform_nato_forces }
						}
						AND = {

							NOT = { has_completed_faction_goal = faction_goal_transform_eap_forces }
						}
					}
				}
			}
			add_to_temp_variable = {
				ROOT.faction_goal_mechanized_battalions_cache_temp = {
					value = 0
					add = num_battalions_with_type@Mech_Inf_Bat
					add = num_battalions_with_type@Arm_Inf_Bat
					add = num_battalions_with_type@Mech_Air_Inf_Bat
					add = num_battalions_with_type@Arm_Air_Inf_Bat
					add = num_battalions_with_type@Mech_Marine_Bat
					add = num_battalions_with_type@Arm_Marine_Bat
				}
			}
		}
		if = {
			limit = {
				ROOT = {
					OR = {
						check_variable = { faction_goal_force_tank_cache = 1 }
						AND = {
							has_faction_goal = faction_goal_transform_nato_forces
							NOT = { has_completed_faction_goal = faction_goal_transform_nato_forces }
						}
						AND = {
							has_faction_goal = faction_goal_transform_eap_forces
							NOT = { has_completed_faction_goal = faction_goal_transform_eap_forces }
						}
						AND = {
							has_faction_goal = faction_goal_an_armored_fist
							NOT = { has_completed_faction_goal = faction_goal_an_armored_fist }
						}
					}
				}
			}
			add_to_temp_variable = {

					value = 0
					add = num_battalions_with_type@armor_Bat

			}
		}
		if = {
			limit = {
				ROOT = {
					OR = {
						check_variable = { faction_goal_force_capital_ships_cache = 1 }
						AND = {
							has_faction_goal = faction_goal_high_seas_fleet
							NOT = { has_completed_faction_goal = faction_goal_high_seas_fleet }
						}
					}
				}
			}
			add_to_temp_variable = {
				ROOT.faction_goal_capital_ships_cache_temp = {
					value = 0
					add = num_ships_with_type@battleship
					add = num_ships_with_type@battle_cruiser

					add = num_ships_with_type@stealth_destroyer
					add = num_ships_with_type@destroyer
					add = num_ships_with_type@heavy_frigate
				}
			}
		}
		add_to_temp_variable = {
			ROOT.faction_goal_marine_battalions_cache_temp = {
				value = 0
				add = num_battalions_with_type@L_Marine_Bat
				add = num_battalions_with_type@L_Marine_Cdo_Bat

		}
		add_to_temp_variable = {
			ROOT.faction_goal_airborne_battalions_cache_temp = {
				value = 0
				add = num_battalions_with_type@Arm_Air_Inf_Bat
				add = num_battalions_with_type@Mech_Air_Inf_Bat
				add = num_battalions_with_type@Mot_Air_Inf_Bat
				add = num_battalions_with_type@L_Air_Inf_Bat
			}
		}
		add_to_temp_variable = {
			ROOT.faction_goal_submarines_cache_temp = {
				value = 0
				add = num_ships_with_type@missile_submarine
				add = num_ships_with_type@attack_submarine
			}
		}
		add_to_temp_variable = {
			ROOT.faction_goal_screen_ships_cache_temp = {

				add = num_ships_with_type@stealth_frigate

				add = num_ships_with_type@stealth_corvette
				add = num_ships_with_type@patrol_boat

		}

		}

			limit = {
				ROOT = {
					OR = {
						check_variable = { faction_goal_force_armies_cache = 1 }
						AND = {
							has_faction_goal = faction_goal_unlock_military
							NOT = { has_completed_faction_goal = faction_goal_unlock_military }
						}
					}
				}
			}

		}

			add_to_temp_variable = { ROOT.faction_goal_fascist_member_count_cache_temp = 1 }
		}

				}
			}
			add_to_temp_variable = { ROOT.faction_goal_anti_communist_member_count_cache_temp = 1 }
		}
		if = {
			limit = { democratic < 0.3 }
			add_to_temp_variable = { ROOT.faction_goal_below_30_democracy_member_count_cache_temp = 1 }
		}
		if = {
			limit = { communism > 0.75 }
			add_to_temp_variable = { ROOT.faction_goal_above_75_communism_member_count_cache_temp = 1 }
		}
		if = {
			limit = { check_variable = { gdp_per_capita > 24.999 } }
			add_to_temp_variable = { ROOT.faction_goal_nato_gdp_eligible_member_count_cache_temp = 1 }
		}
	}
	scope:faction = {
		set_variable = { faction_goal_infantry_battalions_cache = ROOT.faction_goal_infantry_battalions_cache_temp }
		set_variable = { faction_goal_mechanized_battalions_cache = ROOT.faction_goal_mechanized_battalions_cache_temp }
		set_variable = { faction_goal_tank_battalions_cache = ROOT.faction_goal_tank_battalions_cache_temp }
		set_variable = { faction_goal_capital_ships_cache = ROOT.faction_goal_capital_ships_cache_temp }
		set_variable = { faction_goal_marine_battalions_cache = ROOT.faction_goal_marine_battalions_cache_temp }
		set_variable = { faction_goal_airborne_battalions_cache = ROOT.faction_goal_airborne_battalions_cache_temp }
		set_variable = { faction_goal_submarines_cache = ROOT.faction_goal_submarines_cache_temp }
		set_variable = { faction_goal_screen_ships_cache = ROOT.faction_goal_screen_ships_cache_temp }
		set_variable = { faction_goal_fighters_cache = ROOT.faction_goal_fighters_cache_temp }
		set_variable = { faction_goal_strategic_bombers_cache = ROOT.faction_goal_strategic_bombers_cache_temp }
		set_variable = { faction_goal_researched_technologies_cache = ROOT.faction_goal_researched_technologies_cache_temp }
		set_variable = { faction_goal_armies_cache = ROOT.faction_goal_armies_cache_temp }
		set_variable = { faction_goal_military_factories_cache = ROOT.faction_goal_military_factories_cache_temp }
		set_variable = { faction_goal_member_count_cache = ROOT.faction_goal_member_count_cache_temp }
		set_variable = { faction_goal_communist_member_count_cache = ROOT.faction_goal_communist_member_count_cache_temp }
		set_variable = { faction_goal_fascist_member_count_cache = ROOT.faction_goal_fascist_member_count_cache_temp }

faction_goal_select_unlock_research = {
	set_temp_variable = { faction_goal_force_researched_technologies_cache = 1 }
	set_temp_variable = { faction_goal_unlock_research_target_temp = 0 }
	every_faction_member = {
		add_to_temp_variable = {
			ROOT.faction_goal_unlock_research_target_temp = {
				value = num_researched_technologies
				multiply = {
					value = 0
					if = { limit = { value = num_researched_technologies less_than = 50 } add = 1.25 }
					else_if = { limit = { value = num_researched_technologies less_than = 75 } add = 1.2 }
					else_if = { limit = { value = num_researched_technologies less_than = 100 } add = 1.15 }
					else_if = { limit = { value = num_researched_technologies less_than = 150 } add = 1.1 }
					else_if = { limit = { value = num_researched_technologies less_than = 200 } add = 1.05 }
				}
			}
		}
	}
	round_temp_variable = faction_goal_unlock_research_target_temp
	scope:faction = {
		set_variable = { unlock_military_techs_researched_target = ROOT.faction_goal_unlock_research_target_temp }
	}
	faction_goal_update_member_metrics_cache = yes
}

faction_goal_select_unlock_military = {
	set_temp_variable = { faction_goal_force_armies_cache = 1 }
	faction_goal_update_member_metrics_cache = yes
	scope:faction = {
		set_variable = {
			unlock_target = {
				value = faction_goal_armies_cache
				multiply = 1.2
			}
		}
	}
}

faction_goal_update_strategic_bombing_campaign_cache = {
	set_temp_variable = { faction_goal_recently_bombed_states_cache_temp = 0 }
	if = {
		limit = { has_war = yes }
		set_temp_variable = {
			faction_goal_recently_bombed_states_cache_temp = {
				value = 0
				every_collection = {
					named_collection = faction_enemy_owned_core_states_recently_bombed_collection
					add = 1
				}
			}
		}
	}
	scope:faction = {
		set_variable = { faction_goal_recently_bombed_states_cache = ROOT.faction_goal_recently_bombed_states_cache_temp }
	}
}

faction_goal_update_great_fortress_cache = {
	set_temp_variable = { faction_goal_great_fortress_cache_temp = 0 }
	if = {
		limit = {
			count_in_collection = {
				collection = collection:faction_members_collection
				buildings = { bunker coastal_bunker }
				size > 29
			}
		}
		set_temp_variable = { faction_goal_great_fortress_cache_temp = 1 }
	}
	scope:faction = {
		set_variable = { faction_goal_great_fortress_cache = ROOT.faction_goal_great_fortress_cache_temp }
	}
}

faction_goal_update_transform_nato_forces_cache = {
	set_temp_variable = { faction_goal_force_mechanized_cache = 1 }
	set_temp_variable = { faction_goal_force_tank_cache = 1 }
	faction_goal_update_member_metrics_cache = yes
}

faction_goal_update_semiconductor_supremacy_cache = {
	scope:faction = {
		set_variable = { faction_goal_semiconductor_supremacy_cache = 0 }
		if = {
			limit = { check_variable = { faction_goal_microchip_plant_cache > 49 } }
			set_variable = { faction_goal_semiconductor_supremacy_cache = 1 }
		}
	}
}

faction_goal_update_marine_corps_cache = {
	faction_goal_update_member_metrics_cache = yes
}

faction_goal_update_daily_caches = {
	if = {
		limit = { is_faction_leader = yes }
		faction_goal_update_member_metrics_cache = yes
		faction_goal_update_daily_condition_caches = yes
		if = {
			limit = {
				OR = {
					has_war = yes
					AND = {
						has_faction_goal = faction_goal_strategic_bombing_campaign
						NOT = { has_completed_faction_goal = faction_goal_strategic_bombing_campaign }
					}
				}
			}
			faction_goal_update_strategic_bombing_campaign_cache = yes
		}
		if = {
			limit = {
				scope:faction = {
					NOT = { has_variable = faction_goal_technology_sharing_cache }
				}
			}
			faction_goal_update_weekly_condition_caches = yes
		}
	}
}

faction_goal_update_weekly_caches = {

		if = {

				NOT = { has_completed_faction_goal = faction_goal_a_great_fortress }
			}

	}
}
