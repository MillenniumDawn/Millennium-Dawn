add_namespace = STALKER

country_event = {
	id = STALKER.41
	title = STALKER.41.t
	desc = STALKER.41.d
	picture = GFX_nuclear_power_plant
	is_triggered_only = yes

	trigger = {
		STALKER_scenario_enabled = yes
		has_event_target = STALKER_zone_event_target
		event_target:STALKER_zone_event_target = {
			STALKER_is_active_zone_anchor = yes
			is_controlled_by = ROOT
		}
	}

	option = {
		name = STALKER.41.a
		log = "[GetDateText]: [This.GetName]: STALKER.41.a executed"
		custom_effect_tooltip = STALKER.41.a_tt
		hidden_effect = {
			event_target:STALKER_zone_event_target = {
				if = {
					limit = {
						STALKER_is_active_zone_anchor = yes
						is_controlled_by = ROOT
					}
					add_to_variable = { STALKER_zone_exploitation = 5 }
					STALKER_clamp_zone_society = yes
					ROOT = {
						STALKER_refresh_zone_administration = yes
						add_political_power = -25
					}
				}
			}
		}
		ai_chance = {
			base = 60
			modifier = {
				factor = 0
				has_political_power < 25
			}
		}
	}

	option = {
		name = STALKER.41.b
		log = "[GetDateText]: [This.GetName]: STALKER.41.b executed"
		custom_effect_tooltip = STALKER.41.b_tt
		hidden_effect = {
			event_target:STALKER_zone_event_target = {
				if = {
					limit = {
						STALKER_is_active_zone_anchor = yes
						is_controlled_by = ROOT
					}
					add_to_variable = { STALKER_zone_stalkers = -5 }
					STALKER_clamp_zone_society = yes
				}
			}
		}
		ai_chance = {
			base = 40
		}
	}
}
