# Xbox corporate-history scripted effects
# Chain: USA_xbox_events (events/USA_xbox_events.txt)
# Variables: USA_xbox_platform_strength, USA_xbox_developer_ecosystem,
#            USA_xbox_digital_services, USA_xbox_hardware_resilience  [0..10]

USA_xbox_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = USA_xbox_state_initialized } }

		set_variable = { USA_xbox_platform_strength = 3 }
		set_variable = { USA_xbox_developer_ecosystem = 3 }
		set_variable = { USA_xbox_digital_services = 0 }
		set_variable = { USA_xbox_hardware_resilience = 3 }
		set_country_flag = USA_xbox_state_initialized
	}

	USA_xbox_clamp_state = yes
}

USA_xbox_clamp_state = {
	set_temp_variable = { corp_value = USA_xbox_platform_strength }
	corporate_history_clamp_value = yes
	set_variable = { USA_xbox_platform_strength = corp_value }

	set_temp_variable = { corp_value = USA_xbox_developer_ecosystem }
	corporate_history_clamp_value = yes
	set_variable = { USA_xbox_developer_ecosystem = corp_value }

	set_temp_variable = { corp_value = USA_xbox_digital_services }
	corporate_history_clamp_value = yes
	set_variable = { USA_xbox_digital_services = corp_value }

	set_temp_variable = { corp_value = USA_xbox_hardware_resilience }
	corporate_history_clamp_value = yes
	set_variable = { USA_xbox_hardware_resilience = corp_value }
}

USA_xbox_clear_capstone_outcome = {
	remove_ideas = {
		USA_xbox_platform_consolidator
		USA_xbox_hardware_innovator
		USA_xbox_developer_platform
		USA_xbox_cloud_native_platform
		USA_xbox_resilient_games_publisher
	}

	if = {
		limit = { has_country_flag = USA_xbox_outcome_platform_consolidator }
		clr_country_flag = USA_xbox_outcome_platform_consolidator
	}
	if = {
		limit = { has_country_flag = USA_xbox_outcome_hardware_innovator }
		clr_country_flag = USA_xbox_outcome_hardware_innovator
	}
	if = {
		limit = { has_country_flag = USA_xbox_outcome_developer_platform }
		clr_country_flag = USA_xbox_outcome_developer_platform
	}
	if = {
		limit = { has_country_flag = USA_xbox_outcome_cloud_native_platform }
		clr_country_flag = USA_xbox_outcome_cloud_native_platform
	}
	if = {
		limit = { has_country_flag = USA_xbox_outcome_resilient_games_publisher }
		clr_country_flag = USA_xbox_outcome_resilient_games_publisher
	}
	if = {
		limit = { has_country_flag = USA_xbox_capstone_resolved }
		clr_country_flag = USA_xbox_capstone_resolved
	}
}

USA_xbox_resolve_capstone = {
	USA_xbox_initialize_state = yes

	if = {
		limit = { NOT = { has_country_flag = USA_xbox_capstone_resolved } }
		if = {
			limit = {
				check_variable = { var = USA_xbox_platform_strength value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_xbox_digital_services value = 7 compare = greater_than_or_equals }
			}
			USA_xbox_apply_platform_consolidator = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_xbox_hardware_resilience value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_xbox_platform_strength value = 7 compare = greater_than_or_equals }
			}
			USA_xbox_apply_hardware_innovator = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_xbox_developer_ecosystem value = 8 compare = greater_than_or_equals }
				check_variable = { var = USA_xbox_digital_services value = 6 compare = greater_than_or_equals }
			}
			USA_xbox_apply_developer_platform = yes
		}
		else_if = {
			limit = {
				check_variable = { var = USA_xbox_digital_services value = 9 compare = greater_than_or_equals }
			}
			USA_xbox_apply_cloud_native_platform = yes
		}
		else = {
			USA_xbox_apply_resilient_games_publisher = yes
		}
	}
}
