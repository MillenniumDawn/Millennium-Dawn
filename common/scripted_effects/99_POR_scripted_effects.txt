# Target scope, ROOT = POR: writes POR_reward_mult and POR_heat_mult as temps on ROOT.
POR_score_client = {
	set_temp_variable = { ROOT.POR_udg = 0 }
	if = { limit = { has_idea = non_power } add_to_temp_variable = { ROOT.POR_udg = 3 } }
	else_if = { limit = { has_idea = minor_power } add_to_temp_variable = { ROOT.POR_udg = 2 } }
	else_if = { limit = { has_idea = regional_power } add_to_temp_variable = { ROOT.POR_udg = 1 } }

	if = {
		limit = { check_variable = { deployed_army_manpower_k < ROOT.deployed_army_manpower_k } }
		add_to_temp_variable = { ROOT.POR_udg = 1 }
	}
	if = {
		limit = { check_variable = { ROOT.gdp_total > gdp_total } }
		add_to_temp_variable = { ROOT.POR_udg = 1 }
	}
	if = {
		limit = { has_war = yes surrender_progress > 0.20 }
		add_to_temp_variable = { ROOT.POR_udg = 1 }
	}

	set_temp_variable = { ROOT.POR_prh = 0 }
	if = { limit = { has_idea = non_state_actor } add_to_temp_variable = { ROOT.POR_prh = 3 } }
	else_if = { limit = { has_idea = lacks_international_recognition } add_to_temp_variable = { ROOT.POR_prh = 2 } }

	if = { limit = { has_idea = Massive_International_Sanctions } add_to_temp_variable = { ROOT.POR_prh = 3 } }
	else_if = { limit = { has_idea = international_sanctions } add_to_temp_variable = { ROOT.POR_prh = 2 } }
	else_if = {
		limit = { OR = { has_idea = Western_Sanctions has_idea = Reduced_Western_Sanctions } }
		add_to_temp_variable = { ROOT.POR_prh = 1 }
	}

	if = {
		limit = {
			OR = {
				has_idea = unsc_restricted_volunteers_too
				has_dynamic_modifier = { modifier = united_nations_security_council_sanctions }
			}
		}
		add_to_temp_variable = { ROOT.POR_prh = 2 }
	}

	ROOT = {
		clamp_temp_variable = { var = POR_udg min = 0 max = 6 }
		clamp_temp_variable = { var = POR_prh min = 0 max = 6 }

		set_temp_variable = { POR_reward_mult = POR_udg }
		multiply_temp_variable = { POR_reward_mult = 0.10 }
		set_temp_variable = { POR_prh_term = POR_prh }
		multiply_temp_variable = { POR_prh_term = 0.10 }
		add_to_temp_variable = { POR_reward_mult = POR_prh_term }
		add_to_temp_variable = { POR_reward_mult = 1 }
		if = {
			limit = { has_completed_focus = POR_cause_of_the_oppressed }
			add_to_temp_variable = { POR_reward_mult = 0.25 }
		}
		clamp_temp_variable = { var = POR_reward_mult min = 1 max = 2.45 }

		# Only the pariah term raises exposure: backing an underdog is no scandal in Brussels.
		set_temp_variable = { POR_heat_mult = POR_prh }
		multiply_temp_variable = { POR_heat_mult = 0.25 }
		add_to_temp_variable = { POR_heat_mult = 1 }
	}
}

# POR scope. Takes POR_heat_add, scaled by POR_heat_mult when a target was scored.
POR_add_exposure = {
	# Unset when the caller scored no client.
	if = {
		limit = { check_variable = { POR_heat_mult = 0 } }
		set_temp_variable = { POR_heat_mult = 1 }
	}
	multiply_temp_variable = { POR_heat_add = POR_heat_mult }
	add_to_variable = { POR_shadow_exposure = POR_heat_add }
	clamp_variable = { var = POR_shadow_exposure min = 0 max = 100 }
}

POR_update_exposure_tier = {
	add_to_variable = { POR_ops_filter_dirty = 1 }
	clamp_variable = { var = POR_shadow_exposure min = 0 max = 100 }

	if = { limit = { check_variable = { POR_shadow_exposure > 74 } } set_variable = { POR_exposure_tier = 4 } }
	else_if = { limit = { check_variable = { POR_shadow_exposure > 54 } } set_variable = { POR_exposure_tier = 3 } }
	else_if = { limit = { check_variable = { POR_shadow_exposure > 34 } } set_variable = { POR_exposure_tier = 2 } }
	else_if = { limit = { check_variable = { POR_shadow_exposure > 14 } } set_variable = { POR_exposure_tier = 1 } }
	else = { set_variable = { POR_exposure_tier = 0 } }

	if = {
		limit = { check_variable = { POR_exposure_tier = 0 } }
		set_variable = { POR_exposure_breach_bonus = 0 }
		if = {
			limit = { has_dynamic_modifier = { modifier = POR_eu_exposure_modifier } }
			remove_dynamic_modifier = { modifier = POR_eu_exposure_modifier }
		}
	}
	else = {
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = POR_eu_exposure_modifier } } }
			add_dynamic_modifier = { modifier = POR_eu_exposure_modifier }
		}

		set_variable = { POR_exposure_trade_opinion_factor = POR_exposure_tier }
		multiply_variable = { POR_exposure_trade_opinion_factor = -0.05 }
		set_variable = { POR_exposure_political_power_factor = POR_exposure_trade_opinion_factor }

		set_variable = { POR_exposure_foreign_influence_defense_modifier = 0 }
		if = {
			limit = { check_variable = { POR_exposure_tier > 1 } }
			set_variable = { POR_exposure_foreign_influence_defense_modifier = -0.05 }
		}

		set_variable = { POR_exposure_interest_rate_multiplier_modifier = 0 }
		set_variable = { POR_exposure_stability_factor = 0 }
		set_variable = { POR_exposure_breach_bonus = 0 }

		if = {
			limit = { check_variable = { POR_exposure_tier = 3 } }
			set_variable = { POR_exposure_interest_rate_multiplier_modifier = 1.0 }
			set_variable = { POR_exposure_breach_bonus = 4 }
		}
		else_if = {
			limit = { check_variable = { POR_exposure_tier = 4 } }
			set_variable = { POR_exposure_interest_rate_multiplier_modifier = 2.0 }
			set_variable = { POR_exposure_stability_factor = -0.05 }
			set_variable = { POR_exposure_breach_bonus = 6 }
		}
	}

	# Tier 3 is where the breach bonus first crosses the EU serious-breach threshold.
	if = {
		limit = {
			check_variable = { POR_exposure_tier > 2 }
			has_idea = EU_member
			NOT = { has_completed_focus = POR_remain_in_europe }
			NOT = { has_country_flag = POR_ultimatum_issued }
			NOT = { has_country_flag = Article_50 }
		}
		set_country_flag = POR_ultimatum_issued
		activate_mission = POR_brussels_ultimatum
		country_event = { id = portugal.75 hours = 6 }
	}
}

POR_prune_shadow_arrays = {
	clear_temp_array = POR_dropped_ids

	if = {
		limit = { check_variable = { POR_shadow_clients^num > 0 } }
		for_each_scope_loop = {
			array = POR_shadow_clients
			if = {
				limit = {
					OR = {
						NOT = { exists = yes }
						NOT = { has_country_flag = POR_shadow_client }
					}
				}
				add_to_temp_array = { PREV.POR_dropped_ids = THIS.id }
			}
		}
		for_each_loop = {
			array = POR_dropped_ids
			value = v
			remove_from_array = { POR_shadow_clients = v }
		}
	}

	clear_temp_array = POR_dropped_ids

	if = {
		limit = { check_variable = { POR_lusophone_subjects^num > 0 } }
		for_each_scope_loop = {
			array = POR_lusophone_subjects
			if = {
				limit = {
					OR = {
						NOT = { exists = yes }
						NOT = { is_subject_of = PREV }
					}
				}
				add_to_temp_array = { PREV.POR_dropped_ids = THIS.id }
			}
		}
		for_each_loop = {
			array = POR_dropped_ids
			value = v
			remove_from_array = { POR_lusophone_subjects = v }
		}
	}

}

			limit = { ROOT = { network_strength = { target = PREV value > 89 } } }

			limit = { ROOT = { network_strength = { target = PREV value > 79 } } }
			set_variable = { POR_cw_size = 0.8 }
		}
		multiply_variable = { POR_cw_size = 0.55 }
		if = {
			limit = { ROOT = { has_completed_focus = POR_open_road } }
			add_to_variable = { POR_cw_size = 0.08 }
		}
		if = {
			limit = { ROOT = { has_completed_focus = POR_discreet_road } }
			subtract_from_variable = { POR_cw_size = 0.10 }
		}

		set_variable = { POR_cw_underdog = ROOT.POR_reward_mult }
		subtract_from_variable = { POR_cw_underdog = 1 }
		multiply_variable = { POR_cw_underdog = 0.25 }

		set_temp_variable = { POR_client_slot = 1 }

	else_if = {

				size = POR_cw_size
				army_ratio = POR_cw_army
				navy_ratio = POR_cw_size
				air_ratio = POR_cw_size
				set_country_flag = POR_client_fresh
				set_country_flag = POR_shadow_client
			}

			recalculate_party = yes

		}

		}
	}

			limit = { NOT = { has_dynamic_modifier = { modifier = POR_weight_of_distance } } }
			add_dynamic_modifier = { modifier = POR_weight_of_distance }
		}

# Takes POR_ideo_add.
POR_raise_ideological_development = {
	if = {
		limit = { check_variable = { POR_ideo_add = 0 } }
		set_temp_variable = { POR_ideo_add = 2 }
	}
	# Tooltipped by hand: the raw change prints an empty "which grants:" block at tier 0.
	custom_effect_tooltip = POR_ideological_development_up_tt
	hidden_effect = {
		add_to_variable = { POR_ideological_development = POR_ideo_add }
		clamp_variable = { var = POR_ideological_development min = 0 max = 100 }
		clear_variable = POR_ideo_add
		POR_update_ideology_tier = yes
	}
}

POR_update_ideology_tier = {
	if = {
		limit = {
			check_variable = { POR_ideological_development > 0 }
			NOT = { has_dynamic_modifier = { modifier = POR_ideology_modifier } }
		}
		add_dynamic_modifier = { modifier = POR_ideology_modifier }
	}

	if = { limit = { check_variable = { POR_ideological_development > 79 } } set_variable = { POR_ideology_tier = 4 } }
	else_if = { limit = { check_variable = { POR_ideological_development > 59 } } set_variable = { POR_ideology_tier = 3 } }
	else_if = { limit = { check_variable = { POR_ideological_development > 39 } } set_variable = { POR_ideology_tier = 2 } }
	else_if = { limit = { check_variable = { POR_ideological_development > 19 } } set_variable = { POR_ideology_tier = 1 } }
	else = { set_variable = { POR_ideology_tier = 0 } }

	# Shares its field with the ideology focuses, so last week's share comes out first.
	if = {
		limit = { NOT = { has_variable = POR_ideology_tier_drift } }
		set_variable = { POR_ideology_tier_drift = 0 }
	}
	subtract_from_variable = { POR_ideology_drift_defence_factor = POR_ideology_tier_drift }
	set_variable = { POR_ideology_tier_drift = POR_ideology_tier }
	multiply_variable = { POR_ideology_tier_drift = -0.03 }
	add_to_variable = { POR_ideology_drift_defence_factor = POR_ideology_tier_drift }
}

# Not on_government_change: PS, PSD and the liberals share the democratic group, so it never fires.
POR_watch_party_flip = {
	set_party_index_to_ruling_party = yes

	if = {
		limit = { NOT = { has_variable = POR_last_ruling_party } }
		set_variable = { POR_last_ruling_party = party_index }
	}
	if = {
		limit = { NOT = { check_variable = { POR_last_ruling_party = party_index } } }
		set_temp_variable = { POR_ideo_add = 5 }
		# Crossing an outlook group is a bigger shock than an alternation inside one.
		if = {
			limit = {
				OR = {
					AND = {
						check_variable = { POR_last_ruling_party < 4 }
						check_variable = { party_index > 3 }
					}
					AND = {
						check_variable = { POR_last_ruling_party > 3 }
						check_variable = { party_index < 4 }
					}
				}
			}
			set_temp_variable = { POR_ideo_add = 8 }
		}
		POR_raise_ideological_development = yes
		news_event = { id = portugal_news.7 hours = 6 }
	}
	set_variable = { POR_last_ruling_party = party_index }
	clear_variable = party_index
}

# Weekly cache, because focus triggers cannot compute any of this.
POR_update_ideology_support = {
	set_party_index_to_ruling_party = yes
	set_temp_variable = { outlook_party_index = party_index }
	set_outlook_from_party_index = yes
	set_variable = { POR_ruling_outlook = outlook_id }
	set_outlook_popularity_from_id = yes
	set_variable = { POR_ruling_outlook_support = outlook_popularity }

	set_outlook_popularity_array = yes
	set_temp_variable = { POR_rev_best = 0 }
	set_temp_variable = { POR_rev_best_id = -1 }
	for_loop_effect = {
		start = 0
		end = 5
		value = POR_o
		if = {
			limit = {
				NOT = { check_variable = { POR_o = POR_ruling_outlook } }
				check_variable = { outlook_popularity_by_id^POR_o > POR_rev_best }
			}
			set_temp_variable = { POR_rev_best = outlook_popularity_by_id^POR_o }
			set_temp_variable = { POR_rev_best_id = POR_o }
		}
	}
	set_variable = { POR_revolt_support = POR_rev_best }
	set_variable = { POR_revolt_outlook = POR_rev_best_id }

	set_temp_variable = { POR_slot_start = 0 }
	set_temp_variable = { POR_slot_end = 0 }
	if = {
		limit = { check_variable = { POR_revolt_outlook = 0 } }
		set_temp_variable = { POR_slot_start = 0 }
		set_temp_variable = { POR_slot_end = 4 }
	}
	else_if = {
		limit = { check_variable = { POR_revolt_outlook = 1 } }
		set_temp_variable = { POR_slot_start = 4 }
		set_temp_variable = { POR_slot_end = 10 }
	}
	else_if = {
		limit = { check_variable = { POR_revolt_outlook = 2 } }
		set_temp_variable = { POR_slot_start = 10 }
		set_temp_variable = { POR_slot_end = 12 }
	}
	else_if = {
		limit = { check_variable = { POR_revolt_outlook = 3 } }
		set_temp_variable = { POR_slot_start = 12 }
		set_temp_variable = { POR_slot_end = 20 }
	}
	else_if = {
		limit = { check_variable = { POR_revolt_outlook = 4 } }
		set_temp_variable = { POR_slot_start = 20 }
		set_temp_variable = { POR_slot_end = 24 }
	}

	set_temp_variable = { POR_rev_party_pop = 0 }
	set_temp_variable = { POR_rev_party_id = -1 }
	for_loop_effect = {
		start = POR_slot_start
		end = POR_slot_end
		value = POR_p
		if = {
			limit = { check_variable = { party_pop_array^POR_p > POR_rev_party_pop } }
			set_temp_variable = { POR_rev_party_pop = party_pop_array^POR_p }
			set_temp_variable = { POR_rev_party_id = POR_p }
		}
	}
	set_variable = { POR_revolt_party = POR_rev_party_id }

	set_temp_variable = { POR_opp_pop = 0 }
	set_temp_variable = { POR_opp_id = -1 }
	for_loop_effect = {
		start = 0
		end = 24
		value = POR_q
		if = {
			limit = {
				NOT = { check_variable = { POR_q = party_index } }
				check_variable = { party_pop_array^POR_q > POR_opp_pop }
			}
			set_temp_variable = { POR_opp_pop = party_pop_array^POR_q }
			set_temp_variable = { POR_opp_id = POR_q }
		}
	}
	set_variable = { POR_opposition_party = POR_opp_id }
	clear_variable = party_index
}

# Reads the displaced side first, because change_ruling_party_effect rewrites it.
POR_run_revolution = {
	POR_update_ideology_support = yes
	set_temp_variable = { POR_rev_outlook_old = POR_ruling_outlook }
	set_temp_variable = { POR_rev_size = POR_ruling_outlook_support }
	multiply_temp_variable = { POR_rev_size = 0.8 }
	clamp_temp_variable = { var = POR_rev_size min = 0.15 max = 0.55 }

	set_temp_variable = { rul_party_temp = POR_revolt_party }
	set_temp_variable = { change_leader_temp = 0 }
	set_temp_variable = { disable_elections = 0 }
	change_ruling_party_effect = yes

	# A fascist spawn goes to the jihadist chain in on_declare_war, so the far right revolts as nationalist.
	set_country_flag = no_leader_change
	if = {
		limit = { check_variable = { POR_rev_outlook_old = 0 } }
		start_civil_war = { ideology = democratic size = POR_rev_size }
	}
	else_if = {
		limit = { check_variable = { POR_rev_outlook_old = 1 } }
		start_civil_war = { ideology = communism size = POR_rev_size }
	}
	else_if = {
		limit = { check_variable = { POR_rev_outlook_old = 3 } }
		start_civil_war = { ideology = neutrality size = POR_rev_size }
	}
	else = {
		start_civil_war = { ideology = nationalist size = POR_rev_size }
	}
	clr_country_flag = no_leader_change

	add_stability = -0.20
	add_war_support = 0.10
	set_temp_variable = { int_investment_change = -20 }
	modify_international_investment_effect = yes
	set_temp_variable = { debt_change = 20 }
	modify_debt_effect = yes
	set_variable = { protest_strength = 0 }
	set_temp_variable = { POR_ideo_add = 15 }
	POR_raise_ideological_development = yes
	news_event = { id = portugal_news.8 hours = 6 }
}

# One per week, with the claimant passed as FROM so a second landing cannot overwrite it.
POR_check_held_states = {
	if = {
		limit = { check_variable = { POR_held_state_owners^num > 0 } }
		random_scope_in_array = {
			array = POR_held_state_owners
			limit = {
				OR = {
					NOT = { exists = yes }
					AND = { has_war = no has_civil_war = no }
				}
			}
			ROOT = { remove_from_array = { POR_held_state_owners = PREV.id } }
			if = {

# Winner's scope, from on_civil_war_end and civil_war_cleanup_angola; ROOT is not POR here.
POR_resolve_backing = {
	if = {
		limit = {
			has_country_flag = POR_backed_by_portugal
			country_exists = POR
		}
		clr_country_flag = POR_backed_by_portugal
		if = {
			limit = { NOT = { has_variable = POR_backing_mult@POR } }
			set_variable = { POR_backing_mult@POR = 1 }
		}
		set_temp_variable = { POR_pay = 25 }
		multiply_temp_variable = { POR_pay = POR_backing_mult@POR }
		round_temp_variable = POR_pay

		add_opinion_modifier = { target = POR modifier = ideological_alliance }
		reverse_add_opinion_modifier = { target = POR modifier = ideological_alliance }

		POR = {
			set_variable = { POR_backing_pay = PREV.POR_pay }
			set_variable = { POR_backing_winner = PREV.id }
			if = {
				limit = { NOT = { is_in_array = { POR_network_target_array = PREV.id } } }
				add_to_array = { POR_network_target_array = PREV.id }
			}
			country_event = { id = portugal.70 hours = 6 }
		}
		clear_variable = POR_backing_mult@POR
	}
	else_if = {
		limit = {
			has_country_flag = POR_opposed_by_portugal
			country_exists = POR
		}
		clr_country_flag = POR_opposed_by_portugal
		add_opinion_modifier = { target = POR modifier = supported_opposition }
		POR = {
			remove_operation_token = { tag = PREV token = token_civilian }
			remove_operation_token = { tag = PREV token = token_army }
			set_temp_variable = { percent_change = -10 }
			set_temp_variable = { tag_index = THIS.id }
			set_temp_variable = { influence_target = PREV.id }
			change_influence_percentage = yes
			set_temp_variable = { POR_heat_add = 12 }
			POR_add_exposure = yes
			country_event = { id = portugal.71 hours = 6 }
		}
		clear_variable = POR_backing_mult@POR
	}
}

# Weekly: portugal.108 follows seven days after we fully hold state 374 at war with Morocco.
POR_tangier_watch = {
	if = {
		limit = {
			has_country_flag = POR_tangier_goal
			owns_state = 374
		}
		clr_country_flag = POR_tangier_goal
		clr_country_flag = POR_tangier_offered
	}
	else_if = {
		limit = {
			has_country_flag = POR_tangier_goal
			NOT = { has_country_flag = POR_tangier_offered }
			NOT = { has_country_flag = POR_tangier_pending }
			has_war_with = MOR
			374 = { is_fully_controlled_by = ROOT }
		}
		# Outlasts the event delay, so an offer that lapses on recapture can be scheduled again.
		set_country_flag = { flag = POR_tangier_pending value = 1 days = 8 }
		country_event = { id = portugal.108 days = 7 }
	}
	else_if = {
		limit = {
			has_country_flag = POR_tangier_goal
			has_country_flag = POR_tangier_offered
			NOT = { has_war_with = MOR }
		}
		clr_country_flag = POR_tangier_offered
	}
}

# Granter's scope: gives POR the rights to one resource state it does not hold yet.
POR_grant_resource_rights = {
	set_temp_variable = { POR_rights_state = 0 }
	random_owned_controlled_state = {
		limit = {
			NOT = { has_resources_rights = { receiver = POR } }
			OR = {
				has_resources_amount = { resource = oil amount > 0 }
				has_resources_amount = { resource = aluminium amount > 0 }
				has_resources_amount = { resource = rubber amount > 0 }
				has_resources_amount = { resource = tungsten amount > 0 }
				has_resources_amount = { resource = steel amount > 0 }
				has_resources_amount = { resource = chromium amount > 0 }
			}
		}
		set_temp_variable = { POR_rights_state = THIS.id }
	}
	if = {
		limit = { check_variable = { POR_rights_state > 0 } }
		give_resource_rights = { receiver = POR state = POR_rights_state }
	}
}
