NIG_has_been_converted = {
	OR = {
		has_state_flag = NIG_christian_state
		has_state_flag = NIG_islamic_state
	}
}

NIG_ai_alt_path = {
	OR = {
		has_global_flag = NIG_SOCIAL_DEMOCRACY_FOCUS_PATH
		has_global_flag = NIG_CHRISTIAN_NATION_FOCUS_PATH
		has_global_flag = NIG_CALIPHATE_FOCUS_PATH
	}
}

NIG_ai_historical_path = {
	OR = {
		has_global_flag = NIG_HISTORICAL_FOCUS_PATH
		AND = {
			is_historical_focus_on = yes
			NOT = { NIG_ai_alt_path = yes }
		}
	}
}

NIG_ai_not_historical_path = {
	NIG_ai_alt_path = yes
}

NIG_ai_pdp_path = {
	OR = {
		NIG_ai_historical_path = yes
		has_global_flag = NIG_CHRISTIAN_NATION_FOCUS_PATH
	}
}
