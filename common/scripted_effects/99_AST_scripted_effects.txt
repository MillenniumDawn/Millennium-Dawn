bushfires_tick_up = {
	if = {
		limit = { check_variable = { bushfire_ticker < 6 } }
		add_to_variable = { bushfire_ticker = 1 }
	}
}

bushfires_tick_down = {
	if = {
		limit = { check_variable = { bushfire_ticker > 1 } }
		subtract_from_variable = { bushfire_ticker = 1 }
	}
}

AST_fire_bushfire_event = {
	if = {
		limit = { date < 2004.1.1 }
		AST = { country_event = { id = generic_bushfires_ast.2 days = 1 random_days = 360 } }
	}
	else_if = {
		limit = {
			date > 2008.1.1
			date < 2011.1.1
		}
		AST = { country_event = { id = black_saturday.1 days = 1 random_days = 360 } }
	}
	else = {
		AST = { country_event = { id = generic_bushfires_ast.1 days = 1 random_days = 360 } }
	}
}
