add_namespace = STALKER

country_event = {
	id = STALKER.70
	title = STALKER.70.t
	desc = STALKER.70.d
	picture = GFX_libya_militia_training
	is_triggered_only = yes

	trigger = {
		STALKER_scenario_enabled = yes
		has_event_target = STALKER_zone_event_target
		event_target:STALKER_zone_event_target = {
			STALKER_is_active_zone_anchor = yes
			is_controlled_by = ROOT
		}
	}

	immediate = {
		log = "[GetDateText]: [This.GetName]: STALKER.70 immediate executed"
		event_target:STALKER_zone_event_target = {
			add_to_variable = { STALKER_zone_containment = -10 }
			add_to_variable = { STALKER_zone_stalkers = -5 }
			STALKER_clamp_zone_society = yes
		}
	}

	option = {
		name = STALKER.70.a
		log = "[GetDateText]: [This.GetName]: STALKER.70.a executed"
		custom_effect_tooltip = STALKER.70.a_tt
		hidden_effect = {
			if = {
				limit = {
					event_target:STALKER_zone_event_target = {
						STALKER_is_active_zone_anchor = yes
						is_controlled_by = ROOT
					}
				}
				ROOT = { add_political_power = -50 }
				event_target:STALKER_zone_event_target = {
					add_to_variable = { STALKER_zone_monolith = -20 }
					STALKER_monolith_defections = yes
					add_to_variable = { STALKER_zone_duty = 5 }
					STALKER_clamp_zone_society = yes
				}
			}
		}
		ai_chance = {
			base = 60
			modifier = {
				factor = 0
				has_political_power < 50
			}
		}
	}

	option = {
		name = STALKER.70.b
		log = "[GetDateText]: [This.GetName]: STALKER.70.b executed"
		custom_effect_tooltip = STALKER.70.b_tt
		hidden_effect = {
			if = {
				limit = {
					event_target:STALKER_zone_event_target = {
						STALKER_is_active_zone_anchor = yes
						is_controlled_by = ROOT
					}
				}
				event_target:STALKER_zone_event_target = {
					add_to_variable = { STALKER_zone_containment = 5 }
					STALKER_clamp_zone_society = yes
				}
			}
		}
		ai_chance = { base = 40 }
	}
}
