on_actions = {
	on_war_relation_added = {
		effect = {
			if = {
				limit = { original_tag = PMR }
				PMR_partner_withdrawal_check = yes
			}
		}
	}

	# on_annex fires before the annexed country is gone, so the country_exists
	# clauses only resolve once the conference has settled every annexation.
	on_peaceconference_ended = {
		effect = {
			if = {
				limit = { country_exists = PMR }
				PMR = { PMR_partner_withdrawal_check = yes }
			}
		}
	}

	on_subject_annexed = {
		effect = {
			if = {
				limit = { country_exists = PMR }
				PMR = { PMR_partner_withdrawal_check = yes }
			}
		}
	}
}
