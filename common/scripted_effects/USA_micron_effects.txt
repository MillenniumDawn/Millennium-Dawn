USA_micron_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = USA_micron_state_initialized } }

		set_variable = { USA_micron_process_lead = 5 }
		set_variable = { USA_micron_cycle_discipline = 5 }
		set_variable = { USA_micron_us_fab_commitment = 5 }
		set_variable = { USA_micron_hbm_position = 5 }
		set_country_flag = USA_micron_state_initialized
	}

	if = {
		limit = {
			NOT = {
				OR = {
					has_country_flag = USA_stack_txn_lehi_sold_to_ti
					has_country_flag = USA_stack_txn_lehi_retained_by_micron
					has_country_flag = USA_stack_txn_lehi_shared_reserve
				}
			}
		}
		set_country_flag = USA_stack_txn_lehi_retained_by_micron
	}

	USA_micron_clamp_state = yes
	USA_micron_normalize_state = yes
}
