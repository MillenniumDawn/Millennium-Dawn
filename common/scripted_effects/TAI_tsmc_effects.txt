TAI_tsmc_initialize_state = {
	if = {
		limit = { NOT = { has_country_flag = TAI_tsmc_state_initialized } }

		set_variable = { TAI_tsmc_process_leadership = 6 }
		set_variable = { TAI_tsmc_capacity_depth = 5 }
		set_variable = { TAI_tsmc_customer_neutrality = 8 }
		set_variable = { TAI_tsmc_geographic_resilience = 1 }
		set_country_flag = TAI_tsmc_state_initialized
	}

	TAI_tsmc_clamp_state = yes
}

TAI_tsmc_clamp_state = {
	set_temp_variable = { corp_value = TAI_tsmc_process_leadership }
	corporate_history_clamp_value = yes
	set_variable = { TAI_tsmc_process_leadership = corp_value }
	set_temp_variable = { corp_value = TAI_tsmc_capacity_depth }
	corporate_history_clamp_value = yes
	set_variable = { TAI_tsmc_capacity_depth = corp_value }
	set_temp_variable = { corp_value = TAI_tsmc_customer_neutrality }
	corporate_history_clamp_value = yes
	set_variable = { TAI_tsmc_customer_neutrality = corp_value }
	set_temp_variable = { corp_value = TAI_tsmc_geographic_resilience }
	corporate_history_clamp_value = yes
	set_variable = { TAI_tsmc_geographic_resilience = corp_value }
}

TAI_tsmc_resolve_capstone = {
	TAI_tsmc_initialize_state = yes

	if = {
		limit = { NOT = { has_country_flag = TAI_tsmc_capstone_resolved } }
		if = {
			limit = { TAI_tsmc_can_resolve_trusted_allied_network = yes }
			TAI_tsmc_apply_trusted_allied_capstone = yes
		}
		else_if = {
			limit = { TAI_tsmc_can_resolve_systems_foundry_platform = yes }
			TAI_tsmc_apply_systems_foundry_capstone = yes
		}
		else_if = {
			limit = { TAI_tsmc_can_resolve_frontier_island_fortress = yes }
			TAI_tsmc_apply_frontier_island_capstone = yes
		}
		else_if = {
			limit = { TAI_tsmc_can_resolve_pure_play_common_foundry = yes }
			TAI_tsmc_apply_pure_play_capstone = yes
		}
		else = { TAI_tsmc_apply_disciplined_capstone = yes }
	}
}
