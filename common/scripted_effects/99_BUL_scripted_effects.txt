###########BULGARIA - MACEDONIA REVOULT###########
BUL_macedonia_revoult = {
	hidden_effect = {
		create_dynamic_country = {
			original_tag = FYR
			copy_tag = FYR
			transfer_state = 136
			set_popularities = {
				democratic = 5
				neutrality = 0
				fascism = 0
				communism = 5
				nationalist = 90
			}
			inherit_technology = FYR
			set_temp_variable = { rul_party_temp = 23 }
			set_temp_variable = { disable_elections = 1 }
			change_ruling_party_effect = yes
			set_temp_variable = {
				current_cons_popularity = party_pop_array^23
			}
			subtract_from_variable = {
				party_pop_array^23 = current_cons_popularity
			}
			recalculate_party = yes
			retire_country_leader = yes
			FYR = {
				transfer_units_fraction = {
					target = PREV
					size = 0.1
					stockpile_ratio = 0.1
					army_ratio = 0.1
					navy_ratio = 0.1
					air_ratio = 0.1
				}
			}
			declare_war_on = {
				target = FYR
				type = civil_war
			}
			add_civil_war_target = FYR
			country_event = bulg.19
		}
		set_temp_variable = { percent_change = -30 }
		set_temp_variable = { influence_target = FYR }
		change_influence_percentage = yes
	}
}
