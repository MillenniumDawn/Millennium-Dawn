# THIS = the borrower, ROOT = Brazil. True when the borrower can take the BNDES project built for it

	scripted_investment_recipient_available = yes
	set_temp_variable = { investment_slots_needed = 1 }
	set_temp_variable = { investment_slot_building = 11 }
	OR = {
		AND = {
			original_tag = CUB
			capital_scope = {
				is_controlled_by = PREV
				NOT = { is_in_array = { projects_in_state = ROOT } }
				investment_civilian_slot_available = yes
			}
		}
		AND = {
			original_tag = VEN
			capital_scope = {
				is_controlled_by = PREV
				NOT = { is_in_array = { projects_in_state = ROOT } }
				investment_infrastructure_slots_available = yes
			}
		}
		AND = {
			original_tag = ECU
			ROOT = { has_tech = fuel_silos }
			any_controlled_state = {
				is_owned_by = PREV
				NOT = { is_in_array = { projects_in_state = ROOT } }
				investment_single_building_slot_available = yes
			}
		}
	}
}
