KOS_ai_historical_path = {
	OR = {
		is_historical_focus_on = yes
		has_global_flag = KOS_HISTORICAL_FOCUS_PATH
	}
}

KOS_ai_not_historical_path = {
	OR = {
		has_global_flag = KOS_ALBANIAN_UNITY_FOCUS_PATH
		has_global_flag = KOS_YUGOSLAV_REVIVAL_FOCUS_PATH
	}
}

KOS_ai_not_albanian_unity_path = {
	NOT = { has_global_flag = KOS_ALBANIAN_UNITY_FOCUS_PATH }
	OR = {
		is_historical_focus_on = yes
		has_global_flag = KOS_HISTORICAL_FOCUS_PATH
		has_global_flag = KOS_YUGOSLAV_REVIVAL_FOCUS_PATH
	}
}

	}
