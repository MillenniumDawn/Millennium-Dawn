# THIS is the target, FROM is the country picking it
ai_generic_operation_target_scorer = {
	targets = {
		target_array = potential_and_current_enemies
		target_array = enemies_of_allies
		target_array = neighbors
		target_array = global.majors

		score = {
			base = 100

			modifier = {
				OR = {
					exists = no
					has_capitulated = yes
					is_ally_with = FROM
					has_country_flag = generic_operation_timeout_@FROM
				}
				factor = 0
			}

			modifier = {
				has_war_with = FROM
				factor = 4.0
			}

			modifier = {
				OR = {
					is_in_array = { FROM.potential_and_current_enemies = this }
					is_in_array = { FROM.enemies_of_allies = this }
				}
				factor = 1.5
			}

			modifier = {
				NOT = { has_war_with = FROM }
				OR = {
					check_variable = { FROM.ai_strategy_conquer@THIS > 0 }
					check_variable = { FROM.ai_strategy_antagonize@THIS > 0 }
					check_variable = { FROM.ai_strategy_declare_war@THIS > 0 }
					check_variable = { FROM.ai_strategy_prepare_for_war@THIS > 0 }
					check_variable = { FROM.ai_strategy_contain@THIS > 0 }
				}
				factor = 1.3
			}

			modifier = {
				NOT = { has_war_with = FROM }
				OR = {
					check_variable = { FROM.ai_strategy_befriend@THIS > 0 }
					check_variable = { FROM.ai_strategy_protect@THIS > 0 }
					check_variable = { FROM.ai_strategy_alliance@THIS > 0 }
					check_variable = { FROM.ai_strategy_support@THIS > 0 }
				}
				factor = 0.5
			}

			modifier = {
				FROM = {
					has_opinion = {
						target = PREV
						value < -25
					}
				}
				factor = 1.3
			}

			modifier = {
				NOT = { has_war_with = FROM }
				FROM = {
					has_opinion = {
						target = PREV
						value > 50
					}
				}
				factor = 0.6
			}

			modifier = {
				NOT = { has_war_with = FROM }
				ai_has_same_organization = yes
				factor = 0.5
			}

			modifier = {
				has_government = FROM
				NOT = { has_war_with = FROM }
				factor = 0.6
			}

			modifier = {
				is_in_array = { FROM.neighbors = this }
				factor = 1.5
			}

			# A far-off minor outside our region is rarely worth an operative
			modifier = {
				NOT = { has_war_with = FROM }
				NOT = { is_in_array = { FROM.neighbors = this } }
				has_at_least_regional_power_status = no
				NOT = {
					OR = {
						AND = {
							has_country_flag = european_nation_flag
							FROM = { has_country_flag = european_nation_flag }
						}
						AND = {
							has_country_flag = sub_saharan_nation_flag
							FROM = { has_country_flag = sub_saharan_nation_flag }
						}
						AND = {
							has_country_flag = middle_east_nation_flag
							FROM = { has_country_flag = middle_east_nation_flag }
						}
						AND = {
							has_country_flag = asian_nation_flag
							FROM = { has_country_flag = asian_nation_flag }
						}
						AND = {
							has_country_flag = american_nation_flag
							FROM = { has_country_flag = american_nation_flag }
						}
					}
				}
				factor = 0.5
			}

			modifier = {
				has_idea = super_power
				factor = 1.5
			}

			modifier = {
				has_idea = great_power
				factor = 1.3
			}

			modifier = {
				OR = {
					has_idea = large_power
					has_idea = regional_power
				}
				factor = 1.15
			}

			# Our client state: we already see into it through our influence
			modifier = {
				NOT = { has_war_with = FROM }
				check_variable = { influence_array^0 = FROM }
				check_variable = { influence_array_val^0 > 29.999 }
				factor = 0.5
			}

			# Our patron: burning that relationship costs more than the intel is worth
			modifier = {
				NOT = { has_war_with = FROM }
				FROM = {
					check_variable = { influence_array^0 = PREV }
					check_variable = { influence_array_val^0 > 19.999 }
				}
				factor = 0.6
			}

			modifier = {
				set_temp_variable = { temp_score = armies^num }
				divide_temp_variable = { temp_score = 100 }
				add_to_temp_variable = { temp_score = 1.0 }
				clamp_temp_variable = { var = temp_score max = 2.2 }
				factor = temp_score
			}

			modifier = {
				FROM = { check_variable = { generic_operation_target = PREV.id } }
				factor = 2.0
			}

			# Tokens we already hold here unlock the operations that spend them
			modifier = {
				FROM = {
					OR = {
						has_operation_token = { tag = PREV token = token_civilian }
						has_operation_token = { tag = PREV token = token_army }
						has_operation_token = { tag = PREV token = token_navy }
						has_operation_token = { tag = PREV token = token_airforce }
						has_operation_token = { tag = PREV token = token_resistance_contacts }
					}
				}
				factor = 2.0
			}
		}
	}
}
