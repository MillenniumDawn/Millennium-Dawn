MD_sftc_upgrade_school = {
	if = {
		limit = { check_variable = { MD_sftc_school_level < 1 } }
		army_experience = @MD_sftc_upgrade_army_1_neg
		navy_experience = @MD_sftc_upgrade_navy_1_neg
		air_experience = @MD_sftc_upgrade_air_1_neg
		add_ideas = MD_sftc_school_1
	}
	else_if = {
		limit = { check_variable = { MD_sftc_school_level = 1 } }
		army_experience = @MD_sftc_upgrade_army_2_neg
		navy_experience = @MD_sftc_upgrade_navy_2_neg
		air_experience = @MD_sftc_upgrade_air_2_neg
		if = {
			limit = { has_idea = MD_sftc_school_1 }
			swap_ideas = { remove_idea = MD_sftc_school_1 add_idea = MD_sftc_school_2 }
		}
	}
	else = {
		army_experience = @MD_sftc_upgrade_army_3_neg
		navy_experience = @MD_sftc_upgrade_navy_3_neg
		air_experience = @MD_sftc_upgrade_air_3_neg
		if = {
			limit = { has_idea = MD_sftc_school_2 }
			swap_ideas = { remove_idea = MD_sftc_school_2 add_idea = MD_sftc_school_3 }
		}
	}
	hidden_effect = {
		add_to_variable = { MD_sftc_school_level = 1 }
	}
}

MD_sftc_run_qualification = {
	army_experience = @MD_sftc_qualification_army_cost_neg
	navy_experience = @MD_sftc_qualification_navy_cost_neg
	air_experience = @MD_sftc_qualification_air_cost_neg
	add_command_power = @MD_sftc_command_power_cost_neg
	random_army_leader = {
		limit = {
			OR = {
				NOT = { has_trait = commando }
				NOT = { has_trait = ranger }
				NOT = { has_trait = naval_invader }
			}
		}
		random_list = {
			1 = {
				modifier = { factor = 0 has_trait = commando }
				add_unit_leader_trait = commando
			}
			1 = {
				modifier = { factor = 0 has_trait = ranger }
				add_unit_leader_trait = ranger
			}
			1 = {
				modifier = { factor = 0 has_trait = naval_invader }
				add_unit_leader_trait = naval_invader
			}
		}
	}
	add_timed_idea = { idea = MD_sftc_qualification_cooldown days = @MD_sftc_qualification_cooldown_days }
}

MD_sftc_toggle_window = {
	if = {
		limit = { NOT = { has_country_flag = MD_sftc_open } }
		set_country_flag = MD_sftc_open
	}
	else = {
		clr_country_flag = MD_sftc_open
	}
	update_MD_sftc_dirty_variable = yes
}

update_MD_sftc_dirty_variable = {
	if = {
		limit = { check_variable = { global.MD_sftc_dirty_update_var < 10000 } }
		add_to_variable = { global.MD_sftc_dirty_update_var = 1 }
	}
	else = {
		set_variable = { global.MD_sftc_dirty_update_var = 1 }
	}
}
