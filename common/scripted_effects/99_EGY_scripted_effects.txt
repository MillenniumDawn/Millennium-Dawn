EGY_recalculate_regime_balance = {
	set_temp_variable = { reg_influence_total = { value = EGY_military_influence add = EGY_security_influence add = EGY_brotherhood_influence } }
	set_temp_variable = { reg_weighted = 0 }
	if = {
		limit = { check_variable = { reg_influence_total > 0 } }
		set_temp_variable = {
			reg_weighted = {
				value = the_military_opinion
				subtract = 50
				multiply = EGY_military_influence
				add = { value = intelligence_community_opinion subtract = 50 multiply = EGY_security_influence }
				add = { value = the_muslim_brotherhood_opinion subtract = 50 multiply = EGY_brotherhood_influence }
				divide = reg_influence_total
			}
		}
	}
	set_variable = { EGY_mubarak_strength = { value = reg_weighted add = 50 add = EGY_mubarak_authority clamp = { min = 0 max = 100 } round = yes } }
	set_variable = { EGY_regime_pp_var = { value = EGY_mubarak_strength subtract = 50 multiply = 0.004 } }
	set_variable = { EGY_regime_stability_var = { value = EGY_mubarak_strength subtract = 50 multiply = 0.003 } }

	set_temp_variable = { reg_pressure = 0 }
	if = {
		limit = {
			OR = {
				has_idea = defence_07
				has_idea = defence_08
				has_idea = defence_09
			}
		}
		set_temp_variable = { reg_pressure = 1 }
	}
	else_if = {
		limit = {
			OR = {
				has_idea = defence_05
				has_idea = defence_06
			}
		}
		set_temp_variable = { reg_pressure = 0.5 }
	}
	if = {
		limit = { has_idea = depression }
		add_to_temp_variable = { reg_pressure = -1 }
	}
	else_if = {
		limit = { has_idea = recession }
		add_to_temp_variable = { reg_pressure = -0.5 }
	}
	if = {
		limit = { check_variable = { interest_rate > 8 } }
		add_to_temp_variable = { reg_pressure = -0.5 }
	}
	if = {
		limit = {
			NOT = {
				OR = {
					western_autocrats_are_in_power = yes
					emerging_autocracy_are_in_power = yes
					neutrality_neutral_autocracy_are_in_power = yes
					nationalist_fascist_are_in_power = yes
					nationalist_military_junta_are_in_power = yes
				}
			}
		}
		add_to_temp_variable = { reg_pressure = -1 }
	}
	set_variable = { EGY_military_pressure_var = reg_pressure }

	if = {
		limit = { has_idea = the_military }
		set_temp_variable = { reg_mil_tier = 7 }
		if = { limit = { check_variable = { the_military_opinion > 9.99 } } set_temp_variable = { reg_mil_tier = 6 } }
		if = { limit = { check_variable = { the_military_opinion > 19.99 } } set_temp_variable = { reg_mil_tier = 5 } }
		if = { limit = { check_variable = { the_military_opinion > 29.99 } } set_temp_variable = { reg_mil_tier = 4 } }
		if = { limit = { check_variable = { the_military_opinion > 39.99 } } set_temp_variable = { reg_mil_tier = 3 } }
		if = { limit = { check_variable = { the_military_opinion > 49.99 } } set_temp_variable = { reg_mil_tier = 2 } }
		if = { limit = { check_variable = { the_military_opinion > 69.99 } } set_temp_variable = { reg_mil_tier = 1 } }
		if = {
			limit = { NOT = { check_variable = { EGY_current_mil_tier = reg_mil_tier } } }
			set_variable = { EGY_current_mil_tier = reg_mil_tier }
			remove_ideas = {
				EGY_military_happy
				EGY_military_anger1
				EGY_military_anger2
				EGY_military_anger3
				EGY_military_anger4
				EGY_military_anger5
			}
			if = { limit = { check_variable = { reg_mil_tier = 1 } } add_ideas = EGY_military_happy }
			else_if = { limit = { check_variable = { reg_mil_tier = 3 } } add_ideas = EGY_military_anger1 }
			else_if = { limit = { check_variable = { reg_mil_tier = 4 } } add_ideas = EGY_military_anger2 }
			else_if = { limit = { check_variable = { reg_mil_tier = 5 } } add_ideas = EGY_military_anger3 }
			else_if = { limit = { check_variable = { reg_mil_tier = 6 } } add_ideas = EGY_military_anger4 }
			else_if = { limit = { check_variable = { reg_mil_tier = 7 } } add_ideas = EGY_military_anger5 }
		}
	}

	if = {
		limit = { has_idea = intelligence_community }
		set_temp_variable = { reg_sec_tier = 2 }
		if = { limit = { check_variable = { intelligence_community_opinion < 29.99 } } set_temp_variable = { reg_sec_tier = 3 } }
		if = { limit = { check_variable = { intelligence_community_opinion > 69.99 } } set_temp_variable = { reg_sec_tier = 1 } }
		if = {
			limit = { NOT = { check_variable = { EGY_current_sec_tier = reg_sec_tier } } }
			set_variable = { EGY_current_sec_tier = reg_sec_tier }
			remove_ideas = {
				EGY_security_state_loyal
				EGY_security_state_unchecked
			}
			if = { limit = { check_variable = { reg_sec_tier = 1 } } add_ideas = EGY_security_state_loyal }
			else_if = { limit = { check_variable = { reg_sec_tier = 3 } } add_ideas = EGY_security_state_unchecked }
		}
	}

	if = {
		limit = { has_idea = the_muslim_brotherhood }
		set_temp_variable = { reg_mb_tier = 2 }
		if = { limit = { check_variable = { the_muslim_brotherhood_opinion < 29.99 } } set_temp_variable = { reg_mb_tier = 3 } }
		if = { limit = { check_variable = { the_muslim_brotherhood_opinion > 69.99 } } set_temp_variable = { reg_mb_tier = 1 } }
		if = {
			limit = { NOT = { check_variable = { EGY_current_mb_tier = reg_mb_tier } } }
			set_variable = { EGY_current_mb_tier = reg_mb_tier }
			remove_ideas = {
				EGY_brotherhood_quietist
				EGY_brotherhood_agitating
			}
			if = { limit = { check_variable = { reg_mb_tier = 1 } } add_ideas = EGY_brotherhood_quietist }
			else_if = { limit = { check_variable = { reg_mb_tier = 3 } } add_ideas = EGY_brotherhood_agitating }
		}
	}

}

EGY_update_informal_cairo = {
	if = {
		limit = { owns_state = 215 }
		215 = { set_variable = { PREV.EGY_cairo_population_k = state_population_k } }
		clamp_variable = { var = EGY_cairo_population_k min = 1 }
		set_variable = {
			EGY_informal_share = {
				value = EGY_cairo_population_k
				subtract = EGY_cairo_formal_housing_k
				divide = EGY_cairo_population_k
				multiply = 100
				clamp = { min = 0 max = 100 }
			}
		}
	}
	else = {
		set_variable = { EGY_informal_share = 0 }
	}
	set_temp_variable = {
		informal_excess = {
			value = EGY_informal_share
			subtract = 40
			clamp = { min = 0 max = 60 }
		}
	}
	set_variable = { EGY_cairo_country_productivity_growth_modifier = { value = informal_excess multiply = -0.01 } }
	if = {
		limit = { has_completed_focus = EGY_cairo_metro_line_three }
		multiply_variable = { EGY_cairo_country_productivity_growth_modifier = 0.5 }
	}
	set_variable = { EGY_cairo_production_speed_buildings_factor = { value = informal_excess multiply = -0.002 } }
	set_variable = { EGY_cairo_stability_factor = { value = informal_excess multiply = -0.0025 } }
}
