USA_shipyard_setup = {
	set_variable = { shipyard_crisis_level = 8 }
	set_variable = { shipyard_drift_early = 0.2 }
	set_variable = { shipyard_drift_late = 0.35 }
	set_variable = { shipyard_late_year = 2012 }
	set_variable = { shipyard_cap_output = 0.3 }
	set_variable = { shipyard_cap_repair = 0.5 }
	set_variable = { shipyard_cap_refit = 0.35 }
	set_variable = { shipyard_cost_base = 60 }
	set_variable = { shipyard_drydock_stage = 0 }
	set_variable = { shipyard_workforce_stage = 0 }
	set_variable = { shipyard_eq_steps = 0 }
	add_dynamic_modifier = { modifier = shipyard_crisis_modifier }
	shipyard_recalc = yes
}

# One step per 10 crisis points. Submarines and carriers sit in the worst drydock
# queues, so they take the larger share.
USA_shipyard_equipment_step_up = {
	add_equipment_bonus = {
		name = shipyard_crisis_equipment
		bonus = {
			destroyer = {
				instant = yes
				build_cost_ic = 0.03
			}
			stealth_destroyer = {
				instant = yes
				build_cost_ic = 0.03
			}
			cruiser = {
				instant = yes
				build_cost_ic = 0.04
			}
			battle_cruiser = {
				instant = yes
				build_cost_ic = 0.04
			}
			attack_submarine = {
				instant = yes
				build_cost_ic = 0.04
			}
			missile_submarine = {
				instant = yes
				build_cost_ic = 0.04
			}
			carrier = {
				instant = yes
				build_cost_ic = 0.05
			}
		}
	}
}
