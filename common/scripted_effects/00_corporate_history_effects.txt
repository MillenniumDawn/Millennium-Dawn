# Corporate-history framework. Docs: docs/src/content/resources/corporate-history-framework.md
#
# Chains: IBM, Sun/Microsoft, HP, Apple, NVIDIA, Dell, TI, Micron, Motorola (USA),
# Lenovo and Huawei (CHI), Sony (JAP), ATI/AMD and Matrox (CAN), Nokia (FIN), TSMC and Foxconn (TAI),
# Industrial Sovereignty (POL), Siemens (GER), Ericsson (SWE), and BlackBerry (CAN/USA).
# The rule_corporate_history game rule gates the country-owned dispatchers, monthly
# outcome drivers, and popup event triggers. Event-level gates also stop a queued
# Full-mode popup if the rule is no longer Full when delivery occurs.
#
# Start-date invariant: milestone day offsets are scheduled only from a January
# window. Where a chain has local recovery, it reconstructs elapsed milestones
# first, then queues due unresolved events with a short delay; it never replays a
# January offset from a mid-year date.

# Effect: corporate_history_apply_delta
# Purpose: Applies a signed delta to a corporate state value and clamps it to the
#          framework band. The 0..10 bounds live only here.
# Parameters:
# - corp_delta: The signed delta to add before clamping
# - corp_value: The state value, read and written back as a temp variable (optional here only when chained from corporate_history_clamp_value; direct callers must set it)
corporate_history_apply_delta = {
	add_to_temp_variable = { corp_value = corp_delta }
	clamp_temp_variable = { var = corp_value min = 0 max = 10 }
}

# Effect: corporate_history_clamp_value
# Purpose: Clamps a corporate state value to the framework band without changing it.
# Parameters:
# - corp_value: The state value, read and written back as a temp variable
corporate_history_clamp_value = {
	set_temp_variable = { corp_delta = 0 }
	corporate_history_apply_delta = yes
}

# Effect: USA_corporate_history_monthly_outcomes
# Purpose: Outcomes Only driver for USA corporate chains on the first monthly tick
#          after each historical milestone (<= ~31 days lag). In Full mode it owns
#          visible Google/Oracle catch-up, retries the IBM-to-Lenovo handoff, and
#          recovers missed TI/Micron/Motorola/Dell terminal outcomes.
USA_corporate_history_monthly_outcomes = {
	corporate_history_monthly_dispatch = yes
	USA_oem_legacy_monthly_driver = yes
	USA_oem_storage_monthly_driver = yes

	if = {
		limit = {
			original_tag = USA
			NOT = { has_country_flag = collapsed_nation }
		}

		if = {
			limit = { corporate_history_full_enabled = yes }
			USA_ibm_schedule_pc_decision = yes
			USA_ibm_schedule_x86_decision = yes
			USA_google_reconstruct_extension_prerequisite = yes
			if = {
				limit = {
					NOT = { OR = {
						has_country_flag = USA_google_antitrust_settlement
						has_country_flag = USA_google_antitrust_litigation
						has_country_flag = USA_google_antitrust_structural_separation
					} }
				}
				USA_google_extension_catchup = yes
			}
			if = {
				limit = { NOT = { has_country_flag = USA_oracle_event_12_resolved } }
				USA_oracle_chain_catchup = yes
			}
			if = {
				limit = {
					country_exists = CHI
					OR = {
						has_country_flag = USA_ibm_pc_divested
						has_country_flag = USA_ibm_pc_retained
						has_country_flag = USA_ibm_pc_spinout
					}
					CHI = {
						original_tag = CHI
						NOT = { has_country_flag = CHI_lenovo_pc_path_chosen }
					}
				}
				USA_ibm_notify_lenovo_pc_path = yes
			}
		}

		if = {
			limit = {
				corporate_history_outcomes_only_enabled = yes
				OR = {
					NOT = { has_country_flag = USA_microsoft_reconstruct_complete }
					NOT = { has_country_flag = USA_apple_reconstruct_complete }
					NOT = { has_country_flag = USA_ibm_reconstruct_complete }
					NOT = { has_country_flag = USA_nvidia_reconstruct_complete }
					NOT = { has_country_flag = USA_google_reconstruct_complete }
					NOT = { has_country_flag = USA_oracle_reconstruct_complete }
					NOT = { has_country_flag = USA_hp_reconstruct_complete }
					NOT = { has_country_flag = USA_e3_reconstruct_complete }
					NOT = { has_country_flag = USA_dell_reconstruct_complete }
					NOT = { has_country_flag = USA_ti_reconstruct_complete }
					NOT = { has_country_flag = USA_micron_reconstruct_complete }
					NOT = { has_country_flag = USA_motorola_reconstruct_complete }
					NOT = { has_country_flag = USA_aig_reconstruct_complete }
					NOT = { has_country_flag = USA_xbox_reconstruct_complete }
					NOT = { has_country_flag = USA_blackberry_reconstruct_complete }
				}
			}
			USA_microsoft_reconstruct_history = yes
			USA_apple_reconstruct_history = yes
			USA_ibm_reconstruct_history = yes
			USA_nvidia_reconstruct_history = yes
			USA_google_reconstruct_history = yes
			USA_oracle_reconstruct_history = yes
			USA_hp_reconstruct_history = yes
			USA_e3_reconstruct_history = yes
			USA_dell_reconstruct_history = yes
			USA_micron_reconstruct_history = yes
			USA_ti_reconstruct_history = yes
			USA_motorola_reconstruct_history = yes
			USA_aig_reconstruct_history = yes
			USA_xbox_reconstruct_history = yes
			USA_blackberry_reconstruct_history = yes
		}

		if = {
			limit = {
				corporate_history_full_enabled = yes
				date > 2009.1.26
				NOT = { has_country_flag = USA_blackberry_reconstruct_complete }
			}
			USA_blackberry_reconstruct_history = yes

		if = {
			limit = {
				corporate_history_full_enabled = yes
				date > 2025.6.12
				NOT = { has_country_flag = USA_micron_reconstruct_complete }
				NOT = { has_country_flag = USA_micron_capstone_choice_pending }

			USA_micron_reconstruct_history = yes

		if = {
			limit = {
				corporate_history_full_enabled = yes
				date > 2025.8.6
				NOT = { has_country_flag = USA_motorola_reconstruct_complete }
				NOT = { has_country_flag = USA_motorola_capstone_choice_pending }
			}
			USA_motorola_reconstruct_history = yes
		}

		if = {
			limit = {
				corporate_history_full_enabled = yes
				date > 2026.2.4

				NOT = { has_country_flag = USA_ti_capstone_choice_pending }
			}
			USA_ti_reconstruct_history = yes

			limit = {
				corporate_history_full_enabled = yes
				date > 2026.6.15
				NOT = { has_country_flag = USA_dell_reconstruct_complete }
			}
			USA_dell_reconstruct_history = yes
		}

		# MD's Great Financial Crisis is mean-time-to-happen driven, so the 2008 dispatcher
		# can reach AIG's opening date before the Lehman gate is satisfiable.
		if = {
			limit = {
				corporate_history_full_enabled = yes
				date > 2008.9.16
				date < 2010.1.1
				NOT = { has_country_flag = USA_aig_event_1_resolved }
			}
			set_temp_variable = { aig_schedule_mode = 1 }

		}
	}

	if = {
		limit = { corporate_history_enabled = yes }
		USA_corporate_systems_update_economic_bridge = yes
	}
}

		}

		if = {
			limit = { corporate_history_full_enabled = yes }
			CHI_lenovo_schedule_current_year_events = yes

				limit = {
					date > 2014.7.9
					has_country_flag = CHI_lenovo_brand_path_chosen
					NOT = { has_country_flag = CHI_lenovo_systemx_path_chosen }
					NOT = { has_country_flag = CHI_lenovo_event_7_delivered }
					NOT = { has_country_flag = CHI_lenovo_event_7_pending }
					country_exists = USA
					NOT = { has_war_with = USA }
					USA = {
						NOT = { has_country_flag = collapsed_nation }
						OR = {
							has_country_flag = USA_ibm_x86_divested
							has_country_flag = USA_ibm_x86_retained
							has_country_flag = USA_ibm_silicon_retained
						}
					}
				}
				set_country_flag = { flag = CHI_lenovo_event_7_pending value = 1 days = 40 }
				country_event = { id = CHI_lenovo_events.7 days = 5 }
			}
			CHI_lenovo_recover_interrupted_stages = yes
		}

		if = {
			limit = {
				corporate_history_full_enabled = yes
				date > 2026.4.30
				NOT = { has_country_flag = CHI_huawei_reconstruct_complete }
			}
			CHI_huawei_reconstruct_history = yes
		}
	}
}
