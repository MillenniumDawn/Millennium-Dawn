reverse_engineering_study_legacy_family = {
	set_temp_variable = { reverse_engineering_studied_index = -1 }
	for_each_loop = {
		array = reverse_engineering_techs
		value = reverse_engineering_tech
		index = reverse_engineering_index
		set_temp_variable = { reverse_engineering_generation = { value = reverse_engineering_index add = 1 } }
		if = {
			limit = { check_variable = { reverse_engineering_purchase = 1 } }
			meta_effect = {
				text = {
					if = {
						limit = { ROOT = { contract_contains_equipment = [EQUIPMENT] } }
						set_temp_variable = { reverse_engineering_studied_index = reverse_engineering_index }
					}
				}
				EQUIPMENT = "$EQUIPMENT$"
			}
		}
		else = {
			meta_effect = {
				text = {
					if = {
						limit = { ROOT = { has_tech = [TECH] } }
						set_temp_variable = { reverse_engineering_studied_index = reverse_engineering_index }
					}
				}
				TECH = "[?reverse_engineering_tech.GetTokenKey]"
			}
		}
	}
	reverse_engineering_grant_bonuses = yes
}
