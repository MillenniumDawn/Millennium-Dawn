			base = 30
			modifier = {
				factor = 3
				has_global_flag = CHI_HEGEMON_FOCUS_PATH
			}
		}
	}

	option = {
		name = obt_strait.1.b
		log = "[GetDateText]: [This.GetName]: obt_strait.1.b executed"
