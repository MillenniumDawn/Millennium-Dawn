on_actions = {
	on_monthly_ARM = {
		effect = {
			#Tourism Income
			if = {
				limit = {
					has_completed_focus = ARM_economy_tourism
				}
				ARM_tourism_calculate = yes
			}
		}
	}

	on_war_relation_added = {
		effect = {
			if = {
				limit = { ROOT = { original_tag = ARM } }
				ROOT = { ARM_review_foreign_partner_ties = yes }
			}
			if = {
				limit = { FROM = { original_tag = ARM } }
				FROM = { ARM_review_foreign_partner_ties = yes }
			}
		}
	}
}
