# The Belarusian half of the exclusion zone (the Polesie reserve) slips a little each month unless Minsk reinforces it.
STALKER_monthly_polesie_pulse = {
	if = {
		limit = {
			NOT = { has_global_flag = STALKER_kaimanov_outcome }
			698 = { STALKER_is_active_zone_anchor = yes }
			country_exists = BLR
		}
		BLR = {
			if = {
				limit = { NOT = { has_variable = STALKER_polesie_security } }
				set_variable = { STALKER_polesie_security = 40 }
			}
			add_to_variable = { STALKER_polesie_security = -1 }
			clamp_variable = { var = STALKER_polesie_security min = 0 max = 100 }
			if = {
				limit = {
					check_variable = { STALKER_polesie_security < 30 }
					NOT = { has_country_flag = STALKER_polesie_leak_cooldown }
				}
				random = {
					chance = 10
					set_country_flag = { flag = STALKER_polesie_leak_cooldown value = 1 days = 90 }
					country_event = STALKER.60
				}
			}
		}
	}
}

# Success scales with Polesie security; a high roll exposes officials on the smugglers' payroll.
STALKER_run_kgb_investigation = {
	set_temp_variable = { STALKER_kgb_roll = 0 }
	randomize_temp_variable = { var = STALKER_kgb_roll min = 1 max = 100 distribution = uniform }
	set_temp_variable = { STALKER_kgb_success = { value = STALKER_polesie_security divide = 2 add = 25 } }
	if = {
		limit = { check_variable = { STALKER_kgb_roll < STALKER_kgb_success } }
		country_event = STALKER.61
	}
	else_if = {
		limit = { check_variable = { STALKER_kgb_roll > 80 } }
		country_event = STALKER.62
	}
	else = {
		country_event = STALKER.64
	}
}
