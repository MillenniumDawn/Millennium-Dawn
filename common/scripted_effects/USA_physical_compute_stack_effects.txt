USA_physical_compute_stack_resolve = {
	if = {
		limit = {
			has_country_flag = USA_ti_capstone_resolved
			has_country_flag = USA_stack_ti_foundational_viable
		}
		if = {
			limit = { NOT = { has_idea = USA_foundational_semiconductor_base } }
			add_ideas = USA_foundational_semiconductor_base
		}
	}
	else = {
		if = {
			limit = { has_idea = USA_foundational_semiconductor_base }
			remove_ideas = USA_foundational_semiconductor_base
		}
	}
	if = {
		limit = {
			has_country_flag = USA_micron_capstone_resolved
			has_country_flag = USA_stack_micron_memory_viable
		}
		if = {
			limit = { NOT = { has_idea = USA_memory_arsenal } }
			add_ideas = USA_memory_arsenal
		}
	}
	else = {
		if = {
			limit = { has_idea = USA_memory_arsenal }
			remove_ideas = USA_memory_arsenal
		}
	}
	if = {
		limit = {
			has_country_flag = USA_stack_motorola_capstone_resolved
			has_country_flag = USA_stack_motorola_communications_viable
		}
		if = {
			limit = { NOT = { has_idea = USA_mission_critical_networks } }
			add_ideas = USA_mission_critical_networks
		}
	}
	else = {
		if = {
			limit = { has_idea = USA_mission_critical_networks }
			remove_ideas = USA_mission_critical_networks
		}
	}

	set_temp_variable = { USA_stack_seam_count = 0 }
	if = {
		limit = {
			has_country_flag = USA_stack_ti_foundational_viable
			has_country_flag = USA_stack_micron_memory_viable
			OR = {
				has_country_flag = USA_stack_txn_lehi_sold_to_ti
				has_country_flag = USA_stack_txn_lehi_shared_reserve
			}
		}
		set_country_flag = USA_stack_seam_memory_to_signal
	}
	else = { clr_country_flag = USA_stack_seam_memory_to_signal }
	if = {
		limit = { has_country_flag = USA_stack_seam_memory_to_signal }
		add_to_temp_variable = { USA_stack_seam_count = 1 }
	}
	if = {
		limit = {
			has_country_flag = USA_stack_ti_foundational_viable
			has_country_flag = USA_stack_motorola_embedded_viable
		}
		set_country_flag = USA_stack_seam_embedded_interoperability
	}
	else = { clr_country_flag = USA_stack_seam_embedded_interoperability }
	if = {
		limit = { has_country_flag = USA_stack_seam_embedded_interoperability }
		add_to_temp_variable = { USA_stack_seam_count = 1 }
	}
	if = {
		limit = {
			has_country_flag = USA_stack_motorola_communications_viable
			has_country_flag = USA_stack_seam_motorola_interoperable_comms
		}
		add_to_temp_variable = { USA_stack_seam_count = 1 }
	}
	if = {
		limit = {
			has_country_flag = USA_stack_motorola_communications_viable
			has_country_flag = USA_stack_seam_motorola_powerpc
		}
		add_to_temp_variable = { USA_stack_seam_count = 1 }
	}

	if = {
		limit = {
			has_idea = USA_foundational_semiconductor_base
			has_idea = USA_memory_arsenal
			has_idea = USA_mission_critical_networks
			check_variable = { USA_stack_seam_count > 1 }
		}
		if = {
			limit = { NOT = { has_country_flag = USA_stack_physical_capstone_resolved } }
			set_country_flag = USA_stack_physical_capstone_resolved
		}
		if = {
			limit = { NOT = { has_idea = USA_physical_compute_stack } }
			add_ideas = USA_physical_compute_stack
		}
	}
	else = {
		clr_country_flag = USA_stack_physical_capstone_resolved
		if = {
			limit = { has_idea = USA_physical_compute_stack }
			remove_ideas = USA_physical_compute_stack
		}
	}

	# Shared exit for the TI, Micron and Motorola capstones: refresh the national
	# bridge now rather than leaving it stale until the next monthly tick.
	USA_corporate_systems_update_economic_bridge = yes
}
