# Called by the energy owner after its balance calculation, before storage can be mistaken for supply.
ai_race_capture_power_sample = {
	if = {
		limit = {
			OR = {
				ai_race_country_initialized = yes
				has_variable = ai_race_ai_target_stage
			}
		}
		set_variable = { ai_race_power_available = { value = energy_balance add = ai_race_accounted_power max = 0 } }
		set_variable = { ai_race_power_multiplier = ai_race_sampled_power_multiplier }
	}
}

# Prospective countries can sample owners without enrollment or operating demand.
ai_race_refresh_planning_sample = {
	if = {
		limit = { has_country_flag = energy_state_bases_initialized }
		calculate_energy_use = yes
	}
	ai_race_refresh_capacity_sample = yes
}

ai_race_refresh_capacity_sample = {
	set_variable = {
		ai_race_civs_available = {
			value = 0
			every_collection = {
				named_collection = controlled_states_collection
				add = non_damaged_building_level@industrial_complex
			}
		}
	}
	set_variable = {
		ai_race_network_available = {
			value = 0
			every_collection = {
				named_collection = controlled_states_collection
				add = non_damaged_building_level@internet_station
			}
		}
	}
	set_variable = { ai_race_capacity_settled = 0 }
	if = {
		limit = {
			has_country_flag = energy_state_bases_initialized
			has_variable = ai_race_power_multiplier
			OR = {
				NOT = { has_variable = ai_race_stage }
				check_variable = { ai_race_stage = 0 }
				AND = {
					check_variable = { var = global.num_days value = ai_race_demand_settle_day compare = greater_than_or_equals }
					check_variable = { modifier@ai_race_applied_power = ai_race_demand_power }
					check_variable = { modifier@ai_race_applied_microchips = ai_race_demand_microchips }
					check_variable = { modifier@ai_race_applied_composites = ai_race_demand_composites }
					check_variable = { modifier@ai_race_applied_oil = ai_race_demand_oil }
				}
			}
		}
		set_variable = { ai_race_capacity_settled = 1 }
		set_variable = { ai_race_accounted_microchips = modifier@ai_race_applied_microchips }
		set_variable = { ai_race_accounted_composites = modifier@ai_race_applied_composites }
		set_variable = { ai_race_accounted_oil = modifier@ai_race_applied_oil }
	}
	else = {
		# An unsettled modifier never supplies an optimistic add-back after a purchase or trade loss.
		set_variable = { ai_race_accounted_microchips = 0 }
		set_variable = { ai_race_accounted_composites = 0 }
		set_variable = { ai_race_accounted_oil = 0 }
	}
	set_variable = { ai_race_microchips_available = { value = resource@microchips add = ai_race_accounted_microchips max = 0 } }
	set_variable = { ai_race_composites_available = { value = resource@composites add = ai_race_accounted_composites max = 0 } }
	set_variable = { ai_race_oil_available = { value = resource@oil add = ai_race_accounted_oil max = 0 } }
}

ai_race_refresh_operating_state = {
	if = {
		limit = { has_country_flag = energy_state_bases_initialized }
		calculate_energy_use = yes
	}
	if = {
		limit = { ai_race_full_mode = yes }
		set_temp_variable = { ai_race_requirement_stage = ai_race_stage }
		ai_race_load_stage_requirements = yes
		set_variable = { ai_race_current_required_months = ai_race_requirement_months }
		if = {
			limit = { NOT = { check_variable = { ai_race_demand_stage = ai_race_stage } } }
			set_variable = { ai_race_demand_settle_day = { value = global.num_days add = 2 } }
			set_variable = { ai_race_demand_stage = ai_race_stage }
		}
		set_variable = { ai_race_demand_power = ai_race_requirement_power }
		set_variable = { ai_race_demand_microchips = ai_race_requirement_microchips }
		set_variable = { ai_race_demand_composites = ai_race_requirement_composites }
		set_variable = { ai_race_demand_oil = ai_race_requirement_oil }
		ai_race_refresh_capacity_sample = yes
		set_variable = { ai_race_current_readiness = 1 }
		set_variable = { ai_race_current_civs_required = ai_race_requirement_civs }
		set_variable = { ai_race_current_civs_available = ai_race_civs_available }
		set_variable = { ai_race_current_civs_ratio = 1 }
		if = {
			limit = { check_variable = { ai_race_current_civs_required > 0 } }
			set_variable = { ai_race_current_civs_ratio = { value = ai_race_current_civs_available divide = ai_race_current_civs_required clamp = { min = 0 max = 1 } } }
		}
		set_variable = { ai_race_current_readiness = { value = ai_race_current_readiness min = ai_race_current_civs_ratio } }
		set_variable = { ai_race_current_network_required = ai_race_requirement_network }
		set_variable = { ai_race_current_network_available = ai_race_network_available }
		set_variable = { ai_race_current_network_ratio = 1 }
		if = {
			limit = { check_variable = { ai_race_current_network_required > 0 } }
			set_variable = { ai_race_current_network_ratio = { value = ai_race_current_network_available divide = ai_race_current_network_required clamp = { min = 0 max = 1 } } }
		}
		set_variable = { ai_race_current_readiness = { value = ai_race_current_readiness min = ai_race_current_network_ratio } }
		set_variable = { ai_race_current_power_required = ai_race_requirement_power }
		multiply_variable = { ai_race_current_power_required = ai_race_power_multiplier }
		set_variable = { ai_race_current_power_available = ai_race_power_available }
		set_variable = { ai_race_current_power_ratio = 1 }
		if = {
			limit = {
				check_variable = { ai_race_stage > 0 }
				NOT = { has_variable = ai_race_power_multiplier }
			}
			set_variable = { ai_race_current_power_ratio = 0 }
		}
		if = {
			limit = { check_variable = { ai_race_current_power_required > 0 } }
			set_variable = { ai_race_current_power_ratio = { value = ai_race_current_power_available divide = ai_race_current_power_required clamp = { min = 0 max = 1 } } }
		}
		set_variable = { ai_race_current_readiness = { value = ai_race_current_readiness min = ai_race_current_power_ratio } }
		set_variable = { ai_race_current_microchips_required = ai_race_requirement_microchips }
		set_variable = { ai_race_current_microchips_available = ai_race_microchips_available }
		set_variable = { ai_race_current_microchips_ratio = 1 }
		if = {
			limit = { check_variable = { ai_race_current_microchips_required > 0 } }
			set_variable = { ai_race_current_microchips_ratio = { value = ai_race_current_microchips_available divide = ai_race_current_microchips_required clamp = { min = 0 max = 1 } } }
		}
		set_variable = { ai_race_current_readiness = { value = ai_race_current_readiness min = ai_race_current_microchips_ratio } }
		set_variable = { ai_race_current_composites_required = ai_race_requirement_composites }
		set_variable = { ai_race_current_composites_available = ai_race_composites_available }
		set_variable = { ai_race_current_composites_ratio = 1 }
		if = {
			limit = { check_variable = { ai_race_current_composites_required > 0 } }
			set_variable = { ai_race_current_composites_ratio = { value = ai_race_current_composites_available divide = ai_race_current_composites_required clamp = { min = 0 max = 1 } } }
		}
		set_variable = { ai_race_current_readiness = { value = ai_race_current_readiness min = ai_race_current_composites_ratio } }
		set_variable = { ai_race_current_oil_required = ai_race_requirement_oil }
		set_variable = { ai_race_current_oil_available = ai_race_oil_available }
		set_variable = { ai_race_current_oil_ratio = 1 }
		if = {
			limit = { check_variable = { ai_race_current_oil_required > 0 } }
			set_variable = { ai_race_current_oil_ratio = { value = ai_race_current_oil_available divide = ai_race_current_oil_required clamp = { min = 0 max = 1 } } }
		}
		set_variable = { ai_race_current_readiness = { value = ai_race_current_readiness min = ai_race_current_oil_ratio } }
		set_variable = { ai_race_bonus = { value = ai_race_stage multiply = 0.01 } }
		if = {
			limit = {
				check_variable = { ai_race_stage = 4 }
				check_variable = { global.ai_race_first_finisher_id = THIS.id }
			}
			add_to_variable = { ai_race_bonus = 0.01 }
		}
		multiply_variable = { ai_race_bonus = ai_race_current_readiness }
		if = {
			limit = {
				check_variable = { ai_race_stage > 0 }
				NOT = { has_dynamic_modifier = { modifier = ai_race_operating_modifier } }
			}
			add_dynamic_modifier = { modifier = ai_race_operating_modifier }
		}
	}
}

# Operating state refreshes energy first; offers reuse that same sample without accruing expenses.
