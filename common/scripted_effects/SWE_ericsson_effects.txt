SWE_ericsson_schedule_event_1 = {
	if = {
		limit = {
			corporate_history_full_enabled = yes
			original_tag = SWE
			NOT = { has_country_flag = SWE_ericsson_event_1_expected }
			NOT = { has_country_flag = SWE_ericsson_event_1_resolved }
			NOT = { has_country_flag = SWE_sony_ericsson_joint_venture }
			NOT = { has_country_flag = SWE_ericsson_retained_handsets }
		}
		set_country_flag = SWE_ericsson_event_1_expected
		if = {
			limit = { NOT = { has_country_flag = collapsed_nation } }
			if = {
				limit = { has_country_flag = SWE_ericsson_schedule_immediately }
				set_country_flag = { flag = SWE_ericsson_event_1_pending days = 32 value = 1 }
				country_event = { id = SWE_ericsson_events.1 days = 1 }
			}
			else = {
				set_country_flag = { flag = SWE_ericsson_event_1_pending days = 305 value = 1 }
				country_event = { id = SWE_ericsson_events.1 days = 273 }
			}
		}
	}
}

# A milestone crossed after campaign start remains a choice instead of silent history.
SWE_ericsson_schedule_current_year_event = {
	if = {
		limit = {
			date > 2001.9.30
			has_start_date < 2001.10.1
			NOT = { has_country_flag = SWE_ericsson_event_1_expected }
			NOT = { has_country_flag = SWE_ericsson_event_1_resolved }
		}
		set_country_flag = SWE_ericsson_schedule_immediately
		SWE_ericsson_schedule_event_1 = yes
		clr_country_flag = SWE_ericsson_schedule_immediately
	}
	else_if = {
		limit = {
			date > 2012.2.14
			has_start_date < 2012.2.15
			NOT = { has_country_flag = SWE_ericsson_event_2_expected }
			NOT = { has_country_flag = SWE_ericsson_event_2_resolved }
		}
		set_country_flag = SWE_ericsson_schedule_immediately
		SWE_ericsson_schedule_event_2 = yes
		clr_country_flag = SWE_ericsson_schedule_immediately
	}
	else_if = {
		limit = {
			date > 2022.7.20
			has_start_date < 2022.7.21
			NOT = { has_country_flag = SWE_ericsson_event_3_expected }
			NOT = { has_country_flag = SWE_ericsson_event_3_resolved }
		}
		set_country_flag = SWE_ericsson_schedule_immediately
		SWE_ericsson_schedule_event_3 = yes
		clr_country_flag = SWE_ericsson_schedule_immediately
	}
}

# Recovery only requeues deliveries whose normal yearly schedule was interrupted.
SWE_ericsson_recover_interrupted_events = {
	if = {
		limit = {
			date > 2001.9.30
			has_country_flag = SWE_ericsson_event_1_expected
			NOT = { has_country_flag = SWE_ericsson_event_1_pending }
			NOT = { has_country_flag = SWE_ericsson_event_1_resolved }
			NOT = { has_country_flag = SWE_ericsson_event_1_recovery_used }
			NOT = { has_country_flag = SWE_sony_ericsson_joint_venture }
			NOT = { has_country_flag = SWE_ericsson_retained_handsets }
		}
		set_country_flag = SWE_ericsson_event_1_recovery_used
		country_event = SWE_ericsson_events.1
	}
	else_if = {
		limit = {
			date > 2012.2.14
			has_country_flag = SWE_ericsson_event_2_expected
			NOT = { has_country_flag = SWE_ericsson_event_2_pending }
			NOT = { has_country_flag = SWE_ericsson_event_2_resolved }
			NOT = { has_country_flag = SWE_ericsson_event_2_recovery_used }
			NOT = { has_country_flag = SWE_ericsson_exited_handsets }
			NOT = { has_country_flag = SWE_ericsson_full_handset_control }
		}
		set_country_flag = SWE_ericsson_event_2_recovery_used
		country_event = SWE_ericsson_events.2
	}
	else_if = {
		limit = {
			date > 2022.7.20
			has_country_flag = SWE_ericsson_event_3_expected
			NOT = { has_country_flag = SWE_ericsson_event_3_pending }
			NOT = { has_country_flag = SWE_ericsson_event_3_resolved }
			NOT = { has_country_flag = SWE_ericsson_event_3_recovery_used }
			NOT = { has_country_flag = SWE_ericsson_vonage_platform }
			NOT = { has_country_flag = SWE_ericsson_radio_network_focus }
		}
		set_country_flag = SWE_ericsson_event_3_recovery_used
		country_event = SWE_ericsson_events.3
	}
}

SWE_ericsson_monthly_driver = {
	if = {
		limit = {
			corporate_history_enabled = yes
			original_tag = SWE
			NOT = { has_country_flag = collapsed_nation }
			NOT = {
				OR = {
					has_country_flag = SWE_ericsson_vonage_platform
					has_country_flag = SWE_ericsson_radio_network_focus
				}
			}
		}
		if = {
			limit = { corporate_history_outcomes_only_enabled = yes }
			SWE_ericsson_reconstruct_reached_history = yes
		}
		else = {
			if = {
				limit = { NOT = { has_country_flag = SWE_ericsson_history_initialized } }
				set_country_flag = SWE_ericsson_history_initialized
				SWE_ericsson_reconstruct_start_history = yes
			}
			SWE_ericsson_schedule_current_year_event = yes
			SWE_ericsson_recover_interrupted_events = yes
		}
	}
}
