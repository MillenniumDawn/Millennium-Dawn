AI_record_score = {
	set_temp_variable_to_random = {
		var = state_randomizer
		min = -15
		max = 15
		integer = yes
	}
	set_temp_variable = { AI_score = { value = AI_score add = state_randomizer add = state_score_adjustment } }

	if = { limit = { check_variable = { AI_score > AI_best_score } }
		set_temp_variable = { AI_best_score = AI_score }
		set_temp_variable = { AI_best_target = THIS.id }
		set_temp_variable = { AI_best_type = AI_score_type }
	}
}

# Building Type 4: Infrastructure
AI_get_infrastructure_score = {
	if = { limit = { free_building_slots = { building = infrastructure size > 0  include_locked = no } }
		set_temp_variable = { AI_score = 0 }
		set_temp_variable = { planned_infra = { value = building_level@infrastructure add = pending_infra } }

		if = { limit = { check_variable = { planned_infra < 5 } }
			set_temp_variable = { AI_score = { value = 175 subtract = { value = building_level@infrastructure multiply = 20 } } }
			if = {
				limit = {
					OR = {
						has_resources_amount = { resource = aluminium amount > 10 }
						has_resources_amount = { resource = oil amount > 10 }
						has_resources_amount = { resource = steel amount > 10 }
						has_resources_amount = { resource = rubber amount > 10 }
						has_resources_amount = { resource = tungsten amount > 10 }
						has_resources_amount = { resource = chromium amount > 10 }
					}
				}
				add_to_temp_variable = { AI_score = 25 }
			}
			if = {
				limit = {
					OR = {
						has_resources_amount = { resource = aluminium amount > 20 }
						has_resources_amount = { resource = oil amount > 20 }
						has_resources_amount = { resource = steel amount > 20 }
						has_resources_amount = { resource = rubber amount > 20 }
						has_resources_amount = { resource = tungsten amount > 20 }
						has_resources_amount = { resource = chromium amount > 20 }
					}
				}
				add_to_temp_variable = { AI_score = 25 }
			}
			if = { limit = { free_building_slots = { building = industrial_complex size < 2 include_locked = no } }
				subtract_from_temp_variable = { AI_score = 15 }
			}
			if = { limit = { free_building_slots = { building = industrial_complex size < 5 include_locked = no } }
				subtract_from_temp_variable = { AI_score = 15 }
			}
			if = { limit = { free_building_slots = { building = industrial_complex size < 10 include_locked = no } }
				subtract_from_temp_variable = { AI_score = 15 }
			}
		}

		set_temp_variable = { AI_score_type = 4 }
		AI_record_score = yes
	}
}

AI_count_pending_projects = {
	set_temp_variable = { pending_civs = 0 }
	set_temp_variable = { pending_mils = 0 }
	set_temp_variable = { pending_docks = 0 }
	set_temp_variable = { pending_offices = 0 }
	set_temp_variable = { pending_antiair = 0 }
	set_temp_variable = { pending_radar = 0 }
	set_temp_variable = { pending_airbases = 0 }
	set_temp_variable = { pending_fuel_silos = 0 }
	set_temp_variable = { pending_internet = 0 }
	set_temp_variable = { pending_infra = 0 }
	set_temp_variable = { pending_renewable = 0 }
	set_temp_variable = { pending_fossil = 0 }
	set_temp_variable = { pending_nuclear = 0 }
	set_temp_variable = { pending_shared_slot_building = 0 }
	for_each_loop = {
		array = project_type_in_state
		value = v
		index = i
		if = { limit = { check_variable = { project_type_in_state^i = 1 } }
			add_to_temp_variable = { pending_civs = 1 }
			add_to_temp_variable = { pending_shared_slot_building = 1 }
		}
		else_if = { limit = { check_variable = { project_type_in_state^i = 2 } }
			add_to_temp_variable = { pending_mils = 1 }
			add_to_temp_variable = { pending_shared_slot_building = 1 }
		}
		else_if = { limit = { check_variable = { project_type_in_state^i = 3 } }
			add_to_temp_variable = { pending_docks = 1 }
			add_to_temp_variable = { pending_shared_slot_building = 1 }
		}
		else_if = { limit = { check_variable = { project_type_in_state^i = 4 } }
			add_to_temp_variable = { pending_infra = 1 }
		}
		else_if = { limit = { check_variable = { project_type_in_state^i = 5 } }
			add_to_temp_variable = { pending_offices = 1 }
			add_to_temp_variable = { pending_shared_slot_building = 1 }
		}
		else_if = { limit = { check_variable = { project_type_in_state^i = 6 } }
			add_to_temp_variable = { pending_antiair = 1 }
		}
		else_if = { limit = { check_variable = { project_type_in_state^i = 7 } }
			add_to_temp_variable = { pending_radar = 1 }
		}
		else_if = { limit = { check_variable = { project_type_in_state^i = 8 } }
			add_to_temp_variable = { pending_airbases = 1 }
		}
		else_if = { limit = { check_variable = { project_type_in_state^i = 9 } }
			add_to_temp_variable = { pending_fuel_silos = 1 }
		}
		else_if = { limit = { check_variable = { project_type_in_state^i = 10 } }
			add_to_temp_variable = { pending_internet = 1 }
		}
		else_if = { limit = { check_variable = { project_type_in_state^i = 11 } }
			add_to_temp_variable = { pending_renewable = 1 }
		}
		else_if = { limit = { check_variable = { project_type_in_state^i = 12 } }
			add_to_temp_variable = { pending_fossil = 1 }
		}
		else_if = { limit = { check_variable = { project_type_in_state^i = 13 } }
			add_to_temp_variable = { pending_nuclear = 1 }
		}
	}
}

				value = 30
				if = { limit = { value = energy_balance_ratio less_than = 1.00 } add = 10 }
				if = { limit = { value = energy_balance_ratio less_than = 0.95 } add = 15 }
				if = { limit = { value = energy_balance_ratio less_than = 0.90 } add = 20 }
				if = { limit = { value = energy_balance_ratio less_than = 0.85 } add = 25 }
				if = { limit = { value = energy_balance_ratio less_than = 0.80 } add = 30 }
				if = { limit = { value = energy_balance_ratio less_than = 0.70 } add = 40 }
				if = { limit = { value = tgt_gdp_per_capita greater_than = 25 } add = 10 }
				if = { limit = { value = tgt_gdp_per_capita greater_than = 35 } add = 10 }
				if = { limit = { value = tgt_gdp_per_capita greater_than = 45 } add = 5 }
				if = { limit = { value = THIS.state_population_k greater_than = 2000 } add = 5 }
				if = { limit = { value = THIS.state_population_k greater_than = 4000 } add = 10 }
				if = { limit = { value = THIS.state_population_k greater_than = 6000 } add = 5 }
				if = { limit = { value = tgt_oil_imported greater_than = 0 } add = 15 }
				if = { limit = { value = tgt_idea_heavy_power_restrict greater_than = 0 } add = 10 }
				if = { limit = { value = tgt_idea_some_power_restrict greater_than = 0 } add = 5 }
				# Can fuel its own reactors
				if = { limit = { value = tgt_enrichment_facilities greater_than = 0 } add = 25 }
				if = { limit = { value = tgt_nuclear_reactors less_than = 1 } subtract = 75 }
				if = { limit = { value = tgt_gdp_per_capita less_than = 20 } subtract = 100 }
				if = { limit = { value = tgt_gdp_total less_than = 500 } subtract = 75 }
				if = { limit = { value = tgt_interest_rate greater_than = 10.5 } subtract = 60 }
				if = { limit = { value = tgt_interest_rate greater_than = 8.5 } subtract = 40 }
				if = { limit = { value = energy_balance_ratio greater_than = 0.999 } subtract = 1000 }
				subtract = { value = building_level@nuclear_reactor add = pending_nuclear multiply = 50 }
				if = { limit = { value = tgt_idea_heavy_consumption greater_than = 0 } subtract = 30 }
				if = { limit = { value = tgt_idea_some_consumption greater_than = 0 } subtract = 20 }
			}
		}

		set_temp_variable = { AI_score_type = 13 }
		AI_record_score = yes
	}
}

# State-wide score change applied to every building type by AI_record_score
AI_get_state_score_adjustment = {
	set_temp_variable = {
		state_score_adjustment = {
			value = 0
			if = { limit = { value = THIS.state_population_k less_than = 100 } subtract = 20 }
			if = { limit = { value = THIS.state_population_k less_than = 85 } subtract = 40 }
			if = { limit = { value = THIS.state_population_k less_than = 60 } subtract = 80 }
		}
	}
	if = { limit = { has_state_flag = currently_encouraged_investments }
		add_to_temp_variable = { state_score_adjustment = 20 }
	}
	if = { limit = { has_state_flag = HOL_autonomy_economic_zone }
		add_to_temp_variable = { state_score_adjustment = 20 }
	}
	if = { limit = { has_state_flag = ENG_found_invest_northern_ireland_active }
		add_to_temp_variable = { state_score_adjustment = 20 }
	}
}

AI_country_selection_calculation = {
	if = { limit = { has_idea = non_state_actor }
		add_to_temp_variable = { AI_country_score = -60 }
	}
	if = { limit = { has_country_flag = recently_refused_investment@PREV }
		add_to_temp_variable = { AI_country_score = -100 }
	}
	if = {
		limit = { has_country_flag = recently_accepted_an_investment }
		add_to_temp_variable = { AI_country_score = -50 }
	}

	if = { limit = { is_in_faction_with = PREV }
		add_to_temp_variable = { AI_country_score = 25 }
	}
	if = { limit = { has_country_flag = trade_agreement@PREV }
		add_to_temp_variable = { AI_country_score = 25 }
	}
	if = {
		limit = { has_country_flag = mutual_investment_treaty_@PREV }
		add_to_temp_variable = { AI_country_score = 25 }
	}
	if = { limit = { PREV = { check_variable = { active_projects > 0 } } }
		PREV = {
			for_each_loop = {
				array = project_array
				value = v
				index = i
				if = {
					limit = {
						check_variable = { v < 0 }
						check_variable = { project_target_country^i = PREV.id }
					}
					add_to_temp_variable = { AI_country_score = -35 }
				}
			}
		}
	}

	if = { limit = { has_idea = EU_member }
		if = { limit = { check_variable = { inv_has_ECB_president = 1 } }
			add_to_temp_variable = { AI_country_score = 15 }
		}
		if = { limit = { check_variable = { inv_has_EU_finance_minister = 1 } }
			add_to_temp_variable = { AI_country_score = 5 }
		}
	}

	if = { limit = { is_in_array = { array = influence_array value = PREV.id } }
		add_to_temp_variable = { AI_country_score = 40 }
	}
	if = { limit = { PREV = { is_in_array = { array = influence_array value = PREV.id } } }
		add_to_temp_variable = { AI_country_score = 40 }
	}

	set_temp_variable = { opinion_value = opinion@PREV }
	set_temp_variable = {
		AI_country_score = {
			value = AI_country_score
			add = { value = opinion_value multiply = 0.30 clamp = { min = -75 max = 75 } }
			if = { limit = { value = corporate_tax_rate greater_than = 30 } subtract = { value = corporate_tax_rate multiply = 0.50 } }
			# Countries under 30k gdpc gain 2 score per 1k gdpc below 30k, richer ones lose the same
			add = { value = gdp_per_capita multiply = -2 add = 60 }
			# Big economies are slightly less likely to get investments, to thin out the flood into developed European countries
			add = { value = gdp_total multiply = -0.002 clamp = { min = -20 max = 0 } }
		}
	}

	if = { limit = { has_idea = corruption_level_10 }
		add_to_temp_variable = { AI_country_score = -25 }
	}
	else_if = { limit = { has_idea = corruption_level_09 }
		add_to_temp_variable = { AI_country_score = -20 }
	}
	else_if = { limit = { has_idea = corruption_level_08 }
		add_to_temp_variable = { AI_country_score = -17.5 }
	}
	else_if = { limit = { has_idea = corruption_level_07 }
		add_to_temp_variable = { AI_country_score = -15 }
	}
	else_if = { limit = { has_idea = corruption_level_06 }
		add_to_temp_variable = { AI_country_score = -12.5 }
	}
	else_if = { limit = { has_idea = corruption_level_05 }
		add_to_temp_variable = { AI_country_score = -10 }
	}
	else_if = { limit = { has_idea = corruption_level_04 }
		add_to_temp_variable = { AI_country_score = -7.5 }
	}
	else_if = { limit = { has_idea = corruption_level_03 }
		add_to_temp_variable = { AI_country_score = -5 }
	}
	else_if = { limit = { has_idea = corruption_level_02 }
		add_to_temp_variable = { AI_country_score = -2 }
	}

	if = {
		limit = {
			country_exists = PREV
			is_on_same_continent_as = PREV
		}
		add_to_temp_variable = { AI_country_score = 25 }
	}

	# AI Support the other AI
	if = {
		limit = {
			OR = {
				num_of_available_civilian_factories < 15
				check_variable = { treasury_rate < 1.00 }
			}
		}
		add_to_temp_variable = { AI_country_score = 30 }
	}

	# Invest in Enemy of my Enemy
	if = {
		limit = {
			any_enemy_country = {
				OR = {
					has_war_with = ROOT
					has_opinion = { target = ROOT value < -50 }
				}
			}
		}
		add_to_temp_variable = { AI_country_score = 30 }
	}

	set_temp_variable_to_random = {
		var = randomizer_country_score
		min = 25
		max = 75
		integer = yes
	}
	set_temp_variable = {
		AI_country_score = {
			value = AI_country_score
			subtract = randomizer_country_score
			multiply = {
				value = 1
				subtract = modifier@receiving_investment_cost_modifier
				add = inv_cost_modifier
				clamp = { min = 0.50 max = 1.50 }
			}
			multiply = {
				value = 1
				add = modifier@receiving_investment_duration_modifier
				add = inv_duration_modifier
				clamp = { min = 0.50 max = 1.50 }
			}
		}
	}
}
