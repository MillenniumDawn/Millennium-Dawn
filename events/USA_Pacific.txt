# The American side of the Pacific contest.
#
# The Solomon Islands storyline is scored through the Australian competition
# framework held on the Solomon Islands' own country scope. Which configuration
# that competition is running in decides what these events mean:
#
#   Mode 2  Washington is bankrolling Canberra. Everything the United States
#           spends here strengthens an ally's hand rather than its own, and the
#           content reads as burden sharing inside an alliance.
#   Mode 3  Canberra has traded the American guarantee for a Chinese one.
#           Washington stops working through an ally and opens its own file on
#           Honiara, on the Compact of Free Association model that has held
#           Micronesia, the Marshalls and Palau since 1986. That is a bid
#           against Australia, and it is priced like one.
#
# The Compact arc runs across four decisions in USA_solomon_engagement_category
# and reports its beats through the events below.

add_namespace = usa_pacific
add_namespace = usa_pacific_news

	fire_only_once = yes

	option = { # Put the full package on the table
		name = usa_pacific.10.o1

		}

		ai_chance = {
			base = 7
			modifier = {
				factor = 0
				check_variable = { treasury < 20 }
			}
		}

country_event = { # The Compact is signed over our objections
	id = usa_pacific.21
	title = usa_pacific.21.t
	desc = usa_pacific.21.d
	picture = GFX_political_deal

	is_triggered_only = yes
	fire_only_once = yes

	trigger = { original_tag = AST }

	option = { # Recall our ambassador from Washington
		name = usa_pacific.21.o1
		log = "[GetDateText]: [This.GetName]: usa_pacific.21.o1 executed"

		add_political_power = -30
		add_stability = -0.03
		if = {
			limit = { country_exists = USA }
			# The protest at usa_pacific.20 may already have put a grievance on
			# the record. Signing the Compact over it is worse, not twice the
			# same, so the earlier modifier is retired and replaced with the
			# heavier one rather than a second copy being stacked underneath it.
			remove_opinion_modifier = {
				target = USA
				modifier = USA_courting_our_near_region
			}
			add_opinion_modifier = {
				target = USA
				modifier = USA_took_our_near_region
			}
			USA = {
				add_opinion_modifier = {
					target = ROOT
					modifier = usa_fp_severe_confrontation
				}
			}
		}
		# Nothing pushes Canberra further into Beijing's arms than Washington
		# buying the capital Canberra used to speak for.
		if = {
			limit = {
				country_exists = CHI
				NOT = { has_war_with = CHI }
			}
			CHI = {
				add_opinion_modifier = {
					target = ROOT
					modifier = declaration_of_friendship
				}
			}
			add_opinion_modifier = {
				target = CHI
				modifier = declaration_of_friendship
			}
		}

		ai_chance = { base = 6 }
	}

	option = { # Swallow it and keep the relationship
		name = usa_pacific.21.o2
		log = "[GetDateText]: [This.GetName]: usa_pacific.21.o2 executed"

		add_political_power = -10
		add_stability = -0.05
		if = {
			limit = { country_exists = USA }
			add_opinion_modifier = {
				target = USA
				modifier = SOL_abandoned_us
			}
		}

		ai_chance = { base = 3 }
	}
}
