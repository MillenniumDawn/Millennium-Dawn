GRN_ai_historical_path = {
	OR = {
		is_historical_focus_on = yes
		has_global_flag = GRN_HISTORICAL_FOCUS_PATH
	}
}

GRN_ai_expansionist_path = {
	OR = {
		has_global_flag = GRN_SCANDINAVIAN_SERENITY_FOCUS_PATH
		has_global_flag = GRN_CONTINENTAL_HEGEMONY_FOCUS_PATH
	}
}

GRN_ai_not_expansionist_path = {
	NOT = { GRN_ai_expansionist_path = yes }
	GRN_ai_historical_path = yes
}

GRN_ai_not_scandinavian_path = {
	NOT = { has_global_flag = GRN_SCANDINAVIAN_SERENITY_FOCUS_PATH }
	OR = {
		GRN_ai_historical_path = yes
		has_global_flag = GRN_CONTINENTAL_HEGEMONY_FOCUS_PATH
	}
}

GRN_ai_not_continental_path = {
	NOT = { has_global_flag = GRN_CONTINENTAL_HEGEMONY_FOCUS_PATH }
	OR = {
		GRN_ai_historical_path = yes
		has_global_flag = GRN_SCANDINAVIAN_SERENITY_FOCUS_PATH
	}
}
