		visible = {
			has_completed_focus = MER_infrastructure_corridor_program
			has_country_flag = mercosur_joint_tree_member

		}

		available = {
			check_variable = { treasury > 25 }
		}

		days_re_enable = 365

		complete_effect = {
			set_temp_variable = { treasury_change = -25 }
			modify_treasury_effect = yes

			for_each_scope_loop = {
				array = global.mercosur_potential
				if = {
					limit = {
						has_dynamic_modifier = mercosur_member_state_modifier
					}
					add_timed_idea = {
						idea = MER_corridor_investment_round_modifier
						days = 180
					}
				}
			}

			effect_tooltip = {
				add_timed_idea = {
					idea = MER_corridor_investment_round_modifier
					days = 180
				}
			}
		}

		ai_will_do = {
			base = 1
			modifier = {
				factor = 0
				check_variable = { interest_rate > 10 }
			}
		}
